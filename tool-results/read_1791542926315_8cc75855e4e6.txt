    10→/** HTML kaçış — tüm kullanıcı verilerinde kullanılır. */
    11→export function esc(s: unknown): string {
    12→  return String(s ?? '')
    13→    .replace(/&/g, '&amp;')
    14→    .replace(/</g, '&lt;')
    15→    .replace(/>/g, '&gt;')
    16→    .replace(/"/g, '&quot;')
    17→    .replace(/'/g, '&#39;');
    18→}

    20→/** <script> veri adası için JSON — </script> patlamasına karşı < karakteri kodlanır. */
    21→function jsonIsland(data: unknown): string {
    22→  return JSON.stringify(data).replace(/</g, '\\u003c');
    23→}

   319→function pageShell(o: { title: string; desc: string; body: string; scripts?: string[]; extraCss?: string[]; gwPort?: string | null }): string {
   320→  // Gateway (XTransformPort) ile açıldıysa göreli varlık URL'lerine param ekle —
   321→  // aksi hâlde /assets/* istekleri gateway'den CRM'e düşer ve 404 olur.
   322→  const gw = (p: string) =>
   323→    o.gwPort ? p + (p.includes('?') ? '&' : '?') + 'XTransformPort=' + encodeURIComponent(o.gwPort) : p;
   324→  const scripts = (o.scripts || []).map((s) => `<script src="${gw(s)}" defer></script>`).join('\n  ');
   325→  const extraCss = (o.extraCss || []).map((s) => `<link rel="stylesheet" href="${gw(s)}">`).join('\n  ');
   326→  return `<!DOCTYPE html>
   327→<html lang="tr">
   328→<head>
   329→<meta charset="UTF-8">
   330→<meta name="viewport" content="width=device-width, initial-scale=1">
   331→<title>${esc(o.title)}</title>
   332→<meta name="description" content="${esc(o.desc)}">
   333→<link rel="icon" href="${gw(FAVICON)}">
   334→<link rel="stylesheet" href="${gw('/assets/style.css')}">
   335→  ${extraCss}
   336→</head>
   337→<body>
   338→<div class="page">
   339→${o.body}
   340→</div>
   341→<div id="toast" role="status" aria-live="polite"></div>
   342→  ${scripts}
   343→</body>
   344→</html>`;
   345→}
