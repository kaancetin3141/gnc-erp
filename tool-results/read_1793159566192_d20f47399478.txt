   549→  async function handleSave() {

   554→    if (!form.customerPhone.trim()) {

   566→    setSaving(true)

   568→      if (formClosureConflicts.length > 0) {

   578→        })
   579→        setSaving(false)
   580→        return
   581→      }

   583→      const [hh, mm] = form.time.split(':').map(Number)
   584→      const start = new Date(y, m - 1, d, hh, mm, 0, 0)
   585→      const payload = {
   586→        staffId: form.staffId === 'any' ? undefined : form.staffId,
   587→        serviceId: form.serviceId,
   588→        customerName: form.customerName,
   589→        customerPhone: form.customerPhone,

   592→        date: start.toISOString(),
   593→        status: form.status,
   594→        // Çakışma varsa kullanıcı uyarıldı — bilinçli override
   595→        force: formConflicts.length > 0 ? true : undefined,
   596→      }
   597→      if (editAppt) {
   598→        await apiPatch(
   599→          `/api/appointments/providers/${providerId}/appointments/${editAppt.id}`,
   600→          { ...payload, staffId: form.staffId },

   602→        toast.success('Randevu güncellendi')
   603→      } else {
   604→        await apiPost(
   605→          `/api/appointments/providers/${providerId}/appointments`,
   606→          payload,
   607→        )
   608→        toast.success('Randevu oluşturuldu')
   609→      }
   610→      qc.invalidateQueries({ queryKey: ['appointment-appointments', providerId] })
   611→      setDialogOpen(false)
   612→    } catch (e) {
   613→      toast.error(e instanceof Error ? e.message : 'İşlem başarısız')
   614→    } finally {
   615→      setSaving(false)
   616→    }
   617→  }

   648→  async function handleDrop(target: Date, staffColId?: string) {
   649→    const appt = dragAppt
   650→    setDragAppt(null)
   651→    setDragOverKey(null)
   652→    if (!appt) return
   653→    // target = hedef hücrenin TAM tarihi + saati (gün/hafta/personel görünümünden)
   654→    const old = new Date(appt.date)
   655→    const nd = target
   656→    const newStaff = staffColId === undefined ? undefined : staffColId === 'any' ? 'any' : staffColId
   657→    const sameTime = nd.getTime() === old.getTime()
   658→    const sameStaff = newStaff === undefined || newStaff === (appt.staffId ?? 'any')
   659→    if (sameTime && sameStaff) return
   660→    try {
   661→      const payload: Record<string, unknown> = { date: nd.toISOString() }
   662→      if (newStaff !== undefined) payload.staffId = newStaff
   663→      await apiPatch(`/api/appointments/providers/${providerId}/appointments/${appt.id}`, payload)
   664→      qc.invalidateQueries({ queryKey: ['appointment-appointments', providerId] })
   665→      toast.success('Randevu taşındı', {
   666→        description: `${appt.customerName} → ${nd.toLocaleDateString('tr-TR', { day: 'numeric', month: 'short', weekday: 'short' })} ${hhmm(nd)}`,
   667→      })
   668→    } catch (e) {
   669→      toast.error('Taşıma başarısız', {
   670→        description: e instanceof Error ? e.message : 'Saat dilimi dolu olabilir',
   671→      })
   672→    }
   673→  }

   675→  // handleDrop'a her zaman güncel erişim (native document dinleyicileri için)
   676→  const handleDropRef = useRef(handleDrop)
   677→  handleDropRef.current = handleDrop

   679→  // Sürükleme aktifken document-seviyesi native dragover/drop —
   680→  // React'ın root capture-phase drop kaydı untrusted event'lerde tetiklenmediği için
   681→  // native dinleyici kullanılır (gerçek tarayıcı + otomasyon uyumlu).
   682→  useEffect(() => {
   683→    if (!dragAppt) return
   684→    const onDocDragOver = (e: DragEvent) => {
   685→      const cell = (e.target as HTMLElement | null)?.closest?.('[data-appt-cell]') as HTMLElement | null
   686→      if (cell && cell.dataset.inHours === '1') {
   687→        e.preventDefault()
   688→        if (e.dataTransfer) e.dataTransfer.dropEffect = 'move'
   689→        const key = cell.dataset.cellKey ?? null
   690→        setDragOverKey((prev) => (prev === key ? prev : key))
   691→      } else {
   692→        setDragOverKey((prev) => (prev === null ? prev : null))
   693→      }
   694→    }
   695→    const onDocDrop = (e: DragEvent) => {
   696→      const cell = (e.target as HTMLElement | null)?.closest?.('[data-appt-cell]') as HTMLElement | null
   697→      if (!cell) return
   698→      e.preventDefault()
   699→      const dayIso = cell.dataset.dayIso
   700→      const slotMin = Number(cell.dataset.slotMin)
   701→      if (!dayIso || isNaN(slotMin)) return
   702→      const d = new Date(dayIso)
   703→      const staffId = cell.dataset.staffId // undefined = personeli koru (gün/hafta)
   704→      handleDropRef.current(
   705→        new Date(d.getFullYear(), d.getMonth(), d.getDate(), Math.floor(slotMin / 60), slotMin % 60),
   706→        staffId,
   707→      )
   708→    }
   709→    const onDocDragEnd = () => {
   710→      setDragAppt(null)
   711→      setDragOverKey(null)
   712→    }
   713→    document.addEventListener('dragover', onDocDragOver)
   714→    document.addEventListener('drop', onDocDrop)
   715→    document.addEventListener('dragend', onDocDragEnd)
   716→    return () => {
   717→      document.removeEventListener('dragover', onDocDragOver)
   718→      document.removeEventListener('drop', onDocDrop)
   719→      document.removeEventListener('dragend', onDocDragEnd)
   720→    }
   721→  }, [dragAppt])

   724→  function buildDayWeekCells(slot: { min: number }): SlotCellSpec[] {
   725→    return dateRange.map((d) => {
   726→      const dk = dayKeyFromDate(d)
   727→      const sched = workingHours[dk]
   728→      const isClosed = !sched || sched.closed || !sched.start || !sched.end
   729→      const inHours = !isClosed &&
   730→        slot.min >= timeToMinutes(sched!.start!) &&
   731→        slot.min < timeToMinutes(sched!.end!)
   732→      const dayAppts = apptsByDay.get(d.toDateString()) ?? []
   733→      const slotAppts = dayAppts.filter((a) => {
   734→        const ad = new Date(a.date)
   735→        const aMin = ad.getHours() * 60 + ad.getMinutes()
   736→        return aMin >= slot.min && aMin < slot.min + 30
   737→      })
   738→      return {
   739→        key: `${d.toISOString()}-${slot.min}`,
   740→        date: d,
   741→        appts: slotAppts,
   742→        inHours,
   743→        isToday: d.toDateString() === todayKey,
   744→      }
   745→    })
   746→  }

   748→  function buildStaffCells(slot: { min: number }): SlotCellSpec[] {
   749→    const day = dateRange[0]
   750→    const dk = dayKeyFromDate(day)
   751→    const sched = workingHours[dk]
   752→    const inHours = !!sched && !sched.closed && !!sched.start && !!sched.end &&
   753→      slot.min >= timeToMinutes(sched.start) &&
   754→      slot.min < timeToMinutes(sched.end)
   755→    const dayAppts = apptsByDay.get(day.toDateString()) ?? []
   756→    const slotStartMs = day.getTime() + slot.min * 60_000
   757→    const slotEndMs = slotStartMs + 30 * 60_000
   758→    return staffColumns.map((col) => {
   759→      const colAppts = dayAppts.filter((a) => {
   760→        if (col.id === 'unassigned') return !a.staffId
   761→        return a.staffId === col.id
   762→      }).filter((a) => {
   763→        const ad = new Date(a.date)
   764→        const aMin = ad.getHours() * 60 + ad.getMinutes()
   765→        return aMin >= slot.min && aMin < slot.min + 30
   766→      })
   767→      // Bu personel bu slot'ta izinli mi?
   768→      const dayTimeOffs = col.id === 'unassigned' ? [] : (timeOffsByStaffDay.get(`${col.id}|${day.toDateString()}`) ?? [])

   772→        date: day,
   773→        appts: colAppts,
   774→        inHours,
   775→        isToday: day.toDateString() === todayKey,

   777→        timeOffs: slotTimeOffs.length > 0 ? slotTimeOffs : undefined,
