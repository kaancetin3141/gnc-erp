    49→interface OrderDoc {
    50→  id: string
    51→  number: string
    52→  status: string
    53→  orderDate: string
    54→  totalAmount?: number
    55→  currency?: string
    56→  customer?: { id: string; name: string; segment?: string; status?: string } | null
    57→  quote?: { id: string; number: string } | null
    58→  invoice?: { id: string; number: string; status: string; packingListNo?: string | null } | null
    59→  irsaliyeler?: { id: string; number: string; status: string }[]
    60→  exportDocs?: { id: string; type: string; number: string; status: string }[]
    61→  _count?: { trackingSteps: number; productionItems?: number }
    62→}
    63→
    64→interface OrdersResponse {
    65→  items: OrderDoc[]
    66→  total: number
    67→  hidePrices?: boolean
    68→}

    79→// Fatura durumu etiketi (CSV için)
    80→function ORDER_DOC_LABEL(status: string): string {
    81→  switch (status) {
    82→    case 'odeme_bekliyor': return 'Ödeme Bekliyor'
    83→    case 'odendi': return 'Ödendi'
    84→    case 'gecikti': return 'Gecikti'
    85→    case 'iptal': return 'İptal'
    86→    default: return status
    87→  }
    88→}

    95→export function DocumentsView() {
    96→  const qc = useQueryClient()
    97→  const { user } = useAppStore()
    98→  const su = user as SessionUser | null
    99→
   100→  // Yetkiler
   101→  const canSeeInvoice = hasPermission(su, 'invoices.view') || hasPermission(su, 'erp.manage')
   102→  const canSeeIrsaliye = hasPermission(su, 'irsaliye.view')
   103→  const canSeePacking = canSeeIrsaliye // çeki listesi = irsaliye.view sahiplerine açık (müdür + depocu)
   104→  const canSeeExport = canSeeInvoice // ihracat belgeleri = ticari evrak (müdür + admin, depocu hariç)
   105→
   106→  const [search, setSearch] = useState('')
   107→  const [openCompanies, setOpenCompanies] = useState<Record<string, boolean>>({})
   108→  const [generating, setGenerating] = useState<string | null>(null) // "orderId:type"
   109→  const [activeDoc, setActiveDoc] = useState<GeneratedDoc | null>(null)
   110→  const [combinedOrder, setCombinedOrder] = useState<{ id: string; number: string; customerName?: string } | null>(null)
   111→  const [companyPrint, setCompanyPrint] = useState<{ name: string; orders: { id: string; number: string }[] } | null>(null)
   112→
   113→  const canExport = hasPermission(su, 'export.data')
   114→
   115→  const { data, isLoading, isFetching, refetch } = useQuery({
   116→    queryKey: ['documents-orders', search],
   117→    queryFn: () => {
   118→      const p = new URLSearchParams({ limit: '200' })
   119→      if (search) p.set('search', search)
   120→      return apiGet<OrdersResponse>(`/api/orders?${p.toString()}`)
   121→    },
   122→  })
   123→
   124→  const orders = useMemo(() => data?.items ?? [], [data])
   125→  const hidePrices = data?.hidePrices === true
   126→
   127→  // Şirketlere göre grupla
   128→  const companies = useMemo(() => {
   129→    const map = new Map<string, { name: string; segment?: string; orders: OrderDoc[] }>()
   130→    for (const o of orders) {
   131→      const key = o.customer?.id ?? '_'
   132→      if (!map.has(key)) {
   133→        map.set(key, {
   134→          name: o.customer?.name ?? 'Bilinmeyen Şirket',
   135→          segment: o.customer?.segment,
   136→          orders: [],
   137→        })
   138→      }
   139→      map.get(key)!.orders.push(o)
   140→    }
   141→    // Şirket adına göre sırala
   142→    return Array.from(map.values()).sort((a, b) => a.name.localeCompare(b.name, 'tr'))
   143→  }, [orders])
   144→
   145→  // Özet
   146→  const stats = useMemo(() => {
   147→    const invoiced = orders.filter((o) => o.invoice).length
   148→    const irsaliye = orders.filter((o) => (o.irsaliyeler?.length ?? 0) > 0).length
   149→    const packing = orders.filter((o) => o.invoice?.packingListNo).length
   150→    return { orders: data?.total ?? 0, companies: companies.length, invoiced, irsaliye, packing }
   151→  }, [orders, companies, data])
   152→
   153→  // Belge üret + PDF aç
   154→  const generateDoc = async (order: OrderDoc, type: 'invoice' | 'irsaliye' | 'packing_list') => {
   155→    const genKey = `${order.id}:${type}`
   156→    setGenerating(genKey)
   157→    try {
   158→      const res = await apiPost<{
   159→        type: string
   160→        created: boolean
   161→        document: {
   162→          id: string
   163→          number: string
   164→          packingListNo?: string | null
   165→        }
   166→        documentType: 'invoice' | 'irsaliye'
   167→      }>(`/api/orders/${order.id}/generate-document`, { type })
   168→
   169→      if (res.created) {
   170→        toast.success(`Belge otomatik oluşturuldu${res.document.number ? `: ${res.document.number}` : ''}`)
   171→      }
   172→
   173→      if (type === 'irsaliye') {
   174→        setActiveDoc({ kind: 'irsaliye', irsaliyeId: res.document.id, irsaliyeNumber: res.document.number })
   175→      } else if (type === 'packing_list') {
   176→        setActiveDoc({ kind: 'packing', invoiceId: res.document.id })
   177→      } else {
   178→        setActiveDoc({ kind: 'invoice', invoiceId: res.document.id })
   179→      }
   180→
   181→      // Belge durumlarını tazele
   182→      qc.invalidateQueries({ queryKey: ['documents-orders'] })
   183→      qc.invalidateQueries({ queryKey: ['orders'] })
   184→      qc.invalidateQueries({ queryKey: ['irsaliye'] })
   185→      qc.invalidateQueries({ queryKey: ['invoices'] })
   186→    } catch (e) {
   187→      toast.error(e instanceof Error ? e.message : 'Belge oluşturulamadı')
   188→    } finally {
   189→      setGenerating(null)
   190→    }
   191→  }
   192→
   193→  const toggleCompany = (name: string) => {
   194→    setOpenCompanies((prev) => ({ ...prev, [name]: !prev[name] }))
   195→  }
   196→
   197→  // CSV dışa aktarma — siparişler + belge durumları
   198→  const handleExportCSV = () => {
   199→    if (orders.length === 0) {
   200→      toast.error('Dışa aktarılacak sipariş yok')
   201→      return
   202→    }
   203→    const rows = orders.map((o) => ({
   204→      'Sipariş No': o.number,
   205→      'Şirket': o.customer?.name ?? '',
   206→      'Tarih': formatDate(o.orderDate),
   207→      'Durum': ORDER_STATUS_META[o.status]?.label ?? o.status,
   208→      'Fatura No': o.invoice?.number ?? '',
   209→      'Fatura Durumu': o.invoice ? ORDER_DOC_LABEL(o.invoice.status) : 'Yok',
   210→      'İrsaliye No': o.irsaliyeler?.[0]?.number ?? '',
   211→      'İrsaliye Durumu': o.irsaliyeler?.[0] ? ORDER_STATUS_META[o.irsaliyeler[0].status]?.label ?? o.irsaliyeler[0].status : 'Yok',
   212→      'Çeki Listesi No': o.invoice?.packingListNo ?? '',
   213→      ...(hidePrices ? {} : { 'Tutar': o.totalAmount ?? 0, 'Para': o.currency ?? 'TRY' }),
   214→    }))
   215→    const csv = toCSV(rows)
   216→    downloadFile(csv, `belge-yonetimi-${new Date().toISOString().slice(0, 10)}.csv`)
   217→    toast.success(`${orders.length} sipariş dışa aktarıldı`)
   218→  }
   219→
   220→  return (
   221→    <div className="space-y-5 animate-fade-in">
   222→      {/* Header */}
   223→      <div className="flex items-start justify-between flex-wrap gap-3">
   224→        <div>
   225→          <h2 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   226→            <FileStack className="w-6 h-6 text-violet-600" />
   227→            Belge Yönetimi
   228→          </h2>
   229→          <p className="text-sm text-muted-foreground mt-0.5">
   230→            Şirketler ve sipariş belgeleri — fatura, irsaliye, çeki listesi ve ihracat belgeleri otomatik üretilir
   231→            {canSeeIrsaliye && !canSeeInvoice && (
   232→              <span className="ml-2 text-violet-600 font-medium">(görüş alanınız: İrsaliye + Çeki Listesi)</span>
   233→            )}
   234→          </p>
   235→        </div>
   236→        <div className="flex gap-2 flex-wrap">
   237→          {canExport && (
   238→            <Button variant="outline" size="sm" onClick={handleExportCSV} disabled={orders.length === 0}>
   239→              <Download className="w-4 h-4 mr-1.5" />
   240→              CSV Dışa Aktar
   241→            </Button>
   242→          )}
   243→          <Button variant="outline" size="sm" onClick={() => refetch()} disabled={isFetching}>
   244→            <RefreshCw className={cn('w-4 h-4 mr-1.5', isFetching && 'animate-spin')} />
   245→            Yenile
   246→          </Button>
   247→        </div>
   248→      </div>
   249→
   250→      <Tabs defaultValue="orders" className="space-y-4">
   251→        <TabsList className={cn('grid w-full max-w-md', canSeeExport ? 'grid-cols-3' : 'grid-cols-2')}>
   252→          <TabsTrigger value="orders" className="gap-1.5">
   253→            <Package className="w-3.5 h-3.5" /> Sipariş Belgeleri
   254→          </TabsTrigger>
   255→          <TabsTrigger value="irsaliye" className="gap-1.5">
   256→            <Truck className="w-3.5 h-3.5" /> İrsaliye Listesi
   257→          </TabsTrigger>
   258→          {canSeeExport && (
   259→            <TabsTrigger value="ihracat" className="gap-1.5">
   260→              <Globe2 className="w-3.5 h-3.5" /> İhracat
   261→            </TabsTrigger>
   262→          )}
   263→        </TabsList>
   264→
   265→        {/* ==================== TAB 1: SİPARİŞ BELGELERİ ==================== */}
   266→        <TabsContent value="orders" className="space-y-4">
   267→          {/* Stats */}
   268→          <div className="grid grid-cols-2 sm:grid-cols-5 gap-3">
   269→            <Card>
   270→              <CardContent className="p-3">
   271→                <div className="text-[10px] uppercase tracking-wider text-muted-foreground">Sipariş</div>
   272→                <div className="text-xl font-bold">{stats.orders}</div>
   273→              </CardContent>
   274→            </Card>
   275→            <Card>
   276→              <CardContent className="p-3">
   277→                <div className="text-[10px] uppercase tracking-wider text-muted-foreground">Şirket</div>
   278→                <div className="text-xl font-bold">{stats.companies}</div>
   279→              </CardContent>
   280→            </Card>
   281→            {canSeeInvoice && (
   282→              <Card className="bg-emerald-50/50 dark:bg-emerald-950/20">
   283→                <CardContent className="p-3">
   284→                  <div className="text-[10px] uppercase tracking-wider text-emerald-700 dark:text-emerald-400">Faturalı</div>
   285→                  <div className="text-xl font-bold text-emerald-700 dark:text-emerald-400">{stats.invoiced}</div>
   286→                </CardContent>
   287→              </Card>
   288→            )}
   289→            {canSeeIrsaliye && (
   290→              <Card className="bg-violet-50/50 dark:bg-violet-950/20">
   291→                <CardContent className="p-3">
   292→                  <div className="text-[10px] uppercase tracking-wider text-violet-700 dark:text-violet-400">İrsaliyeli</div>
   293→                  <div className="text-xl font-bold text-violet-700 dark:text-violet-400">{stats.irsaliye}</div>
   294→                </CardContent>
   295→              </Card>
   296→            )}
   297→            {canSeePacking && (
   298→              <Card className="bg-amber-50/50 dark:bg-amber-950/20">
   299→                <CardContent className="p-3">
   300→                  <div className="text-[10px] uppercase tracking-wider text-amber-700 dark:text-amber-400">Çeki Listeli</div>
   301→                  <div className="text-xl font-bold text-amber-700 dark:text-amber-400">{stats.packing}</div>
   302→                </CardContent>
   303→              </Card>
   304→            )}
   305→          </div>
   306→
   307→          {/* Arama */}
   308→          <Card>
   309→            <CardContent className="p-4">
   310→              <div className="relative">
   311→                <Search className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground pointer-events-none" />
   312→                <Input
   313→                  value={search}
   314→                  onChange={(e) => setSearch(e.target.value)}
   315→                  placeholder="Şirket veya sipariş no ara..."
   316→                  className="pl-9"
   317→                />
   318→                {search && (
   319→                  <button
   320→                    onClick={() => setSearch('')}
   321→                    className="absolute right-2 top-1/2 -translate-y-1/2 p-1 hover:bg-muted rounded"
   322→                  >
   323→                    <X className="w-3.5 h-3.5 text-muted-foreground" />
   324→                  </button>
   325→                )}
   326→              </div>
   327→            </CardContent>
   328→          </Card>
   329→
   330→          {/* Şirketler → Siparişler */}
   331→          {isLoading ? (
   332→            <div className="space-y-3">
   333→              <Skeleton className="h-16 w-full" />
   334→              <Skeleton className="h-40 w-full" />
   335→              <Skeleton className="h-16 w-full" />
   336→            </div>
   337→          ) : companies.length === 0 ? (
   338→            <Card>
   339→              <CardContent className="p-12 text-center">
   340→                <Building2 className="w-10 h-10 mx-auto text-muted-foreground/50 mb-2" />
   341→                <p className="text-sm text-muted-foreground">Sipariş bulunamadı</p>
   342→              </CardContent>
   343→            </Card>
   344→          ) : (
   345→            <div className="space-y-3">
   346→              {companies.map((company) => {
   347→                const isOpen = openCompanies[company.name] ?? true
   348→                return (
   349→                  <Card key={company.name} className="overflow-hidden">
   350→                    {/* Şirket başlığı — tıklanınca açılır/kapanır, sağdaki butonlar tıklamayı yutmaz */}
   351→                    <div
   352→                      role="button"
   353→                      tabIndex={0}
   354→                      aria-expanded={isOpen}
   355→                      onClick={() => toggleCompany(company.name)}
   356→                      onKeyDown={(e) => {
   357→                        if (e.key === 'Enter' || e.key === ' ') {
   358→                          e.preventDefault()
   359→                          toggleCompany(company.name)
   360→                        }
   361→                      }}
   362→                      className="w-full flex items-center gap-3 p-4 bg-gradient-to-r from-muted/50 to-muted/20 hover:from-muted/70 hover:to-muted/30 transition-all text-left cursor-pointer select-none focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-violet-400"
   363→                    >
   364→                      {isOpen ? (
   365→                        <ChevronDown className="w-4 h-4 text-muted-foreground shrink-0" />
   366→                      ) : (

   368→                      )}

   370→                        <Building2 className="w-4 h-4 text-slate-600 dark:text-slate-300" />
   371→                      </div>
   372→                      <div className="min-w-0 flex-1">
   373→                        <div className="font-semibold text-sm truncate">{company.name}</div>
   374→                        <div className="text-[11px] text-muted-foreground">
   375→                          {company.orders.length} sipariş
   376→                          {company.segment ? ` · ${company.segment}` : ''}
   377→                        </div>
   378→                      </div>
   379→                      {/* Şirket belge özeti */}
   380→                      {(() => {
   381→                        const inv = company.orders.filter((o) => o.invoice).length
   382→                        const irs = company.orders.filter((o) => (o.irsaliyeler?.length ?? 0) > 0).length
   383→                        const pack = company.orders.filter((o) => o.invoice?.packingListNo).length
   384→                        return (
   385→                          <div className="hidden sm:flex items-center gap-1 shrink-0">
   386→                            {canSeeInvoice && inv > 0 && (
   387→                              <Badge variant="outline" className="text-[9px] bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-950/30 dark:text-emerald-300 dark:border-emerald-900/60 gap-0.5">
   388→                                <Receipt className="w-2.5 h-2.5" /> {inv}
   389→                              </Badge>
   390→                            )}
   391→                            {canSeeIrsaliye && irs > 0 && (
   392→                              <Badge variant="outline" className="text-[9px] bg-violet-50 text-violet-700 border-violet-200 dark:bg-violet-950/30 dark:text-violet-300 dark:border-violet-900/60 gap-0.5">
   393→                                <Truck className="w-2.5 h-2.5" /> {irs}
   394→                              </Badge>
   395→                            )}
   396→                            {canSeePacking && pack > 0 && (
   397→                              <Badge variant="outline" className="text-[9px] bg-amber-50 text-amber-700 border-amber-200 dark:bg-amber-950/30 dark:text-amber-300 dark:border-amber-900/60 gap-0.5">
   398→                                <ClipboardList className="w-2.5 h-2.5" /> {pack}
   399→                              </Badge>
   400→                            )}
   401→                          </div>
   402→                        )
   403→                      })()}
   404→                      <Badge variant="outline" className="text-[10px] shrink-0">
   405→                        {company.orders.length}
   406→                      </Badge>
   407→
   408→                      {/* ŞİRKET TOPLU BELGE YAZDIRMA — tüm siparişler tek yazdırmada */}
   409→                      {(canSeeInvoice || canSeeIrsaliye) && (
   410→                        <Tooltip>
   411→                          <TooltipTrigger asChild>
   412→                            <Button
   413→                              variant="outline"
   414→                              size="sm"
   415→                              className="h-7 text-[11px] gap-1 shrink-0 text-violet-700 border-violet-200 hover:bg-violet-50 dark:text-violet-300 dark:border-violet-900/50"
   416→                              onClick={(e) => {
   417→                                e.stopPropagation()

   419→                                  name: company.name,
   420→                                  orders: company.orders.map((o) => ({ id: o.id, number: o.number })),
   421→                                })
   422→                              }}
   423→                            >
   424→                              <Printer className="w-3 h-3" />
   425→                              <span className="hidden md:inline">Tümünü Yazdır</span>
   426→                            </Button>
   427→                          </TooltipTrigger>
   428→                          <TooltipContent>
