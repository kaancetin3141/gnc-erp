    95→    95→export function AppointmentCalendar({ providerId }: CalendarProps) {
    96→    96→  const qc = useQueryClient()
    97→    97→  const [view, setView] = useState<'day' | 'week'>('day')
    98→    98→  const [currentDate, setCurrentDate] = useState(() => {
    99→    99→    const d = new Date()
   100→   100→    d.setHours(0, 0, 0, 0)
   101→   101→    return d
   102→   102→  })
   103→   103→  const [staffFilter, setStaffFilter] = useState<string>('all')
   104→   104→
   105→   105→  // Detail dialog
   106→   106→  const [detailAppt, setDetailAppt] = useState<Appointment | null>(null)
   107→   107→
   108→   108→  // Provider (çalışma saatleri)
   109→   109→  const { data: providerData } = useQuery({
   110→   110→    queryKey: ['appointment-provider', providerId],
   111→   111→    queryFn: () => apiGet<ProviderResponse>(`/api/appointments/providers/${providerId}`),
   112→   112→    enabled: !!providerId,
   113→   113→  })
   114→   114→
   115→   115→  const workingHours: WorkingHours = useMemo(() => {
   116→   116→    if (!providerData?.workingHours) return {}
   117→   117→    try {
   118→   118→      return JSON.parse(providerData.workingHours) as WorkingHours
   119→   119→    } catch {
   120→   120→      return {}
   121→   121→    }
   122→   122→  }, [providerData])
   123→   123→
   124→   124→  const { data: staffData } = useQuery({
   125→   125→    queryKey: ['appointment-staff', providerId],
   126→   126→    queryFn: () => apiGet<StaffResponse>(`/api/appointments/providers/${providerId}/staff`),
   127→   127→    enabled: !!providerId,
   128→   128→  })
   129→   129→  const staffList = staffData?.items ?? []
   130→   130→
   131→   131→  const { data: serviceData } = useQuery({
   132→   132→    queryKey: ['appointment-services', providerId],
   133→   133→    queryFn: () => apiGet<ServiceResponse>(`/api/appointments/providers/${providerId}/services`),
   134→   134→    enabled: !!providerId,
   135→   135→  })
   136→   136→  const services = serviceData?.items ?? []
   137→   137→
   138→   138→  const dateRange = useMemo(() => {
   139→   139→    const start = new Date(currentDate)
   140→   140→    start.setHours(0, 0, 0, 0)
   141→   141→    if (view === 'week') {
   142→   142→      const day = (start.getDay() + 6) % 7
   143→   143→      start.setDate(start.getDate() - day)
   144→   144→    }
   145→   145→    const days: Date[] = []
   146→   146→    const numDays = view === 'week' ? 7 : 1
   147→   147→    for (let i = 0; i < numDays; i++) {
   148→   148→      const d = new Date(start)
   149→   149→      d.setDate(start.getDate() + i)
   150→   150→      days.push(d)
   151→   151→    }
   152→   152→    return days
   153→   153→  }, [currentDate, view])
   154→   154→
   155→   155→  const rangeStart = dateRange[0]
   156→   156→  const rangeEnd = new Date(dateRange[dateRange.length - 1].getTime() + 24 * 60 * 60_000)
   157→   157→
   158→   158→  const { data: apptData, isLoading } = useQuery({
   159→   159→    queryKey: ['appointment-appointments', providerId, rangeStart.toISOString(), rangeEnd.toISOString(), staffFilter],
   160→   160→    queryFn: () => {
   161→   161→      const params = new URLSearchParams({
   162→   162→        startDate: rangeStart.toISOString(),
   163→   163→        endDate: rangeEnd.toISOString(),
   164→   164→      })
   165→   165→      if (staffFilter !== 'all') params.set('staffId', staffFilter)
   166→   166→      return apiGet<ApptsResponse>(
   167→   167→        `/api/appointments/providers/${providerId}/appointments?${params.toString()}`,
   168→   168→      )
   169→   169→    },
   170→   170→    enabled: !!providerId,
   171→   171→  })
   172→   172→  const appointments = useMemo(() => (Array.isArray(apptData) ? apptData : (apptData as unknown as { items?: Appointment[] })?.items ?? []), [apptData])
   173→   173→
   174→   174→  const { visStart, visEnd } = useMemo(() => {
   175→   175→    let minMin = 24 * 60
   176→   176→    let maxMin = 0
   177→   177→    for (const day of dateRange) {
   178→   178→      const dk = dayKeyFromDate(day)
   179→   179→      const sched: DaySchedule | undefined = workingHours[dk]
   180→   180→      if (!sched || sched.closed || !sched.start || !sched.end) continue
   181→   181→      minMin = Math.min(minMin, timeToMinutes(sched.start))
   182→   182→      maxMin = Math.max(maxMin, timeToMinutes(sched.end))
   183→   183→    }
   184→   184→    if (minMin === 24 * 60) minMin = 9 * 60
   185→   185→    if (maxMin === 0) maxMin = 19 * 60
   186→   186→    return { visStart: minMin, visEnd: maxMin }
   187→   187→  }, [dateRange, workingHours])
   188→   188→
   189→   189→  const hourSlots = useMemo(() => {
   190→   190→    const slots: { min: number; label: string }[] = []
   191→   191→    for (let m = visStart; m < visEnd; m += 30) {
   192→   192→      slots.push({ min: m, label: minutesToTime(m) })
   193→   193→    }
   194→   194→    return slots
   195→   195→  }, [visStart, visEnd])
   196→   196→
   197→   197→  const apptsByDay = useMemo(() => {
   198→   198→    const map = new Map<string, Appointment[]>()
   199→   199→    for (const appt of appointments) {
   200→   200→      const d = new Date(appt.date)
   201→   201→      const key = d.toDateString()
   202→   202→      if (!map.has(key)) map.set(key, [])
   203→   203→      map.get(key)!.push(appt)
   204→   204→    }
   205→   205→    return map
   206→   206→  }, [appointments])
   207→   207→
   208→   208→  const [dialogOpen, setDialogOpen] = useState(false)
   209→   209→  const [editAppt, setEditAppt] = useState<Appointment | null>(null)
   210→   210→  const [form, setForm] = useState({
   211→   211→    staffId: 'any',
   212→   212→    serviceId: '',
   213→   213→    customerName: '',
   214→   214→    customerPhone: '',
   215→   215→    customerEmail: '',
   216→   216→    customerNote: '',
   217→   217→    date: '',
   218→   218→    time: '',
   219→   219→    status: 'beklemede' as AppointmentStatus,
   220→   220→  })
   221→   221→  const [saving, setSaving] = useState(false)
   222→   222→  const [deleteTarget, setDeleteTarget] = useState<Appointment | null>(null)
   223→   223→
   224→   224→  function prevPeriod() {
   225→   225→    const d = new Date(currentDate)
   226→   226→    d.setDate(d.getDate() - (view === 'week' ? 7 : 1))
   227→   227→    setCurrentDate(d)
   228→   228→  }
   229→   229→  function nextPeriod() {
   230→   230→    const d = new Date(currentDate)
   231→   231→    d.setDate(d.getDate() + (view === 'week' ? 7 : 1))
   232→   232→    setCurrentDate(d)
   233→   233→  }
   234→   234→  function goToday() {
   235→   235→    const d = new Date()
   236→   236→    d.setHours(0, 0, 0, 0)
   237→   237→    setCurrentDate(d)
   238→   238→  }
   239→   239→
   240→   240→  function openCreate(date?: Date, hour?: number, minute?: number) {
   241→   241→    setEditAppt(null)
   242→   242→    const targetDate = date ?? new Date()
   243→   243→    targetDate.setHours(hour ?? 10, minute ?? 0, 0, 0)
   244→   244→    setForm({
   245→   245→      staffId: staffFilter !== 'all' ? staffFilter : 'any',
   246→   246→      serviceId: services[0]?.id ?? '',
   247→   247→      customerName: '',

   249→   249→      customerEmail: '',
   250→   250→      customerNote: '',
   251→   251→      date: targetDate.toISOString().slice(0, 10),
   252→   252→      time: `${String(targetDate.getHours()).padStart(2, '0')}:${String(targetDate.getMinutes()).padStart(2, '0')}`,
   253→   253→      status: 'beklemede',
   254→   254→    })
   255→   255→    setDialogOpen(true)
   256→   256→  }
   257→   257→
   258→   258→  function openEdit(appt: Appointment) {
   259→   259→    setDetailAppt(null)
   260→   260→    setEditAppt(appt)
   261→   261→    const d = new Date(appt.date)
   262→   262→    setForm({
   263→   263→      staffId: appt.staffId ?? 'any',
   264→   264→      serviceId: appt.serviceId ?? '',
   265→   265→      customerName: appt.customerName,
   266→   266→      customerPhone: appt.customerPhone,
   267→   267→      customerEmail: appt.customerEmail ?? '',
   268→   268→      customerNote: appt.customerNote ?? '',
   269→   269→      date: d.toISOString().slice(0, 10),
   270→   270→      time: `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`,
   271→   271→      status: appt.status as AppointmentStatus,
   272→   272→    })
   273→   273→    setDialogOpen(true)
   274→   274→  }
   275→   275→
   276→   276→  async function handleSave() {
   277→   277→    if (!form.customerName.trim()) {
   278→   278→      toast.error('Müşteri adı gerekli')
   279→   279→      return
   280→   280→    }
   281→   281→    if (!form.customerPhone.trim()) {
   282→   282→      toast.error('Müşteri telefonu gerekli')
   283→   283→      return
   284→   284→    }
   285→   285→    if (!form.serviceId) {
   286→   286→      toast.error('Hizmet seçimi gerekli')
   287→   287→      return
   288→   288→    }
   289→   289→    if (!form.date || !form.time) {
   290→   290→      toast.error('Tarih ve saat gerekli')
   291→   291→      return
   292→   292→    }
   293→   293→    setSaving(true)
   294→   294→    try {
   295→   295→      const [y, m, d] = form.date.split('-').map(Number)
   296→   296→      const [hh, mm] = form.time.split(':').map(Number)
   297→   297→      const start = new Date(y, m - 1, d, hh, mm, 0, 0)
   298→   298→      const payload = {
   299→   299→        staffId: form.staffId === 'any' ? undefined : form.staffId,
   300→   300→        serviceId: form.serviceId,
   301→   301→        customerName: form.customerName,
   302→   302→        customerPhone: form.customerPhone,
   303→   303→        customerEmail: form.customerEmail || undefined,
   304→   304→        customerNote: form.customerNote || undefined,
   305→   305→        date: start.toISOString(),
   306→   306→        status: form.status,
   307→   307→      }
   308→   308→      if (editAppt) {
   309→   309→        await apiPatch(
   310→   310→          `/api/appointments/providers/${providerId}/appointments/${editAppt.id}`,
   311→   311→          { ...payload, staffId: form.staffId },
   312→   312→        )
   313→   313→        toast.success('Randevu güncellendi')
   314→   314→      } else {
   315→   315→        await apiPost(
   316→   316→          `/api/appointments/providers/${providerId}/appointments`,
   317→   317→          payload,
   318→   318→        )
   319→   319→        toast.success('Randevu oluşturuldu')
   320→   320→      }
   321→   321→      qc.invalidateQueries({ queryKey: ['appointment-appointments', providerId] })
   322→   322→      setDialogOpen(false)
   323→   323→    } catch (e) {
   324→   324→      toast.error(e instanceof Error ? e.message : 'İşlem başarısız')
   325→   325→    } finally {
   326→   326→      setSaving(false)
   327→   327→    }
   328→   328→  }
   329→   329→
   330→   330→  async function handleStatusChange(appt: Appointment, status: AppointmentStatus) {

   332→   332→      await apiPatch(
   333→   333→        `/api/appointments/providers/${providerId}/appointments/${appt.id}`,

   338→   338→      toast.success(`Durum: ${getStatusMeta(status).label}`)
   339→   339→    } catch (e) {

   341→   341→    }
   342→   342→  }
   343→   343→
   344→   344→  async function handleDelete() {

   346→   346→    try {
   347→   347→      await apiDelete(
   348→   348→        `/api/appointments/providers/${providerId}/appointments/${deleteTarget.id}`,

   351→   351→      toast.success('Randevu iptal edildi')
   352→   352→      setDeleteTarget(null)
   353→   353→    } catch (e) {
   354→   354→      toast.error(e instanceof Error ? e.message : 'İptal edilemedi')
   355→   355→    }
   356→   356→  }
   357→   357→
   358→   358→  if (isLoading) {
   359→   359→    return (
   360→   360→      <Card>
   361→   361→        <Skeleton className="h-96 w-full" />
   362→   362→      </Card>

   364→   364→  }
   365→   365→
   366→   366→  return (
   367→   367→    <div className="space-y-3">
   368→   368→      {/* Toolbar */}
   369→   369→      <Card>
   370→   370→        <CardContent className="p-3">
   371→   371→          <div className="flex flex-col sm:flex-row sm:items-center gap-2 sm:gap-3">
   372→   372→            <div className="flex items-center gap-1.5 shrink-0">
   373→   373→              <Button size="icon" variant="outline" className="h-9 w-9" onClick={prevPeriod}>
   374→   374→                <ChevronLeft className="w-4 h-4" />
   375→   375→              </Button>
   376→   376→              <Button size="sm" variant="outline" onClick={goToday}>
   377→   377→                Bugün
   378→   378→              </Button>
   379→   379→              <Button size="icon" variant="outline" className="h-9 w-9" onClick={nextPeriod}>
   380→   380→                <ChevronRight className="w-4 h-4" />
   381→   381→              </Button>
   382→   382→            </div>
   383→   383→            <div className="flex items-center gap-2 flex-1 min-w-0">
   384→   384→              <CalendarIcon className="w-4 h-4 text-emerald-600 shrink-0" />
   385→   385→              <span className="font-semibold text-sm truncate">
   386→   386→                {view === 'day'
   387→   387→                  ? currentDate.toLocaleDateString('tr-TR', { day: 'numeric', month: 'long', year: 'numeric', weekday: 'long' })
   388→   388→                  : `${rangeStart.toLocaleDateString('tr-TR', { day: 'numeric', month: 'short' })} - ${new Date(rangeEnd.getTime() - 86400000).toLocaleDateString('tr-TR', { day: 'numeric', month: 'short', year: 'numeric' })}`}
   389→   389→              </span>
   390→   390→            </div>
   391→   391→            <div className="flex items-center gap-2 shrink-0">
   392→   392→              <Select value={staffFilter} onValueChange={setStaffFilter}>
   393→   393→                <SelectTrigger className="h-9 w-40 text-xs">
   394→   394→                  <SelectValue placeholder="Tüm personel" />
   395→   395→                </SelectTrigger>
   396→   396→                <SelectContent>
   397→   397→                  <SelectItem value="all">Tüm Personel</SelectItem>
   398→   398→                  {staffList.map((s) => (
   399→   399→                    <SelectItem key={s.id} value={s.id}>{s.name}</SelectItem>
   400→   400→                  ))}
   401→   401→                </SelectContent>
   402→   402→              </Select>
   403→   403→              <div className="flex border rounded-md overflow-hidden">
   404→   404→                <Button
   405→   405→                  size="sm"
   406→   406→                  variant={view === 'day' ? 'default' : 'ghost'}
   407→   407→                  className={cn('rounded-none h-9', view === 'day' && 'bg-emerald-600 hover:bg-emerald-700')}
   408→   408→                  onClick={() => setView('day')}
   409→   409→                >
   410→   410→                  Gün
   411→   411→                </Button>
   412→   412→                <Button
   413→   413→                  size="sm"
   414→   414→                  variant={view === 'week' ? 'default' : 'ghost'}
   415→   415→                  className={cn('rounded-none h-9', view === 'week' && 'bg-emerald-600 hover:bg-emerald-700')}
   416→   416→                  onClick={() => setView('week')}
   417→   417→                >
   418→   418→                  Hafta
   419→   419→                </Button>
   420→   420→              </div>
   421→   421→              <Button size="sm" onClick={() => openCreate()} className="bg-emerald-600 hover:bg-emerald-700">
   422→   422→                <Plus className="w-4 h-4 mr-1" />
   423→   423→                <span className="hidden sm:inline">Randevu</span>
   424→   424→              </Button>
   425→   425→            </div>
   426→   426→          </div>
   427→   427→        </CardContent>
   428→   428→      </Card>
   429→   429→
   430→   430→      {/* Calendar Grid */}
   431→   431→      <Card>
   432→   432→        <CardContent className="p-0 overflow-x-auto">
   433→   433→          <div className={cn('grid min-w-[640px]', view === 'week' ? 'grid-cols-[60px_repeat(7,1fr)]' : 'grid-cols-[60px_1fr]')}>
   434→   434→            <div className="border-b border-r bg-muted/30 p-2 text-[10px] text-muted-foreground text-right">
   435→   435→              S
   436→   436→            </div>
   437→   437→            {dateRange.map((d) => {
   438→   438→              const isToday = d.toDateString() === new Date().toDateString()
   439→   439→              const dk = dayKeyFromDate(d)
   440→   440→              const sched = workingHours[dk]
   441→   441→              const isClosed = !sched || sched.closed
   442→   442→              return (
   443→   443→                <div
   444→   444→                  key={d.toISOString()}
   445→   445→                  className={cn(
   446→   446→                    'border-b border-r p-2 text-center',
   447→   447→                    isToday && 'bg-emerald-50 dark:bg-emerald-950/20',
   448→   448→                    isClosed && 'bg-muted/20',
   449→   449→                  )}
   450→   450→                >
   451→   451→                  <div className="text-[10px] uppercase tracking-wide text-muted-foreground">
   452→   452→                    {d.toLocaleDateString('tr-TR', { weekday: 'short' })}
   453→   453→                  </div>
   454→   454→                  <div className={cn('text-sm font-semibold', isToday && 'text-emerald-600')}>
   455→   455→                    {d.getDate()}
   456→   456→                  </div>
   457→   457→                  {isClosed && (

   798→   798→function SlotRow({
   799→   799→  slot,
   800→   800→  dateRange,
   801→   801→  apptsByDay,
   802→   802→  workingHours,
   803→   803→  onSlotClick,
   804→   804→  onApptClick,
   805→   805→}: {
   806→   806→  slot: { min: number; label: string }
   807→   807→  dateRange: Date[]
   808→   808→  apptsByDay: Map<string, Appointment[]>
   809→   809→  workingHours: WorkingHours
   810→   810→  onSlotClick: (date: Date) => void
   811→   811→  onApptClick: (appt: Appointment) => void
   812→   812→}) {
   813→   813→  const isHourLine = slot.min % 60 === 0
   814→   814→  return (
   815→   815→    <>
   816→   816→      <div
   817→   817→        className={cn(
   818→   818→          'border-r border-b bg-muted/20 p-1 text-[10px] text-muted-foreground text-right align-top',
   819→   819→          isHourLine ? 'border-b-border' : 'border-b-muted/40',
   820→   820→        )}
   821→   821→      >
   822→   822→        {isHourLine ? slot.label : ''}
   823→   823→      </div>
   824→   824→      {dateRange.map((d) => {
   825→   825→        const dk = dayKeyFromDate(d)
   826→   826→        const sched = workingHours[dk]
   827→   827→        const isClosed = !sched || sched.closed || !sched.start || !sched.end
   828→   828→        const inHours = !isClosed &&
   829→   829→          slot.min >= timeToMinutes(sched!.start!) &&
   830→   830→          slot.min < timeToMinutes(sched!.end!)
   831→   831→
   832→   832→        const dayAppts = apptsByDay.get(d.toDateString()) ?? []
   833→   833→        const slotAppts = dayAppts.filter((a) => {
   834→   834→          const ad = new Date(a.date)
   835→   835→          const aMin = ad.getHours() * 60 + ad.getMinutes()
   836→   836→          return aMin >= slot.min && aMin < slot.min + 30
   837→   837→        })
   838→   838→
   839→   839→        return (
   840→   840→          <div
   841→   841→            key={`${d.toISOString()}-${slot.min}`}
   842→   842→            className={cn(
   843→   843→              'border-r border-b relative h-7 group',
   844→   844→              isHourLine ? 'border-b-border' : 'border-b-muted/40',
   845→   845→              inHours ? 'bg-card hover:bg-emerald-50/30 dark:hover:bg-emerald-950/10 cursor-pointer' : 'bg-muted/10',
   846→   846→            )}
   847→   847→            onClick={() => inHours && onSlotClick(new Date(d.getFullYear(), d.getMonth(), d.getDate(), Math.floor(slot.min / 60), slot.min % 60))}
   848→   848→          >
   849→   849→            {slotAppts.map((appt) => {
   850→   850→              const meta = getStatusMeta(appt.status)
   851→   851→              const isCancelled = appt.status === 'iptal' || appt.status === 'reddedildi'
   852→   852→              return (
   853→   853→                <button
   854→   854→                  key={appt.id}
   855→   855→                  type="button"
   856→   856→                  onClick={(e) => { e.stopPropagation(); onApptClick(appt) }}
   857→   857→                  className={cn(
   858→   858→                    'absolute inset-x-0.5 top-0.5 px-1.5 py-0.5 rounded text-[10px] text-left truncate transition-shadow hover:shadow-md z-10',
   859→   859→                    meta.bg, meta.color,
   860→   860→                    isCancelled && 'opacity-50 line-through',
   861→   861→                  )}
