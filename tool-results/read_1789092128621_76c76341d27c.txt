    98→function ActivityDialog({
    99→  open, onOpenChange, customerId,
   100→}: {
   101→  open: boolean
   102→  onOpenChange: (v: boolean) => void
   103→  customerId: string
   104→}) {
   105→  const qc = useQueryClient()
   106→  const [type, setType] = useState('arama')
   107→  const [subject, setSubject] = useState('')
   108→  const [detail, setDetail] = useState('')
   109→  const [date, setDate] = useState(() => new Date().toISOString().slice(0, 16))
   110→  const [duration, setDuration] = useState('15')
   111→  const [outcome, setOutcome] = useState('basarili')
   112→  const [submitting, setSubmitting] = useState(false)
   113→
   114→  // Reset when opening
   115→  useEffect(() => {
   116→    if (open) {
   117→      setType('arama'); setSubject(''); setDetail('')
   118→      setDate(new Date().toISOString().slice(0, 16))
   119→      setDuration('15'); setOutcome('basarili')
   120→    }
   121→  }, [open])
   122→
   123→  const submit = async () => {
   124→    if (!subject.trim()) {
   125→      toast.error('Konu gerekli')
   126→      return
   127→    }
   128→    setSubmitting(true)
   129→    try {
   130→      await apiPost(`/api/customers/${customerId}/activities`, {
   131→        type, subject: subject.trim(), detail: detail.trim() || null,
   132→        date: new Date(date).toISOString(),
   133→        durationMin: parseInt(duration) || 0, outcome,
   134→      })
   135→      toast.success('Aktivite eklendi')
   136→      qc.invalidateQueries({ queryKey: qk.customer(customerId) })
   137→      qc.invalidateQueries({ queryKey: ['dashboard'] })
   138→      onOpenChange(false)
   139→    } catch (e) {
   140→      toast.error(e instanceof Error ? e.message : 'İşlem başarısız')
   141→    } finally {
   142→      setSubmitting(false)
   143→    }
   144→  }
   145→
   146→  return (
   147→    <Dialog open={open} onOpenChange={onOpenChange}>
   148→      <DialogContent className="sm:max-w-lg">
   149→        <DialogHeader>
   150→          <DialogTitle>Aktivite Ekle</DialogTitle>
   151→          <DialogDescription>Bu müşteri için yeni bir aktivite kaydı oluşturun</DialogDescription>
   152→        </DialogHeader>
   153→        <div className="space-y-3 py-2">
   154→          <div className="grid grid-cols-2 gap-3">
   155→            <div className="space-y-1.5">
   156→              <Label className="text-xs">Tip</Label>
   157→              <Select value={type} onValueChange={setType}>
   158→                <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   159→                <SelectContent>
   160→                  {ACTIVITY_TYPES.map((t) => <SelectItem key={t.value} value={t.value}>{t.label}</SelectItem>)}
   161→                </SelectContent>
   162→              </Select>
   163→            </div>
   164→            <div className="space-y-1.5">
   165→              <Label className="text-xs">Sonuç</Label>
   166→              <Select value={outcome} onValueChange={setOutcome}>
   167→                <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   168→                <SelectContent>
   169→                  {ACTIVITY_OUTCOMES.map((o) => <SelectItem key={o.value} value={o.value}>{o.label}</SelectItem>)}
   170→                </SelectContent>
   171→              </Select>
   172→            </div>
   173→          </div>
   174→          <div className="space-y-1.5">
   175→            <Label className="text-xs">Konu *</Label>
   176→            <Input value={subject} onChange={(e) => setSubject(e.target.value)} placeholder="Örn. Fiyat teklifi görüşmesi" />
   177→          </div>
   178→          <div className="space-y-1.5">
   179→            <Label className="text-xs">Detay</Label>
   180→            <Textarea
   181→              value={detail}
   182→              onChange={(e) => setDetail(e.target.value)}
   183→              rows={3}
   184→              placeholder="Aktivite detayları..."
   185→            />
   186→          </div>
   187→          <div className="grid grid-cols-2 gap-3">
   188→            <div className="space-y-1.5">
   189→              <Label className="text-xs">Tarih & Saat</Label>
   190→              <Input type="datetime-local" value={date} onChange={(e) => setDate(e.target.value)} />
   191→            </div>
   192→            <div className="space-y-1.5">
   193→              <Label className="text-xs">Süre (dk)</Label>
   194→              <Input type="number" value={duration} onChange={(e) => setDuration(e.target.value)} min={0} />
   195→            </div>
   196→          </div>
   197→        </div>
   198→        <DialogFooter>
   199→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={submitting}>İptal</Button>
   200→          <Button onClick={submit} disabled={submitting || !subject.trim()}>
   201→            {submitting && <RefreshCw className="w-4 h-4 mr-1.5 animate-spin" />}
   202→            Kaydet
   203→          </Button>
   204→        </DialogFooter>
   205→      </DialogContent>
   206→    </Dialog>
   207→  )
   208→}

   330→function TaskDialog({
   331→  open, onOpenChange, customerId, users, defaultAssigneeId,
   332→}: {
   333→  open: boolean
   334→  onOpenChange: (v: boolean) => void
   335→  customerId: string
   336→  users: UserListItem[]
   337→  defaultAssigneeId?: string
   338→}) {
   339→  const qc = useQueryClient()
   340→  const [title, setTitle] = useState('')
   341→  const [dueDate, setDueDate] = useState(() => {
   342→    const d = new Date(); d.setDate(d.getDate() + 1)
   343→    return d.toISOString().slice(0, 10)
   344→  })
   345→  const [priority, setPriority] = useState('orta')
   346→  const [assigneeId, setAssigneeId] = useState(defaultAssigneeId || '')
   347→  const [submitting, setSubmitting] = useState(false)
   348→
   349→  useEffect(() => {
   350→    if (open) {
   351→      setTitle(''); setPriority('orta')
   352→      const d = new Date(); d.setDate(d.getDate() + 1)
   353→      setDueDate(d.toISOString().slice(0, 10))
   354→      setAssigneeId(defaultAssigneeId || '')
   355→    }
   356→  }, [open, defaultAssigneeId])
   357→
   358→  const submit = async () => {
   359→    if (!title.trim()) { toast.error('Görev başlığı gerekli'); return }
   360→    if (!dueDate) { toast.error('Bitiş tarihi gerekli'); return }
   361→    setSubmitting(true)
   362→    try {
   363→      await apiPost('/api/tasks', {
   364→        title: title.trim(),
   365→        dueDate: new Date(dueDate).toISOString(),
   366→        assigneeId: assigneeId || null,
   367→        customerId,
   368→        priority,
   369→      })
   370→      toast.success('Görev oluşturuldu')
   371→      qc.invalidateQueries({ queryKey: qk.customer(customerId) })
   372→      qc.invalidateQueries({ queryKey: ['dashboard'] })
   373→      onOpenChange(false)

   376→    } finally {
   377→      setSubmitting(false)
   378→    }
   379→  }
   380→
   381→  return (
   382→    <Dialog open={open} onOpenChange={onOpenChange}>
   383→      <DialogContent className="sm:max-w-lg">
   384→        <DialogHeader>
   385→          <DialogTitle>Görev Oluştur</DialogTitle>

   387→        </DialogHeader>
   388→        <div className="space-y-3 py-2">
   389→          <div className="space-y-1.5">
   390→            <Label className="text-xs">Başlık *</Label>
   391→            <Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="Örn. Teklif takip et" />
   392→          </div>
   393→          <div className="grid grid-cols-2 gap-3">
   394→            <div className="space-y-1.5">
   395→              <Label className="text-xs">Bitiş Tarihi *</Label>
   396→              <Input type="date" value={dueDate} onChange={(e) => setDueDate(e.target.value)} />
   397→            </div>
   398→            <div className="space-y-1.5">
   399→              <Label className="text-xs">Öncelik</Label>
   400→              <Select value={priority} onValueChange={setPriority}>
   401→                <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>

   424→          </Button>
   425→        </DialogFooter>
   426→      </DialogContent>
   427→    </Dialog>

  1102→function formatFileSize(bytes: number): string {
  1103→  if (bytes === 0) return '0 B'
  1104→  const k = 1024
  1105→  const sizes = ['B', 'KB', 'MB', 'GB']
  1106→  const i = Math.floor(Math.log(bytes) / Math.log(k))
  1107→  return `${Math.round((bytes / Math.pow(k, i)) * 10) / 10} ${sizes[i]}`
  1108→}

  1793→function NotesTabWithRef({

  1796→  customer: Customer360Data

  1801→  const [content, setContent] = useState('')

  1806→    if (!content.trim()) return

  1809→      await apiPost(`/api/customers/${customer.id}/notes`, { content: content.trim() })

  1814→      toast.error(e instanceof Error ? e.message : 'İşlem başarısız')

  1816→      setSubmitting(false)

  1823→        noteId: note.id, isPinned: !note.isPinned, content: note.content,
  1824→      })
  1825→      toast.success(note.isPinned ? 'Not sabitlenmesi kaldırıldı' : 'Not sabitlendi')
  1826→      qc.invalidateQueries({ queryKey: qk.customer(customer.id) })

  1860→            <Button size="sm" onClick={addNote} disabled={submitting || !content.trim()}>

  1872→        </div>
  1873→      ) : (
  1874→        <div className="space-y-2">
  1875→          {notes.map((note) => (

  1879→                'hover:shadow-sm transition-shadow',

  1884→                <div className="flex items-start gap-2">
  1885→                  <div className="flex-1 min-w-0">

  1888→                      <Avatar className="w-5 h-5">
