   152→function StatsRow({ users }: { users: UserListItem[] }) {
   153→  const total = users.length
   154→  const active = users.filter((u) => u.status === 'active').length
   155→  const managers = users.filter((u) => u.role === 'manager' || u.role === 'admin' || u.role === 'superadmin').length
   156→  const reps = users.filter((u) => u.role === 'rep').length
   157→
   158→  const stats = [
   159→    { label: 'Toplam Kullanıcı', value: total, icon: Users, color: 'from-emerald-500 to-teal-600', sub: `${active} aktif` },
   160→    { label: 'Aktif', value: active, icon: UserCheck, color: 'from-teal-500 to-cyan-600', sub: `${total - active} pasif` },
   161→    { label: 'Yönetici', value: managers, icon: Crown, color: 'from-violet-500 to-purple-600', sub: 'müdür + admin' },
   162→    { label: 'Temsilci', value: reps, icon: GitBranch, color: 'from-amber-500 to-orange-600', sub: 'satış temsilcisi' },
   163→  ]
   164→  return (
   165→    <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   166→      {stats.map((s) => (
   167→        <Card key={s.label} className="overflow-hidden">
   168→          <CardContent className="p-5">
   169→            <div className="flex items-start justify-between mb-3">
   170→              <div className={cn('w-10 h-10 rounded-lg bg-gradient-to-br flex items-center justify-center shadow-sm', s.color)}>
   171→                <s.icon className="w-5 h-5 text-white" />
   172→              </div>
   173→            </div>
   174→            <div className="text-2xl font-bold tracking-tight">{s.value}</div>
   175→            <div className="text-sm text-muted-foreground mt-0.5">{s.label}</div>
   176→            <div className="text-xs text-muted-foreground/80 mt-1.5">{s.sub}</div>
   177→          </CardContent>
   178→        </Card>
   179→      ))}
   180→    </div>
   181→  )
   182→}

   184→// ─── Add/Edit Dialog ──────────────────────────────────────────────
   185→function UserFormDialog({
   186→  open, onOpenChange, editing, users, actor,
   187→}: {
   188→  open: boolean
   189→  onOpenChange: (v: boolean) => void
   190→  editing: UserListItem | null
   191→  users: UserListItem[]
   192→  actor: SessionUser
   193→}) {
   194→  const qc = useQueryClient()
   195→  const [form, setForm] = useState<UserFormData>(() =>
   196→    editing
   197→      ? {
   198→          name: editing.name,
   199→          email: editing.email,
   200→          role: editing.role,
   201→          title: editing.title ?? '',
   202→          phone: editing.phone ?? '',
   203→          managerId: editing.managerId ?? '',
   204→          status: editing.status,
   205→        }
   206→      : EMPTY_FORM,
   207→  )
   208→  const [submitting, setSubmitting] = useState(false)
   209→  const [showRoleNotice, setShowRoleNotice] = useState(false)
   210→
   211→  const createMut = useMutation({
   212→    mutationFn: (body: UserFormData) => apiPost<UserListItem>('/api/users', body),
   213→    onSuccess: () => {
   214→      toast.success('Kullanıcı davet edildi', { description: `${form.name} sisteme eklendi.` })
   215→      qc.invalidateQueries({ queryKey: qk.users })
   216→      onOpenChange(false)
   217→    },
   218→    onError: (e: ApiError) => toast.error('Hata', { description: e.message }),
   219→    onSettled: () => setSubmitting(false),
   220→  })
   221→
   222→  const updateMut = useMutation({
   223→    mutationFn: (body: Partial<UserFormData>) => apiPatch<UserListItem>(`/api/users/${editing?.id}`, body),
   224→    onSuccess: () => {
   225→      toast.success('Kullanıcı güncellendi', { description: `${form.name} bilgileri kaydedildi.` })
   226→      qc.invalidateQueries({ queryKey: qk.users })
   227→      onOpenChange(false)
   228→    },
   229→    onError: (e: ApiError) => toast.error('Hata', { description: e.message }),
   230→    onSettled: () => setSubmitting(false),
   231→  })
   232→
   233→  const handleSubmit = () => {
   234→    if (!form.name.trim() || !form.email.trim()) {
   235→      toast.error('Eksik bilgi', { description: 'Ad ve e-posta zorunludur.' })
   236→      return
   237→    }
   238→    setSubmitting(true)
   239→    const payload: UserFormData = {
   240→      ...form,
   241→      managerId: form.managerId === 'none' ? '' : form.managerId,
   242→    }
   243→    if (editing) {
   244→      updateMut.mutate(payload)
   245→    } else {
   246→      createMut.mutate(payload)
   247→    }
   248→  }
   249→
   250→  // Manager select — editing kullanıcısını dışla, actor dahil tüm tenant kullanıcıları
   251→  const managerOptions = users.filter((u) => u.id !== editing?.id)
   252→
   253→  // Rol değişince bildirim
   254→  const handleRoleChange = (role: Role) => {
   255→    setForm((f) => ({ ...f, role }))
   256→    setShowRoleNotice(true)
   257→  }
   258→
   259→  const roleDefaultPerms = getRolePermissions(form.role)
   260→
   261→  return (
   262→    <Dialog open={open} onOpenChange={onOpenChange}>
   263→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   264→        <DialogHeader>
   265→          <DialogTitle className="flex items-center gap-2">
   266→            {editing ? <Pencil className="w-4 h-4 text-emerald-600" /> : <UserPlus className="w-4 h-4 text-emerald-600" />}
   267→            {editing ? 'Kullanıcı Düzenle' : 'Kullanıcı Davet Et'}
   268→          </DialogTitle>
   269→          <DialogDescription>
   270→            {editing
   271→              ? `${editing.name} kullanıcısının bilgilerini ve rolünü güncelleyin.`
   272→              : 'Yeni bir kullanıcı ekleyin. Rolüne göre varsayılan yetkiler otomatik atanır.'}
   273→          </DialogDescription>
   274→        </DialogHeader>
   275→
   276→        <div className="grid sm:grid-cols-2 gap-4 py-2">
   277→          <div className="space-y-1.5">
   278→            <Label htmlFor="u-name">Ad Soyad *</Label>
   279→            <Input id="u-name" value={form.name} onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))} placeholder="Örn. Ahmet Yılmaz" />
   280→          </div>
   281→          <div className="space-y-1.5">
   282→            <Label htmlFor="u-email">E-posta *</Label>
   283→            <Input id="u-email" type="email" value={form.email} onChange={(e) => setForm((f) => ({ ...f, email: e.target.value }))} placeholder="ahmet@sirket.com" />
   284→          </div>
   285→          <div className="space-y-1.5">
   286→            <Label htmlFor="u-role">Rol</Label>
   287→            <Select value={form.role} onValueChange={(v) => handleRoleChange(v as Role)}>
   288→              <SelectTrigger id="u-role" className="w-full">
   289→                <SelectValue />
   290→              </SelectTrigger>
   291→              <SelectContent>
   292→                {SELECTABLE_ROLES.map((r) => (
   293→                  <SelectItem key={r} value={r}>{ROLE_LABELS[r]}</SelectItem>
   294→                ))}
   295→              </SelectContent>
   296→            </Select>
   297→          </div>
   298→          <div className="space-y-1.5">
   299→            <Label htmlFor="u-title">Ünvan</Label>
   300→            <Input id="u-title" value={form.title} onChange={(e) => setForm((f) => ({ ...f, title: e.target.value }))} placeholder="Satış Müdürü" />
   301→          </div>
   302→          <div className="space-y-1.5">
   303→            <Label htmlFor="u-phone">Telefon</Label>
   304→            <Input id="u-phone" value={form.phone} onChange={(e) => setForm((f) => ({ ...f, phone: e.target.value }))} placeholder="+90 5xx xxx xx xx" />
   305→          </div>
   306→          <div className="space-y-1.5">
   307→            <Label htmlFor="u-manager">Yönetici</Label>
   308→            <Select value={form.managerId || 'none'} onValueChange={(v) => setForm((f) => ({ ...f, managerId: v === 'none' ? '' : v }))}>
   309→              <SelectTrigger id="u-manager" className="w-full">
   310→                <SelectValue placeholder="Yönetici seçin" />
   311→              </SelectTrigger>
   312→              <SelectContent>
   313→                <SelectItem value="none">— Yönetici yok —</SelectItem>
   314→                {managerOptions.map((m) => (
   315→                  <SelectItem key={m.id} value={m.id}>
   316→                    {m.name} · {ROLE_LABELS[m.role]}
   317→                  </SelectItem>
   318→                ))}
   319→              </SelectContent>
   320→            </Select>
   321→          </div>
   322→          <div className="space-y-1.5">
   323→            <Label htmlFor="u-status">Durum</Label>
   324→            <Select value={form.status} onValueChange={(v) => setForm((f) => ({ ...f, status: v }))}>
   325→              <SelectTrigger id="u-status" className="w-full">
   326→                <SelectValue />
   327→              </SelectTrigger>
   328→              <SelectContent>
   329→                <SelectItem value="active">Aktif</SelectItem>
   330→                <SelectItem value="passive">Pasif</SelectItem>
   331→              </SelectContent>
   332→            </Select>
   333→          </div>
   334→        </div>
   335→
   336→        {/* Rol değişimi bildirimi */}
   337→        {showRoleNotice && (
   338→          <div className="rounded-lg border border-emerald-200 bg-emerald-50/60 dark:bg-emerald-950/20 dark:border-emerald-900/40 p-3 flex gap-2.5">
   339→            <Info className="w-4 h-4 text-emerald-600 shrink-0 mt-0.5" />
   340→            <div className="text-xs text-emerald-800 dark:text-emerald-300 space-y-1">
   341→              <p className="font-medium">
   342→                {ROLE_LABELS[form.role]} rolü için varsayılan yetkiler:
   343→              </p>
   344→              <div className="flex flex-wrap gap-1">
   345→                {roleDefaultPerms.length > 0 ? roleDefaultPerms.slice(0, 6).map((p) => (
   346→                  <span key={p} className="px-1.5 py-0.5 rounded bg-white dark:bg-emerald-950/40 border border-emerald-200 dark:border-emerald-900/50 font-mono text-[10px]">
   347→                    {p}
   348→                  </span>
   349→                )) : <span className="italic">Yetki yok</span>}
   350→                {roleDefaultPerms.length > 6 && (
   351→                  <span className="text-emerald-700 dark:text-emerald-400">+{roleDefaultPerms.length - 6} daha</span>
   352→                )}
   353→              </div>
   354→              <p className="text-[11px] text-emerald-700/80 dark:text-emerald-400/80 pt-0.5">
   355→                Yetkileri özelleştirmek için &ldquo;Yetki Matrisi&rdquo; → kullanıcı seç → düzenle.
   356→              </p>
   357→            </div>
   358→          </div>
   359→        )}
   360→
   361→        {/* Cycle önleme uyarısı */}
   362→        <div className="rounded-lg border border-amber-200 bg-amber-50/40 dark:bg-amber-950/20 dark:border-amber-900/40 p-2.5 flex gap-2">
   363→          <AlertCircle className="w-3.5 h-3.5 text-amber-600 shrink-0 mt-0.5" />
   364→          <p className="text-[11px] text-amber-800 dark:text-amber-300">
   365→            Yönetici ataması döngü oluşturamaz (kullanıcı, kendi astının astı olamaz). Sistem otomatik engeller.
   366→          </p>
   367→        </div>
   368→
   369→        <DialogFooter className="pt-2">
   370→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={submitting}>
   371→            İptal
   372→          </Button>
   373→          <Button
   374→            onClick={handleSubmit}
   375→            disabled={submitting}
   376→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   377→          >
   378→            {submitting ? 'Kaydediliyor…' : editing ? 'Değişiklikleri Kaydet' : 'Kullanıcı Oluştur'}
   379→          </Button>
   380→        </DialogFooter>
   381→      </DialogContent>
   382→    </Dialog>
   383→  )
   384→}

   386→// ─── Yetki Matrisi (read-only) ────────────────────────────────────
   387→function PermissionMatrixReadOnly() {
   388→  const roles: Role[] = ['admin', 'manager', 'rep', 'readonly', 'stock']
   389→  const groups = useMemo(() => {
   390→    const map = new Map<string, typeof ALL_PERMISSIONS>()
   391→    for (const p of ALL_PERMISSIONS) {
   392→      if (!map.has(p.group)) map.set(p.group, [])
   393→      map.get(p.group)!.push(p)
   394→    }
   395→    return Array.from(map.entries())
   396→  }, [])
   397→
   398→  return (
   399→    <div className="rounded-lg border overflow-hidden">
   400→      <div className="overflow-x-auto max-h-[420px] overflow-y-auto">
   401→        <table className="w-full text-sm">
   402→          <thead className="sticky top-0 z-10 bg-muted/80 backdrop-blur">
   403→            <tr className="border-b">
   404→              <th className="text-left font-medium p-3 min-w-[260px]">Yetki</th>
   405→              {roles.map((r) => (
   406→                <th key={r} className="text-center font-medium p-2 min-w-[110px]">
   407→                  <div className="flex flex-col items-center gap-1">
   408→                    <span className="text-xs">{ROLE_LABELS[r]}</span>
   409→                  </div>
   410→                </th>
   411→              ))}
   412→            </tr>
   413→          </thead>
   414→          <tbody>
   415→            {groups.map(([group, perms]) => (
   416→              <Fragment key={`g-${group}`}>
   417→                <tr className="bg-muted/30">
   418→                  <td colSpan={roles.length + 1} className="p-2.5 font-semibold text-xs uppercase tracking-wide text-muted-foreground">
   419→                    {group}
   420→                  </td>
   421→                </tr>
   422→                {perms.map((p) => (
   423→                  <tr key={p.key} className="border-b last:border-0 hover:bg-muted/40 transition-colors">
   424→                    <td className="p-2.5">
   425→                      <div className="font-medium text-[13px]">{p.label}</div>
   426→                      <div className="text-[10px] text-muted-foreground font-mono">{p.key}</div>
   427→                    </td>
   428→                    {roles.map((r) => {
   429→                      const has = ROLE_PERMISSIONS[r].includes(p.key)
   430→                      return (
   431→                        <td key={r} className="p-2 text-center">
   432→                          {has ? (
   433→                            <Tooltip>
   434→                              <TooltipTrigger asChild>
   435→                                <span className="inline-flex items-center justify-center w-6 h-6 rounded-md bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300">
   436→                                  <Check className="w-3.5 h-3.5" />
   437→                                </span>
   438→                              </TooltipTrigger>
   439→                              <TooltipContent>Bu rol bu yetkiye sahip</TooltipContent>
   440→                            </Tooltip>
   441→                          ) : (
   442→                            <span className="inline-flex items-center justify-center w-6 h-6 rounded-md bg-slate-50 text-slate-300 dark:bg-slate-800/40 dark:text-slate-600">
   443→                              <Minus className="w-3.5 h-3.5" />
   444→                            </span>
   445→                          )}
   446→                        </td>
   447→                      )
   448→                    })}
   449→                  </tr>
   450→                ))}
   451→              </Fragment>
   452→            ))}
   453→          </tbody>
   454→        </table>
   455→      </div>
   456→    </div>
   457→  )
   458→}

   460→// ─── Kullanıcı bazlı yetki editörü ────────────────────────────────
   461→function UserPermissionEditor({
   462→  users, actor,
   463→}: {
   464→  users: UserListItem[]
   465→  actor: SessionUser
   466→}) {
   467→  const qc = useQueryClient()
   468→  const [selectedId, setSelectedId] = useState<string>('')
   469→  const [draftPerms, setDraftPerms] = useState<PermissionKey[]>([])
   470→  const [saving, setSaving] = useState(false)
   471→
   472→  const selected = users.find((u) => u.id === selectedId) ?? null
   473→
   474→  // Actor'ın astları (getVisibleUserIds ile, self hariç)
   475→  const subordinates = useMemo(() => {
   476→    const visible = getVisibleUserIds(
   477→      users.map((u) => ({ id: u.id, managerId: u.managerId })),
   478→      actor.id,
   479→    )
   480→    return visible.filter((id) => id !== actor.id)
   481→  }, [users, actor])
   482→
   483→  // Seçim değişince draft'ı hazırla
   484→  const handleSelect = (id: string) => {
   485→    setSelectedId(id)
   486→    const u = users.find((x) => x.id === id)
   487→    setDraftPerms(u?.permissions ?? [])
   488→  }
   489→
   490→  // Belirli bir yetki değiştirilebilir mi? (canDelegatePermission)
   491→  const canEdit = useCallback(
   492→    (perm: PermissionKey): boolean => {
   493→      if (!selected) return false
   494→      return canDelegatePermission(actor, selected.id, perm, subordinates)
   495→    },
   496→    [actor, selected, subordinates],
   497→  )
   498→
   499→  const togglePerm = (perm: PermissionKey) => {
   500→    if (!canEdit(perm)) return
   501→    setDraftPerms((prev) =>
   502→      prev.includes(perm) ? prev.filter((p) => p !== perm) : [...prev, perm],
   503→    )
   504→  }
   505→
   506→  const saveMut = useMutation({
   507→    mutationFn: (perms: PermissionKey[]) =>
   508→      apiPatch<UserListItem>(`/api/users/${selectedId}`, { permissions: perms }),
   509→    onSuccess: () => {
   510→      toast.success('Yetkiler güncellendi', {
   511→        description: `${selected?.name} için yetkiler kaydedildi.`,
   512→      })
   513→      qc.invalidateQueries({ queryKey: qk.users })
   514→    },
   515→    onError: (e: ApiError) => toast.error('Hata', { description: e.message }),
   516→    onSettled: () => setSaving(false),
   517→  })
   518→
   519→  const handleSave = () => {
   520→    setSaving(true)
   521→    saveMut.mutate(draftPerms)
   522→  }
   523→
   524→  const groups = useMemo(() => {
   525→    const map = new Map<string, typeof ALL_PERMISSIONS>()
   526→    for (const p of ALL_PERMISSIONS) {
   527→      if (!map.has(p.group)) map.set(p.group, [])
   528→      map.get(p.group)!.push(p)
   529→    }
   530→    return Array.from(map.entries())
   531→  }, [])
   532→
   533→  // Aktörün kendi yetkileri (delegate edebileceği)
   534→  const actorPerms = actor.permissions
   535→
   536→  return (
   537→    <div className="space-y-4">
   538→      {/* Açıklama */}
   539→      <div className="rounded-lg border border-violet-200 bg-violet-50/50 dark:bg-violet-950/20 dark:border-violet-900/40 p-3 flex gap-2.5">
   540→        <Shield className="w-4 h-4 text-violet-600 shrink-0 mt-0.5" />
   541→        <p className="text-xs text-violet-900 dark:text-violet-300">
   542→          <span className="font-medium">Yetki devri:</span> Bir kullanıcı sadece kendisinde bulunan yetkileri ve sadece kendi astlarına verebilir.
   543→          Astlar hiyerarşide sizin altınızdaki kullanıcılar olarak tanımlanır.
   544→        </p>
   545→      </div>
   546→
   547→      <div className="grid sm:grid-cols-[260px_1fr] gap-4">
   548→        {/* Kullanıcı seçimi */}
   549→        <div className="space-y-2">
   550→          <Label className="text-xs font-medium">Kullanıcı Seç</Label>
   551→          <Select value={selectedId || 'none'} onValueChange={handleSelect}>
   552→            <SelectTrigger className="w-full">
   553→              <SelectValue placeholder="Kullanıcı seçin" />
   554→            </SelectTrigger>
   555→            <SelectContent>
   556→              <SelectItem value="none">— Seçiniz —</SelectItem>
   557→              {users
   558→                .filter((u) => subordinates.includes(u.id))
   559→                .map((u) => (
   560→                  <SelectItem key={u.id} value={u.id}>
   561→                    {u.name} · {ROLE_LABELS[u.role]}
   562→                  </SelectItem>
   563→                ))}
   564→            </SelectContent>
   565→          </Select>
   566→          {selectedId && selectedId !== 'none' && (
   567→            <p className="text-[11px] text-muted-foreground">
   568→              Aktör yetkileri: <span className="font-mono">{actorPerms.length}</span> ·
   569→              Astlar: <span className="font-mono">{subordinates.length}</span>
   570→            </p>
   571→          )}
   572→          {!selectedId || selectedId === 'none' ? (
   573→            <p className="text-[11px] text-muted-foreground/80 italic mt-2">
   574→              Devredilebilir kullanıcı (ast) seçin.
   575→            </p>
   576→          ) : null}
   577→        </div>
   578→
   579→        {/* Yetki listesi */}
   580→        <div className="rounded-lg border">
   581→          {!selected ? (
   582→            <div className="p-8 text-center text-sm text-muted-foreground">
   583→              <Shield className="w-8 h-8 mx-auto mb-2 opacity-30" />
   584→              Düzenlemek için bir ast kullanıcı seçin.
   585→            </div>
   586→          ) : (
   587→            <ScrollArea className="h-[320px]">
   588→              <div className="p-3 space-y-3">
   589→                {groups.map(([group, perms]) => (
   590→                  <div key={group}>
   591→                    <div className="text-xs font-semibold uppercase tracking-wide text-muted-foreground mb-1.5 px-1">
   592→                      {group}
   593→                    </div>
   594→                    <div className="space-y-0.5">
   595→                      {perms.map((p) => {
   596→                        const checked = draftPerms.includes(p.key)
   597→                        const actorHas = actorPerms.includes(p.key)
   598→                        const editable = canEdit(p.key)
   599→                        return (
   600→                          <Tooltip key={p.key}>
   601→                            <TooltipTrigger asChild>
   602→                              <div
   603→                                className={cn(
   604→                                  'flex items-center gap-2.5 px-2.5 py-2 rounded-md transition-colors',
   605→                                  editable ? 'hover:bg-muted cursor-pointer' : 'opacity-60 cursor-not-allowed',
   606→                                )}
   607→                                onClick={() => togglePerm(p.key)}
   608→                              >
   609→                                <Checkbox checked={checked} disabled={!editable} />
   610→                                <div className="flex-1 min-w-0">
   611→                                  <div className="text-[13px] font-medium">{p.label}</div>
   612→                                  <div className="text-[10px] text-muted-foreground font-mono">{p.key}</div>
   613→                                </div>
   614→                                {!actorHas && (
   615→                                  <Badge variant="outline" className="text-[9px] h-4 px-1 bg-slate-50 text-slate-500 border-slate-200">
   616→                                    yetkiniz yok
   617→                                  </Badge>
   618→                                )}
   619→                              </div>
   620→                            </TooltipTrigger>
   621→                            {!editable && (
   622→                              <TooltipContent>
   623→                                {actorHas
   624→                                  ? 'Bu kullanıcı astınız değil — yetki veremezsiniz.'
   625→                                  : 'Bu yetki sizde yok — devredemezsiniz.'}
   626→                              </TooltipContent>
   627→                            )}
   628→                          </Tooltip>
   629→                        )
   630→                      })}
   631→                    </div>
   632→                  </div>
   633→                ))}
   634→              </div>
   635→            </ScrollArea>
   636→          )}
   637→        </div>
   638→      </div>
   639→
   640→      {selected && (
   641→        <div className="flex items-center justify-between gap-2 pt-1">
   642→          <div className="text-xs text-muted-foreground">
   643→            Toplam <span className="font-semibold text-foreground">{draftPerms.length}</span> yetki seçili.
   644→          </div>
   645→          <Button
   646→            onClick={handleSave}
   647→            disabled={saving}
   648→            size="sm"
   649→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   650→          >
   651→            {saving ? 'Kaydediliyor…' : 'Yetkileri Kaydet'}
   652→          </Button>
   653→        </div>
   654→      )}
   655→    </div>
   656→  )
   657→}

   990→// ─── Hiyerarşi ağacı ──────────────────────────────────────────────
   991→interface TreeNode extends UserListItem {
   992→  children: TreeNode[]
   993→}
   994→
   995→function buildTree(users: UserListItem[]): TreeNode[] {
   996→  const map = new Map<string, TreeNode>()
   997→  users.forEach((u) => map.set(u.id, { ...u, children: [] }))
   998→  const roots: TreeNode[] = []
   999→  users.forEach((u) => {
  1000→    const node = map.get(u.id)!
  1001→    if (u.managerId && map.has(u.managerId)) {
  1002→      map.get(u.managerId)!.children.push(node)
  1003→    } else {
  1004→      roots.push(node)
  1005→    }
  1006→  })
  1007→  return roots
  1008→}
  1009→
  1010→function HierarchyNode({
  1011→  node, depth, actor,

  1014→  depth: number
  1015→  actor: SessionUser
  1016→}) {

  1046→        <div className="flex-1 min-w-0 flex items-center gap-2 flex-wrap">

  1052→          </span>

  1055→            <span className="text-xs text-muted-foreground hidden sm:inline">· {node.title}</span>

  1059→              {node._count.subordinates} ast
  1060→            </Badge>

  1062→        </div>

  1065→      {/* Connector line */}

  1075→            ))}
  1076→          </div>
  1077→        </div>
  1078→      )}
  1079→    </div>
  1080→  )
  1081→}
