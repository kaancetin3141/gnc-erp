    63→export function AppointmentsView() {
    64→  const qc = useQueryClient()
    65→  const [selectedProviderId, setSelectedProviderId] = useState<string | null>(null)
    66→  const [tab, setTab] = useState('appointments')
    67→  const [bookOpen, setBookOpen] = useState(false)
    68→  const [selectedDate, setSelectedDate] = useState(new Date().toISOString().slice(0, 10))
    69→  const [statusFilter, setStatusFilter] = useState('all')
    70→
    71→  // Provider list
    72→  const { data: providersData } = useQuery({
    73→    queryKey: ['providers'],
    74→    queryFn: () => apiGet<{ items: Provider[] }>('/api/appointments/providers'),
    75→  })
    76→  const providers = providersData?.items ?? []
    77→  const currentProvider = providers[0] // first provider for now
    78→  const providerId = selectedProviderId || currentProvider?.id
    79→
    80→  // Provider detail (staff + services)
    81→  const { data: provider } = useQuery({
    82→    queryKey: ['provider', providerId],
    83→    queryFn: () => apiGet<Provider>(`/api/appointments/providers/${providerId}`),
    84→    enabled: !!providerId,
    85→  })
    86→
    87→  // Appointments for selected date
    88→  const { data: appointments = [], isLoading } = useQuery({
    89→    queryKey: ['appointments', providerId, selectedDate],
    90→    queryFn: () => apiGet<Appointment[]>(`/api/appointments/providers/${providerId}/appointments?date=${selectedDate}`),
    91→    enabled: !!providerId,
    92→  })
    93→
    94→  // Gün özeti + durum filtresi (program geliştirmesi)
    95→  const dayStats = useMemo(() => {
    96→    const pending = appointments.filter((a) => a.status === 'beklemede')
    97→    const approved = appointments.filter((a) => a.status === 'onaylandi')
    98→    const done = appointments.filter((a) => a.status === 'tamamlandi')
    99→    const cancelled = appointments.filter((a) => a.status === 'iptal' || a.status === 'gelmedi')
   100→    const expectedRevenue = [...pending, ...approved].reduce((sum, a) => sum + (a.price || 0), 0)
   101→    return {
   102→      total: appointments.length,
   103→      pendingCount: pending.length,
   104→      approvedCount: approved.length,
   105→      doneCount: done.length,
   106→      cancelledCount: cancelled.length,
   107→      expectedRevenue,
   108→    }
   109→  }, [appointments])
   110→
   111→  const filteredAppointments = useMemo(() => {
   112→    if (statusFilter === 'all') return appointments
   113→    if (statusFilter === 'iptal') return appointments.filter((a) => a.status === 'iptal' || a.status === 'gelmedi')
   114→    return appointments.filter((a) => a.status === statusFilter)
   115→  }, [appointments, statusFilter])
   116→
   117→  const approveAllPending = async () => {
   118→    const pending = appointments.filter((a) => a.status === 'beklemede')
   119→    if (pending.length === 0 || !providerId) return
   120→    try {
   121→      await Promise.all(pending.map((a) =>
   122→        apiPatch(`/api/appointments/providers/${providerId}/appointments/${a.id}`, { status: 'onaylandi' }),
   123→      ))
   124→      qc.invalidateQueries({ queryKey: ['appointments'] })
   125→      toast.success(`${pending.length} randevu onaylandı`, { description: 'Müşterilere WhatsApp ile bilgi gönderebilirsiniz' })
   126→    } catch (e) {
   127→      toast.error('Toplu onay başarısız', { description: e instanceof Error ? e.message : '' })
   128→    }
   129→  }
   130→
   131→  // Otomatik onay — başlıkta hızlı anahtar (ayrıca İşletme Ayarları'nda da var)
   132→  const autoApprove = provider?.autoApprove ?? true
   133→  const [autoApproveSaving, setAutoApproveSaving] = useState(false)
   134→  const toggleAutoApprove = async (checked: boolean) => {
   135→    if (!providerId) return
   136→    setAutoApproveSaving(true)
   137→    try {
   138→      await apiPatch(`/api/appointments/providers/${providerId}`, { autoApprove: checked })
   139→      qc.invalidateQueries({ queryKey: ['provider', providerId] })
   140→      qc.invalidateQueries({ queryKey: ['appointment-provider', providerId] })
   141→      qc.invalidateQueries({ queryKey: ['providers'] })
   142→      qc.invalidateQueries({ queryKey: ['appointment-providers'] })
   143→      toast.success(
   144→        checked
   145→          ? 'Otomatik onay açıldı — yeni randevular onaysız kabul edilir'
   146→          : 'Otomatik onay kapatıldı — yeni randevular onay bekler',
   147→        { icon: checked ? '⚡' : '⏳' },
   148→      )
   149→    } catch (e) {
   150→      toast.error('Ayar değiştirilemedi', { description: e instanceof Error ? e.message : '' })
   151→    } finally {
   152→      setAutoApproveSaving(false)
   153→    }
   154→  }
   155→
   156→  if (!providersData) {
   157→    return <div className="p-8"><Skeleton className="h-32" /></div>
   158→  }
   159→
   160→  if (providers.length === 0) {
   161→    return (
   162→      <div className="space-y-5 animate-fade-in">
   163→        <h2 className="text-2xl font-bold">Randevu Sistemi</h2>
   164→        <Card><CardContent className="p-8 text-center">
   165→          <Store className="w-12 h-12 mx-auto mb-3 text-muted-foreground/30" />
   166→          <p className="text-sm text-muted-foreground mb-4">Henüz işletme kaydı yok. İlk işletmenizi oluşturun.</p>
   167→          <ProviderCreateDialog tenantId="" onCreated={() => qc.invalidateQueries({ queryKey: ['providers'] })} />
   168→        </CardContent></Card>
   169→      </div>
   170→    )
   171→  }
   172→
   173→  return (
   174→    <div className="space-y-5 animate-fade-in">
   175→      {/* Header */}
   176→      <div className="flex items-center justify-between flex-wrap gap-3">
   177→        <div>
   178→          <h2 className="text-2xl font-bold tracking-tight">Randevu Sistemi</h2>
   179→          <p className="text-sm text-muted-foreground mt-0.5">
   180→            {provider?.name} · {PROVIDER_TYPES.find((t) => t.value === provider?.type)?.label}
   181→          </p>
   182→        </div>
   183→        <div className="flex gap-2 items-center flex-wrap">
   184→          {/* Otomatik onay anahtarı — Randevu Ekle'nin yanı sıra */}
   185→          <div
   186→            className={cn(
   187→              'flex items-center gap-2 h-9 px-3 rounded-lg border transition-colors',
   188→              autoApprove
   189→                ? 'bg-emerald-50 border-emerald-200 dark:bg-emerald-950/30 dark:border-emerald-900/60'
   190→                : 'bg-amber-50 border-amber-200 dark:bg-amber-950/30 dark:border-amber-900/60',
   191→            )}
   192→          >
   193→            {autoApprove
   194→              ? <Zap className="w-4 h-4 text-emerald-600 shrink-0" />
   195→              : <ShieldCheck className="w-4 h-4 text-amber-600 shrink-0" />}
   196→            <span className={cn('text-xs font-medium whitespace-nowrap', autoApprove ? 'text-emerald-700 dark:text-emerald-400' : 'text-amber-700 dark:text-amber-400')}>
   197→              {autoApprove ? 'Otomatik onay' : 'Onay gerekli'}
   198→            </span>
   199→            <Switch
   200→              checked={autoApprove}
   201→              onCheckedChange={toggleAutoApprove}
   202→              disabled={autoApproveSaving}
   203→              aria-label="Randevuları otomatik onayla"
   204→            />
   205→          </div>
   206→          <Input type="date" value={selectedDate} onChange={(e) => setSelectedDate(e.target.value)} className="w-40" />
   207→          <Button size="sm" className="bg-emerald-600 hover:bg-emerald-700" onClick={() => setBookOpen(true)}>
   208→            <Plus className="w-4 h-4 mr-1.5" /> Randevu Ekle
   209→          </Button>
   210→        </div>
   211→      </div>
   212→
   213→      <Tabs value={tab} onValueChange={(v) => {
   214→        setTab(v)
   215→        // Personel/hizmet değişiklikleri Müşteri Görünümü'ne ve listeye yansısın
   216→        if (v === 'booking' || v === 'appointments') qc.invalidateQueries({ queryKey: ['provider', providerId] })
   217→      }}>
   218→        <TabsList>
   219→          <TabsTrigger value="appointments" className="text-xs"><Calendar className="w-3.5 h-3.5 mr-1" /> Randevular</TabsTrigger>
   220→          <TabsTrigger value="calendar" className="text-xs"><CalendarDays className="w-3.5 h-3.5 mr-1" /> Takvim</TabsTrigger>
   221→          <TabsTrigger value="staff" className="text-xs"><Users className="w-3.5 h-3.5 mr-1" /> Personel</TabsTrigger>
   222→          <TabsTrigger value="services" className="text-xs"><Scissors className="w-3.5 h-3.5 mr-1" /> Hizmetler</TabsTrigger>
   223→          <TabsTrigger value="booking" className="text-xs"><Store className="w-3.5 h-3.5 mr-1" /> Müşteri Görünümü</TabsTrigger>
   224→          <TabsTrigger value="settings" className="text-xs"><Settings2 className="w-3.5 h-3.5 mr-1" /> İşletme Ayarları</TabsTrigger>
   225→        </TabsList>
   226→
   227→        {/* Randevular tab */}
   228→        <TabsContent value="appointments" className="mt-4 space-y-3">
   229→          {/* Gün özeti kartları */}
   230→          {!isLoading && appointments.length > 0 && (
   231→            <div className="grid grid-cols-2 md:grid-cols-4 gap-2">
   232→              <Card className="shadow-soft"><CardContent className="p-3">
   233→                <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground"><Calendar className="w-3.5 h-3.5" /> Bugün</div>
   234→                <div className="text-xl font-bold mt-0.5 tabular-nums">{dayStats.total}</div>
   235→                <div className="text-[10px] text-muted-foreground">randevu</div>
   236→              </CardContent></Card>
   237→              <Card className="shadow-soft"><CardContent className="p-3">
   238→                <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground"><Clock className="w-3.5 h-3.5 text-amber-500" /> Bekleyen</div>
   239→                <div className="text-xl font-bold mt-0.5 tabular-nums text-amber-600">{dayStats.pendingCount}</div>
   240→                <div className="text-[10px] text-muted-foreground">onay bekliyor</div>
   241→              </CardContent></Card>
   242→              <Card className="shadow-soft"><CardContent className="p-3">
   243→                <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground"><CheckCircle2 className="w-3.5 h-3.5 text-emerald-500" /> Onaylı</div>
   244→                <div className="text-xl font-bold mt-0.5 tabular-nums text-emerald-600">{dayStats.approvedCount}</div>
   245→                <div className="text-[10px] text-muted-foreground">{dayStats.doneCount} tamamlandı</div>
   246→              </CardContent></Card>
   247→              <Card className="shadow-soft"><CardContent className="p-3">
   248→                <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground"><Wallet className="w-3.5 h-3.5 text-teal-500" /> Beklenen</div>
   249→                <div className="text-xl font-bold mt-0.5 tabular-nums text-teal-600">{formatCurrency(dayStats.expectedRevenue)}</div>
   250→                <div className="text-[10px] text-muted-foreground">ciro potansiyeli</div>
   251→              </CardContent></Card>
   252→            </div>
   253→          )}
   254→
   255→          {/* Onay bekleyen uyarı bandı — otomatik onay kapalıyken */}
   256→          {!isLoading && !autoApprove && dayStats.pendingCount > 0 && (
   257→            <div className="flex items-center justify-between gap-3 flex-wrap rounded-lg border border-amber-200 bg-amber-50 dark:bg-amber-950/30 dark:border-amber-900/60 px-4 py-3">
   258→              <div className="flex items-center gap-2">
   259→                <AlertCircle className="w-4 h-4 text-amber-600 shrink-0" />
   260→                <span className="text-sm text-amber-800 dark:text-amber-300">
   261→                  <strong>{dayStats.pendingCount} randevu</strong> onayınızı bekliyor — otomatik onay kapalı
   262→                </span>
   263→              </div>
   264→              <Button size="sm" className="h-8 bg-emerald-600 hover:bg-emerald-700 text-white" onClick={approveAllPending}>
   265→                <CheckCheck className="w-3.5 h-3.5 mr-1.5" /> Tümünü Onayla
   266→              </Button>
   267→            </div>
   268→          )}
   269→
   270→          {/* Durum filtre çipleri */}
   271→          {!isLoading && appointments.length > 0 && (
   272→            <div className="flex items-center gap-1.5 flex-wrap">
   273→              {([
   274→                { key: 'all', label: 'Tümü', count: dayStats.total, cls: 'bg-primary text-primary-foreground border-transparent' },
   275→                { key: 'beklemede', label: 'Beklemede', count: dayStats.pendingCount, cls: 'bg-amber-100 text-amber-800 border-amber-300 dark:bg-amber-950/40 dark:text-amber-300 dark:border-amber-900' },
   276→                { key: 'onaylandi', label: 'Onaylı', count: dayStats.approvedCount, cls: 'bg-emerald-100 text-emerald-800 border-emerald-300 dark:bg-emerald-950/40 dark:text-emerald-300 dark:border-emerald-900' },
   277→                { key: 'tamamlandi', label: 'Tamamlanan', count: dayStats.doneCount, cls: 'bg-teal-100 text-teal-800 border-teal-300 dark:bg-teal-950/40 dark:text-teal-300 dark:border-teal-900' },
   278→                { key: 'iptal', label: 'İptal/Gelmedi', count: dayStats.cancelledCount, cls: 'bg-red-100 text-red-800 border-red-300 dark:bg-red-950/40 dark:text-red-300 dark:border-red-900' },
   279→              ] as const).map((c) => (
   280→                <button
   281→                  key={c.key}
   282→                  onClick={() => setStatusFilter(c.key)}
   283→                  className={cn(
   284→                    'h-7 px-2.5 rounded-full border text-[11px] font-medium transition-colors',
   285→                    statusFilter === c.key ? c.cls : 'bg-background text-muted-foreground border-border hover:bg-accent',
   286→                  )}
   287→                >
   288→                  {c.label} · {c.count}
   289→                </button>
   290→              ))}
   291→            </div>
   292→          )}
   293→
   294→          {isLoading ? (
   295→            <div className="space-y-2">{[...Array(5)].map((_, i) => <Skeleton key={i} className="h-16" />)}</div>
   296→          ) : filteredAppointments.length === 0 ? (
   297→            <Card><CardContent className="py-12 text-center">
   298→              <Calendar className="w-10 h-10 mx-auto mb-2 text-muted-foreground/30" />
   299→              <p className="text-sm text-muted-foreground">
   300→                {appointments.length === 0
   301→                  ? `${formatDate(selectedDate)} tarihinde randevu yok`
   302→                  : 'Bu filtreye uyan randevu yok'}
   303→              </p>
   304→            </CardContent></Card>
   305→          ) : (
   306→            <div className="space-y-2">
   307→              {filteredAppointments.map((apt) => {
   308→                const st = APPT_STATUS.find((s) => s.value === apt.status)
   309→                return (
   310→                  <Card key={apt.id} className="shadow-soft">
   311→                    <CardContent className="p-3 flex items-center gap-3">
   312→                      <div className="text-center shrink-0 w-16">
   313→                        <div className="text-lg font-bold tabular-nums">{formatTime(apt.date)}</div>
   314→                        <div className="text-[10px] text-muted-foreground">{apt.endTime ? formatTime(apt.endTime) : ''}</div>
   315→                      </div>
   316→                      <div className="flex-1 min-w-0">
   317→                        <div className="flex items-center gap-2">
   318→                          <span className="font-medium text-sm">{apt.customerName}</span>
   319→                          <Badge variant="outline" className={cn('text-[10px]', st?.color ?? '')}>{st?.label}</Badge>
   320→                        </div>
   321→                        <div className="text-xs text-muted-foreground mt-0.5">
   322→                          {apt.service?.name ?? '—'} · {apt.staff?.name ?? 'Herhangi'} · {formatCurrency(apt.price)}
   323→                        </div>
   324→                      </div>
   325→                      <div className="flex items-center gap-1 shrink-0 flex-wrap justify-end">
   326→                        {/* Beklemede → hızlı onay */}
   327→                        {apt.status === 'beklemede' && (
   328→                          <Button size="sm" className="h-8 text-[11px] bg-emerald-600 hover:bg-emerald-700 text-white"
   329→                            onClick={async () => { await apiPatch(`/api/appointments/providers/${providerId}/appointments/${apt.id}`, { status: 'onaylandi' }); qc.invalidateQueries({ queryKey: ['appointments'] }); toast.success('Randevu onaylandı — WhatsApp ile bilgi gönderebilirsiniz') }}>
   330→                            <CheckCircle2 className="w-3.5 h-3.5 mr-1" /> Onayla
   331→                          </Button>
   332→                        )}
   333→                        {/* Onaylandı → WhatsApp ile bilgi gönder */}
   334→                        {apt.status === 'onaylandi' && (
   335→                          <Button
   336→                            size="sm"
   337→                            variant={apt.reminderSent ? 'outline' : 'default'}
   338→                            className={cn(
   339→                              'h-8 text-[11px]',
   340→                              !apt.reminderSent && 'bg-[#25D366] hover:bg-[#1fb857] text-white border-0',
   341→                              apt.reminderSent && 'text-[#25D366] border-[#25D366]/40',
   342→                            )}
   343→                            title="Müşteriye WhatsApp ile onay bilgisi gönder"
   344→                            onClick={async () => {
   345→                              const svc = apt.service?.name ? apt.service.name : 'randevu'
   346→                              const stf = apt.staff?.name ? ` (${apt.staff.name})` : ''
   347→                              const msg = `Merhaba ${apt.customerName}, 🎉\n\n${formatDateTime(apt.date)} tarihli ${svc}${stf} randevunuz ONAYLANDI. ✅\n\nGörüşmek üzere! 🙌`
   348→                              window.open(whatsappLink(apt.customerPhone, msg), '_blank', 'noopener,noreferrer')
   349→                              try {
   350→                                await apiPatch(`/api/appointments/providers/${providerId}/appointments/${apt.id}`, { reminderSent: true })
   351→                                qc.invalidateQueries({ queryKey: ['appointments'] })
   352→                              } catch { /* sessiz */ }
   353→                              toast.success('WhatsApp onay mesajı hazırlandı')
   354→                            }}
   355→                          >
   356→                            <MessageCircle className="w-3.5 h-3.5 mr-1" />
   357→                            {apt.reminderSent ? 'Bilgi Gönderildi' : 'WhatsApp ile Bilgi Gönder'}
   358→                          </Button>
   359→                        )}
   360→                        <a href={whatsappLink(apt.customerPhone, `Merhaba ${apt.customerName}, ${formatDate(apt.date)} ${formatTime(apt.date)} randevunuz hatırlatmasıdır.`)} target="_blank" rel="noopener">
   361→                          <Button variant="ghost" size="icon" className="h-8 w-8 text-green-600"><MessageCircle className="w-4 h-4" /></Button>
   362→                        </a>
   363→                        <a href={`tel:${apt.customerPhone}`}><Button variant="ghost" size="icon" className="h-8 w-8"><Phone className="w-4 h-4" /></Button></a>
