   141→                    </Button>

   467→function InvoicePdfDialog({ invoice, detail, onClose }: {
   468→  invoice: Invoice
   469→  detail: Invoice | null
   470→  onClose: () => void
   471→}) {
   472→  const d = detail ?? invoice
   473→  const lines = (d as Invoice & { lines?: { id: string; description: string; qty: number; unitPrice: number; taxRate: number; lineTotal: number; weightPerUnit?: number | null; weightUnit?: string | null; totalWeight?: number | null; product?: { name: string; sku: string | null } | null }[] }).lines ?? []
   474→  // Ağırlık kontrolü (F4)
   475→  const hasWeight = lines.some((l) => l.totalWeight != null && l.totalWeight > 0)
   476→  const totalNetWeight = hasWeight
   477→    ? calculateTotalWeight(lines.map((l) => ({ qty: l.qty, weightPerUnit: l.weightPerUnit ?? null })))
   478→    : null
