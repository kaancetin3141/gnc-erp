   485→function StockMovementForm({
   486→  productId, unit, currentStock, onDone,
   487→}: {
   488→  productId: string
   489→  unit: string
   490→  currentStock: number
   491→  onDone: () => void
   492→}) {
   493→  const qc = useQueryClient()
   494→  const [type, setType] = useState<'giris' | 'cikis' | 'duzeltme'>('giris')
   495→  const [quantity, setQuantity] = useState('')
   496→  const [reason, setReason] = useState('')
   497→  const [submitting, setSubmitting] = useState(false)
   498→
   499→  const handleSubmit = async () => {
   500→    const q = parseInt(quantity)
   501→    if (!Number.isFinite(q)) {
   502→      toast.error('Miktar girin')
   503→      return
   504→    }
   505→    if (type === 'giris' && q <= 0) {
   506→      toast.error('Giriş miktarı pozitif olmalı')
   507→      return
   508→    }
   509→    if (type === 'cikis' && q <= 0) {
   510→      toast.error('Çıkış miktarı pozitif olmalı')
   511→      return
   512→    }
   513→    if (type === 'cikis' && q > currentStock) {
   514→      toast.error(`Yetersiz stok (mevcut: ${currentStock} ${unit})`)
   515→      return
   516→    }
   517→    if (type === 'duzeltme' && q < 0) {
   518→      toast.error('Düzeltme sonrası stok negatif olamaz')
   519→      return
   520→    }
   521→    setSubmitting(true)
   522→    try {
   523→      await apiPost(`/api/products/${productId}/stock`, {
   524→        quantity: q,
   525→        type,
   526→        reason: reason.trim() || null,
   527→        refType: 'manual',
   528→      })
   529→      toast.success('Stok hareketi eklendi')
   530→      setQuantity('')
   531→      setReason('')
   532→      setType('giris')
   533→      qc.invalidateQueries({ queryKey: ['product', productId] })
   534→      qc.invalidateQueries({ queryKey: ['products'] })
   535→      onDone()
   536→    } catch (e) {
   537→      toast.error(e instanceof Error ? e.message : 'Hareket eklenemedi')
   538→    } finally {
   539→      setSubmitting(false)
   540→    }
   541→  }
   542→
   543→  return (
   544→    <div className="space-y-3">
   545→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   546→        <div className="space-y-1">
   547→          <Label className="text-xs">Hareket Tipi</Label>
   548→          <Select value={type} onValueChange={(v) => setType(v as typeof type)}>
   549→            <SelectTrigger className="w-full h-9"><SelectValue /></SelectTrigger>
   550→            <SelectContent>
   551→              <SelectItem value="giris">Giriş (+)</SelectItem>
   552→              <SelectItem value="cikis">Çıkış (-)</SelectItem>
   553→              <SelectItem value="duzeltme">Düzeltme (=)</SelectItem>
   554→            </SelectContent>
   555→          </Select>
   556→        </div>
   557→        <div className="space-y-1">
   558→          <Label className="text-xs">Miktar ({unit})</Label>
   559→          <Input
   560→            type="number"
   561→            min="0"
   562→            value={quantity}
   563→            onChange={(e) => setQuantity(e.target.value)}
   564→            placeholder="0"
   565→          />
   566→        </div>
   567→        <div className="space-y-1 col-span-2">
   568→          <Label className="text-xs">Açıklama / Sebep</Label>
   569→          <Input
   570→            value={reason}
   571→            onChange={(e) => setReason(e.target.value)}
   572→            placeholder="Örn. Alım, Satış, Sayım farkı..."
   573→          />
   574→        </div>
   575→      </div>
   576→      <div className="flex items-center justify-between gap-2">
   577→        <div className="text-xs text-muted-foreground">
