   678→function pageShell(o: { title: string; desc: string; body: string; scripts?: string[]; extraCss?: string[]; gwPort?: string | null; bodyClass?: string }): string {
   679→  // Sürüm parametresi: CSS/JS içeriğinin djb2 özeti — içerik değişince URL değişir,
   680→  // tarayıcı/heuristik önbellek eski stili servis edemez (cache-busting).
   681→  let _vh = 5381;
   682→  const _verSrc = STYLE_CSS + '\u0000' + HOME_JS + '\u0000' + DETAIL_JS;
   683→  for (let i = 0; i < _verSrc.length; i++) _vh = ((_vh << 5) + _vh + _verSrc.charCodeAt(i)) >>> 0;
   684→  const VER = _vh.toString(36);
   685→  // Gateway (XTransformPort) ile açıldıysa göreli varlık URL'lerine param ekle —
   686→  // aksi hâlde /assets/* istekleri gateway'den CRM'e düşer ve 404 olur.
   687→  const gw = (p: string) => {
   688→    const withVer = p + (p.includes('?') ? '&' : '?') + 'v=' + VER;
   689→    return o.gwPort ? withVer + '&XTransformPort=' + encodeURIComponent(o.gwPort) : withVer;
   690→  };
   691→  const scripts = (o.scripts || []).map((s) => `<script src="${gw(s)}" defer></script>`).join('\n  ');
   692→  const extraCss = (o.extraCss || []).map((s) => `<link rel="stylesheet" href="${gw(s)}">`).join('\n  ');
   693→  return `<!DOCTYPE html>
   694→<html lang="tr">
   695→<head>
   696→<meta charset="UTF-8">
   697→<meta name="viewport" content="width=device-width, initial-scale=1">
   698→<title>${esc(o.title)}</title>
   699→<meta name="description" content="${esc(o.desc)}">
   700→<link rel="icon" href="${gw(FAVICON)}">
   701→<link rel="stylesheet" href="${gw('/assets/style.css')}">
   702→  ${extraCss}

   705→<div class="page">
   706→${o.body}
   707→</div>
   708→<div id="toast" role="status" aria-live="polite"></div>
