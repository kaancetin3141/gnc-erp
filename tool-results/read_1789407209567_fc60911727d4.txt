   287→function NotificationsTab() {
   288→  const [settings, setSettings] = useState<Record<string, string>>({})
   289→  const [saving, setSaving] = useState(false)
   290→
   291→  const { data, isLoading } = useQuery({
   292→    queryKey: ['settings'],
   293→    queryFn: () => apiGet<Record<string, string>>(`/api/settings`),
   294→  })
   295→
   296→  useEffect(() => {
   297→    if (data) setSettings(data)
   298→  }, [data])
   299→
   300→  const updateSetting = async (key: string, value: string) => {
   301→    setSettings((s) => ({ ...s, [key]: value }))
   302→    setSaving(true)
   303→    try {
   304→      await apiPatch('/api/settings', { key, value })
   305→      toast.success('Ayar güncellendi')
   306→    } catch (e) {
   307→      toast.error('Ayar kaydedilemedi', { description: e instanceof Error ? e.message : '' })
   308→    } finally {
   309→      setSaving(false)
   310→    }
   311→  }
   312→
   313→  const emailNotif = settings['notification.email'] !== 'false'
   314→  const inAppNotif = settings['notification.inApp'] !== 'false'
   315→  const staleAlert = settings['notification.staleCustomerAlert'] !== 'false'
   316→  const staleDays = settings['automation.thresholdDays'] || '30'
   317→  const taskReminder = settings['notification.taskReminder'] !== 'false'
   318→
   319→  return (
   320→    <Card>
   321→      <CardHeader>
   322→        <CardTitle className="text-base flex items-center gap-2">
   323→          <Bell className="w-4 h-4 text-amber-600" />
   324→          Bildirim Tercihleri
   325→          {saving && <div className="w-3.5 h-3.5 border-2 border-amber-500 border-t-transparent rounded-full animate-spin" />}
   326→        </CardTitle>
   327→        <CardDescription>
   328→          E-posta ve uygulama içi bildirimlerinizi yönetin. Değişiklikler otomatik kaydedilir.
   329→        </CardDescription>
   330→      </CardHeader>
   331→      <CardContent className="space-y-1">
   332→        <NotificationRow
   333→          icon={<Bell className="w-4 h-4" />}
   334→          title="E-posta Bildirimleri"
   335→          desc="Önemli olaylar için e-posta alın (yeni müşteri, fırsat kazanıldı, görev hatırlatma)."
   336→          checked={emailNotif}
   337→          onChange={(v) => updateSetting('notification.email', String(v))}
   338→        />
   339→        <Separator />
   340→        <NotificationRow
   341→          icon={<Bell className="w-4 h-4" />}
   342→          title="Uygulama İçi Bildirimler"
   343→          desc="Tarayıcı içerinde anlık bildirimler göster."
   344→          checked={inAppNotif}
   345→          onChange={(v) => updateSetting('notification.inApp', String(v))}
   346→        />
   347→        <Separator />
   348→        <NotificationRow
   349→          icon={<AlertTriangle className="w-4 h-4" />}
   350→          title="İletişimsiz Müşteri Uyarısı"
   351→          desc="Belirli gün sayısı boyunca iletişim kurulmayan müşteriler için uyarı göster."
   352→          checked={staleAlert}
   353→          onChange={(v) => updateSetting('notification.staleCustomerAlert', String(v))}
   354→          extra={
   355→            <div className="flex items-center gap-2 mt-3">
   356→              <Label className="text-xs text-muted-foreground">Eşik (gün):</Label>
   357→              <Input
   358→                type="number"
   359→                value={staleDays}
   360→                onChange={(e) => updateSetting('automation.thresholdDays', e.target.value)}
   361→                className="w-20 h-8"
   362→                min={1}
   363→                max={365}
   364→                disabled={!staleAlert}
   365→              />
   366→            </div>
   367→          }
   368→        />
   369→        <Separator />
   370→        <NotificationRow
   371→          icon={<Clock className="w-4 h-4" />}
   372→          title="Görev Hatırlatma Süresi"
   373→          desc="Vade tarihinden belirli saat önce hatırlatma gönder."
   374→          checked={taskReminder}
   375→          onChange={(v) => updateSetting('notification.taskReminder', String(v))}
   376→          extra={
   377→            <div className="flex items-center gap-2 mt-3">
   378→              <Label className="text-xs text-muted-foreground">Hatırlatma (saat):</Label>
   379→              <Input
   380→                type="number"
   381→                value={settings['notification.taskReminderHours'] || '24'}
   382→                onChange={(e) => updateSetting('notification.taskReminderHours', e.target.value)}
   383→                className="w-20 h-8"
   384→                min={1}
   385→                max={168}
   386→                disabled={!taskReminder}
   387→              />
   388→            </div>
   389→          }
   390→        />
   391→        <Separator />
   392→        <div className="flex items-center justify-between pt-4 text-xs text-muted-foreground">
   393→          <span>{isLoading ? 'Ayarlar yükleniyor...' : 'Değişiklikler otomatik kaydedilir'}</span>
   394→        </div>
   395→      </CardContent>
   396→    </Card>
   397→  )
   398→}
