    37→type Entry = { expires: number; data: Json };
    38→const mem = new Map<string, Entry>();
    39→
    40→function cacheGet(key: string): Json | null {
    41→  const e = mem.get(key);
    42→  if (!e) return null;
    43→  if (Date.now() > e.expires) {
    44→    mem.delete(key);
    45→    return null;
    46→  }
    47→  return e.data;
    48→}
    49→
    50→function cacheSet(key: string, data: Json, ttlMs: number) {
    51→  mem.set(key, { expires: Date.now() + ttlMs, data });
    52→  if (mem.size > 200) {
    53→    const now = Date.now();
    54→    for (const [k, v] of mem) if (now > v.expires) mem.delete(k);
    55→  }
    56→}

    62→async function upstream(path: string, init?: RequestInit & { timeoutMs?: number }) {
    63→  const timeoutMs = init?.timeoutMs ?? 8000;
    64→  const ctrl = new AbortController();
    65→  const timer = setTimeout(() => ctrl.abort(), timeoutMs);
    66→  try {
    67→    const res = await fetch(UPSTREAM + path, {
    68→      ...init,
    69→      signal: ctrl.signal,
    70→      headers: { accept: 'application/json', ...(init?.headers || {}) },
    71→    });
    72→    const text = await res.text();
    73→    let data: any = null;
    74→    try {
    75→      data = JSON.parse(text);
    76→    } catch {
    77→      data = null;
    78→    }
    79→    return { ok: res.ok, status: res.status, data };
    80→  } catch {
    81→    return { ok: false, status: 502, data: null as any };
    82→  } finally {
    83→    clearTimeout(timer);
    84→  }
    85→}

    89→function normBiz(it: Json): Json {
    90→  return {
    91→    id: it.id ?? '',
    92→    slug: it.slug ?? '',
    93→    name: it.name ?? '',
    94→    type: it.type ?? '',
    95→    address: it.address ?? '',
    96→    city: it.city ?? '',
    97→    district: it.district ?? '',
    98→    photo: it.photo ?? null,
    99→    about: typeof it.about === 'string' ? it.about : null,
   100→    phone: it.phone ?? '',
   101→    email: it.email ?? '',
   102→    lat: typeof it.lat === 'number' && !isNaN(it.lat) ? it.lat : it.lat != null && !isNaN(Number(it.lat)) ? Number(it.lat) : null,
   103→    lng: typeof it.lng === 'number' && !isNaN(it.lng) ? it.lng : it.lng != null && !isNaN(Number(it.lng)) ? Number(it.lng) : null,
   104→    workingHours: it.workingHours && typeof it.workingHours === 'object' ? it.workingHours : {},
   105→    services: Array.isArray(it.services) ? it.services : [],
   106→    // değerlendirme özeti (ana sayfa kartındaki ★ rozeti)
   107→    ratingAvg: typeof it.ratingAvg === 'number' ? it.ratingAvg : 0,
   108→    ratingCount: typeof it.ratingCount === 'number' ? it.ratingCount : 0,
   109→  };
   110→}

   112→const trLower = (s: unknown) => String(s ?? '').toLocaleLowerCase('tr');
   113→
   114→function filterBiz(items: Json[], q: string, type: string, city: string): Json[] {
   115→  const nq = trLower(q.trim());
   116→  return items.filter((b) => {
   117→    if (type && b.type !== type) return false;
   118→    if (city && b.city !== city) return false;
   119→    if (nq) {
   120→      const svcNames = (b.services as Json[]).map((s) => trLower(s.name)).join(' ');
   121→      const hay = trLower([b.name, b.address, b.district, b.city].join(' ') + ' ' + svcNames);
   122→      if (!hay.includes(nq)) return false;
   123→    }
   124→    return true;
   125→  });
   126→}

   128→async function getBusinesses(): Promise<Json[]> {
   129→  const hit = cacheGet('businesses');
   130→  if (hit && Array.isArray(hit.items)) return hit.items;
   131→  const r = await upstream('/api/public/providers');
   132→  if (!r.ok || !r.data || !Array.isArray(r.data.items)) {
   133→    throw new UpstreamError('İşletme listesi şu anda alınamıyor — ana uygulama yanıt vermiyor olabilir. Birkaç saniye sonra tekrar dene.');
   134→  }
   135→  const items = (r.data.items as Json[]).map(normBiz);
   136→  cacheSet('businesses', { items }, 60_000);
   137→  return items;
   138→}

   140→// 404 öneri sıralaması: arama terimi işletme ADI yanında HİZMET adı/etiketinde de
   141→// aranır; hizmet eşleşmesi en güçlü sinyaldir ("manik" → Manikür hizmetli kuaför).
   142→// Eşleşen hizmet adı _svcMatch alanına yazılır (arayüzde chip olarak gösterilir).
   143→function rankSuggestions(items: Json[], q: string, excludeSlug?: string, limit = 6): Json[] {
   144→  const term = trLower(q).trim().slice(0, 64);
   145→  const pool = excludeSlug ? items.filter((b) => b.slug !== excludeSlug) : items;
   146→  if (!term) return pool.slice(0, limit);
   147→  const scored = pool.map((b) => {
   148→    let score = 0;
   149→    let svcMatch = '';
   150→    const services = Array.isArray(b.services) ? (b.services as Json[]) : [];
   151→    for (const sv of services) {
   152→      const nm = trLower(sv.name);
   153→      const cat = trLower(sv.category);
   154→      if (nm === term) { score = Math.max(score, 100); svcMatch = String(sv.name || ''); break; }
   155→      if (nm.includes(term)) { score = Math.max(score, 60); if (!svcMatch) svcMatch = String(sv.name || ''); }
   156→      else if (cat && (cat === term || cat.includes(term))) { score = Math.max(score, 40); if (!svcMatch) svcMatch = String(sv.category || ''); }
   157→    }
   158→    if (trLower(b.name).includes(term)) score += 30;
   159→    const hay = trLower([b.address, b.district, b.city, b.type].join(' '));
   160→    if (hay.includes(term)) score += 10;
   161→    return { b, score, svcMatch };
   162→  });
   163→  scored.sort((x, y) => y.score - x.score);
   164→  return scored
   165→    .filter((x) => x.score > 0)
   166→    .slice(0, limit)
   167→    .map((x) => (x.svcMatch ? { ...x.b, _svcMatch: x.svcMatch } : x.b));
   168→}

   200→async function renderDetail(slug: string, subdomain: boolean, gwPort?: string | null, q?: string, embed?: boolean): Promise<{ body: string; status: number }> {
   201→  const r = await upstream(`/api/public/providers/${encodeURIComponent(slug)}`);
   202→  if (r.status === 404) {
   203→    // Alt alan adı modunda "tüm işletmelere dön" hedefi: randevu.ANA_DOMAIN ana liste
   204→    const randevuHome = subdomain && BASE_DOMAIN ? `https://${listHost()}` : null;
   205→    // Hata sayfasını zenginleştir: terim hizmet adlarında da aranır (hizmet eşleşmesi öne çıkar)
   206→    let sugg: Json[] = [];
   207→    try {
   208→      const all = await getBusinesses();
   209→      sugg = rankSuggestions(all, q || '', slug);
   210→    } catch {
   211→      // liste alınamadıysa önerisiz 404 gösterilir
   212→    }
   213→    return { body: notFoundPage(slug, gwPort, randevuHome, sugg, subdomain && BASE_DOMAIN ? BASE_DOMAIN : undefined, q), status: 404 };
   214→  }
   215→  if (!r.ok || !r.data || !r.data.provider) {
   216→    throw new UpstreamError('İşletme bilgileri şu anda alınamıyor — ana uygulama yanıt vermiyor olabilir.');
   217→  }
   218→  const services: Json[] = Array.isArray(r.data.services) ? r.data.services : [];
   219→  // vitrin (tur4): portföy fotoğrafları + ekip — upstream hatasına dayanıklı
   220→  const photos: Json[] = Array.isArray(r.data.photos) ? r.data.photos : [];
   221→  const staff: Json[] = Array.isArray(r.data.staff) ? r.data.staff : [];
   222→  const provider = normBiz({ ...r.data.provider, services });
   223→  // Değerlendirmeler (server-side çekilir; hata olursa bölüm boş düşer)
   224→  let reviews: { items: Json[]; summary: Json } | null = null;
   225→  try {
   226→    const rr = await upstream(`/api/public/providers/${encodeURIComponent(slug)}/reviews?limit=20`);
   227→    const rd = rr.data as Json | null;
   228→    if (rr.ok && rd && Array.isArray(rd.items)) {
   229→      reviews = {
   230→        items: rd.items,
   231→        summary: (rd.summary ?? { average: 0, count: 0, distribution: {} }) as Json,
   232→      };
   233→    }
   234→  } catch {
   235→    // değerlendirmeler olmadan da sayfa render edilir
   236→  }
   237→  // Gateway modunda ana sayfa bağlantısına param ekle (CRM'e düşmesin)
   238→  // Düzeltme: "Tüm işletmeler" her zaman randevu LİSTESİNE gider (ana siteye değil)
   239→  const homeUrl = subdomain && BASE_DOMAIN ? `https://${listHost()}` : gwq('/')
   240→  return { body: detailPage({ provider, services, photos, staff, reviews, homeUrl, gwPort, embed }), status: 200 };
   241→
   242→  // BASE_DOMAIN=randevu.x biçimindeyse aynen; bare domain ise randevu. öneki ekle
   243→  function listHost(): string {
   244→    return BASE_DOMAIN.startsWith('randevu.') ? BASE_DOMAIN : `randevu.${BASE_DOMAIN}`;
   245→  }
   246→
   247→  function gwq(p: string) {
   248→    return gwPort ? p + (p.indexOf('?') === -1 ? '?' : '&') + 'XTransformPort=' + encodeURIComponent(gwPort) : p;
   249→  }
   250→}

   254→Bun.serve({

   257→  async fetch(req) {
   258→    const url = new URL(req.url);
   259→    const path = url.pathname !== '/' ? url.pathname.replace(/\/+$/, '') : '/';

   266→      const s = host.slice(0, host.length - BASE_DOMAIN.length - 1);

   270→    try {

   284→      // --- API: işletme listesi (60 sn cache + filtre)

   398→        const key = `reviews:${slug}:${lim}`;

   401→        const r = await upstream(`/api/public/providers/${encodeURIComponent(slug)}/reviews?limit=${encodeURIComponent(lim)}`);
   402→        if (r.status === 404) return json({ error: 'İşletme bulunamadı.' }, 404);
   403→        if (!r.ok || !r.data) return json({ error: 'Değerlendirmeler şu anda alınamıyor. Birazdan tekrar dene.' }, 502);
   404→        // admin onayı yeni yorum yayınlandığında görünsün — kısa TTL
   405→        cacheSet(key, r.data, 15_000);
   406→        return json(r.data);
   407→      }
   408→
   409→      // --- API: bekleme listesi proxy'si (CRM /api/public/waitlist → same-origin)
   410→      // POST /api/waitlist → beklemeye gir (üye token'ı varsa Bearer iletilir)
   411→      // GET  /api/waitlist?phone=  → "Bekleme Durumum" sorgusu
   412→      // Gerçek istemci IP'si (x-forwarded-for) rate limit için taşınır —
   413→      // aksi halde tüm müşteri sitesi istekleri tek 'unknown' kovasına düşerdi.
   414→      if (path === '/api/waitlist') {
   415→        const auth = req.headers.get('authorization');
   416→        const headers: Record<string, string> = { accept: 'application/json' };
   417→        if (auth) headers['authorization'] = auth;
   418→        const xff = req.headers.get('x-forwarded-for') || req.headers.get('x-real-ip');
   419→        if (xff) headers['x-forwarded-for'] = xff;
   420→        if (method === 'GET') {
   421→          const qs = url.searchParams.toString();
   422→          const r = await upstream('/api/public/waitlist' + (qs ? '?' + qs : ''), { headers });
   423→          if (r.data) return json(r.data, r.ok ? 200 : r.status);
   424→          return json({ error: 'Bekleme durumu şu anda alınamıyor. Birazdan tekrar dene.' }, 502);
   425→        }
   426→        if (method !== 'POST') return json({ error: 'Bu uç için GET veya POST kullan.' }, 405);
   427→        let wbody: Json;
   428→        try {
   429→          const raw = await req.text();
   430→          wbody = (raw.trim() ? JSON.parse(raw) : {}) as Json;
   431→        } catch {
   432→          return json({ error: 'İstek gövdesi okunamadı.' }, 400);
   433→        }
   434→        const r = await upstream('/api/public/waitlist', {
   435→          method: 'POST',
   436→          headers: { ...headers, 'content-type': 'application/json' },
   437→          body: JSON.stringify(wbody),
   438→          timeoutMs: 12000,
   439→        });
   440→        // 201 (oluştu) dahil olmak üzere status aynen iletilir
   441→        if (r.data) return json(r.data, r.status);
   442→        return json({ error: 'Bekleme servisi şu anda yanıt vermiyor. Birazdan tekrar dene.' }, 502);
   443→      }

   447→      // /api/reviews proxy deseni). Kartlar yalnız aktif damga programı

   458→      // --- API: ICS takvim dosyası proxy'si (CRM /api/public/appointments/ics → same-origin)
   459→      // "📅 Takvime ekle" butonları bu ucu kullanır: ?code=XXXXXXXX (misafir) ya da
   460→      // ?id=<appointmentId> + Authorization: Bearer (üye). Token varsa upstream'e
   461→      // iletilir; gerçek istemci IP'si (x-forwarded-for) rate limit için taşınır.
   462→      // text/calendar yanıtı olduğu gibi (Content-Disposition dahil) aktarılır;
   463→      // JSON hataları (400/401/404/409/429) status koduyla iletilir.
   464→      if (path === '/api/appointments/ics' && method === 'GET') {
   465→        const qs = url.searchParams.toString();
   466→        const fwdHeaders: Record<string, string> = { accept: 'text/calendar, application/json' };
   467→        const auth = req.headers.get('authorization');
   468→        if (auth) fwdHeaders['authorization'] = auth;
   469→        const xff = req.headers.get('x-forwarded-for') || req.headers.get('x-real-ip');
   470→        if (xff) fwdHeaders['x-forwarded-for'] = xff;
   471→        try {
   472→          const ctrl = new AbortController();
   473→          const timer = setTimeout(() => ctrl.abort(), 8000);
   474→          const res = await fetch(UPSTREAM + '/api/public/appointments/ics' + (qs ? '?' + qs : ''), {
   475→            headers: fwdHeaders,
   476→            signal: ctrl.signal,
   477→          });
   478→          clearTimeout(timer);
   479→          const ct = res.headers.get('content-type') || '';
   480→          if (ct.includes('text/calendar')) {
   481→            const body = await res.text();
   482→            return new Response(body, {
   483→              status: res.status,
   484→              headers: {
   485→                'content-type': 'text/calendar; charset=utf-8',
   486→                'content-disposition':
   487→                  res.headers.get('content-disposition') || 'attachment; filename="randevu.ics"',
   488→                'cache-control': 'no-store',
   489→              },
   490→            });
   491→          }
   492→          // hata yanıtı: JSON (404/429/400/…) — olduğu gibi aktar
   493→          let data: Json | null = null;
   494→          try {
   495→            data = (await res.json()) as Json;
   496→          } catch {
   497→            data = null;
   498→          }
   499→          return json(data ?? { error: 'Takvim dosyası alınamadı.' }, res.status);
   500→        } catch {
   501→          return json({ error: 'Takvim servisi şu anda yanıt vermiyor. Birazdan tekrar dene.' }, 502);
   502→        }
   503→      }

   506→      if (path === '/api/book') {
   507→        if (method !== 'POST') return json({ success: false, error: 'Bu uç için POST kullan.' }, 405);
   508→        let body: Json;
   509→        try {
   510→          body = (await req.json()) as Json;
   511→        } catch {

   515→        const missing = required.filter((k) => !body[k] || typeof body[k] !== 'string' || !String(body[k]).trim());
   516→        if (missing.length) {

   518→        }

   520→          return json({ success: false, error: 'Tarih biçimi geçersiz.' }, 400);
   521→        }

   523→          method: 'POST',
   524→          headers: { 'content-type': 'application/json' },

   526→          timeoutMs: 12000,
   527→        });
   528→        if (r.ok && r.data) return json({ success: true, data: r.data });
   529→        if (r.data && typeof r.data.error === 'string') {
   530→          const status = [400, 401, 403, 404, 409, 422].includes(r.status) ? r.status : 400;
   531→          return json({ success: false, error: r.data.error }, status);
   532→        }
   533→        return json({ success: false, error: 'Randevu servisi şu anda yanıt vermiyor. Lütfen birkaç saniye sonra tekrar dene.' }, 502);
   534→      }

   537→      if (path === '/' && method === 'GET') return htmlRes(homePage(BASE_DOMAIN, gwPort));
   538→      if (path === '/uyelik' && method === 'GET') return htmlRes(memberPage(BASE_DOMAIN, gwPort));
   539→

   541→      if (mDetail && method === 'GET') {
   542→        const slug = safeDecode(mDetail[1]);
   543→        // ?embed=1 → siteye göm modu (iframe içi kompakt: nav/footer gizli)
   544→        const embed = url.searchParams.get('embed') === '1';

   547→      }
   548→

   551→      // Aynı kamu randevu API'lerini kullanır (ek saldırı yüzeyi yok).
   552→      const mWidget = path.match(/^\/widget\/([^/]+)$/);
   553→      if (mWidget && method === 'GET') {
   554→        const slug = safeDecode(mWidget[1]);
   555→        const page = await renderDetail(slug, false, gwPort, undefined, true);
   556→        return htmlRes(page.body, page.status);
   557→      }
   558→
   559→      // --- gömülü varlıklar
   560→      if (path === '/assets/style.css' && method === 'GET') return assetRes(STYLE_CSS, 'text/css');
   561→      if (path === '/assets/home.js' && method === 'GET') return assetRes(HOME_JS, 'text/javascript');
   562→      if (path === '/assets/detail.js' && method === 'GET') return assetRes(DETAIL_JS, 'text/javascript');

   564→
   565→      // --- leaflet varlıkları (self-hosted harita kütüphanesi)
   566→      const lf = LEAFLET_FILES[path];
   567→      if (lf && (method === 'GET' || method === 'HEAD')) return leafletRes(lf);
   568→
   569→      // Bilinmeyen sayfa: önerili 404 — ?q= terimi hizmet adlarında da aranır
   570→      const nfq = (url.searchParams.get('q') || '').trim();
   571→      let sugg: Json[] = [];
   572→      try {
   573→        sugg = rankSuggestions(await getBusinesses(), nfq);
   574→      } catch {
   575→        // öneri alınamadı
   576→      }
   577→      return htmlRes(notFoundPage(undefined, gwPort, null, sugg, subSlug && BASE_DOMAIN ? BASE_DOMAIN : undefined, nfq || undefined), 404);
   578→    } catch (e: any) {
   579→      const isUp = e instanceof UpstreamError;
   580→      console.error(`[customer-page] ${method} ${path} →`, isUp ? e.message : e?.message || e);
   581→      return htmlRes(
   582→        serverErrorPage(isUp ? e.message : 'Beklenmeyen bir sorun oluştu. Lütfen sayfayı yenile.', url.searchParams.get('XTransformPort')),
   583→        isUp ? 502 : 500
   584→      );
   585→    }
   586→  },
