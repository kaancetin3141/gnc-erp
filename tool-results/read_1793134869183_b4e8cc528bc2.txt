    37→type Entry = { expires: number; data: Json };
    38→const mem = new Map<string, Entry>();
    39→
    40→function cacheGet(key: string): Json | null {
    41→  const e = mem.get(key);
    42→  if (!e) return null;
    43→  if (Date.now() > e.expires) {
    44→    mem.delete(key);
    45→    return null;
    46→  }
    47→  return e.data;
    48→}
    49→
    50→function cacheSet(key: string, data: Json, ttlMs: number) {
    51→  mem.set(key, { expires: Date.now() + ttlMs, data });
    52→  if (mem.size > 200) {
    53→    const now = Date.now();
    54→    for (const [k, v] of mem) if (now > v.expires) mem.delete(k);
    55→  }
    56→}

    62→async function upstream(path: string, init?: RequestInit & { timeoutMs?: number }) {
    63→  const timeoutMs = init?.timeoutMs ?? 8000;
    64→  const ctrl = new AbortController();
    65→  const timer = setTimeout(() => ctrl.abort(), timeoutMs);
    66→  try {
    67→    const res = await fetch(UPSTREAM + path, {
    68→      ...init,
    69→      signal: ctrl.signal,
    70→      headers: { accept: 'application/json', ...(init?.headers || {}) },
    71→    });
    72→    const text = await res.text();
    73→    let data: any = null;
    74→    try {
    75→      data = JSON.parse(text);
    76→    } catch {
    77→      data = null;
    78→    }
    79→    return { ok: res.ok, status: res.status, data };
    80→  } catch {
    81→    return { ok: false, status: 502, data: null as any };
    82→  } finally {
    83→    clearTimeout(timer);
    84→  }
    85→}

    89→function normBiz(it: Json): Json {
    90→  return {
    91→    id: it.id ?? '',
    92→    slug: it.slug ?? '',
    93→    name: it.name ?? '',
    94→    type: it.type ?? '',
    95→    address: it.address ?? '',
    96→    city: it.city ?? '',
    97→    district: it.district ?? '',
    98→    photo: it.photo ?? null,
    99→    phone: it.phone ?? '',
   100→    email: it.email ?? '',
   101→    lat: typeof it.lat === 'number' && !isNaN(it.lat) ? it.lat : it.lat != null && !isNaN(Number(it.lat)) ? Number(it.lat) : null,
   102→    lng: typeof it.lng === 'number' && !isNaN(it.lng) ? it.lng : it.lng != null && !isNaN(Number(it.lng)) ? Number(it.lng) : null,
   103→    workingHours: it.workingHours && typeof it.workingHours === 'object' ? it.workingHours : {},
   104→    services: Array.isArray(it.services) ? it.services : [],
   105→    // değerlendirme özeti (ana sayfa kartındaki ★ rozeti)
   106→    ratingAvg: typeof it.ratingAvg === 'number' ? it.ratingAvg : 0,
   107→    ratingCount: typeof it.ratingCount === 'number' ? it.ratingCount : 0,
   108→  };

   111→const trLower = (s: unknown) => String(s ?? '').toLocaleLowerCase('tr');
   112→
   113→function filterBiz(items: Json[], q: string, type: string, city: string): Json[] {
   114→  const nq = trLower(q.trim());
   115→  return items.filter((b) => {
   116→    if (type && b.type !== type) return false;
   117→    if (city && b.city !== city) return false;
   118→    if (nq) {
   119→      const svcNames = (b.services as Json[]).map((s) => trLower(s.name)).join(' ');
   120→      const hay = trLower([b.name, b.address, b.district, b.city].join(' ') + ' ' + svcNames);
   121→      if (!hay.includes(nq)) return false;
   122→    }
   123→    return true;
   124→  });
   125→}

   127→async function getBusinesses(): Promise<Json[]> {
   128→  const hit = cacheGet('businesses');
   129→  if (hit && Array.isArray(hit.items)) return hit.items;
   130→  const r = await upstream('/api/public/providers');
   131→  if (!r.ok || !r.data || !Array.isArray(r.data.items)) {
   132→    throw new UpstreamError('İşletme listesi şu anda alınamıyor — ana uygulama yanıt vermiyor olabilir. Birkaç saniye sonra tekrar dene.');
   133→  }
   134→  const items = (r.data.items as Json[]).map(normBiz);
   135→  cacheSet('businesses', { items }, 60_000);
   136→  return items;
   137→}

   139→// 404 öneri sıralaması: arama terimi işletme ADI yanında HİZMET adı/etiketinde de
   140→// aranır; hizmet eşleşmesi en güçlü sinyaldir ("manik" → Manikür hizmetli kuaför).
   141→// Eşleşen hizmet adı _svcMatch alanına yazılır (arayüzde chip olarak gösterilir).
   142→function rankSuggestions(items: Json[], q: string, excludeSlug?: string, limit = 6): Json[] {
   143→  const term = trLower(q).trim().slice(0, 64);
   144→  const pool = excludeSlug ? items.filter((b) => b.slug !== excludeSlug) : items;
   145→  if (!term) return pool.slice(0, limit);
   146→  const scored = pool.map((b) => {
   147→    let score = 0;
   148→    let svcMatch = '';
   149→    const services = Array.isArray(b.services) ? (b.services as Json[]) : [];
   150→    for (const sv of services) {
   151→      const nm = trLower(sv.name);
   152→      const cat = trLower(sv.category);
   153→      if (nm === term) { score = Math.max(score, 100); svcMatch = String(sv.name || ''); break; }
   154→      if (nm.includes(term)) { score = Math.max(score, 60); if (!svcMatch) svcMatch = String(sv.name || ''); }
   155→      else if (cat && (cat === term || cat.includes(term))) { score = Math.max(score, 40); if (!svcMatch) svcMatch = String(sv.category || ''); }
   156→    }
   157→    if (trLower(b.name).includes(term)) score += 30;
   158→    const hay = trLower([b.address, b.district, b.city, b.type].join(' '));
   159→    if (hay.includes(term)) score += 10;
   160→    return { b, score, svcMatch };
   161→  });
   162→  scored.sort((x, y) => y.score - x.score);
   163→  return scored
   164→    .filter((x) => x.score > 0)
   165→    .slice(0, limit)
   166→    .map((x) => (x.svcMatch ? { ...x.b, _svcMatch: x.svcMatch } : x.b));
   167→}

   199→async function renderDetail(slug: string, subdomain: boolean, gwPort?: string | null, q?: string, embed?: boolean): Promise<{ body: string; status: number }> {
   200→  const r = await upstream(`/api/public/providers/${encodeURIComponent(slug)}`);
   201→  if (r.status === 404) {
   202→    // Alt alan adı modunda "tüm işletmelere dön" hedefi: randevu.ANA_DOMAIN ana liste
   203→    const randevuHome = subdomain && BASE_DOMAIN ? `https://${listHost()}` : null;
   204→    // Hata sayfasını zenginleştir: terim hizmet adlarında da aranır (hizmet eşleşmesi öne çıkar)
   205→    let sugg: Json[] = [];
   206→    try {
   207→      const all = await getBusinesses();
   208→      sugg = rankSuggestions(all, q || '', slug);
   209→    } catch {
   210→      // liste alınamadıysa önerisiz 404 gösterilir
   211→    }
   212→    return { body: notFoundPage(slug, gwPort, randevuHome, sugg, subdomain && BASE_DOMAIN ? BASE_DOMAIN : undefined, q), status: 404 };
   213→  }
   214→  if (!r.ok || !r.data || !r.data.provider) {
   215→    throw new UpstreamError('İşletme bilgileri şu anda alınamıyor — ana uygulama yanıt vermiyor olabilir.');
   216→  }
   217→  const services: Json[] = Array.isArray(r.data.services) ? r.data.services : [];
   218→  const provider = normBiz({ ...r.data.provider, services });
   219→  // Değerlendirmeler (server-side çekilir; hata olursa bölüm boş düşer)
   220→  let reviews: { items: Json[]; summary: Json } | null = null;
   221→  try {
   222→    const rr = await upstream(`/api/public/providers/${encodeURIComponent(slug)}/reviews?limit=20`);
   223→    const rd = rr.data as Json | null;
   224→    if (rr.ok && rd && Array.isArray(rd.items)) {
   225→      reviews = {
   226→        items: rd.items,
   227→        summary: (rd.summary ?? { average: 0, count: 0, distribution: {} }) as Json,
   228→      };
   229→    }
   230→  } catch {
   231→    // değerlendirmeler olmadan da sayfa render edilir
   232→  }
   233→  // Gateway modunda ana sayfa bağlantısına param ekle (CRM'e düşmesin)
   234→  // Düzeltme: "Tüm işletmeler" her zaman randevu LİSTESİNE gider (ana siteye değil)
   235→  const homeUrl = subdomain && BASE_DOMAIN ? `https://${listHost()}` : gwq('/')
   236→  return { body: detailPage({ provider, services, reviews, homeUrl, gwPort, embed }), status: 200 };
   237→
   238→  // BASE_DOMAIN=randevu.x biçimindeyse aynen; bare domain ise randevu. öneki ekle
   239→  function listHost(): string {
   240→    return BASE_DOMAIN.startsWith('randevu.') ? BASE_DOMAIN : `randevu.${BASE_DOMAIN}`;
   241→  }
   242→
   243→  function gwq(p: string) {
   244→    return gwPort ? p + (p.indexOf('?') === -1 ? '?' : '&') + 'XTransformPort=' + encodeURIComponent(gwPort) : p;
   245→  }
   246→}

   403→      }

   405→      // --- API: bekleme listesi proxy'si (CRM /api/public/waitlist → same-origin)
   406→      // POST /api/waitlist → beklemeye gir (üye token'ı varsa Bearer iletilir)
   407→      // GET  /api/waitlist?phone=  → "Bekleme Durumum" sorgusu
   408→      // Gerçek istemci IP'si (x-forwarded-for) rate limit için taşınır —
   409→      // aksi halde tüm müşteri sitesi istekleri tek 'unknown' kovasına düşerdi.
   410→      if (path === '/api/waitlist') {
   411→        const auth = req.headers.get('authorization');
   412→        const headers: Record<string, string> = { accept: 'application/json' };
   413→        if (auth) headers['authorization'] = auth;
   414→        const xff = req.headers.get('x-forwarded-for') || req.headers.get('x-real-ip');
   415→        if (xff) headers['x-forwarded-for'] = xff;
   416→        if (method === 'GET') {
   417→          const qs = url.searchParams.toString();
   418→          const r = await upstream('/api/public/waitlist' + (qs ? '?' + qs : ''), { headers });
   419→          if (r.data) return json(r.data, r.ok ? 200 : r.status);
   420→          return json({ error: 'Bekleme durumu şu anda alınamıyor. Birazdan tekrar dene.' }, 502);
   421→        }
   422→        if (method !== 'POST') return json({ error: 'Bu uç için GET veya POST kullan.' }, 405);
   423→        let wbody: Json;
   424→        try {
   425→          const raw = await req.text();
   426→          wbody = (raw.trim() ? JSON.parse(raw) : {}) as Json;
   427→        } catch {
   428→          return json({ error: 'İstek gövdesi okunamadı.' }, 400);
   429→        }
   430→        const r = await upstream('/api/public/waitlist', {
   431→          method: 'POST',
   432→          headers: { ...headers, 'content-type': 'application/json' },
   433→          body: JSON.stringify(wbody),
   434→          timeoutMs: 12000,
   435→        });
   436→        // 201 (oluştu) dahil olmak üzere status aynen iletilir
   437→        if (r.data) return json(r.data, r.status);
   438→        return json({ error: 'Bekleme servisi şu anda yanıt vermiyor. Birazdan tekrar dene.' }, 502);
   439→      }
   440→
   441→      // --- API: damga kartı proxy'si (CRM /api/public/stampcard → same-origin)
   442→      // GET /api/stampcard — üyenin damga kartları (Bearer token iletilir;
   443→      // /api/reviews proxy deseni). Kartlar yalnız aktif damga programı
   444→      // olan işletmeler için döner.
   445→      if (path === '/api/stampcard' && method === 'GET') {
   446→        const auth = req.headers.get('authorization');
   447→        const headers: Record<string, string> = { accept: 'application/json' };
   448→        if (auth) headers['authorization'] = auth;
   449→        const r = await upstream('/api/public/stampcard', { headers });
   450→        if (r.data) return json(r.data, r.ok ? 200 : r.status);
   451→        return json({ error: 'Damga kartların şu anda alınamıyor. Birazdan tekrar dene.' }, 502);
   452→      }
   453→
   454→      // --- API: ICS takvim dosyası proxy'si (CRM /api/public/appointments/ics → same-origin)
   455→      // "📅 Takvime ekle" butonları bu ucu kullanır: ?code=XXXXXXXX (misafir) ya da
   456→      // ?id=<appointmentId> + Authorization: Bearer (üye). Token varsa upstream'e
   457→      // iletilir; gerçek istemci IP'si (x-forwarded-for) rate limit için taşınır.
   458→      // text/calendar yanıtı olduğu gibi (Content-Disposition dahil) aktarılır;
   459→      // JSON hataları (400/401/404/409/429) status koduyla iletilir.
   460→      if (path === '/api/appointments/ics' && method === 'GET') {
   461→        const qs = url.searchParams.toString();
   462→        const fwdHeaders: Record<string, string> = { accept: 'text/calendar, application/json' };
   463→        const auth = req.headers.get('authorization');
   464→        if (auth) fwdHeaders['authorization'] = auth;
   465→        const xff = req.headers.get('x-forwarded-for') || req.headers.get('x-real-ip');
   466→        if (xff) fwdHeaders['x-forwarded-for'] = xff;
   467→        try {
   468→          const ctrl = new AbortController();
   469→          const timer = setTimeout(() => ctrl.abort(), 8000);
   470→          const res = await fetch(UPSTREAM + '/api/public/appointments/ics' + (qs ? '?' + qs : ''), {
   471→            headers: fwdHeaders,
   472→            signal: ctrl.signal,
   473→          });
   474→          clearTimeout(timer);
   475→          const ct = res.headers.get('content-type') || '';
   476→          if (ct.includes('text/calendar')) {
   477→            const body = await res.text();
   478→            return new Response(body, {
   479→              status: res.status,
   480→              headers: {
   481→                'content-type': 'text/calendar; charset=utf-8',
   482→                'content-disposition':
   483→                  res.headers.get('content-disposition') || 'attachment; filename="randevu.ics"',
   484→                'cache-control': 'no-store',
   485→              },
   486→            });
   487→          }
   488→          // hata yanıtı: JSON (404/429/400/…) — olduğu gibi aktar
   489→          let data: Json | null = null;
   490→          try {
   491→            data = (await res.json()) as Json;
   492→          } catch {
   493→            data = null;
   494→          }
   495→          return json(data ?? { error: 'Takvim dosyası alınamadı.' }, res.status);
   496→        } catch {
   497→          return json({ error: 'Takvim servisi şu anda yanıt vermiyor. Birazdan tekrar dene.' }, 502);
   498→        }
   499→      }
   500→
   501→      // --- API: randevu oluştur (upstream'e ilet)
   502→      if (path === '/api/book') {
   503→        if (method !== 'POST') return json({ success: false, error: 'Bu uç için POST kullan.' }, 405);
   504→        let body: Json;
   505→        try {
   506→          body = (await req.json()) as Json;
   507→        } catch {
   508→          return json({ success: false, error: 'İstek gövdesi okunamadı.' }, 400);
   509→        }
   510→        const required: Array<keyof Json> = ['providerSlug', 'serviceId', 'date', 'time', 'customerName', 'customerPhone'];
   511→        const missing = required.filter((k) => !body[k] || typeof body[k] !== 'string' || !String(body[k]).trim());
   512→        if (missing.length) {
   513→          return json({ success: false, error: 'Lütfen hizmet, tarih, saat, ad ve telefon bilgilerini eksiksiz doldur.' }, 400);
   514→        }
   515→        if (!/^\d{4}-\d{2}-\d{2}$/.test(String(body.date))) {
   516→          return json({ success: false, error: 'Tarih biçimi geçersiz.' }, 400);
   517→        }
   518→        const r = await upstream('/api/public/appointments', {
   519→          method: 'POST',
   520→          headers: { 'content-type': 'application/json' },
   521→          body: JSON.stringify(body),
   522→          timeoutMs: 12000,
   523→        });
   524→        if (r.ok && r.data) return json({ success: true, data: r.data });
   525→        if (r.data && typeof r.data.error === 'string') {
   526→          const status = [400, 401, 403, 404, 409, 422].includes(r.status) ? r.status : 400;
   527→          return json({ success: false, error: r.data.error }, status);
   528→        }
   529→        return json({ success: false, error: 'Randevu servisi şu anda yanıt vermiyor. Lütfen birkaç saniye sonra tekrar dene.' }, 502);
   530→      }
   531→
   532→      // --- sayfalar
   533→      if (path === '/' && method === 'GET') return htmlRes(homePage(BASE_DOMAIN, gwPort));
   534→      if (path === '/uyelik' && method === 'GET') return htmlRes(memberPage(BASE_DOMAIN, gwPort));
   535→
   536→      const mDetail = path.match(/^\/isletme\/([^/]+)$/);
   537→      if (mDetail && method === 'GET') {
   538→        const slug = safeDecode(mDetail[1]);
   539→        // ?embed=1 → siteye göm modu (iframe içi kompakt: nav/footer gizli)
   540→        const embed = url.searchParams.get('embed') === '1';
   541→        const page = await renderDetail(slug, false, gwPort, url.searchParams.get('q') || undefined, embed);
   542→        return htmlRes(page.body, page.status);
   543→      }
   544→
   545→      // --- 🔗 siteye göm (widget) — işletmelerin kendi sitelerine eklediği
   546→      // iframe kaynağı: /widget/:slug → detail sayfasının çerçevesiz hâli.
   547→      // Aynı kamu randevu API'lerini kullanır (ek saldırı yüzeyi yok).
   548→      const mWidget = path.match(/^\/widget\/([^/]+)$/);
   549→      if (mWidget && method === 'GET') {
   550→        const slug = safeDecode(mWidget[1]);
   551→        const page = await renderDetail(slug, false, gwPort, undefined, true);
   552→        return htmlRes(page.body, page.status);
   553→      }
   554→
   555→      // --- gömülü varlıklar
   556→      if (path === '/assets/style.css' && method === 'GET') return assetRes(STYLE_CSS, 'text/css');
   557→      if (path === '/assets/home.js' && method === 'GET') return assetRes(HOME_JS, 'text/javascript');
   558→      if (path === '/assets/detail.js' && method === 'GET') return assetRes(DETAIL_JS, 'text/javascript');
   559→      if (path === '/assets/member.js' && method === 'GET') return assetRes(MEMBER_JS, 'text/javascript');
   560→
   561→      // --- leaflet varlıkları (self-hosted harita kütüphanesi)
   562→      const lf = LEAFLET_FILES[path];
   563→      if (lf && (method === 'GET' || method === 'HEAD')) return leafletRes(lf);
   564→
   565→      // Bilinmeyen sayfa: önerili 404 — ?q= terimi hizmet adlarında da aranır
   566→      const nfq = (url.searchParams.get('q') || '').trim();
   567→      let sugg: Json[] = [];
   568→      try {
   569→        sugg = rankSuggestions(await getBusinesses(), nfq);
   570→      } catch {
   571→        // öneri alınamadı
   572→      }
   573→      return htmlRes(notFoundPage(undefined, gwPort, null, sugg, subSlug && BASE_DOMAIN ? BASE_DOMAIN : undefined, nfq || undefined), 404);
   574→    } catch (e: any) {
   575→      const isUp = e instanceof UpstreamError;
   576→      console.error(`[customer-page] ${method} ${path} →`, isUp ? e.message : e?.message || e);
   577→      return htmlRes(
   578→        serverErrorPage(isUp ? e.message : 'Beklenmeyen bir sorun oluştu. Lütfen sayfayı yenile.', url.searchParams.get('XTransformPort')),
   579→        isUp ? 502 : 500
   580→      );
   581→    }
   582→  },
   583→});

   585→function safeDecode(s: string): string {
   586→  try {
   587→    return decodeURIComponent(s);
   588→  } catch {
   589→    return s;
   590→  }
