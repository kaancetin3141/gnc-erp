   151→function StatsRow({ users }: { users: UserListItem[] }) {
   152→  const total = users.length
   153→  const active = users.filter((u) => u.status === 'active').length
   154→  const managers = users.filter((u) => u.role === 'manager' || u.role === 'admin' || u.role === 'superadmin').length
   155→  const reps = users.filter((u) => u.role === 'rep').length
   156→
   157→  const stats = [
   158→    { label: 'Toplam Kullanıcı', value: total, icon: Users, color: 'from-emerald-500 to-teal-600', sub: `${active} aktif` },
   159→    { label: 'Aktif', value: active, icon: UserCheck, color: 'from-teal-500 to-cyan-600', sub: `${total - active} pasif` },
   160→    { label: 'Yönetici', value: managers, icon: Crown, color: 'from-violet-500 to-purple-600', sub: 'müdür + admin' },
   161→    { label: 'Temsilci', value: reps, icon: GitBranch, color: 'from-amber-500 to-orange-600', sub: 'satış temsilcisi' },
   162→  ]
   163→  return (
   164→    <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   165→      {stats.map((s) => (
   166→        <Card key={s.label} className="overflow-hidden">
   167→          <CardContent className="p-5">
   168→            <div className="flex items-start justify-between mb-3">
   169→              <div className={cn('w-10 h-10 rounded-lg bg-gradient-to-br flex items-center justify-center shadow-sm', s.color)}>
   170→                <s.icon className="w-5 h-5 text-white" />
   171→              </div>
   172→            </div>
   173→            <div className="text-2xl font-bold tracking-tight">{s.value}</div>
   174→            <div className="text-sm text-muted-foreground mt-0.5">{s.label}</div>
   175→            <div className="text-xs text-muted-foreground/80 mt-1.5">{s.sub}</div>
   176→          </CardContent>
   177→        </Card>
   178→      ))}
   179→    </div>
   180→  )
   181→}

   183→// ─── Add/Edit Dialog ──────────────────────────────────────────────
   184→function UserFormDialog({
   185→  open, onOpenChange, editing, users, actor,
   186→}: {
   187→  open: boolean
   188→  onOpenChange: (v: boolean) => void
   189→  editing: UserListItem | null
   190→  users: UserListItem[]
   191→  actor: SessionUser
   192→}) {
   193→  const qc = useQueryClient()
   194→  const [form, setForm] = useState<UserFormData>(() =>
   195→    editing
   196→      ? {
   197→          name: editing.name,
   198→          email: editing.email,
   199→          role: editing.role,
   200→          title: editing.title ?? '',
   201→          phone: editing.phone ?? '',
   202→          managerId: editing.managerId ?? '',
   203→          status: editing.status,
   204→        }
   205→      : EMPTY_FORM,
   206→  )
   207→  const [submitting, setSubmitting] = useState(false)
   208→  const [showRoleNotice, setShowRoleNotice] = useState(false)
   209→
   210→  const createMut = useMutation({
   211→    mutationFn: (body: UserFormData) => apiPost<UserListItem>('/api/users', body),
   212→    onSuccess: () => {
   213→      toast.success('Kullanıcı davet edildi', { description: `${form.name} sisteme eklendi.` })
   214→      qc.invalidateQueries({ queryKey: qk.users })
   215→      onOpenChange(false)
   216→    },
   217→    onError: (e: ApiError) => toast.error('Hata', { description: e.message }),
   218→    onSettled: () => setSubmitting(false),
   219→  })
   220→
   221→  const updateMut = useMutation({
   222→    mutationFn: (body: Partial<UserFormData>) => apiPatch<UserListItem>(`/api/users/${editing?.id}`, body),
   223→    onSuccess: () => {
   224→      toast.success('Kullanıcı güncellendi', { description: `${form.name} bilgileri kaydedildi.` })
   225→      qc.invalidateQueries({ queryKey: qk.users })
   226→      onOpenChange(false)
   227→    },
   228→    onError: (e: ApiError) => toast.error('Hata', { description: e.message }),
   229→    onSettled: () => setSubmitting(false),
   230→  })
   231→
   232→  const handleSubmit = () => {
   233→    if (!form.name.trim() || !form.email.trim()) {
   234→      toast.error('Eksik bilgi', { description: 'Ad ve e-posta zorunludur.' })
   235→      return
   236→    }
   237→    setSubmitting(true)
   238→    const payload: UserFormData = {
   239→      ...form,
   240→      managerId: form.managerId === 'none' ? '' : form.managerId,
   241→    }
   242→    if (editing) {
   243→      updateMut.mutate(payload)
   244→    } else {
   245→      createMut.mutate(payload)
   246→    }
   247→  }
   248→
   249→  // Manager select — editing kullanıcısını dışla, actor dahil tüm tenant kullanıcıları
   250→  const managerOptions = users.filter((u) => u.id !== editing?.id)
   251→
   252→  // Rol değişince bildirim
   253→  const handleRoleChange = (role: Role) => {
   254→    setForm((f) => ({ ...f, role }))
   255→    setShowRoleNotice(true)
   256→  }
   257→
   258→  const roleDefaultPerms = getRolePermissions(form.role)
   259→
   260→  return (
   261→    <Dialog open={open} onOpenChange={onOpenChange}>
   262→      <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto">
   263→        <DialogHeader>
   264→          <DialogTitle className="flex items-center gap-2">
   265→            {editing ? <Pencil className="w-4 h-4 text-emerald-600" /> : <UserPlus className="w-4 h-4 text-emerald-600" />}
   266→            {editing ? 'Kullanıcı Düzenle' : 'Kullanıcı Davet Et'}
   267→          </DialogTitle>
   268→          <DialogDescription>
   269→            {editing
   270→              ? `${editing.name} kullanıcısının bilgilerini ve rolünü güncelleyin.`
   271→              : 'Yeni bir kullanıcı ekleyin. Rolüne göre varsayılan yetkiler otomatik atanır.'}
   272→          </DialogDescription>
   273→        </DialogHeader>
   274→
   275→        <div className="grid sm:grid-cols-2 gap-4 py-2">
   276→          <div className="space-y-1.5">
   277→            <Label htmlFor="u-name">Ad Soyad *</Label>
   278→            <Input id="u-name" value={form.name} onChange={(e) => setForm((f) => ({ ...f, name: e.target.value }))} placeholder="Örn. Ahmet Yılmaz" />
   279→          </div>
   280→          <div className="space-y-1.5">
   281→            <Label htmlFor="u-email">E-posta *</Label>
   282→            <Input id="u-email" type="email" value={form.email} onChange={(e) => setForm((f) => ({ ...f, email: e.target.value }))} placeholder="ahmet@sirket.com" />
   283→          </div>
   284→          <div className="space-y-1.5">
   285→            <Label htmlFor="u-role">Rol</Label>
   286→            <Select value={form.role} onValueChange={(v) => handleRoleChange(v as Role)}>
   287→              <SelectTrigger id="u-role" className="w-full">
   288→                <SelectValue />
   289→              </SelectTrigger>
   290→              <SelectContent>
   291→                {SELECTABLE_ROLES.map((r) => (
   292→                  <SelectItem key={r} value={r}>{ROLE_LABELS[r]}</SelectItem>
   293→                ))}
   294→              </SelectContent>
   295→            </Select>
   296→          </div>
   297→          <div className="space-y-1.5">
   298→            <Label htmlFor="u-title">Ünvan</Label>
   299→            <Input id="u-title" value={form.title} onChange={(e) => setForm((f) => ({ ...f, title: e.target.value }))} placeholder="Satış Müdürü" />
   300→          </div>
   301→          <div className="space-y-1.5">
   302→            <Label htmlFor="u-phone">Telefon</Label>
   303→            <Input id="u-phone" value={form.phone} onChange={(e) => setForm((f) => ({ ...f, phone: e.target.value }))} placeholder="+90 5xx xxx xx xx" />
   304→          </div>
   305→          <div className="space-y-1.5">
   306→            <Label htmlFor="u-manager">Yönetici</Label>
   307→            <Select value={form.managerId || 'none'} onValueChange={(v) => setForm((f) => ({ ...f, managerId: v === 'none' ? '' : v }))}>
   308→              <SelectTrigger id="u-manager" className="w-full">
   309→                <SelectValue placeholder="Yönetici seçin" />
   310→              </SelectTrigger>
   311→              <SelectContent>
   312→                <SelectItem value="none">— Yönetici yok —</SelectItem>
   313→                {managerOptions.map((m) => (
   314→                  <SelectItem key={m.id} value={m.id}>
   315→                    {m.name} · {ROLE_LABELS[m.role]}
   316→                  </SelectItem>
   317→                ))}
   318→              </SelectContent>
   319→            </Select>
   320→          </div>
   321→          <div className="space-y-1.5">
   322→            <Label htmlFor="u-status">Durum</Label>
   323→            <Select value={form.status} onValueChange={(v) => setForm((f) => ({ ...f, status: v }))}>
   324→              <SelectTrigger id="u-status" className="w-full">
   325→                <SelectValue />
   326→              </SelectTrigger>
   327→              <SelectContent>
   328→                <SelectItem value="active">Aktif</SelectItem>
   329→                <SelectItem value="passive">Pasif</SelectItem>
   330→              </SelectContent>
   331→            </Select>
   332→          </div>
   333→        </div>
   334→
   335→        {/* Rol değişimi bildirimi */}
   336→        {showRoleNotice && (
   337→          <div className="rounded-lg border border-emerald-200 bg-emerald-50/60 dark:bg-emerald-950/20 dark:border-emerald-900/40 p-3 flex gap-2.5">
   338→            <Info className="w-4 h-4 text-emerald-600 shrink-0 mt-0.5" />
   339→            <div className="text-xs text-emerald-800 dark:text-emerald-300 space-y-1">
   340→              <p className="font-medium">
   341→                {ROLE_LABELS[form.role]} rolü için varsayılan yetkiler:
   342→              </p>
   343→              <div className="flex flex-wrap gap-1">
   344→                {roleDefaultPerms.length > 0 ? roleDefaultPerms.slice(0, 6).map((p) => (
   345→                  <span key={p} className="px-1.5 py-0.5 rounded bg-white dark:bg-emerald-950/40 border border-emerald-200 dark:border-emerald-900/50 font-mono text-[10px]">
   346→                    {p}
   347→                  </span>
   348→                )) : <span className="italic">Yetki yok</span>}
   349→                {roleDefaultPerms.length > 6 && (
   350→                  <span className="text-emerald-700 dark:text-emerald-400">+{roleDefaultPerms.length - 6} daha</span>
   351→                )}
   352→              </div>
   353→              <p className="text-[11px] text-emerald-700/80 dark:text-emerald-400/80 pt-0.5">
   354→                Yetkileri özelleştirmek için &ldquo;Yetki Matrisi&rdquo; → kullanıcı seç → düzenle.
   355→              </p>
   356→            </div>
   357→          </div>

   359→
   360→        {/* Cycle önleme uyarısı */}
   361→        <div className="rounded-lg border border-amber-200 bg-amber-50/40 dark:bg-amber-950/20 dark:border-amber-900/40 p-2.5 flex gap-2">
   362→          <AlertCircle className="w-3.5 h-3.5 text-amber-600 shrink-0 mt-0.5" />
   363→          <p className="text-[11px] text-amber-800 dark:text-amber-300">
   364→            Yönetici ataması döngü oluşturamaz (kullanıcı, kendi astının astı olamaz). Sistem otomatik engeller.
   365→          </p>
   366→        </div>
   367→
   368→        <DialogFooter className="pt-2">
   369→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={submitting}>
   370→            İptal
   371→          </Button>
   372→          <Button
   373→            onClick={handleSubmit}
   374→            disabled={submitting}
   375→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   376→          >
   377→            {submitting ? 'Kaydediliyor…' : editing ? 'Değişiklikleri Kaydet' : 'Kullanıcı Oluştur'}
   378→          </Button>
   379→        </DialogFooter>
   380→      </DialogContent>
   381→    </Dialog>
   382→  )
   383→}

   385→// ─── Yetki Matrisi (read-only) ────────────────────────────────────
   386→function PermissionMatrixReadOnly() {
   387→  const roles: Role[] = ['admin', 'manager', 'rep', 'readonly', 'stock']
   388→  const groups = useMemo(() => {
   389→    const map = new Map<string, typeof ALL_PERMISSIONS>()
   390→    for (const p of ALL_PERMISSIONS) {
   391→      if (!map.has(p.group)) map.set(p.group, [])
   392→      map.get(p.group)!.push(p)
   393→    }
   394→    return Array.from(map.entries())
   395→  }, [])
   396→
   397→  return (
   398→    <div className="rounded-lg border overflow-hidden">
   399→      <div className="overflow-x-auto max-h-[420px] overflow-y-auto">
   400→        <table className="w-full text-sm">
   401→          <thead className="sticky top-0 z-10 bg-muted/80 backdrop-blur">
   402→            <tr className="border-b">
   403→              <th className="text-left font-medium p-3 min-w-[260px]">Yetki</th>
   404→              {roles.map((r) => (
   405→                <th key={r} className="text-center font-medium p-2 min-w-[110px]">
   406→                  <div className="flex flex-col items-center gap-1">
   407→                    <span className="text-xs">{ROLE_LABELS[r]}</span>
   408→                  </div>
   409→                </th>
   410→              ))}
   411→            </tr>
   412→          </thead>
   413→          <tbody>
   414→            {groups.map(([group, perms]) => (
   415→              <Fragment key={`g-${group}`}>
   416→                <tr className="bg-muted/30">
   417→                  <td colSpan={roles.length + 1} className="p-2.5 font-semibold text-xs uppercase tracking-wide text-muted-foreground">
   418→                    {group}
   419→                  </td>
   420→                </tr>
   421→                {perms.map((p) => (
   422→                  <tr key={p.key} className="border-b last:border-0 hover:bg-muted/40 transition-colors">
   423→                    <td className="p-2.5">
   424→                      <div className="font-medium text-[13px]">{p.label}</div>
   425→                      <div className="text-[10px] text-muted-foreground font-mono">{p.key}</div>
   426→                    </td>
   427→                    {roles.map((r) => {
   428→                      const has = ROLE_PERMISSIONS[r].includes(p.key)
   429→                      return (
   430→                        <td key={r} className="p-2 text-center">
   431→                          {has ? (
   432→                            <Tooltip>
   433→                              <TooltipTrigger asChild>
   434→                                <span className="inline-flex items-center justify-center w-6 h-6 rounded-md bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300">
   435→                                  <Check className="w-3.5 h-3.5" />
   436→                                </span>
   437→                              </TooltipTrigger>
   438→                              <TooltipContent>Bu rol bu yetkiye sahip</TooltipContent>
   439→                            </Tooltip>
   440→                          ) : (
   441→                            <span className="inline-flex items-center justify-center w-6 h-6 rounded-md bg-slate-50 text-slate-300 dark:bg-slate-800/40 dark:text-slate-600">
   442→                              <Minus className="w-3.5 h-3.5" />
   443→                            </span>
   444→                          )}
   445→                        </td>
   446→                      )
   447→                    })}
   448→                  </tr>
   449→                ))}
   450→              </Fragment>
   451→            ))}
   452→          </tbody>
   453→        </table>
   454→      </div>
   455→    </div>
   456→  )
   457→}

   560→                    {u.name} · {ROLE_LABELS[u.role]}

   887→// ─── Hiyerarşi ağacı ──────────────────────────────────────────────
   888→interface TreeNode extends UserListItem {
   889→  children: TreeNode[]
   890→}
   891→
   892→function buildTree(users: UserListItem[]): TreeNode[] {
   893→  const map = new Map<string, TreeNode>()
   894→  users.forEach((u) => map.set(u.id, { ...u, children: [] }))
   895→  const roots: TreeNode[] = []
   896→  users.forEach((u) => {
   897→    const node = map.get(u.id)!
   898→    if (u.managerId && map.has(u.managerId)) {
   899→      map.get(u.managerId)!.children.push(node)
   900→    } else {
   901→      roots.push(node)
   902→    }
   903→  })
   904→  return roots
   905→}
