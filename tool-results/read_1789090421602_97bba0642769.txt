    82→function isOverdue(t: Task): boolean {
    83→  if (t.status !== 'acik') return false
    84→  return new Date(t.dueDate).getTime() < Date.now()
    85→}
    86→
    87→function isDueToday(t: Task): boolean {
    88→  const d = new Date(t.dueDate)
    89→  const now = new Date()
    90→  return d.getFullYear() === now.getFullYear()
    91→    && d.getMonth() === now.getMonth()
    92→    && d.getDate() === now.getDate()
    93→}
    94→
    95→function isCompletedThisWeek(t: Task): boolean {
    96→  if (t.status !== 'tamamlandi' || !t.completedAt) return false
    97→  const completed = new Date(t.completedAt).getTime()
    98→  const sevenDaysAgo = Date.now() - 7 * 24 * 60 * 60 * 1000
    99→  return completed >= sevenDaysAgo
   100→}
   101→
   102→function isDueThisWeek(t: Task): boolean {
   103→  if (t.status !== 'acik') return false
   104→  const due = new Date(t.dueDate).getTime()
   105→  const now = new Date()
   106→  const startOfToday = new Date(now)
   107→  startOfToday.setHours(0, 0, 0, 0)
   108→  const endOfWeek = new Date(startOfToday)
   109→  endOfWeek.setDate(endOfWeek.getDate() + 7)
   110→  return due >= startOfToday.getTime() && due <= endOfWeek.getTime()
   111→}

   127→function getTaskGroup(t: Task): GroupKey {
   128→  if (t.status === 'tamamlandi') return 'tamamlanan'
   129→  if (t.status === 'iptal') return 'iptal'
   130→  const now = new Date()
   131→  const due = new Date(t.dueDate)
   132→  const todayStart = new Date(now); todayStart.setHours(0, 0, 0, 0)
   133→  const todayEnd = new Date(now); todayEnd.setHours(23, 59, 59, 999)
   134→  const tomorrowEnd = new Date(todayEnd); tomorrowEnd.setDate(tomorrowEnd.getDate() + 1)
   135→  const weekEnd = new Date(todayEnd); weekEnd.setDate(weekEnd.getDate() + 7)
   136→
   137→  if (due.getTime() < todayStart.getTime()) return 'gecikmis'
   138→  if (due.getTime() >= todayStart.getTime() && due.getTime() <= todayEnd.getTime()) return 'bugun'
   139→  if (due.getTime() > todayEnd.getTime() && due.getTime() <= tomorrowEnd.getTime()) return 'yarin'
   140→  if (due.getTime() > tomorrowEnd.getTime() && due.getTime() <= weekEnd.getTime()) return 'buhafta'
   141→  return 'sonra'
   142→}

   453→interface TaskFormDialogProps {
   454→  open: boolean
   455→  onOpenChange: (open: boolean) => void

   457→  initialTask?: Task
   458→  defaultAssigneeId: string

   462→
   463→function TaskFormDialog({
   464→  open, onOpenChange, mode, initialTask, defaultAssigneeId, users, onSaved,
   465→}: TaskFormDialogProps) {
   466→  const qc = useQueryClient()
   467→  const [form, setForm] = useState<TaskFormState>(() =>
   468→    initialTask ? formFromTask(initialTask) : emptyForm(defaultAssigneeId),
   469→  )
   470→  const [deleteOpen, setDeleteOpen] = useState(false)
   471→
   472→  // Reset form when dialog opens or task changes
   473→  const taskKey = initialTask?.id ?? 'new'
   474→  const [lastKey, setLastKey] = useState<string>(taskKey)
   475→  if (open && lastKey !== taskKey) {
   476→    setLastKey(taskKey)
   477→    setForm(initialTask ? formFromTask(initialTask) : emptyForm(defaultAssigneeId))
   478→  }
   479→
   480→  const set = <K extends keyof TaskFormState>(k: K, v: TaskFormState[K]) => {
   481→    setForm((p) => ({ ...p, [k]: v }))
   482→  }
   483→
   484→  const createMutation = useMutation({
   485→    mutationFn: (payload: Record<string, unknown>) =>
   486→      apiPost<Task>('/api/tasks', payload),
   487→    onSuccess: () => {
   488→      toast.success('Görev oluşturuldu')
   489→      qc.invalidateQueries({ queryKey: ['tasks'] })
   490→      qc.invalidateQueries({ queryKey: ['dashboard'] })
   491→      onOpenChange(false)
   492→      onSaved?.()
   493→    },
   494→    onError: (e: ApiError) => toast.error(e.message || 'Görev oluşturulamadı'),
   495→  })
   496→
   497→  const updateMutation = useMutation({
   498→    mutationFn: ({ id, payload }: { id: string; payload: Record<string, unknown> }) =>
   499→      apiPatch<Task>(`/api/tasks/${id}`, payload),
   500→    onSuccess: () => {
   501→      toast.success('Görev güncellendi')
   502→      qc.invalidateQueries({ queryKey: ['tasks'] })
   503→      qc.invalidateQueries({ queryKey: ['dashboard'] })
   504→      onOpenChange(false)
   505→      onSaved?.()
   506→    },
   507→    onError: (e: ApiError) => toast.error(e.message || 'Güncelleme başarısız'),
   508→  })
   509→
   510→  const deleteMutation = useMutation({
   511→    mutationFn: (id: string) => apiDelete(`/api/tasks/${id}`),
   512→    onSuccess: () => {
   513→      toast.success('Görev silindi')
   514→      qc.invalidateQueries({ queryKey: ['tasks'] })
   515→      qc.invalidateQueries({ queryKey: ['dashboard'] })
   516→      setDeleteOpen(false)
   517→      onOpenChange(false)
   518→      onSaved?.()
   519→    },
   520→    onError: (e: ApiError) => toast.error(e.message || 'Silme başarısız'),
   521→  })
   522→
   523→  const handleSubmit = (e: React.FormEvent) => {
   524→    e.preventDefault()
   525→    if (!form.title.trim()) {
   526→      toast.error('Görev başlığı gerekli')
   527→      return
   528→    }
   529→    if (!form.dueDate) {
   530→      toast.error('Bitiş tarihi gerekli')
   531→      return
   532→    }
   533→    const payload: Record<string, unknown> = {
   534→      title: form.title.trim(),
   535→      description: form.description.trim() || null,
   536→      dueDate: new Date(form.dueDate).toISOString(),
   537→      assigneeId: form.assigneeId || undefined,
   538→      customerId: form.customerId || null,
   539→      priority: form.priority,
   540→      reminderTime: form.reminderTime || null,
   541→    }
   542→    if (mode === 'edit') {
   543→      payload.status = form.status
   544→      updateMutation.mutate({ id: initialTask!.id, payload })
   545→    } else {
   546→      createMutation.mutate(payload)
   547→    }
   548→  }
   549→
   550→  const isPending = createMutation.isPending || updateMutation.isPending
   551→
