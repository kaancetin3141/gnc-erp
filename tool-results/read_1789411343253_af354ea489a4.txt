   605→function CreateSiteDialog({ onCreated }: { onCreated: () => void }) {
   606→  const [open, setOpen] = useState(false)
   607→  const [form, setForm] = useState({ name: '', address: '', city: '', phone: '', managerName: '', dueDay: '5', defaultDueAmount: '500' })
   608→  const [saving, setSaving] = useState(false)
   609→
   610→  const handleSave = async () => {
   611→    setSaving(true)
   612→    try {
   613→      await apiPost('/api/site', form)
   614→      toast.success('Site oluşturuldu')
   615→      onCreated()
   616→      setOpen(false)
   617→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Oluşturulamadı') }
   618→    finally { setSaving(false) }
   619→  }
   620→
   621→  return (
   622→    <>
   623→      <Button size="sm" className="bg-emerald-600 hover:bg-emerald-700" onClick={() => setOpen(true)}>
   624→        <Plus className="w-4 h-4 mr-1.5" /> Site Ekle
   625→      </Button>
   626→      <Dialog open={open} onOpenChange={setOpen}>
   627→        <DialogContent className="max-w-md">
   628→          <DialogHeader><DialogTitle>Yeni Site</DialogTitle></DialogHeader>
   629→          <div className="space-y-3">
   630→            <div><Label className="text-xs">Site Adı *</Label><Input value={form.name} onChange={e => setForm({ ...form, name: e.target.value })} /></div>
   631→            <div><Label className="text-xs">Adres</Label><Input value={form.address} onChange={e => setForm({ ...form, address: e.target.value })} /></div>
   632→            <div className="grid grid-cols-2 gap-2">
   633→              <div><Label className="text-xs">Şehir</Label><Input value={form.city} onChange={e => setForm({ ...form, city: e.target.value })} /></div>
   634→              <div><Label className="text-xs">Telefon</Label><Input value={form.phone} onChange={e => setForm({ ...form, phone: e.target.value })} /></div>
   635→            </div>
   636→            <div><Label className="text-xs">Yönetici</Label><Input value={form.managerName} onChange={e => setForm({ ...form, managerName: e.target.value })} /></div>
   637→            <div className="grid grid-cols-2 gap-2">
   638→              <div><Label className="text-xs">Aidat Günü</Label><Input type="number" value={form.dueDay} onChange={e => setForm({ ...form, dueDay: e.target.value })} /></div>
   639→              <div><Label className="text-xs">Aidat Tutarı (₺)</Label><Input type="number" value={form.defaultDueAmount} onChange={e => setForm({ ...form, defaultDueAmount: e.target.value })} /></div>
   640→            </div>
   641→          </div>
   642→          <DialogFooter>
   643→            <Button variant="outline" onClick={() => setOpen(false)}>İptal</Button>
   644→            <Button className="bg-emerald-600 hover:bg-emerald-700" onClick={handleSave} disabled={saving || !form.name}>{saving ? 'Kaydediliyor...' : 'Oluştur'}</Button>
   645→          </DialogFooter>
   646→        </DialogContent>
   647→      </Dialog>
   648→    </>
   649→  )
