   301→// Müşteri görünümü — public booking widget
   302→function PublicBooking({ provider, onBooked }: { provider: Provider; onBooked: () => void }) {
   303→  const [step, setStep] = useState(1)
   304→  const [selectedService, setSelectedService] = useState<Service | null>(null)
   305→  const [selectedStaff, setSelectedStaff] = useState<string>('')
   306→  const [selectedDate, setSelectedDate] = useState(new Date().toISOString().slice(0, 10))
   307→  const [selectedTime, setSelectedTime] = useState('')
   308→  const [customer, setCustomer] = useState({ name: '', phone: '', email: '', note: '' })
   309→  const [booking, setBooking] = useState(false)
   310→
   311→  // Basit slot üretimi (09:00-18:00, 30dk aralık)
   312→  const slots = useMemo(() => {
   313→    const result: string[] = []
   314→    for (let h = 9; h < 18; h++) {
   315→      result.push(`${String(h).padStart(2, '0')}:00`)
   316→      result.push(`${String(h).padStart(2, '0')}:30`)
   317→    }
   318→    return result
   319→  }, [])
   320→
   321→  const handleBook = async () => {
   322→    setBooking(true)
   323→    try {
   324→      const dateTime = new Date(`${selectedDate}T${selectedTime}:00`)
   325→      await apiPost(`/api/appointments/providers/${provider.id}/appointments`, {
   326→        serviceId: selectedService?.id,
   327→        staffId: selectedStaff || undefined,
   328→        customerName: customer.name,
   329→        customerPhone: customer.phone,
   330→        customerEmail: customer.email || undefined,
   331→        customerNote: customer.note || undefined,
   332→        date: dateTime.toISOString(),
   333→        source: 'web',
   334→      })
   335→      toast.success('Randevu oluşturuldu!', { description: `${formatDate(selectedDate)} ${selectedTime}` })
   336→      setStep(5) // success
   337→      onBooked()
   338→    } catch (e) {
   339→      toast.error('Randevu oluşturulamadı', { description: e instanceof Error ? e.message : '' })
   340→    } finally {
   341→      setBooking(false)
   342→    }
   343→  }
   344→
   345→  return (
   346→    <Card className="shadow-soft max-w-2xl mx-auto">
   347→      <CardContent className="p-6">
   348→        {/* Progress */}
   349→        <div className="flex items-center justify-between mb-6">
   350→          {['Hizmet', 'Personel', 'Tarih', 'Bilgiler', 'Tamam'].map((label, i) => (
   351→            <div key={i} className="flex items-center flex-1">
   352→              <div className={cn('w-7 h-7 rounded-full flex items-center justify-center text-xs font-semibold shrink-0',
   353→                step > i + 1 ? 'bg-emerald-500 text-white' : step === i + 1 ? 'bg-emerald-600 text-white' : 'bg-muted text-muted-foreground')}>
   354→                {step > i + 1 ? '✓' : i + 1}
   355→              </div>
   356→              {i < 4 && <div className={cn('flex-1 h-0.5 mx-1', step > i + 1 ? 'bg-emerald-500' : 'bg-muted')} />}
   357→            </div>
   358→          ))}
   359→        </div>
   360→
   361→        {/* Step 1: Service */}
   362→        {step === 1 && (
   363→          <div className="space-y-3">
   364→            <h3 className="font-semibold text-lg">Hizmet Seçin</h3>
   365→            <div className="grid grid-cols-1 md:grid-cols-2 gap-2">
   366→              {provider.services.map((s) => (
   367→                <button key={s.id} onClick={() => { setSelectedService(s); setStep(2) }}
   368→                  className="flex items-center gap-3 p-3 rounded-lg border border-border hover:border-emerald-300 hover:bg-emerald-50/50 dark:hover:bg-emerald-950/20 transition-colors text-left">
   369→                  <div className="w-10 h-10 rounded-lg bg-emerald-50 dark:bg-emerald-950/30 flex items-center justify-center shrink-0">
   370→                    <Scissors className="w-5 h-5 text-emerald-600" />
   371→                  </div>
   372→                  <div className="flex-1 min-w-0">
   373→                    <div className="font-medium text-sm">{s.name}</div>
   374→                    <div className="text-xs text-muted-foreground">{s.duration}dk · {formatCurrency(s.price)}</div>
   375→                  </div>
   376→                </button>
   377→              ))}
   378→            </div>
   379→          </div>
   380→        )}
   381→
   382→        {/* Step 2: Staff */}
   383→        {step === 2 && (
   384→          <div className="space-y-3">
   385→            <h3 className="font-semibold text-lg">Personel Seçin</h3>
   386→            <button onClick={() => { setSelectedStaff(''); setStep(3) }}
   387→              className="w-full flex items-center gap-3 p-3 rounded-lg border border-emerald-300 bg-emerald-50/50 dark:bg-emerald-950/20 transition-colors text-left">
   388→              <div className="w-10 h-10 rounded-full bg-muted flex items-center justify-center"><User className="w-5 h-5" /></div>
   389→              <span className="font-medium text-sm">Herhangi biri</span>
   390→            </button>
   391→            {provider.staff.map((s) => (
   392→              <button key={s.id} onClick={() => { setSelectedStaff(s.id); setStep(3) }}
