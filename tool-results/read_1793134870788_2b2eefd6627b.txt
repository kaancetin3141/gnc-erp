    62→    62→async function upstream(path: string, init?: RequestInit & { timeoutMs?: number }) {
    63→    63→  const timeoutMs = init?.timeoutMs ?? 8000;
    64→    64→  const ctrl = new AbortController();
    65→    65→  const timer = setTimeout(() => ctrl.abort(), timeoutMs);
    66→    66→  try {
    67→    67→    const res = await fetch(UPSTREAM + path, {
    68→    68→      ...init,
    69→    69→      signal: ctrl.signal,
    70→    70→      headers: { accept: 'application/json', ...(init?.headers || {}) },
    71→    71→    });
    72→    72→    const text = await res.text();
    73→    73→    let data: any = null;
    74→    74→    try {
    75→    75→      data = JSON.parse(text);
    76→    76→    } catch {
    77→    77→      data = null;
    78→    78→    }
    79→    79→    return { ok: res.ok, status: res.status, data };
    80→    80→  } catch {
    81→    81→    return { ok: false, status: 502, data: null as any };
    82→    82→  } finally {
    83→    83→    clearTimeout(timer);
    84→    84→  }

   113→   113→function filterBiz(items: Json[], q: string, type: string, city: string): Json[] {
   114→   114→  const nq = trLower(q.trim());
   115→   115→  return items.filter((b) => {
   116→   116→    if (type && b.type !== type) return false;
   117→   117→    if (city && b.city !== city) return false;
   118→   118→    if (nq) {
   119→   119→      const svcNames = (b.services as Json[]).map((s) => trLower(s.name)).join(' ');
   120→   120→      const hay = trLower([b.name, b.address, b.district, b.city].join(' ') + ' ' + svcNames);
   121→   121→      if (!hay.includes(nq)) return false;
   122→   122→    }
   123→   123→    return true;
   124→   124→  });

   141→   141→// Eşleşen hizmet adı _svcMatch alanına yazılır (arayüzde chip olarak gösterilir).
   142→   142→function rankSuggestions(items: Json[], q: string, excludeSlug?: string, limit = 6): Json[] {
   143→   143→  const term = trLower(q).trim().slice(0, 64);
   144→   144→  const pool = excludeSlug ? items.filter((b) => b.slug !== excludeSlug) : items;
   145→   145→  if (!term) return pool.slice(0, limit);
   146→   146→  const scored = pool.map((b) => {
   147→   147→    let score = 0;
   148→   148→    let svcMatch = '';
   149→   149→    const services = Array.isArray(b.services) ? (b.services as Json[]) : [];
   150→   150→    for (const sv of services) {
   151→   151→      const nm = trLower(sv.name);
   152→   152→      const cat = trLower(sv.category);
   153→   153→      if (nm === term) { score = Math.max(score, 100); svcMatch = String(sv.name || ''); break; }
   154→   154→      if (nm.includes(term)) { score = Math.max(score, 60); if (!svcMatch) svcMatch = String(sv.name || ''); }
   155→   155→      else if (cat && (cat === term || cat.includes(term))) { score = Math.max(score, 40); if (!svcMatch) svcMatch = String(sv.category || ''); }
   156→   156→    }
   157→   157→    if (trLower(b.name).includes(term)) score += 30;
   158→   158→    const hay = trLower([b.address, b.district, b.city, b.type].join(' '));
   159→   159→    if (hay.includes(term)) score += 10;
   160→   160→    return { b, score, svcMatch };
   161→   161→  });
   162→   162→  scored.sort((x, y) => y.score - x.score);
   163→   163→  return scored
   164→   164→    .filter((x) => x.score > 0)
   165→   165→    .slice(0, limit)
   166→   166→    .map((x) => (x.svcMatch ? { ...x.b, _svcMatch: x.svcMatch } : x.b));
   167→   167→}
