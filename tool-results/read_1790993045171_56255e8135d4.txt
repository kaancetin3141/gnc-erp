   132→// ─── Helper: format time HH:mm ───────────────────────────────────
   133→function fmtTime(dateStr: string): string {
   134→  const d = new Date(dateStr)
   135→  return `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`
   136→}
   137→
   138→function fmtDateLabel(dateStr: string): string {
   139→  const d = new Date(dateStr)
   140→  const now = new Date()
   141→  const isToday = d.toDateString() === now.toDateString()
   142→  const yest = new Date(now.getTime() - 24 * 60 * 60 * 1000)
   143→  const isYesterday = d.toDateString() === yest.toDateString()
   144→  if (isToday) return 'Bugün'
   145→  if (isYesterday) return 'Dün'
   146→  return d.toLocaleDateString('tr-TR', { day: 'numeric', month: 'long' })
   147→}

   150→// Format: "📎 Fatura FAT-2026-001 — Müşteri Adı — 1.234,56 TRY"
   151→function parseAttachment(content: string): { type: ChatDocumentType; number: string; rest: string } | null {
   152→  const match = /^📎\s*(Fatura|Sipariş|Proforma|Teklif)\s+([A-Za-z0-9çğıöşüÇĞİÖŞÜ._-]+)\s*—\s*([\s\S]*)$/.exec(content)
   153→  if (!match) return null
   154→  return { type: match[1] as ChatDocumentType, number: match[2], rest: match[3].trim() }
   155→}

   427→// ─── Belge seçici diyaloğu (gerçek kayıtlar, aramalı) ────────────
   428→function DocumentPickerDialog({
   429→  open,
   430→  onOpenChange,
   431→  initialType,
   432→  onPick,
   433→}: {
   434→  open: boolean
   435→  onOpenChange: (open: boolean) => void
   436→  initialType: ChatDocumentType
   437→  onPick: (doc: ChatDocument) => void
   438→}) {
   439→  const [typeFilter, setTypeFilter] = useState<ChatDocumentType | 'Tümü'>(initialType)
   440→  const [search, setSearch] = useState('')
   441→
   442→  useEffect(() => {
   443→    if (open) {
   444→      setTypeFilter(initialType)
   445→      setSearch('')
   446→    }
   447→  }, [open, initialType])
   448→
   449→  const { data, isLoading } = useQuery({
   450→    queryKey: ['chat-documents'],
   451→    queryFn: () => apiGet<DocumentsResponse>('/api/messages?type=documents'),
   452→    enabled: open,
   453→    staleTime: 30_000,
   454→  })
   455→
   456→  const filtered = useMemo(() => {
   457→    let list = data?.items ?? []
   458→    if (typeFilter !== 'Tümü') list = list.filter((d) => d.type === typeFilter)
   459→    if (search.trim()) {
   460→      const q = search.toLowerCase().trim()
   461→      list = list.filter(
   462→        (d) =>
   463→          d.number.toLowerCase().includes(q) ||
   464→          d.customerName.toLowerCase().includes(q),
   465→      )
   466→    }
   467→    return list
   468→  }, [data, typeFilter, search])
   469→
   470→  const types: (ChatDocumentType | 'Tümü')[] = ['Tümü', 'Fatura', 'Sipariş', 'Proforma', 'Teklif']
   471→
   472→  return (
   473→    <Dialog open={open} onOpenChange={onOpenChange}>
   474→      <DialogContent className="sm:max-w-md">
   475→        <DialogHeader>
   476→          <DialogTitle>Belge Gönder</DialogTitle>
   477→          <DialogDescription>
   478→            Gerçek bir belge seç — numarası mesaja ek olarak gider.
   479→          </DialogDescription>
   480→        </DialogHeader>
   481→
   482→        <div className="flex flex-wrap gap-1.5">
   483→          {types.map((t) => (
   484→            <button
   485→              key={t}
   486→              type="button"
   487→              onClick={() => setTypeFilter(t)}
   488→              className={cn(
   489→                'rounded-full border px-2.5 py-1 text-xs font-medium transition-colors',
   490→                typeFilter === t
   491→                  ? 'border-emerald-600 bg-emerald-600 text-white'
   492→                  : 'border-border text-muted-foreground hover:bg-muted',
   493→              )}
   494→            >
   495→              {t}
   496→            </button>
   497→          ))}
   498→        </div>
   499→
   500→        <div className="relative">
   501→          <Search className="w-4 h-4 absolute left-2.5 top-1/2 -translate-y-1/2 text-muted-foreground" />
   502→          <Input
   503→            value={search}
   504→            onChange={(e) => setSearch(e.target.value)}
   505→            placeholder="Numara veya müşteri ara…"
   506→            className="pl-9 h-9 text-sm"
   507→          />
   508→        </div>
   509→
   510→        <div className="max-h-80 overflow-y-auto custom-scroll space-y-1.5 -mx-1 px-1">
   511→          {isLoading ? (
   512→            <div className="py-8 flex items-center justify-center gap-2 text-xs text-muted-foreground">
   513→              <Loader2 className="w-4 h-4 animate-spin" />
   514→              Belgeler yükleniyor…
   515→            </div>
   516→          ) : filtered.length === 0 ? (
   517→            <div className="py-8 text-center text-xs text-muted-foreground">
   518→              Bu kriterde belge bulunamadı.
   519→            </div>
   520→          ) : (
   521→            filtered.map((doc) => {
   522→              const Icon = DOC_TYPE_ICON[doc.type] ?? FileText
   523→              return (
   524→                <button
   525→                  key={`${doc.type}-${doc.id}`}
   526→                  type="button"
   527→                  onClick={() => onPick(doc)}
   528→                  className="w-full flex items-center gap-3 rounded-lg border p-2.5 text-left hover:bg-muted/60 transition-colors"
   529→                >
   530→                  <span className="w-8 h-8 rounded-md bg-emerald-50 dark:bg-emerald-950/40 text-emerald-600 flex items-center justify-center shrink-0">
   531→                    <Icon className="w-4 h-4" />
   532→                  </span>
   533→                  <span className="flex-1 min-w-0">
   534→                    <span className="block text-sm font-medium truncate">{doc.number}</span>
   535→                    <span className="block text-xs text-muted-foreground truncate">
   536→                      {doc.customerName} · {formatDate(doc.createdAt)}
   537→                    </span>
   538→                  </span>
   539→                  <span className="text-xs font-semibold tabular-nums shrink-0">
   540→                    {formatCurrency(doc.total, doc.currency)}
   541→                  </span>
   542→                </button>
   543→              )
   544→            })
   545→          )}
   546→        </div>
   547→      </DialogContent>
   548→    </Dialog>
   549→  )

   552→// ─── Conversation panel ──────────────────────────────────────────
   553→function ConversationPanel({
   554→  otherUser,
   555→  messages,
   556→  loading,
   557→  currentUserId,
   558→  onlineUserIds,
   559→  isTyping,
   560→  sendTyping,
   561→  sendReadReceipt,
   562→  onBack,
   563→}: {
   564→  otherUser: UsersResponse['items'][0] | null
   565→  messages: ChatMessage[]
   566→  loading: boolean
   567→  currentUserId: string
   568→  onlineUserIds: ReadonlySet<string>
   569→  isTyping: boolean
   570→  sendTyping: (toUserId: string) => void
   571→  sendReadReceipt: (senderId: string) => void
   572→  onBack: () => void
   573→}) {
   574→  const qc = useQueryClient()
   575→  const [input, setInput] = useState('')
   576→  const [docPickerOpen, setDocPickerOpen] = useState(false)
   577→  const [docPickerType, setDocPickerType] = useState<ChatDocumentType>('Fatura')
   578→
   579→  // ── Akıllı kaydırma state'leri ──
   580→  const scrollRef = useRef<HTMLDivElement>(null)
   581→  const atBottomRef = useRef(true) // kullanıcı altta mı? (alttan <120px)
   582→  const forceScrollRef = useRef(false) // kendi mesajını gönderdi → her zaman en alta
   583→  const [showJumpPill, setShowJumpPill] = useState(false)
   584→  const [newBelowCount, setNewBelowCount] = useState(0)
   585→
   586→  // En alta kaydır — çift requestAnimationFrame ile (DOM paint'ten sonra)
   587→  const scrollToBottom = useCallback((behavior: ScrollBehavior = 'auto') => {
   588→    requestAnimationFrame(() => {
   589→      requestAnimationFrame(() => {
   590→        const el = scrollRef.current
   591→        if (!el) return
   592→        el.scrollTo({ top: el.scrollHeight, behavior })
   593→        atBottomRef.current = true
   594→        setShowJumpPill(false)
   595→        setNewBelowCount(0)
   596→      })
   597→    })
   598→  }, [])
   599→
   600→  // Kullanıcı kaydırınca pozisyonu izle (alttan >120px → otomatik kaydırma durur)
   601→  const handleScroll = useCallback(() => {
   602→    const el = scrollRef.current
   603→    if (!el) return
   604→    const distance = el.scrollHeight - el.scrollTop - el.clientHeight
   605→    const atBottom = distance < 120
   606→    atBottomRef.current = atBottom
   607→    if (atBottom) {
   608→      setShowJumpPill(false)
   609→      setNewBelowCount(0)
   610→    } else {
   611→      setShowJumpPill(true)
   612→    }
   613→  }, [])
   614→
   615→  // Konuşma değişince (otherUser.id) anında en alta + göstergeleri sıfırla
   616→  useEffect(() => {
   617→    atBottomRef.current = true
   618→    setShowJumpPill(false)
   619→    setNewBelowCount(0)
   620→    scrollToBottom('auto')
   621→  }, [otherUser?.id, scrollToBottom])
   622→
   623→  // Mesaj listesi değiştiğinde (array identity + son mesaj id) kaydır:
   624→  //  - kendi gönderdiğimiz mesaj → HER ZAMAN en alta
   625→  //  - zaten alttaysak → sessizce takip et
   626→  //  - yukarıda okuyorsak → KAYDIRMA, "yeni mesaj" sayacını artır
   627→  const lastMessageId = messages.length > 0 ? messages[messages.length - 1].id : ''
   628→  useEffect(() => {
   629→    if (messages.length === 0) return
   630→    if (forceScrollRef.current) {
   631→      forceScrollRef.current = false
   632→      scrollToBottom('smooth')
   633→      return
   634→    }
   635→    const last = messages[messages.length - 1]
   636→    if (atBottomRef.current) {
   637→      scrollToBottom('auto')
   638→    } else if (last.senderId !== currentUserId) {
   639→      setNewBelowCount((c) => c + 1)
   640→    }
   641→  }, [messages, lastMessageId, currentUserId, scrollToBottom])
   642→
   643→  const sendMut = useMutation({
   644→    mutationFn: (body: {
   645→      receiverId: string
   646→      content: string
   647→      attachmentType?: string
   648→      attachmentId?: string
   649→      attachmentName?: string
   650→    }) => apiPost<ChatMessage>('/api/messages', body),
   651→    onSuccess: (msg) => {
   652→      setInput('')
   653→      // Optimistic enjeksiyon (socket event'i id ile tekilleştirir)
   654→      if (otherUser?.id) {
   655→        const conv = qc.getQueryData<MessagesResponse>(['conversation', otherUser.id])
   656→        if (conv && !conv.items.some((m) => m.id === msg.id)) {
   657→          qc.setQueryData<MessagesResponse>(['conversation', otherUser.id], {
   658→            ...conv,
   659→            items: [...conv.items, msg],
   660→            total: conv.total + 1,
   661→          })
   662→        }
   663→        const all = qc.getQueryData<MessagesResponse>(['all-messages'])
   664→        if (all && !all.items.some((m) => m.id === msg.id)) {
   665→          qc.setQueryData<MessagesResponse>(['all-messages'], {
   666→            ...all,
   667→            items: [...all.items, msg],
   668→            total: all.total + 1,
   669→          })
   670→        }
   671→      }
   672→      qc.invalidateQueries({ queryKey: ['all-messages'] })
   673→      qc.invalidateQueries({ queryKey: ['unread-messages'] })
   674→    },
   675→    onError: (e: ApiError) => {
   676→      forceScrollRef.current = false
   677→      toast.error('Mesaj gönderilemedi', { description: e.message })
   678→    },
   679→  })
   680→
   681→  const deleteMut = useMutation({
   682→    mutationFn: (id: string) => apiDelete<{ success: boolean }>(`/api/messages/${id}`),
   683→    onSuccess: () => {
   684→      qc.invalidateQueries({ queryKey: ['all-messages'] })
   685→      qc.invalidateQueries({ queryKey: ['conversation', otherUser?.id] })
   686→      qc.invalidateQueries({ queryKey: ['unread-messages'] })
   687→    },
   688→    onError: (e: ApiError) => {
   689→      toast.error('Mesaj silinemedi', { description: e.message })
   690→    },
   691→  })
   692→
   693→  const handleSend = () => {
   694→    if (!input.trim() || !otherUser) return
   695→    // Mesaj gönderince HER ZAMAN en alta (kullanıcı yukarıdayken bile)
   696→    forceScrollRef.current = true
   697→    scrollToBottom('smooth')
   698→    sendMut.mutate({ receiverId: otherUser.id, content: input.trim() })
   699→  }
   700→
   701→  // Gerçek belge gönder (fatura/sipariş/proforma/teklif — gerçek kayıtlar)
   702→  const handleSendDocument = (doc: ChatDocument) => {
   703→    if (!otherUser) return
   704→    forceScrollRef.current = true
   705→    scrollToBottom('smooth')
   706→    sendMut.mutate({
   707→      receiverId: otherUser.id,
   708→      content: `📎 ${doc.type} ${doc.number} — ${doc.customerName} — ${formatCurrency(doc.total, doc.currency)}`,
   709→      attachmentType: doc.type,
   710→      attachmentId: doc.id,
   711→      attachmentName: doc.number,
   712→    })
   713→    toast.success(`${doc.type} gönderildi`, {
   714→      description: `${doc.number} · ${doc.customerName}`,
   715→    })
   716→    setDocPickerOpen(false)
   717→  }
   718→
   719→  const handleKeyDown = (e: React.KeyboardEvent) => {
   720→    if (e.key === 'Enter' && !e.shiftKey) {
   721→      e.preventDefault()
   722→      handleSend()
   723→    }
   724→  }
   725→
   726→  const handleDelete = (id: string) => {
   727→    deleteMut.mutate(id)
   728→  }
   729→
   730→  const copyToClipboard = (text: string) => {
   731→    navigator.clipboard?.writeText(text).then(
   732→      () => toast.success('Kopyalandı', { description: text }),
   733→      () => toast.error('Kopyalanamadı'),
   734→    )
   735→  }
   736→
   737→  // ── Görüldü: gelen okunmamışları işaretle (PATCH = DB doğruluğu, socket = anında sinyal)
   738→  const messagesRef = useRef(messages)
   739→  useEffect(() => {
   740→    messagesRef.current = messages
   741→  }, [messages])
   742→  const unreadCount = useMemo(
   743→    () => messages.reduce((n, m) => n + (m.receiverId === currentUserId && !m.isRead ? 1 : 0), 0),
   744→    [messages, currentUserId],
   745→  )
   746→  const otherId = otherUser?.id ?? null
   747→  useEffect(() => {
   748→    if (unreadCount === 0 || !otherId) return
   749→    const mark = async () => {
   750→      sendReadReceipt(otherId) // anında görüldü sinyali (socket)
   751→      const targets = messagesRef.current.filter(
   752→        (m) => m.receiverId === currentUserId && !m.isRead,
   753→      )
   754→      await Promise.all(
   755→        targets.map((m) => apiPatch(`/api/messages/${m.id}`, {}).catch(() => null)),
   756→      )
   757→      qc.invalidateQueries({ queryKey: ['all-messages'] })
   758→      qc.invalidateQueries({ queryKey: ['conversation', otherId] })
   759→      qc.invalidateQueries({ queryKey: ['unread-messages'] })
   760→    }
   761→    const t = setTimeout(mark, 400)
   762→    return () => clearTimeout(t)
   763→  }, [unreadCount, otherId, currentUserId, qc, sendReadReceipt])
   764→
   765→  if (!otherUser) {
   766→    return <EmptyConversation />

   769→  // Group messages by date
   770→  const groups: { label: string; messages: ChatMessage[] }[] = []
   771→  let currentLabel = ''

   773→    const label = fmtDateLabel(m.createdAt)
   774→    if (label !== currentLabel) {
   775→      groups.push({ label, messages: [] })
   776→      currentLabel = label
   777→    }
   778→    groups[groups.length - 1].messages.push(m)
   779→  }
   780→
   781→  const online = onlineUserIds.has(otherUser.id)
   782→
   783→  return (
   784→    <>
   785→      {/* Header */}
   786→      <div className="flex items-center gap-3 p-3 border-b bg-background">
   787→        <Button
   788→          variant="ghost"
   789→          size="icon"
   790→          className="md:hidden h-8 w-8"
   791→          onClick={onBack}
   792→        >
   793→          <ArrowLeft className="w-4 h-4" />
   794→        </Button>
   795→        <div className="relative">
   796→          <Avatar className="w-9 h-9">
   797→            {otherUser.avatarUrl ? <AvatarImage src={otherUser.avatarUrl} alt={otherUser.name} /> : null}
   798→            <AvatarFallback
   799→              className={cn(
   800→                'bg-gradient-to-br text-white text-xs font-semibold',
   801→                ROLE_AVATAR_GRADIENT[otherUser.role],
   802→              )}
   803→            >
   804→              {initials(otherUser.name)}
   805→            </AvatarFallback>
   806→          </Avatar>
   807→          {online && (
   808→            <span className="absolute -bottom-0.5 -right-0.5 w-2.5 h-2.5 rounded-full bg-emerald-500 border-2 border-background" />
   809→          )}
   810→        </div>
   811→        <div className="flex-1 min-w-0">
   812→          <div className="text-sm font-semibold truncate">{otherUser.name}</div>
   813→          <div className="text-[11px] text-muted-foreground flex items-center gap-1.5">
   814→            {otherUser.title && <span className="truncate">{otherUser.title}</span>}

   823→        <div className="flex items-center gap-1 px-4 py-1.5 border-b bg-emerald-50/60 dark:bg-emerald-950/20 text-[11px] text-emerald-700 dark:text-emerald-300">
   824→          <span className="typing-dot" />
   825→          <span className="typing-dot [animation-delay:150ms]" />
   826→          <span className="typing-dot [animation-delay:300ms]" />
   827→          <span className="ml-1">{otherUser.name} yazıyor…</span>
   828→        </div>
   829→      )}
   830→
   831→      {/* Messages */}
   832→      <div
   833→        ref={scrollRef}
   834→        onScroll={handleScroll}
   835→        className="flex-1 overflow-y-auto custom-scroll p-4 space-y-4 bg-muted/10"
   836→      >
   837→        {loading ? (
   838→          <div className="flex items-center justify-center h-full text-xs text-muted-foreground">
   839→            Mesajlar yükleniyor…
   840→          </div>
   841→        ) : messages.length === 0 ? (
   842→          <div className="flex flex-col items-center justify-center h-full text-center">
   843→            <div className="w-12 h-12 rounded-full bg-emerald-50 dark:bg-emerald-950/30 flex items-center justify-center mb-3">
   844→              <MessageCircle className="w-6 h-6 text-emerald-600" />
   845→            </div>
   846→            <p className="text-sm text-muted-foreground max-w-xs">
   847→              {otherUser.name} ile henüz mesajlaşmadın. İlk mesajı gönder!
   848→            </p>
   849→          </div>
   850→        ) : (
   851→          groups.map((g) => (
   852→            <div key={g.label} className="space-y-2">
   853→              <div className="flex items-center justify-center">
   854→                <span className="text-[10px] text-muted-foreground bg-background px-2 py-0.5 rounded-full border">
   855→                  {g.label}
   856→                </span>
   857→              </div>
   858→              {g.messages.map((m) => {
   859→                const mine = m.senderId === currentUserId
   860→                const att = parseAttachment(m.content)
   861→                const AttIcon = att ? DOC_TYPE_ICON[att.type] : null
   862→                return (
   863→                  <div
   864→                    key={m.id}
   865→                    className={cn(
   866→                      'flex items-end gap-2 group',
   867→                      mine ? 'justify-end' : 'justify-start',
   868→                    )}
   869→                  >
   870→                    {!mine && (
   871→                      <Avatar className="w-7 h-7 shrink-0">
   872→                        {otherUser.avatarUrl ? <AvatarImage src={otherUser.avatarUrl} alt={otherUser.name} /> : null}
   873→                        <AvatarFallback
   874→                          className={cn(
   875→                            'bg-gradient-to-br text-white text-[10px] font-semibold',
   876→                            ROLE_AVATAR_GRADIENT[otherUser.role],
   877→                          )}
   878→                        >
   879→                          {initials(otherUser.name)}
   880→                        </AvatarFallback>
   881→                      </Avatar>
   882→                    )}
   883→                    <div
   884→                      className={cn(
   885→                        'max-w-[75%] rounded-2xl px-3.5 py-2 text-sm relative space-y-1.5',
   886→                        mine
   887→                          ? 'bg-emerald-600 text-white rounded-br-sm'
   888→                          : 'bg-background border rounded-bl-sm',
   889→                      )}
   890→                    >
   891→                      {att && AttIcon ? (
   892→                        // Belge eki chip'i — tıklayınca numara panoya kopyalanır
   893→                        <button
   894→                          type="button"
   895→                          onClick={() => copyToClipboard(att.number)}
   896→                          title="Numarayı panoya kopyala"
   897→                          className={cn(
   898→                            'flex items-center gap-1.5 rounded-lg px-2 py-1 text-xs font-medium transition-colors w-fit',
   899→                            mine
   900→                              ? 'bg-white/15 hover:bg-white/25 text-white'
   901→                              : 'bg-muted hover:bg-accent text-foreground',
   902→                          )}
   903→                        >
   904→                          <AttIcon className="w-3.5 h-3.5 shrink-0" />
   905→                          <span>{att.type} · {att.number}</span>
   906→                          <Copy className="w-3 h-3 opacity-60 shrink-0" />
   907→                        </button>
   908→                      ) : null}

   914→                          'flex items-center gap-1 mt-1 text-[10px]',

   921→                            {m.isRead && <span className="text-[9px]">Görüldü</span>}
   922→                            <CheckCheck
   923→                              className={cn('w-3 h-3', m.isRead ? 'text-emerald-100' : 'text-emerald-300/70')}
   924→                            />
   925→                          </span>
   926→                        )}
   927→                      </div>
   928→                      {mine && (
   929→                        <Tooltip>
   930→                          <TooltipTrigger asChild>
   931→                            <button
   932→                              onClick={() => handleDelete(m.id)}
   933→                              className="absolute -top-2 -right-2 w-5 h-5 rounded-full bg-background border shadow-sm flex items-center justify-center opacity-0 group-hover:opacity-100 transition-opacity hover:bg-red-50 hover:text-red-600"
   934→                            >
   935→                              <Trash2 className="w-2.5 h-2.5" />
   936→                            </button>
   937→                          </TooltipTrigger>
   938→                          <TooltipContent>Mesajı sil</TooltipContent>
   939→                        </Tooltip>
   940→                      )}
   941→                    </div>
   942→                  </div>
   943→                )
   944→              })}
   945→            </div>
   946→          ))
   947→        )}
   948→      </div>
   949→
   950→      {/* "↓ Yeni mesaj" pill — kullanıcı yukarı okurken görünür */}
   951→      {showJumpPill && (
   952→        <button
   953→          type="button"
   954→          onClick={() => scrollToBottom('smooth')}
   955→          className="absolute bottom-24 right-5 z-10 flex items-center gap-1.5 rounded-full bg-emerald-600 px-3.5 py-2 text-xs font-medium text-white shadow-lg shadow-emerald-600/25 hover:bg-emerald-700 transition-colors animate-fade-in"
   956→        >
   957→          <ArrowDown className="w-3.5 h-3.5" />
   958→          {newBelowCount > 0 ? 'Yeni mesaj' : 'En alta git'}
   959→          {newBelowCount > 0 && (
   960→            <span className="unread-pulse flex h-5 min-w-5 items-center justify-center rounded-full bg-white px-1.5 text-[10px] font-bold text-emerald-700">
   961→              {newBelowCount}
   962→            </span>
   963→          )}
   964→        </button>
   965→      )}
   966→
   967→      {/* Input */}
   968→      <div className="p-3 border-t bg-background">
   969→        <div className="flex items-center gap-2">
   970→          <DropdownMenu>
   971→            <DropdownMenuTrigger asChild>
   972→              <Button variant="ghost" size="icon" className="shrink-0 h-9 w-9" title="Belge gönder">
   973→                <Paperclip className="w-4 h-4" />
   974→              </Button>
   975→            </DropdownMenuTrigger>
   976→            <DropdownMenuContent align="start">
   977→              <DropdownMenuLabel>Belge Gönder</DropdownMenuLabel>
   978→              <DropdownMenuSeparator />
   979→              <DropdownMenuItem onClick={() => { setDocPickerType('Fatura'); setDocPickerOpen(true) }}>
   980→                🧾 Fatura
   981→              </DropdownMenuItem>
   982→              <DropdownMenuItem onClick={() => { setDocPickerType('Sipariş'); setDocPickerOpen(true) }}>
   983→                📦 Sipariş
   984→              </DropdownMenuItem>
   985→              <DropdownMenuItem onClick={() => { setDocPickerType('Proforma'); setDocPickerOpen(true) }}>
   986→                📄 Proforma
   987→              </DropdownMenuItem>
   988→              <DropdownMenuItem onClick={() => { setDocPickerType('Teklif'); setDocPickerOpen(true) }}>
   989→                📋 Teklif
   990→              </DropdownMenuItem>
   991→            </DropdownMenuContent>
   992→          </DropdownMenu>
   993→          <Input
   994→            value={input}
   995→            onChange={(e) => {
   996→              setInput(e.target.value)
   997→              if (otherUser) sendTyping(otherUser.id)
   998→            }}
   999→            onKeyDown={handleKeyDown}
  1000→            placeholder={`${otherUser.name} kişisine mesaj yaz…`}
  1001→            disabled={sendMut.isPending}
  1002→            className="flex-1"
  1003→          />
  1004→          <Button
  1005→            onClick={handleSend}
  1006→            disabled={!input.trim() || sendMut.isPending}
  1007→            size="icon"
  1008→            className="bg-emerald-600 hover:bg-emerald-700 text-white shrink-0"
  1009→          >
  1010→            <Send className="w-4 h-4" />
  1011→          </Button>
  1012→        </div>
  1013→        <div className="text-[10px] text-muted-foreground mt-1.5 text-center">
  1014→          Enter ile gönder · Shift+Enter ile yeni satır
  1015→        </div>
  1016→      </div>
  1017→
  1018→      {/* Gerçek belge seçici */}
  1019→      <DocumentPickerDialog
  1020→        open={docPickerOpen}
  1021→        onOpenChange={setDocPickerOpen}
  1022→        initialType={docPickerType}
  1023→        onPick={handleSendDocument}
  1024→      />
  1025→    </>
  1026→  )
  1027→}
