   152→function TreePanel({

   179→        </div>
   180→        <CollapsibleContent>

   182→            <TreeRow
   183→              active={selection.kind === 'all'}
   184→              onClick={() => onSelect({ kind: 'all' })}
   185→              icon={<BarChart3 className="w-4 h-4" />}
   186→              label="Tüm Müşteriler"
   187→              count={totalCustomers}
   188→              tone="text-slate-600 dark:text-slate-300"
   189→              dotClass="bg-slate-400"

   192→              const Icon = t.icon
   193→              const count = overview?.customerByType[t.value] ?? 0
   194→              return (
   195→                <TreeRow
   196→                  key={t.value}
   197→                  active={selection.kind === 'customerType' && selection.value === t.value}
   198→                  onClick={() => onSelect({ kind: 'customerType', value: t.value })}

   201→                  count={count}
   202→                  tone={t.color.split(' ')[0]}
   203→                  dotClass={t.dot}
