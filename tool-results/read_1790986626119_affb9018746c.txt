     1→'use client'

     4→import { useQuery, useQueryClient } from '@tanstack/react-query'
     5→import { apiGet, apiPatch, apiDelete, apiPost } from '@/lib/api-client'
     6→
     7→import { Card, CardContent } from '@/components/ui/card'
     8→import { Button } from '@/components/ui/button'
     9→import { Skeleton } from '@/components/ui/skeleton'
    10→import { Badge } from '@/components/ui/badge'
    11→import { Separator } from '@/components/ui/separator'
    12→import {
    13→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    14→} from '@/components/ui/table'
    15→import {
    16→  Dialog, DialogContent, DialogDescription, DialogHeader,
    17→  DialogTitle,
    18→} from '@/components/ui/dialog'
    19→import {
    20→  AlertDialog, AlertDialogAction, AlertDialogCancel,
    21→  AlertDialogContent, AlertDialogDescription, AlertDialogFooter,
    22→  AlertDialogHeader, AlertDialogTitle,
    23→} from '@/components/ui/alert-dialog'
    24→import {
    25→  Tooltip, TooltipContent, TooltipTrigger,
    26→} from '@/components/ui/tooltip'
    27→import { toast } from 'sonner'
    28→import {
    29→  Receipt, Pencil, Trash2, RefreshCw, User, Calendar,
    30→  Clock, CheckCircle2, AlertTriangle, TrendingUp, FileText,
    31→  Printer, Plus, CircleCheckBig, Undo2, MessageCircle, CalendarClock,
    32→  HandCoins, Loader2,
    33→} from 'lucide-react'
    34→import { formatCurrency, formatDate } from '@/lib/format'
    35→import { sendWhatsAppTracked } from '@/lib/whatsapp-hub'
    36→import { cn } from '@/lib/utils'
    37→import { formatWeight, calculateTotalWeight } from '@/lib/weight-utils'
    38→import {
    39→  useInvoiceTemplate, TemplateA4Page, PdfHeader, PdfFooter, DocVerifyQr,
    40→} from '@/components/pdf/pdf-template'
    41→import type { Invoice, InvoiceLine } from './types'
    42→import {
    43→  INVOICE_STATUSES, getInvoiceStatusMeta,
    44→  buildInvoiceWhatsAppMessage, overdueDays,
    45→} from './invoice-utils'
    46→
    47→// ============================================================
    48→// Fatura Detay Dialog
    49→// ============================================================
    50→
    51→export function InvoiceDetailDialog({
    52→  invoice, open, onOpenChange, onEdit,
    53→}: {
    54→  invoice: Invoice | null
    55→  open: boolean
    56→  onOpenChange: (v: boolean) => void
    57→  onEdit: (inv: Invoice) => void
    58→}) {
    59→  const qc = useQueryClient()
    60→  const [deleteOpen, setDeleteOpen] = useState(false)
    61→  const [deleting, setDeleting] = useState(false)
    62→  const [changing, setChanging] = useState(false)
    63→  const [pdfOpen, setPdfOpen] = useState(false)
    64→  const [quickBusy, setQuickBusy] = useState<string | null>(null)
    65→
    66→  // Detay sorgu
    67→  const { data: detail, isLoading } = useQuery({
    68→    queryKey: ['invoice', invoice?.id],
    69→    queryFn: () => apiGet<Invoice>(`/api/invoices/${invoice!.id}`),
    70→    enabled: !!invoice && open,
    71→  })
    72→
    73→  if (!invoice) return null
    74→
    75→  const d = detail ?? invoice
    76→  const status = getInvoiceStatusMeta(d.status)
    77→
    78→  const changeStatus = async (newStatus: string) => {
    79→    setChanging(true)
    80→    try {
    81→      await apiPatch(`/api/invoices/${invoice.id}`, { status: newStatus })
    82→      toast.success('Durum güncellendi')
    83→      qc.invalidateQueries({ queryKey: ['invoice', invoice.id] })
    84→      qc.invalidateQueries({ queryKey: ['invoices'] })
    85→    } catch (e) {
    86→      toast.error(e instanceof Error ? e.message : 'Güncelleme başarısız')
    87→    } finally {
    88→      setChanging(false)
    89→    }
    90→  }
    91→
    92→  const handleDelete = async () => {
    93→    setDeleting(true)
    94→    try {
    95→      await apiDelete(`/api/invoices/${invoice.id}`)
    96→      toast.success('Fatura silindi')
    97→      qc.invalidateQueries({ queryKey: ['invoices'] })
    98→      setDeleteOpen(false)
    99→      onOpenChange(false)
   100→    } catch (e) {
   101→      toast.error(e instanceof Error ? e.message : 'Silme başarısız')
   102→    } finally {
   103→      setDeleting(false)
   104→    }
   105→  }
   106→
   107→  // Gecikmiş mi? — canlı detay (d) üzerinden okunur; hızlı aksiyonlar
   108→  // sonrası prop güncellenmeden bile arayüz doğru duruma geçer
   109→  const isOverdue = d.status === 'odeme_bekliyor' && d.dueDate
   110→    ? new Date(d.dueDate).getTime() < Date.now()
   111→    : false
   112→
   113→  // ----- Hızlı tahsilat aksiyonları (liste menüsündeki aksiyonların detaya taşınması) -----
   114→  const isPending = d.status === 'odeme_bekliyor' || d.status === 'gecikti'
   115→  const quickActionInvalid = (e: unknown, fallback: string) => {
   116→    toast.error(e instanceof Error ? e.message : fallback)
   117→  }
   118→
   119→  // Ödendi işaretle / geri al
   120→  const handleQuickStatus = async (status: 'odendi' | 'odeme_bekliyor') => {
   121→    setQuickBusy(status)
   122→    try {
   123→      await apiPatch(`/api/invoices/${invoice.id}`, { status })
   124→      qc.invalidateQueries({ queryKey: ['invoice', invoice.id] })
   125→      qc.invalidateQueries({ queryKey: ['invoices'] })
   126→      toast.success(
   127→        status === 'odendi'
   128→          ? `${invoice.number} ödendi olarak işaretlendi 🎉`
   129→          : `${invoice.number} ödeme bekliyor durumuna alındı`,
   130→      )
   131→    } catch (e) {
   132→      quickActionInvalid(e, 'İşlem başarısız')
   133→    } finally {
   134→      setQuickBusy(null)
   135→    }
   136→  }
   137→
   138→  // Vade uzat +7 gün
   139→  const handleExtendDue = async () => {
   140→    setQuickBusy('extend')
   141→    try {
   142→      const base = invoice.dueDate && !isNaN(new Date(invoice.dueDate).getTime())
   143→        ? new Date(invoice.dueDate)
   144→        : new Date()
   145→      base.setDate(base.getDate() + 7)
   146→      await apiPatch(`/api/invoices/${invoice.id}`, { dueDate: base.toISOString() })
   147→      qc.invalidateQueries({ queryKey: ['invoice', invoice.id] })
   148→      qc.invalidateQueries({ queryKey: ['invoices'] })
   149→      toast.success(`${invoice.number} vadesi ${formatDate(base.toISOString())} olarak uzatıldı`)
   150→    } catch (e) {
   151→      quickActionInvalid(e, 'Vade uzatılamadı')
   152→    } finally {
   153→      setQuickBusy(null)
   154→    }
   155→  }
   156→
   157→  // Ödeme hatırlatma (WhatsApp) — detaydaki müşteri telefonuyla wa.me linki + aktivite kaydı
   158→  const handleRemind = async () => {
   159→    setQuickBusy('remind')
   160→    try {
   161→      const phone = d.customer?.phone ?? null
   162→      if (!phone) {
   163→        toast.error(`${d.customer?.name ?? 'Müşteri'} için telefon numarası kayıtlı değil`)
   164→        return
   165→      }
   166→      const overdue = isOverdue || d.status === 'gecikti'
   167→      const msg = buildInvoiceWhatsAppMessage({
   168→        number: invoice.number,
   169→        customerName: d.customer?.name ?? '',
   170→        total: invoice.total,
   171→        currency: invoice.currency,
   172→        dueDate: invoice.dueDate,
   173→        isOverdue: overdue,
   174→      })
   175→      // WhatsApp Mesaj Merkezi kaydı + wa.me sekmesi + gönderildi işaretleme
   176→      const result = await sendWhatsAppTracked({
   177→        phone,
   178→        body: msg,
   179→        title: `Ödeme Hatırlatma — ${invoice.number}`,
   180→        contextType: 'fatura_hatirlatma',
   181→        contextId: invoice.id,
   182→        contextNo: invoice.number,
   183→        customerId: d.customer?.id ?? invoice.customerId,
   184→        customerName: d.customer?.name ?? null,
   185→        amount: invoice.total,
   186→        currency: invoice.currency,
   187→      })
   188→      toast.success(result.popupOpened
   189→        ? `${invoice.number} için hatırlatma mesajı hazırlandı`
   190→        : `${invoice.number} hatırlatması kuyruğa alındı — Mesaj Merkezi'nden gönderebilirsiniz`)
   191→
   192→      // Müşteri 360 zaman tüneliğine kaydet
   193→      const customerId = d.customer?.id ?? invoice.customerId
   194→      if (customerId) {
   195→        try {
   196→          await apiPost('/api/customers/' + customerId + '/activities', {
   197→            type: 'whatsapp',
   198→            subject: `Ödeme hatırlatması gönderildi: ${invoice.number}`,
   199→            detail: `${invoice.total.toLocaleString('tr-TR')} ${invoice.currency} · Vade: ${invoice.dueDate ? formatDate(invoice.dueDate) : '—'}`,
   200→            outcome: overdue ? null : 'basarili',
   201→          })
   202→        } catch { /* aktivite kaydı akışı etkilemez */ }
   203→      }
   204→    } catch (e) {
   205→      quickActionInvalid(e, 'Hatırlatma hazırlanamadı')
   206→    } finally {
   207→      setQuickBusy(null)
   208→    }
   209→  }
   210→
   211→  const lines: InvoiceLine[] = d.lines ?? []
   212→  // Ağırlık kontrolü (F4)
   213→  const hasWeight = lines.some((l) => l.totalWeight != null && l.totalWeight > 0)
   214→  const totalNetWeight = hasWeight
   215→    ? calculateTotalWeight(lines.map((l) => ({ qty: l.qty, weightPerUnit: l.weightPerUnit ?? null })))
   216→    : null
   217→
   218→  return (
   219→    <>
   220→      <Dialog open={open} onOpenChange={onOpenChange}>
   221→        <DialogContent className="sm:max-w-2xl max-h-[92vh] overflow-y-auto custom-scroll">
   222→          <DialogHeader className="print:hidden">
   223→            <div className="flex items-start justify-between gap-3">
   224→              <div className="min-w-0">
   225→                <DialogTitle className="flex items-center gap-2 flex-wrap">
   226→                  <Receipt className="w-5 h-5 text-amber-600 shrink-0" />
   227→                  <span className="font-mono">{invoice.number}</span>
   228→                  <Badge variant="outline" className={cn('text-[10px]', status.color)}>
   229→                    <status.icon className="w-3 h-3 mr-1" />
   230→                    {status.label}
   231→                  </Badge>
   232→                  {isOverdue && (
   233→                    <Badge variant="outline" className="text-[10px] text-red-700 bg-red-50 border-red-200">
   234→                      <AlertTriangle className="w-3 h-3 mr-1" />
   235→                      Vade Geçti
   236→                    </Badge>
   237→                  )}
   238→                </DialogTitle>
   239→                <DialogDescription className="mt-1 flex items-center gap-2 flex-wrap text-xs">
   240→                  <User className="w-3 h-3" />
   241→                  {invoice.customer?.name ?? '—'}
   242→                </DialogDescription>
   243→              </div>
   244→              <div className="flex items-center gap-1 shrink-0">
   245→                <Tooltip>
   246→                  <TooltipTrigger asChild>
   247→                    <Button variant="outline" size="sm" onClick={() => setPdfOpen(true)}>
   248→                      <FileText className="w-3.5 h-3.5 mr-1" /> PDF Görüntüle
   249→                    </Button>
   250→                  </TooltipTrigger>
   251→                  <TooltipContent>Fatura PDF önizleme</TooltipContent>
   252→                </Tooltip>
   253→                <Tooltip>
   254→                  <TooltipTrigger asChild>
   255→                    <Button variant="outline" size="sm" onClick={() => onEdit(invoice)}>
   256→                      <Pencil className="w-3.5 h-3.5 mr-1" /> Düzenle
   257→                    </Button>
   258→                  </TooltipTrigger>
   259→                  <TooltipContent>Faturayı düzenle</TooltipContent>
   260→                </Tooltip>
   261→                <Button
   262→                  variant="outline"
   263→                  size="sm"
   264→                  className="text-red-600 hover:text-red-700 hover:bg-red-50 dark:hover:bg-red-950/30 border-red-200 dark:border-red-900/50"
   265→                  onClick={() => setDeleteOpen(true)}
   266→                >
   267→                  <Trash2 className="w-3.5 h-3.5" />
   268→                </Button>
   269→              </div>
   270→            </div>
   271→          </DialogHeader>
   272→
   273→          {isLoading ? (
   274→            <div className="space-y-3">
   275→              <Skeleton className="h-20 w-full" />
   276→              <Skeleton className="h-20 w-full" />
   277→            </div>
   278→          ) : (
   279→            <>
   280→              {/* Bilgi kartları */}
   281→              <div className="grid grid-cols-2 sm:grid-cols-4 gap-3 py-1">
   282→                <Card className="bg-muted/30">
   283→                  <CardContent className="p-3">
   284→                    <div className="text-[10px] uppercase tracking-wider text-muted-foreground flex items-center gap-1">
   285→                      <Calendar className="w-3 h-3" /> Düzenleme
   286→                    </div>
   287→                    <div className="text-sm font-bold mt-0.5">{formatDate(d.issueDate)}</div>
   288→                  </CardContent>
   289→                </Card>
   290→                <Card className="bg-muted/30">
   291→                  <CardContent className="p-3">
   292→                    <div className="text-[10px] uppercase tracking-wider text-muted-foreground flex items-center gap-1">
   293→                      <Clock className="w-3 h-3" /> Vade
   294→                    </div>
   295→                    <div className="text-sm font-bold mt-0.5">{formatDate(d.dueDate)}</div>
   296→                  </CardContent>
   297→                </Card>
   298→                <Card className="bg-muted/30">
   299→                  <CardContent className="p-3">
   300→                    <div className="text-[10px] uppercase tracking-wider text-muted-foreground flex items-center gap-1">
   301→                      <CheckCircle2 className="w-3 h-3" /> Ödeme Tarihi
   302→                    </div>
   303→                    <div className="text-sm font-bold mt-0.5">{formatDate(d.paidDate)}</div>
   304→                  </CardContent>
   305→                </Card>
   306→                <Card className="bg-muted/30">
   307→                  <CardContent className="p-3">
   308→                    <div className="text-[10px] uppercase tracking-wider text-muted-foreground">Para Birimi</div>
   309→                    <div className="text-sm font-bold mt-0.5">{d.currency}</div>
   310→                  </CardContent>
   311→                </Card>
   312→              </div>
   313→
   314→              {/* Toplamlar */}
   315→              <div className="grid grid-cols-3 gap-3">
   316→                <Card className="bg-muted/20">
   317→                  <CardContent className="p-3">
   318→                    <div className="text-[10px] uppercase tracking-wider text-muted-foreground">Ara Toplam</div>
   319→                    <div className="text-base font-bold mt-0.5 tabular-nums">{formatCurrency(d.subtotal, d.currency)}</div>
   320→                  </CardContent>
   321→                </Card>
   322→                <Card className="bg-muted/20">
   323→                  <CardContent className="p-3">
   324→                    <div className="text-[10px] uppercase tracking-wider text-muted-foreground">KDV</div>
   325→                    <div className="text-base font-bold mt-0.5 tabular-nums">{formatCurrency(d.taxTotal, d.currency)}</div>
   326→                  </CardContent>
   327→                </Card>
   328→                <Card className="bg-amber-50/50 dark:bg-amber-950/20 border-amber-200 dark:border-amber-900/50">
   329→                  <CardContent className="p-3">
   330→                    <div className="text-[10px] uppercase tracking-wider text-amber-700 dark:text-amber-400">Genel Toplam</div>
   331→                    <div className="text-base font-bold mt-0.5 tabular-nums text-amber-700 dark:text-amber-400">{formatCurrency(d.total, d.currency)}</div>
   332→                  </CardContent>
   333→                </Card>
   334→              </div>
   335→
   336→              <Separator />
   337→
   338→              {/* Satır detayları (lines) */}
   339→              {lines.length > 0 && (
   340→                <div className="space-y-2">
   341→                  <div className="flex items-center gap-2">
   342→                    <FileText className="w-4 h-4 text-muted-foreground" />
   343→                    <h4 className="text-sm font-semibold">Kalemler ({lines.length})</h4>
   344→                  </div>
   345→                  <div className="rounded-lg border border-border overflow-hidden">
   346→                    <Table>
   347→                      <TableHeader>
   348→                        <TableRow className="bg-muted/50">
   349→                          <TableHead className="text-xs uppercase tracking-wider text-muted-foreground py-2">Açıklama</TableHead>
   350→                          <TableHead className="text-xs uppercase tracking-wider text-muted-foreground py-2 text-right">Miktar</TableHead>
   351→                          <TableHead className="text-xs uppercase tracking-wider text-muted-foreground py-2 text-right">Birim</TableHead>
   352→                          <TableHead className="text-xs uppercase tracking-wider text-muted-foreground py-2 text-right">KDV</TableHead>
   353→                          {hasWeight && (
   354→                            <TableHead className="text-xs uppercase tracking-wider text-muted-foreground py-2 text-right">Ağırlık</TableHead>
   355→                          )}
   356→                          <TableHead className="text-xs uppercase tracking-wider text-muted-foreground py-2 text-right">Tutar</TableHead>
   357→                        </TableRow>
   358→                      </TableHeader>
   359→                      <TableBody>
   360→                        {lines.map((line) => (
   361→                          <TableRow key={line.id} className="text-sm">
   362→                            <TableCell className="py-2">
   363→                              {line.description}
   364→                              {line.product && (
   365→                                <span className="text-[10px] text-muted-foreground ml-1">({line.product.sku || line.product.name})</span>
   366→                              )}
   367→                            </TableCell>
   368→                            <TableCell className="text-right py-2 tabular-nums">{line.qty}</TableCell>
   369→                            <TableCell className="text-right py-2 tabular-nums">{formatCurrency(line.unitPrice, d.currency)}</TableCell>
   370→                            <TableCell className="text-right py-2 tabular-nums">%{line.taxRate}</TableCell>
   371→                            {hasWeight && (
   372→                              <TableCell className="text-right py-2 tabular-nums">
   373→                                {line.totalWeight != null && line.totalWeight > 0
   374→                                  ? formatWeight(line.totalWeight, line.weightUnit || 'kg')
   375→                                  : '—'}
   376→                              </TableCell>
   377→                            )}
   378→                            <TableCell className="text-right py-2 tabular-nums font-medium">{formatCurrency(line.lineTotal, d.currency)}</TableCell>
   379→                          </TableRow>
   380→                        ))}
   381→                      </TableBody>
   382→                    </Table>
   383→                  </div>
   384→                  {totalNetWeight != null && (
   385→                    <div className="flex justify-end">
   386→                      <div className="text-xs text-muted-foreground">
   387→                        Toplam Net Ağırlık: <span className="font-medium text-foreground tabular-nums">{formatWeight(totalNetWeight, 'kg')}</span>
   388→                      </div>
   389→                    </div>
   390→                  )}
   391→                </div>
   392→              )}
   393→
   394→              {/* Durum aksiyonları */}
   395→              <div className="space-y-2">
   396→                <div className="flex items-center gap-2">
   397→                  <TrendingUp className="w-4 h-4 text-amber-600" />
   398→                  <h4 className="text-sm font-semibold">Durum Yönetimi</h4>
   399→                </div>
   400→                <div className="flex flex-wrap gap-1.5">
   401→                  {INVOICE_STATUSES.map((s) => (
   402→                    <Button
   403→                      key={s.value}
   404→                      variant={d.status === s.value ? 'default' : 'outline'}
   405→                      size="sm"
   406→                      disabled={changing || d.status === s.value}
   407→                      onClick={() => changeStatus(s.value)}
   408→                      className={cn(
   409→                        d.status === s.value && 'bg-amber-600 hover:bg-amber-700',
   410→                      )}
   411→                    >
   412→                      <s.icon className="w-3.5 h-3.5 mr-1" />
   413→                      {s.label}
   414→                    </Button>
   415→                  ))}
   416→                </div>
   417→
   418→                {d.status === 'odendi' && d.paidDate && (
   419→                  <div className="p-2 rounded-md bg-emerald-50 dark:bg-emerald-950/30 border border-emerald-200 dark:border-emerald-900/50 text-xs text-emerald-700 dark:text-emerald-300 flex items-center gap-2">
   420→                    <CheckCircle2 className="w-4 h-4" />
   421→                    Ödendi — {formatDate(d.paidDate)}
   422→                  </div>
   423→                )}
   424→                {isOverdue && (
   425→                  <div className="p-2 rounded-md bg-red-50 dark:bg-red-950/30 border border-red-200 dark:border-red-900/50 text-xs text-red-700 dark:text-red-300 flex items-center gap-2">
   426→                    <AlertTriangle className="w-4 h-4" />
   427→                    Vade tarihi geçti. &quot;Gecikti&quot; olarak işaretleyebilirsiniz.
   428→                  </div>
   429→                )}
   430→
   431→                {/* Hızlı tahsilat aksiyonları — listedeki menünün aynısı, detaydan erişilebilir */}
   432→                {(isPending || d.status === 'odendi') && (
   433→                  <div className="pt-1">
   434→                    <div className="flex items-center gap-1.5 mb-1.5">
   435→                      <HandCoins className="w-3.5 h-3.5 text-fuchsia-600" />
   436→                      <span className="text-[10px] uppercase tracking-wider text-muted-foreground font-medium">Hızlı Tahsilat</span>
   437→                    </div>
   438→                    <div className="flex flex-wrap gap-1.5">
   439→                      {isPending && (
   440→                        <>
   441→                          <Button
   442→                            variant="outline"
   443→                            size="sm"
   444→                            disabled={quickBusy !== null}
   445→                            onClick={() => handleQuickStatus('odendi')}
   446→                            className="text-emerald-700 hover:text-emerald-800 hover:bg-emerald-50 dark:hover:bg-emerald-950/30 border-emerald-200 dark:border-emerald-900/50 h-8 text-xs"
   447→                          >
   448→                            {quickBusy === 'odendi'
   449→                              ? <Loader2 className="w-3.5 h-3.5 mr-1 animate-spin" />
   450→                              : <CircleCheckBig className="w-3.5 h-3.5 mr-1" />}
   451→                            Ödendi İşaretle
   452→                          </Button>
   453→                          <Button
   454→                            variant="outline"
   455→                            size="sm"
   456→                            disabled={quickBusy !== null}
   457→                            onClick={handleRemind}
   458→                            className="h-8 text-xs border-sky-200 dark:border-sky-900/50 hover:bg-sky-50 dark:hover:bg-sky-950/30"
   459→                          >
   460→                            {quickBusy === 'remind'
   461→                              ? <Loader2 className="w-3.5 h-3.5 mr-1 animate-spin" />
   462→                              : <MessageCircle className="w-3.5 h-3.5 mr-1" />}
   463→                            Hatırlat (WhatsApp)
   464→                          </Button>
   465→                          <Button
   466→                            variant="outline"
   467→                            size="sm"
