    76→    76→function ORDER_DOC_LABEL(status: string): string {
    77→    77→  switch (status) {
    78→    78→    case 'odeme_bekliyor': return 'Ödeme Bekliyor'
    79→    79→    case 'odendi': return 'Ödendi'
    80→    80→    case 'gecikti': return 'Gecikti'
    81→    81→    case 'iptal': return 'İptal'
    82→    82→    default: return status
    83→    83→  }

    91→    91→export function DocumentsView() {
    92→    92→  const qc = useQueryClient()
    93→    93→  const { user } = useAppStore()
    94→    94→  const su = user as SessionUser | null
    95→    95→
    96→    96→  // Yetkiler
    97→    97→  const canSeeInvoice = hasPermission(su, 'invoices.view') || hasPermission(su, 'erp.manage')
    98→    98→  const canSeeIrsaliye = hasPermission(su, 'irsaliye.view')
    99→    99→  const canSeePacking = canSeeIrsaliye // çeki listesi = irsaliye.view sahiplerine açık (müdür + depocu)
   100→   100→
   101→   101→  const [search, setSearch] = useState('')
   102→   102→  const [openCompanies, setOpenCompanies] = useState<Record<string, boolean>>({})
   103→   103→  const [generating, setGenerating] = useState<string | null>(null) // "orderId:type"
   104→   104→  const [activeDoc, setActiveDoc] = useState<GeneratedDoc | null>(null)
   105→   105→
   106→   106→  const canExport = hasPermission(su, 'export.data')
   107→   107→
   108→   108→  const { data, isLoading, isFetching, refetch } = useQuery({
   109→   109→    queryKey: ['documents-orders', search],
   110→   110→    queryFn: () => {
   111→   111→      const p = new URLSearchParams({ limit: '200' })
   112→   112→      if (search) p.set('search', search)
   113→   113→      return apiGet<OrdersResponse>(`/api/orders?${p.toString()}`)
   114→   114→    },
   115→   115→  })
   116→   116→
   117→   117→  const orders = useMemo(() => data?.items ?? [], [data])
   118→   118→  const hidePrices = data?.hidePrices === true
   119→   119→
   120→   120→  // Şirketlere göre grupla
   121→   121→  const companies = useMemo(() => {
   122→   122→    const map = new Map<string, { name: string; segment?: string; orders: OrderDoc[] }>()
   123→   123→    for (const o of orders) {
   124→   124→      const key = o.customer?.id ?? '_'
   125→   125→      if (!map.has(key)) {
   126→   126→        map.set(key, {
   127→   127→          name: o.customer?.name ?? 'Bilinmeyen Şirket',
   128→   128→          segment: o.customer?.segment,
   129→   129→          orders: [],
   130→   130→        })
   131→   131→      }
   132→   132→      map.get(key)!.orders.push(o)
   133→   133→    }
   134→   134→    // Şirket adına göre sırala
   135→   135→    return Array.from(map.values()).sort((a, b) => a.name.localeCompare(b.name, 'tr'))
   136→   136→  }, [orders])
   137→   137→
   138→   138→  // Özet
   139→   139→  const stats = useMemo(() => {
   140→   140→    const invoiced = orders.filter((o) => o.invoice).length
   141→   141→    const irsaliye = orders.filter((o) => (o.irsaliyeler?.length ?? 0) > 0).length
   142→   142→    const packing = orders.filter((o) => o.invoice?.packingListNo).length
   143→   143→    return { orders: data?.total ?? 0, companies: companies.length, invoiced, irsaliye, packing }
   144→   144→  }, [orders, companies, data])
   145→   145→
   146→   146→  // Belge üret + PDF aç
   147→   147→  const generateDoc = async (order: OrderDoc, type: 'invoice' | 'irsaliye' | 'packing_list') => {
   148→   148→    const genKey = `${order.id}:${type}`
   149→   149→    setGenerating(genKey)
   150→   150→    try {
   151→   151→      const res = await apiPost<{
   152→   152→        type: string
   153→   153→        created: boolean
   154→   154→        document: {
   155→   155→          id: string
   156→   156→          number: string
   157→   157→          packingListNo?: string | null
   158→   158→        }
   159→   159→        documentType: 'invoice' | 'irsaliye'
   160→   160→      }>(`/api/orders/${order.id}/generate-document`, { type })
   161→   161→
   162→   162→      if (res.created) {
   163→   163→        toast.success(`Belge otomatik oluşturuldu${res.document.number ? `: ${res.document.number}` : ''}`)
   164→   164→      }
   165→   165→
   166→   166→      if (type === 'irsaliye') {
   167→   167→        setActiveDoc({ kind: 'irsaliye', irsaliyeId: res.document.id, irsaliyeNumber: res.document.number })
   168→   168→      } else if (type === 'packing_list') {
   169→   169→        setActiveDoc({ kind: 'packing', invoiceId: res.document.id })
   170→   170→      } else {
   171→   171→        setActiveDoc({ kind: 'invoice', invoiceId: res.document.id })
   172→   172→      }
   173→   173→
   174→   174→      // Belge durumlarını tazele
   175→   175→      qc.invalidateQueries({ queryKey: ['documents-orders'] })
   176→   176→      qc.invalidateQueries({ queryKey: ['orders'] })
   177→   177→      qc.invalidateQueries({ queryKey: ['irsaliye'] })
   178→   178→      qc.invalidateQueries({ queryKey: ['invoices'] })
   179→   179→    } catch (e) {
   180→   180→      toast.error(e instanceof Error ? e.message : 'Belge oluşturulamadı')
   181→   181→    } finally {
   182→   182→      setGenerating(null)
   183→   183→    }
   184→   184→  }
   185→   185→
   186→   186→  const toggleCompany = (name: string) => {
   187→   187→    setOpenCompanies((prev) => ({ ...prev, [name]: !prev[name] }))
   188→   188→  }
   189→   189→
   190→   190→  // CSV dışa aktarma — siparişler + belge durumları
   191→   191→  const handleExportCSV = () => {
   192→   192→    if (orders.length === 0) {
   193→   193→      toast.error('Dışa aktarılacak sipariş yok')
   194→   194→      return
   195→   195→    }
   196→   196→    const rows = orders.map((o) => ({
   197→   197→      'Sipariş No': o.number,
   198→   198→      'Şirket': o.customer?.name ?? '',
   199→   199→      'Tarih': formatDate(o.orderDate),
   200→   200→      'Durum': ORDER_STATUS_META[o.status]?.label ?? o.status,
   201→   201→      'Fatura No': o.invoice?.number ?? '',
   202→   202→      'Fatura Durumu': o.invoice ? ORDER_DOC_LABEL(o.invoice.status) : 'Yok',
   203→   203→      'İrsaliye No': o.irsaliyeler?.[0]?.number ?? '',
   204→   204→      'İrsaliye Durumu': o.irsaliyeler?.[0] ? ORDER_STATUS_META[o.irsaliyeler[0].status]?.label ?? o.irsaliyeler[0].status : 'Yok',
   205→   205→      'Çeki Listesi No': o.invoice?.packingListNo ?? '',
   206→   206→      ...(hidePrices ? {} : { 'Tutar': o.totalAmount ?? 0, 'Para': o.currency ?? 'TRY' }),
   207→   207→    }))
   208→   208→    const csv = toCSV(rows)
   209→   209→    downloadFile(csv, `belge-yonetimi-${new Date().toISOString().slice(0, 10)}.csv`)
   210→   210→    toast.success(`${orders.length} sipariş dışa aktarıldı`)
   211→   211→  }
   212→   212→
   213→   213→  return (
   214→   214→    <div className="space-y-5 animate-fade-in">
   215→   215→      {/* Header */}
   216→   216→      <div className="flex items-start justify-between flex-wrap gap-3">
   217→   217→        <div>
   218→   218→          <h2 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   219→   219→            <FileStack className="w-6 h-6 text-violet-600" />
   220→   220→            Belge Yönetimi
   221→   221→          </h2>
   222→   222→          <p className="text-sm text-muted-foreground mt-0.5">
   223→   223→            Şirketler ve sipariş belgeleri — fatura, irsaliye ve çeki listesi otomatik üretilir
   224→   224→            {canSeeIrsaliye && !canSeeInvoice && (
   225→   225→              <span className="ml-2 text-violet-600 font-medium">(görüş alanınız: İrsaliye + Çeki Listesi)</span>
   226→   226→            )}
   227→   227→          </p>
   228→   228→        </div>
   229→   229→        <div className="flex gap-2 flex-wrap">
   230→   230→          {canExport && (
   231→   231→            <Button variant="outline" size="sm" onClick={handleExportCSV} disabled={orders.length === 0}>
   232→   232→              <Download className="w-4 h-4 mr-1.5" />
   233→   233→              CSV Dışa Aktar
   234→   234→            </Button>
   235→   235→          )}
   236→   236→          <Button variant="outline" size="sm" onClick={() => refetch()} disabled={isFetching}>
   237→   237→            <RefreshCw className={cn('w-4 h-4 mr-1.5', isFetching && 'animate-spin')} />
   238→   238→            Yenile
   239→   239→          </Button>
   240→   240→        </div>
   241→   241→      </div>
   242→   242→
   243→   243→      <Tabs defaultValue="orders" className="space-y-4">
   244→   244→        <TabsList className="grid w-full max-w-md grid-cols-2">
   245→   245→          <TabsTrigger value="orders" className="gap-1.5">
   246→   246→            <Package className="w-3.5 h-3.5" /> Sipariş Belgeleri
   247→   247→          </TabsTrigger>
   248→   248→          <TabsTrigger value="irsaliye" className="gap-1.5">
   249→   249→            <Truck className="w-3.5 h-3.5" /> İrsaliye Listesi
   250→   250→          </TabsTrigger>
   251→   251→        </TabsList>
   252→   252→
   253→   253→        {/* ==================== TAB 1: SİPARİŞ BELGELERİ ==================== */}
   254→   254→        <TabsContent value="orders" className="space-y-4">
   255→   255→          {/* Stats */}
   256→   256→          <div className="grid grid-cols-2 sm:grid-cols-5 gap-3">
   257→   257→            <Card>
   258→   258→              <CardContent className="p-3">
   259→   259→                <div className="text-[10px] uppercase tracking-wider text-muted-foreground">Sipariş</div>
   260→   260→                <div className="text-xl font-bold">{stats.orders}</div>
   261→   261→              </CardContent>
   262→   262→            </Card>
   263→   263→            <Card>
   264→   264→              <CardContent className="p-3">
   265→   265→                <div className="text-[10px] uppercase tracking-wider text-muted-foreground">Şirket</div>
   266→   266→                <div className="text-xl font-bold">{stats.companies}</div>
   267→   267→              </CardContent>
   268→   268→            </Card>
   269→   269→            {canSeeInvoice && (
   270→   270→              <Card className="bg-emerald-50/50 dark:bg-emerald-950/20">
   271→   271→                <CardContent className="p-3">
   272→   272→                  <div className="text-[10px] uppercase tracking-wider text-emerald-700 dark:text-emerald-400">Faturalı</div>
   273→   273→                  <div className="text-xl font-bold text-emerald-700 dark:text-emerald-400">{stats.invoiced}</div>
   274→   274→                </CardContent>
   275→   275→              </Card>
   276→   276→            )}
   277→   277→            {canSeeIrsaliye && (
   278→   278→              <Card className="bg-violet-50/50 dark:bg-violet-950/20">
   279→   279→                <CardContent className="p-3">
   280→   280→                  <div className="text-[10px] uppercase tracking-wider text-violet-700 dark:text-violet-400">İrsaliyeli</div>
   281→   281→                  <div className="text-xl font-bold text-violet-700 dark:text-violet-400">{stats.irsaliye}</div>
   282→   282→                </CardContent>
   283→   283→              </Card>
   284→   284→            )}
   285→   285→            {canSeePacking && (
   286→   286→              <Card className="bg-amber-50/50 dark:bg-amber-950/20">
   287→   287→                <CardContent className="p-3">
   288→   288→                  <div className="text-[10px] uppercase tracking-wider text-amber-700 dark:text-amber-400">Çeki Listeli</div>
   289→   289→                  <div className="text-xl font-bold text-amber-700 dark:text-amber-400">{stats.packing}</div>

   298→   298→              <div className="relative">
   299→   299→                <Search className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground pointer-events-none" />
   300→   300→                <Input
   301→   301→                  value={search}
   302→   302→                  onChange={(e) => setSearch(e.target.value)}
   303→   303→                  placeholder="Şirket veya sipariş no ara..."
   304→   304→                  className="pl-9"

   306→   306→                {search && (
   307→   307→                  <button
   308→   308→                    onClick={() => setSearch('')}
   309→   309→                    className="absolute right-2 top-1/2 -translate-y-1/2 p-1 hover:bg-muted rounded"
   310→   310→                  >
   311→   311→                    <X className="w-3.5 h-3.5 text-muted-foreground" />

   314→   314→              </div>
   315→   315→            </CardContent>
   316→   316→          </Card>
   317→   317→
   318→   318→          {/* Şirketler → Siparişler */}
   319→   319→          {isLoading ? (
   320→   320→            <div className="space-y-3">
   321→   321→              <Skeleton className="h-16 w-full" />
   322→   322→              <Skeleton className="h-40 w-full" />
   323→   323→              <Skeleton className="h-16 w-full" />
   324→   324→            </div>
   325→   325→          ) : companies.length === 0 ? (
   326→   326→            <Card>
   327→   327→              <CardContent className="p-12 text-center">
   328→   328→                <Building2 className="w-10 h-10 mx-auto text-muted-foreground/50 mb-2" />

   331→   331→            </Card>
   332→   332→          ) : (
   333→   333→            <div className="space-y-3">
   334→   334→              {companies.map((company) => {
   335→   335→                const isOpen = openCompanies[company.name] ?? true

   337→   337→                  <Card key={company.name} className="overflow-hidden">
   338→   338→                    {/* Şirket başlığı */}
   339→   339→                    <button
   340→   340→                      onClick={() => toggleCompany(company.name)}
   341→   341→                      className="w-full flex items-center gap-3 p-4 bg-gradient-to-r from-muted/50 to-muted/20 hover:from-muted/70 hover:to-muted/30 transition-all text-left"
   342→   342→                    >
   343→   343→                      {isOpen ? (
   344→   344→                        <ChevronDown className="w-4 h-4 text-muted-foreground shrink-0" />
   345→   345→                      ) : (
   346→   346→                        <ChevronRight className="w-4 h-4 text-muted-foreground shrink-0" />
   347→   347→                      )}

   349→   349→                        <Building2 className="w-4 h-4 text-slate-600 dark:text-slate-300" />
   350→   350→                      </div>
   351→   351→                      <div className="min-w-0 flex-1">
   352→   352→                        <div className="font-semibold text-sm truncate">{company.name}</div>
   353→   353→                        <div className="text-[11px] text-muted-foreground">

   355→   355→                          {company.segment ? ` · ${company.segment}` : ''}
   356→   356→                        </div>
   357→   357→                      </div>
   358→   358→                      {/* Şirket belge özeti */}
   359→   359→                      {(() => {

   363→   363→                        return (
