    12→// ---------- Deterministik RNG ----------
    13→function mulberry32(seed: number) {
    14→  let a = seed
    15→  return function () {
    16→    a |= 0
    17→    a = (a + 0x6d2b79f5) | 0
    18→    let t = Math.imul(a ^ (a >>> 15), 1 | a)
    19→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t
    20→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296
    21→  }
    22→}
    23→const rng = mulberry32(20261002)
    24→const pick = <T,>(arr: readonly T[]): T => arr[Math.floor(rng() * arr.length)]
    25→const ri = (min: number, max: number) => min + Math.floor(rng() * (max - min + 1))
    26→const rf = (min: number, max: number) => min + rng() * (max - min)
    27→const chance = (p: number) => rng() < p

    31→  let s = 'c'
    32→  for (let i = 0; i < 24; i++) s += chars[Math.floor(rng() * chars.length)]
    33→  return s
    34→}
