   259→// Müşteri görünümü — public booking widget
   260→function PublicBooking({ provider, onBooked }: { provider: Provider; onBooked: () => void }) {
   261→  const [step, setStep] = useState(1)
   262→  const [selectedService, setSelectedService] = useState<Service | null>(null)
   263→  const [selectedStaff, setSelectedStaff] = useState<string>('')
   264→  const [selectedDate, setSelectedDate] = useState(new Date().toISOString().slice(0, 10))
   265→  const [selectedTime, setSelectedTime] = useState('')
   266→  const [customer, setCustomer] = useState({ name: '', phone: '', email: '', note: '' })
   267→  const [booking, setBooking] = useState(false)
   268→
   269→  // Basit slot üretimi (09:00-18:00, 30dk aralık)
   270→  const slots = useMemo(() => {
   271→    const result: string[] = []
   272→    for (let h = 9; h < 18; h++) {
   273→      result.push(`${String(h).padStart(2, '0')}:00`)
   274→      result.push(`${String(h).padStart(2, '0')}:30`)
   275→    }
   276→    return result
   277→  }, [])
   278→
   279→  const handleBook = async () => {
   280→    setBooking(true)
   281→    try {
   282→      const dateTime = new Date(`${selectedDate}T${selectedTime}:00`)
   283→      await apiPost(`/api/appointments/providers/${provider.id}/appointments`, {
   284→        serviceId: selectedService?.id,
   285→        staffId: selectedStaff || undefined,
   286→        customerName: customer.name,
   287→        customerPhone: customer.phone,
   288→        customerEmail: customer.email || undefined,
   289→        customerNote: customer.note || undefined,
   290→        date: dateTime.toISOString(),
   291→        source: 'web',
   292→      })
   293→      toast.success('Randevu oluşturuldu!', { description: `${formatDate(selectedDate)} ${selectedTime}` })
   294→      setStep(5) // success
   295→      onBooked()
   296→    } catch (e) {
   297→      toast.error('Randevu oluşturulamadı', { description: e instanceof Error ? e.message : '' })
   298→    } finally {
   299→      setBooking(false)
   300→    }
   301→  }
   302→
   303→  return (
   304→    <Card className="shadow-soft max-w-2xl mx-auto">
   305→      <CardContent className="p-6">
   306→        {/* Progress */}
   307→        <div className="flex items-center justify-between mb-6">
   308→          {['Hizmet', 'Personel', 'Tarih', 'Bilgiler', 'Tamam'].map((label, i) => (
   309→            <div key={i} className="flex items-center flex-1">
   310→              <div className={cn('w-7 h-7 rounded-full flex items-center justify-center text-xs font-semibold shrink-0',
   311→                step > i + 1 ? 'bg-emerald-500 text-white' : step === i + 1 ? 'bg-emerald-600 text-white' : 'bg-muted text-muted-foreground')}>
   312→                {step > i + 1 ? '✓' : i + 1}
   313→              </div>
   314→              {i < 4 && <div className={cn('flex-1 h-0.5 mx-1', step > i + 1 ? 'bg-emerald-500' : 'bg-muted')} />}
   315→            </div>
   316→          ))}
   317→        </div>
   318→
   319→        {/* Step 1: Service */}
   320→        {step === 1 && (
   321→          <div className="space-y-3">
   322→            <h3 className="font-semibold text-lg">Hizmet Seçin</h3>
   323→            <div className="grid grid-cols-1 md:grid-cols-2 gap-2">
   324→              {provider.services.map((s) => (
   325→                <button key={s.id} onClick={() => { setSelectedService(s); setStep(2) }}
   326→                  className="flex items-center gap-3 p-3 rounded-lg border border-border hover:border-emerald-300 hover:bg-emerald-50/50 dark:hover:bg-emerald-950/20 transition-colors text-left">
   327→                  <div className="w-10 h-10 rounded-lg bg-emerald-50 dark:bg-emerald-950/30 flex items-center justify-center shrink-0">
   328→                    <Scissors className="w-5 h-5 text-emerald-600" />
   329→                  </div>
   330→                  <div className="flex-1 min-w-0">
   331→                    <div className="font-medium text-sm">{s.name}</div>
   332→                    <div className="text-xs text-muted-foreground">{s.duration}dk · {formatCurrency(s.price)}</div>
   333→                  </div>
   334→                </button>
   335→              ))}
   336→            </div>
   337→          </div>
   338→        )}
   339→
   340→        {/* Step 2: Staff */}
   341→        {step === 2 && (
   342→          <div className="space-y-3">
   343→            <h3 className="font-semibold text-lg">Personel Seçin</h3>
   344→            <button onClick={() => { setSelectedStaff(''); setStep(3) }}
   345→              className="w-full flex items-center gap-3 p-3 rounded-lg border border-emerald-300 bg-emerald-50/50 dark:bg-emerald-950/20 transition-colors text-left">
   346→              <div className="w-10 h-10 rounded-full bg-muted flex items-center justify-center"><User className="w-5 h-5" /></div>
   347→              <span className="font-medium text-sm">Herhangi biri</span>
   348→            </button>
   349→            {provider.staff.map((s) => (
   350→              <button key={s.id} onClick={() => { setSelectedStaff(s.id); setStep(3) }}
   351→                className="w-full flex items-center gap-3 p-3 rounded-lg border border-border hover:border-emerald-300 hover:bg-emerald-50/50 dark:hover:bg-emerald-950/20 transition-colors text-left">
   352→                <div className="w-10 h-10 rounded-full bg-muted flex items-center justify-center text-sm font-semibold">{s.name.charAt(0)}</div>
   353→                <div><div className="font-medium text-sm">{s.name}</div><div className="text-xs text-muted-foreground">{s.title}</div></div>
   354→              </button>
   355→            ))}
   356→            <Button variant="ghost" size="sm" onClick={() => setStep(1)}>← Geri</Button>

   359→
   360→        {/* Step 3: Date & Time */}
   361→        {step === 3 && (

   369→                  {time}
   370→                </button>
   371→              ))}
   372→            </div>
   373→            <Button variant="ghost" size="sm" onClick={() => setStep(2)}>← Geri</Button>
   374→          </div>
   375→        )}
   376→
   377→        {/* Step 4: Customer info */}
   378→        {step === 4 && (
   379→          <div className="space-y-3">
   380→            <h3 className="font-semibold text-lg">Bilgilerinizi Girin</h3>
   381→            <div className="space-y-2">

   384→              <div><Label className="text-xs">E-posta (opsiyonel)</Label><Input value={customer.email} onChange={(e) => setCustomer({ ...customer, email: e.target.value })} placeholder="email@ornek.com" /></div>

   386→            </div>
   387→            <div className="p-3 rounded-lg bg-muted/30 text-sm">
   388→              <div className="font-medium">Özet</div>
   389→              <div className="text-xs text-muted-foreground mt-1">
   390→                {selectedService?.name} · {selectedStaff ? provider.staff.find((s) => s.id === selectedStaff)?.name : 'Herhangi'} · {formatDate(selectedDate)} {selectedTime} · {formatCurrency(selectedService?.price ?? 0)}
