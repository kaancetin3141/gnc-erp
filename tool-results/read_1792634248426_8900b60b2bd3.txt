   387→   387→function pageShell(o: { title: string; desc: string; body: string; scripts?: string[]; extraCss?: string[]; gwPort?: string | null }): string {
   388→   388→  // Sürüm parametresi: CSS/JS içeriğinin djb2 özeti — içerik değişince URL değişir,
   389→   389→  // tarayıcı/heuristik önbellek eski stili servis edemez (cache-busting).
   390→   390→  let _vh = 5381;
   391→   391→  const _verSrc = STYLE_CSS + '\u0000' + HOME_JS + '\u0000' + DETAIL_JS;
   392→   392→  for (let i = 0; i < _verSrc.length; i++) _vh = ((_vh << 5) + _vh + _verSrc.charCodeAt(i)) >>> 0;
   393→   393→  const VER = _vh.toString(36);
   394→   394→  // Gateway (XTransformPort) ile açıldıysa göreli varlık URL'lerine param ekle —
   395→   395→  // aksi hâlde /assets/* istekleri gateway'den CRM'e düşer ve 404 olur.
   396→   396→  const gw = (p: string) => {
   397→   397→    const withVer = p + (p.includes('?') ? '&' : '?') + 'v=' + VER;
   398→   398→    return o.gwPort ? withVer + '&XTransformPort=' + encodeURIComponent(o.gwPort) : withVer;
   399→   399→  };
   400→   400→  const scripts = (o.scripts || []).map((s) => `<script src="${gw(s)}" defer></script>`).join('\n  ');
   401→   401→  const extraCss = (o.extraCss || []).map((s) => `<link rel="stylesheet" href="${gw(s)}">`).join('\n  ');
   402→   402→  return `<!DOCTYPE html>
   403→   403→<html lang="tr">
   404→   404→<head>
   405→   405→<meta charset="UTF-8">
   406→   406→<meta name="viewport" content="width=device-width, initial-scale=1">
   407→   407→<title>${esc(o.title)}</title>
   408→   408→<meta name="description" content="${esc(o.desc)}">
   409→   409→<link rel="icon" href="${gw(FAVICON)}">
   410→   410→<link rel="stylesheet" href="${gw('/assets/style.css')}">
   411→   411→  ${extraCss}
   412→   412→</head>
   413→   413→<body>
   414→   414→<div class="page">
   415→   415→${o.body}
   416→   416→</div>
   417→   417→<div id="toast" role="status" aria-live="polite"></div>
   418→   418→  ${scripts}
