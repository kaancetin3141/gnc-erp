   122→
   123→export function CafeOrderScreen({ cafeId }: { cafeId: string }) {
   124→  const qc = useQueryClient()
   125→  const [tab, setTab] = useState<'new' | 'active'>('new')
   126→  const [selectedTableId, setSelectedTableId] = useState<string | null>(null)
   127→  const [cart, setCart] = useState<CartItem[]>([])
   128→  const [orderType, setOrderType] = useState<'dine_in' | 'takeaway' | 'delivery'>('dine_in')
   129→  const [customerName, setCustomerName] = useState('')
   130→  const [orderNotes, setOrderNotes] = useState('')
   131→  const [sending, setSending] = useState(false)
   132→  const [activeOrder, setActiveOrder] = useState<CafeOrder | null>(null)
   133→  const [activeOrderDialog, setActiveOrderDialog] = useState(false)
   134→  const [cancelTarget, setCancelTarget] = useState<CafeOrder | null>(null)
   135→
   136→  const { data: tablesData, isLoading: tablesLoading } = useQuery({
   137→    queryKey: ['cafe-tables', cafeId],
   138→    queryFn: () => apiGet<{ items: CafeTable[] }>(`/api/cafe/${cafeId}/tables`),
   139→  })
   140→  const { data: menuData, isLoading: menuLoading } = useQuery({
   141→    queryKey: ['cafe-menu', cafeId],
   142→    queryFn: () => apiGet<{ items: MenuCategory[] }>(`/api/cafe/${cafeId}/menu`),
   143→  })
   144→  const { data: ordersData, isLoading: ordersLoading } = useQuery({
   145→    queryKey: ['cafe-orders', cafeId, 'active'],
   146→    queryFn: () => apiGet<{ items: CafeOrder[] }>(`/api/cafe/${cafeId}/orders?status=acik&status=hazirlaniyor&status=hazir`),
   147→    refetchInterval: 10_000,
   148→  })
   149→
   150→  const tables = tablesData?.items ?? []
   151→  const categories = menuData?.items ?? []
   152→  const orders = ordersData?.items ?? []
   153→
   154→  // Flat item list with category info
   155→  const allItems = useMemo(() => {
   156→    const list: (MenuItem & { categoryName: string })[] = []
   157→    for (const cat of categories) {
   158→      for (const it of cat.items) {
   159→        list.push({ ...it, categoryName: cat.name })
   160→      }
   161→    }
   162→    return list.filter((i) => i.isAvailable)
   163→  }, [categories])
   164→
   165→  const cartTotal = useMemo(() => cart.reduce((s, c) => s + c.menuItem.price * c.qty, 0), [cart])
   166→
   167→  function addToCart(item: MenuItem) {
   168→    setCart((prev) => {
   169→      const ex = prev.find((c) => c.menuItem.id === item.id && c.notes === '')
   170→      if (ex) {
   171→        return prev.map((c) =>
   172→          c.menuItem.id === item.id && c.notes === '' ? { ...c, qty: c.qty + 1 } : c,
   173→        )
   174→      }
   175→      return [...prev, { menuItem: item, qty: 1, notes: '' }]
   176→    })
   177→  }
   178→
   179→  function updateQty(idx: number, delta: number) {
   180→    setCart((prev) => {
   181→      const next = [...prev]
   182→      const newQty = next[idx].qty + delta
   183→      if (newQty <= 0) {
   184→        next.splice(idx, 1)
   185→      } else {
   186→        next[idx] = { ...next[idx], qty: newQty }
   187→      }
   188→      return next
   189→    })
   190→  }
   191→
   192→  function updateNotes(idx: number, notes: string) {
   193→    setCart((prev) => {
   194→      const next = [...prev]
   195→      next[idx] = { ...next[idx], notes }
   196→      return next
   197→    })
   198→  }
   199→
   200→  function removeFromCart(idx: number) {
   201→    setCart((prev) => {
   202→      const next = [...prev]
   203→      next.splice(idx, 1)
   204→      return next
   205→    })
   206→  }
   207→
   208→  function clearCart() {
   209→    setCart([])
   210→    setOrderNotes('')
   211→    setCustomerName('')
   212→  }
   213→
   214→  async function handleSendOrder() {
   215→    if (cart.length === 0) {
   216→      toast.error('Sepet boş')
   217→      return
   218→    }
   219→    if (orderType === 'dine_in' && !selectedTableId) {
   220→      toast.error('Masa seçimi gerekli')
   221→      return
   222→    }
   223→    setSending(true)
   224→    try {
   225→      await apiPost(`/api/cafe/${cafeId}/orders`, {
   226→        tableId: orderType === 'dine_in' ? selectedTableId : null,
   227→        type: orderType,
   228→        customerName: customerName || undefined,
   229→        notes: orderNotes || undefined,
   230→        items: cart.map((c) => ({
   231→          menuItemId: c.menuItem.id,
   232→          qty: c.qty,
   233→          notes: c.notes || undefined,
   234→        })),
   235→      })
   236→      qc.invalidateQueries({ queryKey: ['cafe-orders', cafeId] })
   237→      qc.invalidateQueries({ queryKey: ['cafe-tables', cafeId] })
   238→      toast.success('Sipariş gönderildi')
   239→      clearCart()
   240→      setSelectedTableId(null)
   241→      setTab('active')
   242→    } catch (e) {
   243→      toast.error(e instanceof Error ? e.message : 'Sipariş gönderilemedi')
   244→    } finally {
   245→      setSending(false)
   246→    }
   247→  }
   248→
   249→  async function openOrderDetail(o: CafeOrder) {
   250→    try {
   251→      const detail = await apiGet<CafeOrder>(`/api/cafe/${cafeId}/orders/${o.id}`)
   252→      setActiveOrder(detail)
   253→      setActiveOrderDialog(true)
   254→    } catch (e) {
   255→      toast.error(e instanceof Error ? e.message : 'Sipariş yüklenemedi')
   256→    }
   257→  }
   258→
   259→  async function handleCancelOrder() {
   260→    if (!cancelTarget) return
   261→    try {
   262→      await apiDelete(`/api/cafe/${cafeId}/orders/${cancelTarget.id}`)
   263→      qc.invalidateQueries({ queryKey: ['cafe-orders', cafeId] })
   264→      qc.invalidateQueries({ queryKey: ['cafe-tables', cafeId] })
   265→      toast.success('Sipariş iptal edildi')
   266→      setCancelTarget(null)
   267→      if (activeOrder?.id === cancelTarget.id) setActiveOrderDialog(false)
   268→    } catch (e) {
   269→      toast.error(e instanceof Error ? e.message : 'İptal başarısız')
   270→    }
   271→  }
   272→
   273→  return (
   274→    <div className="space-y-4">
   275→      <Tabs value={tab} onValueChange={(v) => setTab(v as 'new' | 'active')}>
   276→        <TabsList>
   277→          <TabsTrigger value="new" className="text-xs sm:text-sm">
   278→            <Plus className="w-3.5 h-3.5 mr-1" />
   279→            Yeni Sipariş
   280→          </TabsTrigger>
   281→          <TabsTrigger value="active" className="text-xs sm:text-sm">
   282→            <Receipt className="w-3.5 h-3.5 mr-1" />
   283→            Aktif Siparişler
   284→            {orders.length > 0 && (
   285→              <span className="ml-1 px-1.5 py-0.5 rounded-full bg-emerald-100 dark:bg-emerald-950/40 text-emerald-700 dark:text-emerald-300 text-[10px] font-bold">
   286→                {orders.length}
   287→              </span>
   288→            )}
   289→          </TabsTrigger>
   290→        </TabsList>
   291→      </Tabs>
   292→
   293→      {/* New order tab */}
   294→      {tab === 'new' && (
   295→        <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   296→          {/* Left: menu */}
   297→          <div className="lg:col-span-2 space-y-3">
   298→            {/* Order type selector */}
   299→            <Card>
   300→              <CardContent className="p-3">
   301→                <div className="space-y-3">
   302→                  <div className="grid grid-cols-3 gap-2">
   303→                    {[
   304→                      { v: 'dine_in', label: 'Masa', icon: Table2 },
   305→                      { v: 'takeaway', label: 'Paket', icon: Receipt },
   306→                      { v: 'delivery', label: 'Gel-Al', icon: User },
   307→                    ].map((t) => (
   308→                      <button
   309→                        key={t.v}
   310→                        onClick={() => setOrderType(t.v as 'dine_in' | 'takeaway' | 'delivery')}
   311→                        className={cn(
   312→                          'flex flex-col items-center gap-1 p-2.5 rounded-lg border-2 transition-all',
   313→                          orderType === t.v
   314→                            ? 'border-emerald-500 bg-emerald-50 dark:bg-emerald-950/30 text-emerald-700 dark:text-emerald-300'
   315→                            : 'border-border hover:border-emerald-300',
   316→                        )}
   317→                      >
   318→                        <t.icon className="w-4 h-4" />
   319→                        <span className="text-xs">{t.label}</span>
   320→                      </button>
   321→                    ))}
   322→                  </div>
   323→
   324→                  {orderType === 'dine_in' && (
   325→                    <div>
   326→                      <Label className="text-xs text-muted-foreground">Masa</Label>
   327→                      {tablesLoading ? (
   328→                        <Skeleton className="h-9 w-full mt-1" />
   329→                      ) : tables.length === 0 ? (
   330→                        <p className="text-xs text-amber-600 mt-1">Önce kafe yöneticisi masa eklemeli.</p>
   331→                      ) : (
   332→                        <Select value={selectedTableId ?? ''} onValueChange={setSelectedTableId}>
   333→                          <SelectTrigger className="mt-1">
   334→                            <SelectValue placeholder="Masa seçin..." />
   335→                          </SelectTrigger>
   336→                          <SelectContent>
   337→                            {tables
   338→                              .filter((t) => t.status !== 'siparis' || t.id === selectedTableId)
   339→                              .map((t) => (
   340→                                <SelectItem key={t.id} value={t.id} disabled={t.status === 'siparis' && t.id !== selectedTableId}>
   341→                                  Masa {t.number} {t.status === 'siparis' ? '(dolu)' : `· ${t.status}`}
   342→                                </SelectItem>
   343→                              ))}
   344→                          </SelectContent>
   345→                        </Select>
   346→                      )}
   347→                    </div>
   348→                  )}
   349→
   350→                  {(orderType === 'takeaway' || orderType === 'delivery') && (
   351→                    <div>
   352→                      <Label className="text-xs text-muted-foreground">Müşteri Adı</Label>
   353→                      <Input
   354→                        value={customerName}
   355→                        onChange={(e) => setCustomerName(e.target.value)}
   356→                        placeholder="Müşteri adı (opsiyonel)"
   357→                        className="mt-1 h-9"
   358→                      />
   359→                    </div>
   360→                  )}
   361→                </div>
   362→              </CardContent>
   363→            </Card>
   364→
   365→            {/* Menu items */}
   366→            <Card>
   367→              <CardContent className="p-3">
   368→                <div className="text-xs font-semibold uppercase tracking-wider text-muted-foreground mb-2">
   369→                  Menü
   370→                </div>
   371→                {menuLoading ? (
   372→                  <div className="space-y-2">
   373→                    {[...Array(6)].map((_, i) => <Skeleton key={i} className="h-12 w-full" />)}
   374→                  </div>
   375→                ) : allItems.length === 0 ? (
   376→                  <div className="text-center py-8 text-sm text-muted-foreground">
   377→                    <Coffee className="w-8 h-8 mx-auto mb-2 opacity-50" />
   378→                    Menüde ürün yok
   379→                  </div>
   380→                ) : (
   381→                  <div className="space-y-4 max-h-[60vh] overflow-y-auto custom-scroll pr-1">
   382→                    {categories.map((cat) => {
   383→                      const items = cat.items.filter((i) => i.isAvailable)
   384→                      if (items.length === 0) return null
   385→                      return (
   386→                        <div key={cat.id}>
   387→                          <div className="flex items-center gap-1.5 text-xs font-semibold text-muted-foreground mb-2 px-1">
   388→                            {cat.icon ? <span>{cat.icon}</span> : <Coffee className="w-3.5 h-3.5" />}
   389→                            {cat.name}
   390→                            <Badge variant="outline" className="text-[10px] h-4">{items.length}</Badge>
   391→                          </div>
   392→                          <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
   393→                            {items.map((item) => {
   394→                              const Icon = STATION_ICON[item.station] ?? Coffee
   395→                              return (
   396→                                <button
   397→                                  key={item.id}
   398→                                  onClick={() => addToCart(item)}
   399→                                  className="group p-2.5 rounded-lg border border-border hover:border-emerald-400 hover:bg-emerald-50/50 dark:hover:bg-emerald-950/20 text-left transition-all"
   400→                                >
   401→                                  <div className="flex items-start gap-2">
   402→                                    <div className="w-10 h-10 rounded-md bg-muted flex items-center justify-center shrink-0 overflow-hidden">
   403→                                      {item.photo ? (
   404→                                        // eslint-disable-next-line @next/next/no-img-element
   405→                                        <img src={item.photo} alt={item.name} className="w-full h-full object-cover" />
   406→                                      ) : (
   407→                                        <Icon className="w-4 h-4 text-muted-foreground" />
   408→                                      )}
   409→                                    </div>
   410→                                    <div className="flex-1 min-w-0">
   411→                                      <div className="text-xs font-medium leading-tight truncate">{item.name}</div>
   412→                                      <div className="text-xs text-emerald-600 dark:text-emerald-400 font-semibold mt-0.5">
   413→                                        {formatCurrency(item.price, item.currency)}
   414→                                      </div>
   415→                                    </div>
   416→                                  </div>
   417→                                </button>
   418→                              )
   419→                            })}
   420→                          </div>
   421→                        </div>
   422→                      )
   423→                    })}
   424→                  </div>
   425→                )}
   426→              </CardContent>
   427→            </Card>
   428→          </div>
   429→
   430→          {/* Right: cart */}
   431→          <Card className="lg:sticky lg:top-4 h-fit">
   432→            <CardContent className="p-3">
   433→              <div className="flex items-center justify-between mb-3">
   434→                <div className="flex items-center gap-2">
   435→                  <ShoppingCart className="w-4 h-4 text-emerald-600" />
   436→                  <span className="text-sm font-semibold">Sepet</span>
   437→                  {cart.length > 0 && (
   438→                    <Badge variant="outline" className="text-[10px] h-5">{cart.length}</Badge>
   439→                  )}
   440→                </div>
   441→                {cart.length > 0 && (
   442→                  <Button size="sm" variant="ghost" onClick={clearCart} className="h-7 text-xs text-red-600 hover:text-red-700">
   443→                    <Trash2 className="w-3 h-3 mr-1" />
   444→                    Temizle
   445→                  </Button>
   446→                )}
   447→              </div>
   448→
   449→              {cart.length === 0 ? (
   450→                <div className="text-center py-8 text-sm text-muted-foreground">
   451→                  <ShoppingCart className="w-8 h-8 mx-auto mb-2 opacity-50" />
   452→                  Sepet boş
   453→                </div>
   454→              ) : (
   455→                <div className="space-y-2 max-h-[40vh] overflow-y-auto custom-scroll">
   456→                  {cart.map((c, idx) => {
   457→                    const Icon = STATION_ICON[c.menuItem.station] ?? Coffee
   458→                    return (
   459→                      <div key={idx} className="p-2 rounded-lg bg-muted/50">
   460→                        <div className="flex items-start gap-2">
   461→                          <Icon className="w-3.5 h-3.5 mt-0.5 text-muted-foreground shrink-0" />
   462→                          <div className="flex-1 min-w-0">
   463→                            <div className="flex items-start justify-between gap-2">
   464→                              <span className="text-xs font-medium leading-tight">{c.menuItem.name}</span>
   465→                              <button
   466→                                onClick={() => removeFromCart(idx)}
   467→                                className="text-red-500 hover:text-red-700 shrink-0"
   468→                              >
   469→                                <X className="w-3.5 h-3.5" />
   470→                              </button>
   471→                            </div>
   472→                            <div className="text-[10px] text-muted-foreground">
   473→                              {formatCurrency(c.menuItem.price, c.menuItem.currency)} × {c.qty} ={' '}
   474→                              <span className="text-emerald-600 dark:text-emerald-400 font-semibold">
   475→                                {formatCurrency(c.menuItem.price * c.qty, c.menuItem.currency)}
   476→                              </span>
   477→                            </div>
   478→                            <Input
   479→                              value={c.notes}
   480→                              onChange={(e) => updateNotes(idx, e.target.value)}
   481→                              placeholder="Not: az şekerli..."
   482→                              className="h-7 mt-1 text-[11px]"
   483→                            />
   484→                          </div>
   485→                        </div>
   486→                        <div className="flex items-center justify-end gap-1.5 mt-1.5">
   487→                          <Button
   488→                            size="sm"
   489→                            variant="outline"
   490→                            className="h-6 w-6 p-0"
   491→                            onClick={() => updateQty(idx, -1)}
   492→                          >
   493→                            <Minus className="w-3 h-3" />
   494→                          </Button>
   495→                          <span className="text-xs font-semibold w-6 text-center">{c.qty}</span>
   496→                          <Button
   497→                            size="sm"
   498→                            variant="outline"
   499→                            className="h-6 w-6 p-0"
   500→                            onClick={() => updateQty(idx, 1)}
   501→                          >
   502→                            <Plus className="w-3 h-3" />
   503→                          </Button>
   504→                        </div>
   505→                      </div>
   506→                    )
   507→                  })}
   508→                </div>
   509→              )}
   510→
   511→              {cart.length > 0 && (
   512→                <div className="space-y-2 mt-3 pt-3 border-t">
   513→                  <Input
   514→                    value={orderNotes}
   515→                    onChange={(e) => setOrderNotes(e.target.value)}
   516→                    placeholder="Sipariş notu (genel)"
   517→                    className="h-8 text-xs"
   518→                  />
   519→                  <div className="flex items-center justify-between">
