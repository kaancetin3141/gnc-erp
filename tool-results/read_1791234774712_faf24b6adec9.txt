   696→// IRSALİYE (DISPATCH NOTE) — Sevk irsaliyesi, ağırlık takibi, palet
   697→// ============================================================
   698→// Numara formatı: IRS-2026-001, IRS-2026-002 (yıl + tenant bazlı sıra)
   699→// Durum: taslak | hazir | sevk_edildi | teslim_edildi | iptal
   700→// Ağırlık hesabı:
   701→//   totalNetWeight       = sum(line.weightPerUnit * line.qty)
   702→//   totalPackagingWeight = sum(product.packagingWeight * line.qty) (opsiyonel)
   703→//   palletWeight         = kullanıcı girişi (varsayılan 20 kg)
   704→//   totalGrossWeight     = net + packaging + pallet
   705→model Irsaliye {
   706→  id                 String   @id @default(cuid())
   707→  tenantId           String
   708→  tenant             Tenant   @relation(fields: [tenantId], references: [id], onDelete: Cascade)
   709→  orderId            String?
   710→  order              Order?   @relation(fields: [orderId], references: [id], onDelete: SetNull)
   711→  customerId         String
   712→  customer           Customer @relation(fields: [customerId], references: [id])
   713→  number             String   // IRS-2026-001
   714→  date               DateTime @default(now())
   715→  status             String   @default("taslak") // taslak | hazir | sevk_edildi | teslim_edildi | iptal
   716→  // Ağırlık takibi
   717→  totalNetWeight         Float?   // sum of line.weightPerUnit * line.qty
   718→  totalPackagingWeight    Float?   // sum of product.packagingWeight * line.qty
   719→  palletWeight            Float?   // user-entered pallet weight (default 20 kg)
   720→  totalGrossWeight        Float?   // net + packaging + pallet
   721→  palletCount             Int?
   722→  palletType              String?  // tahta | plastik | tek_kullanilik | eur_pallet | other
   723→  // Sevkiyat bilgileri
   724→  shippingAddress         String?
   725→  carrier                 String?  // yurtici | aras | mng | ptt | ups | fedex | dhl | other
   726→  trackingNo              String?
   727→  // Notlar
   728→  notes                   String?
   729→  createdById             String?
   730→  createdAt               DateTime @default(now())
   731→  updatedAt               DateTime @updatedAt
   732→  deliveredAt             DateTime?
   733→
   734→  lines IrsaliyeLine[]
   735→
   736→  @@unique([tenantId, number])
   737→  @@index([tenantId, status])
   738→}
