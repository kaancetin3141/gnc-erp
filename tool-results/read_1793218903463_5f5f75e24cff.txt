    98→function agingLabel(days: number | null): { text: string; cls: string } {
    99→  if (days == null) return { text: 'Yok', cls: 'text-slate-500 bg-slate-100 dark:bg-slate-900/50' }
   100→  if (days <= 30) return { text: `${days} gün`, cls: 'text-emerald-700 bg-emerald-100 dark:bg-emerald-950/40' }
   101→  if (days <= 60) return { text: `${days} gün`, cls: 'text-amber-700 bg-amber-100 dark:bg-amber-950/40' }
   102→  if (days <= 90) return { text: `${days} gün`, cls: 'text-orange-700 bg-orange-100 dark:bg-orange-950/40' }
   103→  return { text: `${days} gün`, cls: 'text-red-700 bg-red-100 dark:bg-red-950/40' }
   104→}

   338→function CustomerFormDialog({ open, onOpenChange, marketId, editCustomer }: {
   339→  open: boolean; onOpenChange: (v: boolean) => void; marketId: string
   340→  editCustomer: { id: string; name: string; phone: string | null; note: string | null; creditLimit: number | null } | null
   341→}) {
   342→  const qc = useQueryClient()
   343→  const [saving, setSaving] = useState(false)
   344→  const [form, setForm] = useState({ name: '', phone: '', note: '', creditLimit: '' })
   345→  const [initializedFor, setInitializedFor] = useState<string | null>(null)
   346→
   347→  // Edit moduna geçildiğinde formu doldur
   348→  if (open && editCustomer && initializedFor !== editCustomer.id) {
   349→    setForm({
   350→      name: editCustomer.name,
   351→      phone: editCustomer.phone ?? '',
   352→      note: editCustomer.note ?? '',
   353→      creditLimit: editCustomer.creditLimit != null ? String(editCustomer.creditLimit) : '',
   354→    })
   355→    setInitializedFor(editCustomer.id)
   356→  }
   357→  if (!open && initializedFor) setInitializedFor(null)
   358→
   359→  const handleSave = async () => {
   360→    if (!form.name.trim()) { toast.error('Müşteri adı gerekli'); return }
   361→    setSaving(true)
   362→    try {
   363→      if (editCustomer) {
   364→        await apiPatch(`/api/market/${marketId}/credit/${editCustomer.id}`, {
   365→          name: form.name,
   366→          phone: form.phone || null,
   367→          note: form.note || null,
   368→          creditLimit: form.creditLimit ? Number(form.creditLimit) : null,
   369→        })
   370→        toast.success('Müşteri güncellendi')
   371→      } else {
   372→        await apiPost(`/api/market/${marketId}/credit`, {
   373→          name: form.name,
   374→          phone: form.phone || undefined,
   375→          note: form.note || undefined,
   376→          creditLimit: form.creditLimit ? Number(form.creditLimit) : undefined,
   377→        })
   378→        toast.success('Veresiye müşterisi eklendi')
   379→      }
   380→      void qc.invalidateQueries({ queryKey: ['market-credit'] })
   381→      void qc.invalidateQueries({ queryKey: ['market-credit-detail'] })
   382→      onOpenChange(false)
   383→      setForm({ name: '', phone: '', note: '', creditLimit: '' })
   384→    } catch (e) {
   385→      toast.error(e instanceof Error ? e.message : 'Kaydedilemedi')
   386→    } finally {
   387→      setSaving(false)
   388→    }
   389→  }
   390→
   391→  return (
   392→    <Dialog open={open} onOpenChange={onOpenChange}>
   393→      <DialogContent className="sm:max-w-md">
   394→        <DialogHeader>
   395→          <DialogTitle>{editCustomer ? 'Müşteriyi Düzenle' : 'Veresiye Müşterisi Ekle'}</DialogTitle>
   396→          <DialogDescription>{editCustomer ? 'Bilgileri güncelleyin' : 'Borç defterine yeni kişi kaydet — POS&apos;ta veresiye ödemesi bu kişiye bağlanır'}</DialogDescription>
   397→        </DialogHeader>
   398→        <div className="space-y-3">
   399→          <div>
   400→            <Label className="text-xs">Ad Soyad / İşletme *</Label>
   401→            <Input value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} placeholder="Örn: Kahveci Mehmet" className="mt-1" />
   402→          </div>
   403→          <div>
   404→            <Label className="text-xs">Telefon (hatırlatma için)</Label>
   405→            <Input value={form.phone} onChange={(e) => setForm({ ...form, phone: e.target.value })} placeholder="0532..." className="mt-1" />
   406→          </div>
   407→          <div className="grid grid-cols-2 gap-3">
   408→            <div>
   409→              <Label className="text-xs">Kredi Limiti (₺)</Label>
   410→              <Input type="number" min={0} value={form.creditLimit} onChange={(e) => setForm({ ...form, creditLimit: e.target.value })} placeholder="Boş = limitsiz" className="mt-1" />
   411→            </div>
   412→            <div>
   413→              <Label className="text-xs">Not</Label>
   414→              <Input value={form.note} onChange={(e) => setForm({ ...form, note: e.target.value })} placeholder="Mahalle, iş yeri..." className="mt-1" />
   415→            </div>
   416→          </div>
   417→        </div>
   418→        <DialogFooter>
   419→          <Button variant="outline" onClick={() => onOpenChange(false)}>İptal</Button>
   420→          <Button className="bg-violet-600 hover:bg-violet-700 text-white" onClick={handleSave} disabled={saving || !form.name.trim()}>
   421→            {saving ? 'Kaydediliyor...' : editCustomer ? 'Güncelle' : 'Ekle'}
   422→          </Button>
   423→        </DialogFooter>
   424→      </DialogContent>
   425→    </Dialog>
   426→  )
   427→}

   445→  const { data: detail, isLoading } = useQuery<CustomerDetail>({
   446→    queryKey: ['market-credit-detail', marketId, customerId],
   447→    queryFn: () => apiGet(`/api/market/${marketId}/credit/${customerId}`),
   448→  })
   449→
   450→  const invalidate = () => {
   451→    void qc.invalidateQueries({ queryKey: ['market-credit-detail'] })
   452→    void qc.invalidateQueries({ queryKey: ['market-credit'] })
   453→  }
   454→
   455→  const addEntry = async () => {
   456→    const amt = Number(amount)
   457→    if (!amt || amt <= 0) { toast.error('Tutar girin'); return }
   458→    if (entryType === 'borc' && dueDate && isNaN(new Date(dueDate).getTime())) { toast.error('Geçersiz vade tarihi'); return }
   459→    setSaving(true)
   460→    try {
   461→      await apiPost(`/api/market/${marketId}/credit/${customerId}`, {
   462→        type: entryType,
   463→        amount: amt,
   464→        method: entryType === 'odeme' ? method : undefined,
   465→        dueDate: entryType === 'borc' && dueDate ? new Date(dueDate).toISOString() : undefined,
   466→        note: note || undefined,
   467→      })
   468→      toast.success(entryType === 'odeme' ? 'Ödeme kaydedildi' : 'Borç kaydedildi')
   469→      setAmount(''); setNote(''); setDueDate('')
   470→      invalidate()
   471→    } catch (e) {
   472→      toast.error(e instanceof Error ? e.message : 'Kaydedilemedi')
   473→    } finally {
   474→      setSaving(false)
   475→    }
   476→  }
   477→
   478→  const toggleActive = async () => {
   479→    if (!detail) return
   480→    try {
   481→      await apiPatch(`/api/market/${marketId}/credit/${customerId}`, { isActive: !detail.isActive })
   482→      toast.success(detail.isActive ? 'Müşteri pasifleştirildi' : 'Müşteri aktifleştirildi')
   483→      invalidate()
   484→    } catch (e) {
   485→      toast.error(e instanceof Error ? e.message : 'Güncellenemedi')
   486→    }
   487→  }
   488→
   489→  // GERÇEK PDF — cari ekstresini sunucuda pdf-lib ile üretip indir
   490→  const [pdfBusy, setPdfBusy] = useState(false)
   491→  const downloadEkstre = async () => {
   492→    setPdfBusy(true)
   493→    try {
   494→      const name = ((detail as { name?: string } | undefined)?.name ?? 'musteri').replace(/[^\w.-]+/g, '-')
   495→      await downloadFile(`/api/market/${marketId}/credit/${customerId}/pdf`, `Veresiye-Ekstre-${name}.pdf`)
   496→      toast.success('Cari ekstresi PDF indirildi', { description: 'Tüm borç/ödeme hareketleri + bakiye' })
   497→    } catch (e) {
   498→      toast.error(e instanceof Error ? e.message : 'PDF indirilemedi')
   499→    } finally {
   500→      setPdfBusy(false)
   501→    }
   502→  }
   503→
   504→  const detailCustomer = detail as (CustomerDetail & { name?: string; phone?: string | null; note?: string | null; creditLimit?: number | null; isActive?: boolean; balance?: number }) | undefined
   505→
   506→  const reminderText = detailCustomer
   507→    ? `Merhaba ${detailCustomer.name}, defterimize göre ${formatCurrency(detailCustomer.balance ?? 0)} borcunuz bulunmaktadır. Uygun olduğunda geçirebilirseniz memnun oluruz. Teşekkürler 🙏`
   508→    : ''
   509→
   510→  return (
   511→    <Sheet open onOpenChange={(v) => { if (!v) onClose() }}>
   512→      <SheetContent className="w-full sm:max-w-lg overflow-y-auto custom-scroll p-0">
   513→        <SheetHeader className="p-4 pb-2">
   514→          <SheetTitle className="flex items-center gap-2">
   515→            <UserRound className="w-5 h-5 text-violet-600" />
   516→            {isLoading ? 'Yükleniyor...' : detailCustomer?.name}
   517→          </SheetTitle>
   518→          {detailCustomer && (
   519→            <div className="flex flex-wrap items-center gap-2 text-xs text-muted-foreground mt-1">
   520→              <span className="font-semibold text-base text-foreground">
   521→                Bakiye: <span className={cn((detailCustomer.balance ?? 0) > 0 ? 'text-amber-700 dark:text-amber-400' : (detailCustomer.balance ?? 0) < 0 ? 'text-emerald-700 dark:text-emerald-400' : '')}>
   522→                  {formatCurrency(detailCustomer.balance ?? 0)}
   523→                </span>
   524→              </span>
   525→              {detailCustomer.creditLimit != null && <span>· Limit {formatCurrency(detailCustomer.creditLimit)}</span>}
   526→              {detailCustomer.phone && (
   527→                <>
   528→                  <span>· <a className="underline" href={`tel:${detailCustomer.phone}`}>{detailCustomer.phone}</a></span>
   529→                  <a href={whatsappLink(detailCustomer.phone, reminderText)} target="_blank" rel="noopener noreferrer"
   530→                    className="inline-flex items-center gap-1 text-emerald-700 dark:text-emerald-400 hover:underline">
   531→                    <MessageCircle className="w-3.5 h-3.5" /> WhatsApp hatırlat
   532→                  </a>
   533→                </>
   534→              )}
   535→              <button className="inline-flex items-center gap-1 underline hover:text-foreground disabled:opacity-50" onClick={() => void downloadEkstre()} disabled={pdfBusy || isLoading}>
   536→                {pdfBusy ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <FileDown className="w-3.5 h-3.5" />} Ekstre PDF
   537→              </button>
   538→              <button className="underline hover:text-foreground" onClick={() => onEdit({ id: customerId, name: detailCustomer.name ?? '', phone: detailCustomer.phone ?? null, note: detailCustomer.note ?? null, creditLimit: detailCustomer.creditLimit ?? null })}>düzenle</button>
   539→              <button className={cn('underline hover:text-foreground', detailCustomer.isActive && 'text-red-600')} onClick={() => void toggleActive()}>
   540→                {detailCustomer.isActive ? 'pasifleştir' : 'aktifleştir'}
   541→              </button>
   542→            </div>
   543→          )}
   544→        </SheetHeader>
   545→
   546→        <div className="px-4 space-y-4 pb-8">
   547→          {/* Hızlı hareket girişi */}
   548→          <Card className="border shadow-sm">
   549→            <CardContent className="p-3 space-y-2.5">
   550→              <div className="flex gap-1.5">
   551→                <Button size="sm" variant={entryType === 'odeme' ? 'default' : 'outline'}
   552→                  className={cn('flex-1 h-9', entryType === 'odeme' && 'bg-emerald-600 hover:bg-emerald-700 text-white')}
   553→                  onClick={() => setEntryType('odeme')}>
   554→                  <ArrowDownCircle className="w-4 h-4 mr-1.5" /> Ödeme Aldı
   555→                </Button>
   556→                <Button size="sm" variant={entryType === 'borc' ? 'default' : 'outline'}
   557→                  className={cn('flex-1 h-9', entryType === 'borc' && 'bg-amber-600 hover:bg-amber-700 text-white')}
   558→                  onClick={() => setEntryType('borc')}>
   559→                  <ArrowUpCircle className="w-4 h-4 mr-1.5" /> Borç Yaz
   560→                </Button>
   561→              </div>
   562→              <div className="grid grid-cols-2 gap-2">
   563→                <div>
   564→                  <Label className="text-xs">Tutar (₺) *</Label>
   565→                  <Input type="number" min={0} step="0.01" value={amount} onChange={(e) => setAmount(e.target.value)} className="mt-1" placeholder="0" />
   566→                </div>
   567→                {entryType === 'odeme' ? (
   568→                  <div>
   569→                    <Label className="text-xs">Yöntem</Label>
   570→                    <Select value={method} onValueChange={setMethod}>
   571→                      <SelectTrigger className="mt-1"><SelectValue /></SelectTrigger>
   572→                      <SelectContent>
   573→                        <SelectItem value="nakit">Nakit</SelectItem>
   574→                        <SelectItem value="kart">Kart</SelectItem>
   575→                        <SelectItem value="havale">Havale/EFT</SelectItem>
   576→                      </SelectContent>
   577→                    </Select>
   578→                  </div>
   579→                ) : (
   580→                  <div>
   581→                    <Label className="text-xs">Vade (opsiyonel)</Label>
   582→                    <Input type="date" value={dueDate} onChange={(e) => setDueDate(e.target.value)} className="mt-1" />
   583→                  </div>
   584→                )}
   585→              </div>
   586→              <Input value={note} onChange={(e) => setNote(e.target.value)} placeholder="Not (opsiyonel)" className="h-9" />
   587→              <Button className="w-full h-9 bg-violet-600 hover:bg-violet-700 text-white" onClick={() => void addEntry()} disabled={saving || !amount}>
   588→                {saving ? 'Kaydediliyor...' : entryType === 'odeme' ? 'Ödemeyi Kaydet' : 'Borcu Kaydet'}
   589→              </Button>
   590→            </CardContent>
   591→          </Card>
   592→
   593→          {/* Hareket defteri */}
   594→          <div>
   595→            <div className="text-sm font-medium mb-2 flex items-center gap-1.5">
   596→              <Receipt className="w-4 h-4 text-violet-600" /> Hareket Defteri
   597→            </div>
   598→            {isLoading ? (
   599→              <div className="space-y-2">{[1, 2, 3].map((i) => <Skeleton key={i} className="h-12 w-full rounded-lg" />)}</div>
   600→            ) : !detailCustomer?.entries?.length ? (
   601→              <Card className="border-dashed"><CardContent className="py-6 text-center text-sm text-muted-foreground">Henüz hareket yok</CardContent></Card>
   602→            ) : (
   603→              <div className="space-y-1.5 max-h-80 overflow-y-auto custom-scroll pr-1">
   604→                {detailCustomer.entries.map((e) => (
   605→                  <div key={e.id} className={cn('flex items-center gap-2.5 rounded-lg border px-3 py-2',
   606→                    e.type === 'borc' ? 'border-amber-200 bg-amber-50/50 dark:border-amber-900/60 dark:bg-amber-950/20' : 'border-emerald-200 bg-emerald-50/50 dark:border-emerald-900/60 dark:bg-emerald-950/20')}>
   607→                    {e.type === 'borc'
   608→                      ? <ArrowUpCircle className="w-4 h-4 text-amber-600 shrink-0" />
   609→                      : <ArrowDownCircle className="w-4 h-4 text-emerald-600 shrink-0" />}
   610→                    <div className="flex-1 min-w-0">
   611→                      <div className="flex items-center gap-1.5 text-xs font-medium">
   612→                        {e.type === 'borc' ? 'Borç' : `Ödeme${e.method ? ` · ${METHOD_LABELS[e.method] ?? e.method}` : ''}`}
   613→                        {e.refSaleNumber && (
   614→                          <Badge variant="outline" className="text-[9px] px-1 py-0">POS {e.refSaleNumber}</Badge>
   615→                        )}
   616→                      </div>
   617→                      <div className="text-[10px] text-muted-foreground">
   618→                        {formatDateTime(e.createdAt)}
   619→                        {e.dueDate && ` · vade ${formatDateTime(e.dueDate)}`}
   620→                        {e.note && ` · ${e.note}`}
   621→                      </div>
   622→                    </div>
   623→                    <span className={cn('text-sm font-bold tabular-nums shrink-0', e.type === 'borc' ? 'text-amber-700 dark:text-amber-400' : 'text-emerald-700 dark:text-emerald-400')}>
   624→                      {e.type === 'borc' ? '+' : '−'}{formatCurrency(e.amount)}
   625→                    </span>
   626→                  </div>
   627→                ))}
   628→              </div>
   629→            )}
   630→          </div>
   631→
   632→          {/* İpucu */}
   633→          <div className="flex items-start gap-2 rounded-lg bg-muted/50 p-3 text-[11px] text-muted-foreground">
   634→            <HandCoins className="w-4 h-4 shrink-0 mt-0.5 text-amber-600" />
   635→            POS&apos;ta ödeme adımında &quot;Veresiye&quot; seçilerek yapılan satışlar bu deftere otomatik borç olarak işlenir.
   636→          </div>
   637→        </div>
   638→      </SheetContent>
   639→    </Sheet>
   640→  )
   641→}
