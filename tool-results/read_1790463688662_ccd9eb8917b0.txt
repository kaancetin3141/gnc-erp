    48→interface OrderDoc {
    49→  id: string
    50→  number: string
    51→  status: string
    52→  orderDate: string
    53→  totalAmount?: number
    54→  currency?: string
    55→  customer?: { id: string; name: string; segment?: string; status?: string } | null
    56→  quote?: { id: string; number: string } | null
    57→  invoice?: { id: string; number: string; status: string; packingListNo?: string | null } | null
    58→  irsaliyeler?: { id: string; number: string; status: string }[]
    59→  _count?: { trackingSteps: number; productionItems?: number }
    60→}
    61→
    62→interface OrdersResponse {
    63→  items: OrderDoc[]
    64→  total: number
    65→  hidePrices?: boolean
    66→}

    77→// Fatura durumu etiketi (CSV için)
    78→function ORDER_DOC_LABEL(status: string): string {
    79→  switch (status) {
    80→    case 'odeme_bekliyor': return 'Ödeme Bekliyor'
    81→    case 'odendi': return 'Ödendi'
    82→    case 'gecikti': return 'Gecikti'
    83→    case 'iptal': return 'İptal'
    84→    default: return status
    85→  }
    86→}

    93→export function DocumentsView() {
    94→  const qc = useQueryClient()
    95→  const { user } = useAppStore()
    96→  const su = user as SessionUser | null
    97→
    98→  // Yetkiler
    99→  const canSeeInvoice = hasPermission(su, 'invoices.view') || hasPermission(su, 'erp.manage')
   100→  const canSeeIrsaliye = hasPermission(su, 'irsaliye.view')
   101→  const canSeePacking = canSeeIrsaliye // çeki listesi = irsaliye.view sahiplerine açık (müdür + depocu)
   102→
   103→  const [search, setSearch] = useState('')
   104→  const [openCompanies, setOpenCompanies] = useState<Record<string, boolean>>({})
   105→  const [generating, setGenerating] = useState<string | null>(null) // "orderId:type"
   106→  const [activeDoc, setActiveDoc] = useState<GeneratedDoc | null>(null)
   107→  const [combinedOrder, setCombinedOrder] = useState<{ id: string; number: string; customerName?: string } | null>(null)
   108→  const [companyPrint, setCompanyPrint] = useState<{ name: string; orders: { id: string; number: string }[] } | null>(null)
   109→
   110→  const canExport = hasPermission(su, 'export.data')
   111→
   112→  const { data, isLoading, isFetching, refetch } = useQuery({
   113→    queryKey: ['documents-orders', search],
   114→    queryFn: () => {
   115→      const p = new URLSearchParams({ limit: '200' })
   116→      if (search) p.set('search', search)
   117→      return apiGet<OrdersResponse>(`/api/orders?${p.toString()}`)
   118→    },
   119→  })
   120→
   121→  const orders = useMemo(() => data?.items ?? [], [data])
   122→  const hidePrices = data?.hidePrices === true
   123→
   124→  // Şirketlere göre grupla
   125→  const companies = useMemo(() => {
   126→    const map = new Map<string, { name: string; segment?: string; orders: OrderDoc[] }>()
   127→    for (const o of orders) {
   128→      const key = o.customer?.id ?? '_'
   129→      if (!map.has(key)) {
   130→        map.set(key, {
   131→          name: o.customer?.name ?? 'Bilinmeyen Şirket',
   132→          segment: o.customer?.segment,
   133→          orders: [],
   134→        })
   135→      }
   136→      map.get(key)!.orders.push(o)
   137→    }
   138→    // Şirket adına göre sırala
   139→    return Array.from(map.values()).sort((a, b) => a.name.localeCompare(b.name, 'tr'))
   140→  }, [orders])
   141→
   142→  // Özet
   143→  const stats = useMemo(() => {
   144→    const invoiced = orders.filter((o) => o.invoice).length
   145→    const irsaliye = orders.filter((o) => (o.irsaliyeler?.length ?? 0) > 0).length
   146→    const packing = orders.filter((o) => o.invoice?.packingListNo).length
   147→    return { orders: data?.total ?? 0, companies: companies.length, invoiced, irsaliye, packing }
   148→  }, [orders, companies, data])
   149→
   150→  // Belge üret + PDF aç
   151→  const generateDoc = async (order: OrderDoc, type: 'invoice' | 'irsaliye' | 'packing_list') => {
   152→    const genKey = `${order.id}:${type}`
   153→    setGenerating(genKey)
   154→    try {
   155→      const res = await apiPost<{
   156→        type: string
   157→        created: boolean
   158→        document: {
   159→          id: string
   160→          number: string
   161→          packingListNo?: string | null
   162→        }
   163→        documentType: 'invoice' | 'irsaliye'
   164→      }>(`/api/orders/${order.id}/generate-document`, { type })
   165→
   166→      if (res.created) {
   167→        toast.success(`Belge otomatik oluşturuldu${res.document.number ? `: ${res.document.number}` : ''}`)
   168→      }
   169→
   170→      if (type === 'irsaliye') {
   171→        setActiveDoc({ kind: 'irsaliye', irsaliyeId: res.document.id, irsaliyeNumber: res.document.number })
   172→      } else if (type === 'packing_list') {
   173→        setActiveDoc({ kind: 'packing', invoiceId: res.document.id })
   174→      } else {
   175→        setActiveDoc({ kind: 'invoice', invoiceId: res.document.id })
   176→      }
   177→
   178→      // Belge durumlarını tazele
   179→      qc.invalidateQueries({ queryKey: ['documents-orders'] })
   180→      qc.invalidateQueries({ queryKey: ['orders'] })
   181→      qc.invalidateQueries({ queryKey: ['irsaliye'] })
   182→      qc.invalidateQueries({ queryKey: ['invoices'] })
   183→    } catch (e) {
   184→      toast.error(e instanceof Error ? e.message : 'Belge oluşturulamadı')
   185→    } finally {
   186→      setGenerating(null)
   187→    }
   188→  }
   189→
   190→  const toggleCompany = (name: string) => {
   191→    setOpenCompanies((prev) => ({ ...prev, [name]: !prev[name] }))
   192→  }
   193→
   194→  // CSV dışa aktarma — siparişler + belge durumları
   195→  const handleExportCSV = () => {
   196→    if (orders.length === 0) {
   197→      toast.error('Dışa aktarılacak sipariş yok')
   198→      return
   199→    }
   200→    const rows = orders.map((o) => ({
   201→      'Sipariş No': o.number,
   202→      'Şirket': o.customer?.name ?? '',
   203→      'Tarih': formatDate(o.orderDate),
   204→      'Durum': ORDER_STATUS_META[o.status]?.label ?? o.status,
   205→      'Fatura No': o.invoice?.number ?? '',
   206→      'Fatura Durumu': o.invoice ? ORDER_DOC_LABEL(o.invoice.status) : 'Yok',
   207→      'İrsaliye No': o.irsaliyeler?.[0]?.number ?? '',
   208→      'İrsaliye Durumu': o.irsaliyeler?.[0] ? ORDER_STATUS_META[o.irsaliyeler[0].status]?.label ?? o.irsaliyeler[0].status : 'Yok',
   209→      'Çeki Listesi No': o.invoice?.packingListNo ?? '',
   210→      ...(hidePrices ? {} : { 'Tutar': o.totalAmount ?? 0, 'Para': o.currency ?? 'TRY' }),
   211→    }))
   212→    const csv = toCSV(rows)
   213→    downloadFile(csv, `belge-yonetimi-${new Date().toISOString().slice(0, 10)}.csv`)
   214→    toast.success(`${orders.length} sipariş dışa aktarıldı`)
   215→  }
   216→
   217→  return (
   218→    <div className="space-y-5 animate-fade-in">
   219→      {/* Header */}
   220→      <div className="flex items-start justify-between flex-wrap gap-3">
   221→        <div>
   222→          <h2 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   223→            <FileStack className="w-6 h-6 text-violet-600" />
   224→            Belge Yönetimi
   225→          </h2>
   226→          <p className="text-sm text-muted-foreground mt-0.5">
   227→            Şirketler ve sipariş belgeleri — fatura, irsaliye ve çeki listesi otomatik üretilir
   228→            {canSeeIrsaliye && !canSeeInvoice && (
   229→              <span className="ml-2 text-violet-600 font-medium">(görüş alanınız: İrsaliye + Çeki Listesi)</span>
   230→            )}
   231→          </p>
   232→        </div>
   233→        <div className="flex gap-2 flex-wrap">
   234→          {canExport && (
   235→            <Button variant="outline" size="sm" onClick={handleExportCSV} disabled={orders.length === 0}>
   236→              <Download className="w-4 h-4 mr-1.5" />
   237→              CSV Dışa Aktar
   238→            </Button>
   239→          )}
   240→          <Button variant="outline" size="sm" onClick={() => refetch()} disabled={isFetching}>
   241→            <RefreshCw className={cn('w-4 h-4 mr-1.5', isFetching && 'animate-spin')} />
   242→            Yenile
   243→          </Button>
   244→        </div>
   245→      </div>
   246→
   247→      <Tabs defaultValue="orders" className="space-y-4">
   248→        <TabsList className="grid w-full max-w-md grid-cols-2">
   249→          <TabsTrigger value="orders" className="gap-1.5">
   250→            <Package className="w-3.5 h-3.5" /> Sipariş Belgeleri
   251→          </TabsTrigger>
   252→          <TabsTrigger value="irsaliye" className="gap-1.5">
   253→            <Truck className="w-3.5 h-3.5" /> İrsaliye Listesi
   254→          </TabsTrigger>
   255→        </TabsList>
   256→
   257→        {/* ==================== TAB 1: SİPARİŞ BELGELERİ ==================== */}
   258→        <TabsContent value="orders" className="space-y-4">
   259→          {/* Stats */}
   260→          <div className="grid grid-cols-2 sm:grid-cols-5 gap-3">
   261→            <Card>
   262→              <CardContent className="p-3">
   263→                <div className="text-[10px] uppercase tracking-wider text-muted-foreground">Sipariş</div>
   264→                <div className="text-xl font-bold">{stats.orders}</div>
   265→              </CardContent>
   266→            </Card>
   267→            <Card>
   268→              <CardContent className="p-3">
   269→                <div className="text-[10px] uppercase tracking-wider text-muted-foreground">Şirket</div>
   270→                <div className="text-xl font-bold">{stats.companies}</div>
   271→              </CardContent>
   272→            </Card>
   273→            {canSeeInvoice && (
   274→              <Card className="bg-emerald-50/50 dark:bg-emerald-950/20">
   275→                <CardContent className="p-3">
   276→                  <div className="text-[10px] uppercase tracking-wider text-emerald-700 dark:text-emerald-400">Faturalı</div>
   277→                  <div className="text-xl font-bold text-emerald-700 dark:text-emerald-400">{stats.invoiced}</div>
   278→                </CardContent>
   279→              </Card>
   280→            )}
   281→            {canSeeIrsaliye && (
   282→              <Card className="bg-violet-50/50 dark:bg-violet-950/20">
   283→                <CardContent className="p-3">
   284→                  <div className="text-[10px] uppercase tracking-wider text-violet-700 dark:text-violet-400">İrsaliyeli</div>
   285→                  <div className="text-xl font-bold text-violet-700 dark:text-violet-400">{stats.irsaliye}</div>
   286→                </CardContent>
   287→              </Card>
   288→            )}
   289→            {canSeePacking && (
   290→              <Card className="bg-amber-50/50 dark:bg-amber-950/20">
   291→                <CardContent className="p-3">
   292→                  <div className="text-[10px] uppercase tracking-wider text-amber-700 dark:text-amber-400">Çeki Listeli</div>
   293→                  <div className="text-xl font-bold text-amber-700 dark:text-amber-400">{stats.packing}</div>
   294→                </CardContent>
   295→              </Card>
   296→            )}
   297→          </div>
   298→
   299→          {/* Arama */}
   300→          <Card>
   301→            <CardContent className="p-4">
   302→              <div className="relative">
   303→                <Search className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground pointer-events-none" />
   304→                <Input
   305→                  value={search}
   306→                  onChange={(e) => setSearch(e.target.value)}
   307→                  placeholder="Şirket veya sipariş no ara..."
   308→                  className="pl-9"
   309→                />
   310→                {search && (
   311→                  <button
   312→                    onClick={() => setSearch('')}
   313→                    className="absolute right-2 top-1/2 -translate-y-1/2 p-1 hover:bg-muted rounded"
   314→                  >
   315→                    <X className="w-3.5 h-3.5 text-muted-foreground" />
   316→                  </button>
   317→                )}
   318→              </div>
   319→            </CardContent>
   320→          </Card>
   321→
   322→          {/* Şirketler → Siparişler */}
   323→          {isLoading ? (
   324→            <div className="space-y-3">
   325→              <Skeleton className="h-16 w-full" />
   326→              <Skeleton className="h-40 w-full" />
   327→              <Skeleton className="h-16 w-full" />
   328→            </div>
   329→          ) : companies.length === 0 ? (
   330→            <Card>
   331→              <CardContent className="p-12 text-center">
   332→                <Building2 className="w-10 h-10 mx-auto text-muted-foreground/50 mb-2" />
   333→                <p className="text-sm text-muted-foreground">Sipariş bulunamadı</p>
   334→              </CardContent>
   335→            </Card>
   336→          ) : (
   337→            <div className="space-y-3">
   338→              {companies.map((company) => {
   339→                const isOpen = openCompanies[company.name] ?? true
   340→                return (
   341→                  <Card key={company.name} className="overflow-hidden">
   342→                    {/* Şirket başlığı — tıklanınca açılır/kapanır, sağdaki butonlar tıklamayı yutmaz */}
   343→                    <div
   344→                      role="button"
   345→                      tabIndex={0}
   346→                      aria-expanded={isOpen}
   347→                      onClick={() => toggleCompany(company.name)}
   348→                      onKeyDown={(e) => {
   349→                        if (e.key === 'Enter' || e.key === ' ') {
   350→                          e.preventDefault()
   351→                          toggleCompany(company.name)
   352→                        }
   353→                      }}
   354→                      className="w-full flex items-center gap-3 p-4 bg-gradient-to-r from-muted/50 to-muted/20 hover:from-muted/70 hover:to-muted/30 transition-all text-left cursor-pointer select-none focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-violet-400"
   355→                    >
   356→                      {isOpen ? (
   357→                        <ChevronDown className="w-4 h-4 text-muted-foreground shrink-0" />
   358→                      ) : (
   359→                        <ChevronRight className="w-4 h-4 text-muted-foreground shrink-0" />
   360→                      )}

   362→                        <Building2 className="w-4 h-4 text-slate-600 dark:text-slate-300" />

   364→                      <div className="min-w-0 flex-1">
   365→                        <div className="font-semibold text-sm truncate">{company.name}</div>

   367→                          {company.orders.length} sipariş
   368→                          {company.segment ? ` · ${company.segment}` : ''}

   372→                      {(() => {
   373→                        const inv = company.orders.filter((o) => o.invoice).length
   374→                        const irs = company.orders.filter((o) => (o.irsaliyeler?.length ?? 0) > 0).length
   375→                        const pack = company.orders.filter((o) => o.invoice?.packingListNo).length
   376→                        return (
   377→                          <div className="hidden sm:flex items-center gap-1 shrink-0">
   378→                            {canSeeInvoice && inv > 0 && (
   379→                              <Badge variant="outline" className="text-[9px] bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-950/30 dark:text-emerald-300 dark:border-emerald-900/60 gap-0.5">
   380→                                <Receipt className="w-2.5 h-2.5" /> {inv}
   381→                              </Badge>
   382→                            )}
   383→                            {canSeeIrsaliye && irs > 0 && (
   384→                              <Badge variant="outline" className="text-[9px] bg-violet-50 text-violet-700 border-violet-200 dark:bg-violet-950/30 dark:text-violet-300 dark:border-violet-900/60 gap-0.5">
   385→                                <Truck className="w-2.5 h-2.5" /> {irs}
   386→                              </Badge>
   387→                            )}
   388→                            {canSeePacking && pack > 0 && (
   389→                              <Badge variant="outline" className="text-[9px] bg-amber-50 text-amber-700 border-amber-200 dark:bg-amber-950/30 dark:text-amber-300 dark:border-amber-900/60 gap-0.5">
   390→                                <ClipboardList className="w-2.5 h-2.5" /> {pack}
   391→                              </Badge>
   392→                            )}
   393→                          </div>
   394→                        )
   395→                      })()}
   396→                      <Badge variant="outline" className="text-[10px] shrink-0">
   397→                        {company.orders.length}
   398→                      </Badge>
   399→
   400→                      {/* ŞİRKET TOPLU BELGE YAZDIRMA — tüm siparişler tek yazdırmada */}
   401→                      {(canSeeInvoice || canSeeIrsaliye) && (
   402→                        <Tooltip>
   403→                          <TooltipTrigger asChild>
   404→                            <Button
   405→                              variant="outline"
   406→                              size="sm"
   407→                              className="h-7 text-[11px] gap-1 shrink-0 text-violet-700 border-violet-200 hover:bg-violet-50 dark:text-violet-300 dark:border-violet-900/50"
   408→                              onClick={(e) => {
   409→                                e.stopPropagation()

   411→                                  name: company.name,
   412→                                  orders: company.orders.map((o) => ({ id: o.id, number: o.number })),
   413→                                })

   415→                            >
   416→                              <Printer className="w-3 h-3" />
   417→                              <span className="hidden md:inline">Tümünü Yazdır</span>
   418→                            </Button>
   419→                          </TooltipTrigger>
   420→                          <TooltipContent>
   421→                            Şirketin tüm sipariş belgelerini tek yazdırmada birleştir
   422→                          </TooltipContent>
   423→                        </Tooltip>
   424→                      )}
   425→                    </div>
   426→
   427→                    {/* Sipariş listesi */}
   428→                    {isOpen && (
   429→                      <div className="overflow-x-auto border-t">
   430→                        <Table>
   431→                          <TableHeader>
   432→                            <TableRow className="bg-muted/20 hover:bg-muted/20">
   433→                              <TableHead className="pl-4 min-w-[120px]">Sipariş No</TableHead>
   434→                              <TableHead className="min-w-[100px]">Tarih</TableHead>
