     9→import { useState } from 'react'
    10→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
    11→import { apiGet, apiPatch, apiPost } from '@/lib/api-client'
    12→import { toast } from 'sonner'
    13→import { cn } from '@/lib/utils'
    14→import { ROLE_LABELS, isSuperAdmin } from '@/lib/rbac'
    15→import { SECTOR_META } from '@/lib/tenant-sector'
    16→import { useAppStore } from '@/store/app-store'
    17→import type { Role } from '@/types'
    18→import {
    19→  Card, CardContent, CardHeader, CardTitle, CardDescription,
    20→} from '@/components/ui/card'
    21→import { Button } from '@/components/ui/button'
    22→import { Input } from '@/components/ui/input'
    23→import { Badge } from '@/components/ui/badge'
    24→import { Skeleton } from '@/components/ui/skeleton'
    25→import {
    26→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    27→} from '@/components/ui/table'
    28→import {
    29→  Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle,
    30→} from '@/components/ui/dialog'
    31→import {
    32→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    33→} from '@/components/ui/select'
    34→import {
    35→  Building2, Search, RefreshCw, Users, Store, CalendarClock,
    36→  Receipt, ListTree, ShoppingCart, Save, ShieldCheck, Globe, Clock, Loader2, Plus,
    37→} from 'lucide-react'

    39→interface TenantListItem {
    40→  id: string
    41→  name: string
    42→  plan: string
    43→  defaultCurrency: string
    44→  country: string
    45→  createdAt: string
    46→  sector: keyof typeof SECTOR_META
    47→  counts: {
    48→    users: number; customers: number; providers: number; orders: number
    49→    invoices: number; tasks: number; products: number; appointments: number
    50→  }
    51→}
    52→
    53→interface TenantDetail {
    54→  tenant: {
    55→    id: string; name: string; plan: string; defaultCurrency: string
    56→    country: string; createdAt: string; sector: keyof typeof SECTOR_META
    57→  }
    58→  roleGroups: Record<string, number>
    59→  users: Array<{
    60→    id: string; name: string; email: string; role: Role; status: string
    61→    title?: string | null; phone?: string | null; createdAt: string
    62→    _count: { ownedCustomers: number; assignedTasks: number }
    63→  }>
    64→  providers: Array<{
    65→    id: string; name: string; slug?: string | null; type: string
    66→    city?: string | null; district?: string | null; isActive: boolean
    67→    lat?: number | null; lng?: number | null
    68→    _count: { services: number; staff: number; appointments: number }
    69→  }>
    70→  settings: Array<{ key: string; value: unknown; updatedAt: string }>
    71→  counts: Record<string, number>
    72→  recentAudit: Array<{
    73→    id: string; action: string; entity: string; createdAt: string
    74→    actor?: { id: string; name: string } | null
    75→  }>
    76→}
    77→
    78→const VALID_ROLES: Role[] = ['superadmin', 'admin', 'manager', 'rep', 'readonly', 'stock']
    79→
    80→function StatMini({ icon: Icon, label, value, color }: {
    81→  icon: React.ComponentType<{ className?: string }>
    82→  label: string; value: number | undefined; color: string
    83→}) {
    84→  return (
    85→    <div className="flex items-center gap-2 rounded-lg border bg-card p-2.5">
    86→      <div className={cn('flex h-8 w-8 items-center justify-center rounded-md', color)}>
    87→        <Icon className="h-4 w-4" />
    88→      </div>
    89→      <div className="min-w-0">
    90→        <div className="text-sm font-semibold tabular-nums leading-none">{value ?? '—'}</div>
    91→        <div className="text-[11px] text-muted-foreground mt-0.5 truncate">{label}</div>
    92→      </div>
    93→    </div>
    94→  )
    95→}
    96→
    97→export function TenantsTab() {
    98→  const qc = useQueryClient()
    99→  const { user: sessionUser } = useAppStore()
   100→  const superAdmin = isSuperAdmin(sessionUser?.role ?? '')
   101→  const [search, setSearch] = useState('')
   102→  const [selectedId, setSelectedId] = useState<string | null>(null)
   103→
   104→  const listQ = useQuery({
   105→    queryKey: ['admin-tenants'],
   106→    queryFn: () => apiGet<{ items: TenantListItem[]; total: number; scope: string }>('/api/admin/tenants'),
   107→  })
   108→
   109→  const items = listQ.data?.items ?? []
   110→  const filtered = items.filter((t) =>
   111→    t.name.toLowerCase().includes(search.toLowerCase()),
   112→  )
   113→  const activeId = selectedId && items.some((t) => t.id === selectedId)
   114→    ? selectedId
   115→    : (items[0]?.id ?? null)
   116→
   117→  const detailQ = useQuery({
   118→    queryKey: ['admin-tenant-detail', activeId],
   119→    queryFn: () => apiGet<TenantDetail>(`/api/admin/tenants/${activeId}`),
   120→    enabled: !!activeId,
   121→  })
   122→
   123→  // İşletme bilgi formu
   124→  const [form, setForm] = useState<{ name: string; plan: string; defaultCurrency: string; country: string } | null>(null)
   125→  const detail = detailQ.data
   126→  const formKey = `${detail?.tenant.id}:${detail?.tenant.name}:${detail?.tenant.plan}:${detail?.tenant.defaultCurrency}:${detail?.tenant.country}`
   127→  const [formKeyState, setFormKeyState] = useState('')
   128→  if (detail && formKeyState !== formKey) {
   129→    setFormKeyState(formKey)
   130→    setForm({
   131→      name: detail.tenant.name,
   132→      plan: detail.tenant.plan,
   133→      defaultCurrency: detail.tenant.defaultCurrency,
   134→      country: detail.tenant.country,
   135→    })
   136→  }
   137→
   138→  const saveTenant = useMutation({
   139→    mutationFn: (patch: Record<string, string>) =>
   140→      apiPatch<{ tenant: { name: string } }>(`/api/admin/tenants/${activeId}`, patch),
   141→    onSuccess: (res) => {
   142→      toast.success(`İşletme güncellendi: ${res.tenant.name}`)
   143→      qc.invalidateQueries({ queryKey: ['admin-tenants'] })
   144→      qc.invalidateQueries({ queryKey: ['admin-tenant-detail', activeId] })
   145→    },
   146→    onError: (e: Error) => toast.error(e.message),
   147→  })
   148→
   149→  const assignRole = useMutation({
   150→    mutationFn: (p: { userId: string; role: Role }) =>
   151→      apiPost('/api/admin/assign-role', p),
   152→    onSuccess: () => {
   153→      toast.success('Rol güncellendi')
   154→      qc.invalidateQueries({ queryKey: ['admin-tenant-detail', activeId] })
   155→    },
   156→    onError: (e: Error) => toast.error(e.message),
   157→  })
   158→
   159→  // --- Yeni İşletme sihirbazı (yalnız superadmin) ---
   160→  const EMPTY_CREATE = {
   161→    name: '', plan: 'business', defaultCurrency: 'TRY', country: 'TR',
   162→    adminName: '', adminEmail: '', adminPassword: '',
   163→  }
   164→  const [createOpen, setCreateOpen] = useState(false)
   165→  const [createForm, setCreateForm] = useState(EMPTY_CREATE)
   166→  const [createError, setCreateError] = useState('')
   167→
   168→  const createTenant = useMutation({
   169→    mutationFn: (payload: Record<string, unknown>) =>
   170→      apiPost<{ tenant: { id: string; name: string; sector: string }; admin: { email: string } | null }>(
   171→        '/api/admin/tenants',
   172→        payload,
   173→      ),
   174→    onSuccess: (res) => {
   175→      toast.success(
   176→        res.admin
   177→          ? `İşletme oluşturuldu: ${res.tenant.name} · İlk admin: ${res.admin.email}`
   178→          : `İşletme oluşturuldu: ${res.tenant.name}`,
   179→      )
   180→      qc.invalidateQueries({ queryKey: ['admin-tenants'] })
   181→      setCreateOpen(false)
   182→      setCreateForm(EMPTY_CREATE)
   183→      setCreateError('')
   184→    },
   185→    onError: (e: Error) => setCreateError(e.message),
   186→  })
   187→
   188→  const submitCreate = () => {
   189→    setCreateError('')
   190→    const name = createForm.name.trim()
   191→    if (!name) {
   192→      setCreateError('İşletme adı gerekli')
   193→      return
   194→    }
   195→    if (createForm.country && !/^[A-Z]{2}$/.test(createForm.country.trim().toUpperCase())) {
   196→      setCreateError('Ülke kodu 2 harf olmalı (örn. TR)')
   197→      return
   198→    }
   199→    const adminName = createForm.adminName.trim()
   200→    const adminEmail = createForm.adminEmail.trim()
   201→    const adminPassword = createForm.adminPassword
   202→    const hasAdmin = Boolean(adminName || adminEmail || adminPassword)
   203→    if (hasAdmin) {
   204→      if (!adminName || !adminEmail || !adminPassword) {
   205→        setCreateError('İlk admin için ad, e-posta ve şifre alanlarının tamamı gerekli (ya da üçünü de boş bırakın)')
   206→        return
   207→      }
   208→      if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(adminEmail)) {
   209→        setCreateError('Geçerli bir e-posta adresi girin')
   210→        return
   211→      }
   212→      if (adminPassword.length < 8) {
   213→        setCreateError('İlk admin şifresi en az 8 karakter olmalı')
   214→        return
   215→      }
   216→    }
   217→    createTenant.mutate({
   218→      name,
   219→      plan: createForm.plan,
   220→      defaultCurrency: createForm.defaultCurrency,
   221→      country: createForm.country.trim().toUpperCase(),
   222→      ...(hasAdmin
   223→        ? { adminUser: { name: adminName, email: adminEmail, password: adminPassword } }
   224→        : {}),
   225→    })
   226→  }
   227→
   228→  const sectorMeta = detail ? SECTOR_META[detail.tenant.sector] : null
   229→
   230→  return (
   231→    <div className="space-y-4">
   232→      <div className="flex items-center justify-between flex-wrap gap-2">
   233→        <div>
   234→          <h3 className="text-lg font-semibold flex items-center gap-2">
   235→            <Building2 className="h-5 w-5 text-emerald-600" />
   236→            İşletmeler
   237→            <Badge variant="outline" className="text-[11px]">
   238→              {superAdmin ? 'Tüm platform' : 'Sadece kendi işletmeniz'}
   239→            </Badge>
   240→          </h3>
   241→          <p className="text-xs text-muted-foreground mt-0.5">
   242→            İşletmeyi seçin; bilgilerini, rollerini ve modüllerini buradan yönetin
   243→          </p>
   244→        </div>
   245→        <div className="flex items-center gap-2">
   246→          {superAdmin && (
   247→            <Button size="sm" onClick={() => { setCreateForm(EMPTY_CREATE); setCreateError(''); setCreateOpen(true) }}>
   248→              <Plus className="h-4 w-4 mr-1.5" />
   249→              Yeni İşletme
   250→            </Button>
   251→          )}
   252→          <Button variant="outline" size="sm" onClick={() => listQ.refetch()} disabled={listQ.isFetching}>
   253→            <RefreshCw className={cn('h-4 w-4 mr-1.5', listQ.isFetching && 'animate-spin')} />
   254→            Yenile
   255→          </Button>
   256→        </div>
   257→      </div>
   258→
   259→      <div className="grid grid-cols-1 lg:grid-cols-12 gap-4">
   260→        {/* Sol: işletme listesi */}
   261→        <Card className="lg:col-span-4 xl:col-span-3 h-fit">
   262→          <CardHeader className="pb-2">
   263→            <CardTitle className="text-sm flex items-center gap-1.5">
   264→              <Store className="h-4 w-4 text-emerald-600" />
   265→              İşletme Listesi
   266→              <Badge variant="secondary" className="ml-auto text-[11px] tabular-nums">{items.length}</Badge>
   267→            </CardTitle>
   268→            <div className="relative mt-2">
   269→              <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   270→              <Input
   271→                placeholder="İşletme ara…"
   272→                value={search}
   273→                onChange={(e) => setSearch(e.target.value)}
   274→                className="pl-8 h-9"
   275→              />
   276→            </div>
   277→          </CardHeader>
   278→          <CardContent className="pt-0 max-h-[520px] overflow-y-auto custom-scroll space-y-1.5">
   279→            {listQ.isLoading ? (
   280→              [...Array(4)].map((_, i) => <Skeleton key={i} className="h-14 w-full" />)
   281→            ) : filtered.length === 0 ? (
   282→              <p className="text-xs text-muted-foreground py-6 text-center">İşletme bulunamadı</p>
   283→            ) : (
   284→              filtered.map((t) => {
   285→                const sm = SECTOR_META[t.sector]
   286→                const selected = t.id === activeId
   287→                return (
   288→                  <button
   289→                    key={t.id}
   290→                    onClick={() => { setSelectedId(t.id); setFormKeyState('') }}
   291→                    className={cn(
   292→                      'w-full text-left rounded-lg border p-2.5 transition-colors',
   293→                      selected
   294→                        ? 'border-emerald-300 bg-emerald-50 dark:border-emerald-700 dark:bg-emerald-950/30'
   295→                        : 'hover:bg-muted/60',
   296→                    )}
   297→                  >
   298→                    <div className="flex items-center gap-2">
   299→                      <span className="text-base leading-none">{sm?.emoji ?? '🏢'}</span>
   300→                      <span className="text-sm font-medium truncate flex-1">{t.name}</span>
   301→                      <Badge variant="outline" className="text-[10px] px-1.5 shrink-0">{t.plan}</Badge>
   302→                    </div>
   303→                    <div className="mt-1 flex items-center gap-2 text-[11px] text-muted-foreground">
   304→                      <span className={cn('font-medium', sm?.accent)}>{sm?.shortLabel ?? t.sector}</span>
   305→                      <span>·</span>
   306→                      <span className="tabular-nums">{t.counts.users} kullanıcı</span>
   307→                      <span>·</span>
   308→                      <span className="tabular-nums">{t.counts.customers} müşteri</span>
   309→                    </div>
   310→                  </button>
   311→                )
   312→              })
   313→            )}
   314→          </CardContent>
   315→        </Card>
   316→
   317→        {/* Sağ: detay */}
   318→        <div className="lg:col-span-8 xl:col-span-9 space-y-4">
   319→          {!activeId ? (
   320→            <Card>
   321→              <CardContent className="py-16 text-center text-sm text-muted-foreground">
   322→                <Building2 className="h-10 w-10 mx-auto mb-3 opacity-30" />
   323→                Detay için soldan bir işletme seçin
   324→              </CardContent>
   325→            </Card>
   326→          ) : detailQ.isLoading || !detail || !form ? (
   327→            <div className="space-y-4">
   328→              <Skeleton className="h-28 w-full" />
   329→              <Skeleton className="h-64 w-full" />
   330→            </div>
   331→          ) : (
   332→            <>
   333→              {/* Bilgi + düzenleme */}
   334→              <Card>
   335→                <CardHeader className="pb-3">
   336→                  <CardTitle className="text-base flex items-center flex-wrap gap-2">
   337→                    <span className="text-xl">{sectorMeta?.emoji}</span>
   338→                    {detail.tenant.name}
   339→                    <Badge variant="outline" className={cn('text-[11px]', sectorMeta?.badge)}>
   340→                      {sectorMeta?.label}
   341→                    </Badge>
   342→                    <Badge variant="secondary" className="text-[11px]">{detail.tenant.plan}</Badge>
   343→                  </CardTitle>
   344→                  <CardDescription className="text-xs">
   345→                    Oluşturma: {new Date(detail.tenant.createdAt).toLocaleDateString('tr-TR')} ·
   346→                    ID: <span className="font-mono text-[10px]">{detail.tenant.id}</span>
   347→                  </CardDescription>
   348→                </CardHeader>
   349→                <CardContent>
   350→                  <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   351→                    <div className="space-y-1">
   352→                      <label className="text-xs font-medium text-muted-foreground">İşletme Adı</label>
   353→                      <Input value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} className="h-9" />
   354→                    </div>
   355→                    <div className="space-y-1">
   356→                      <label className="text-xs font-medium text-muted-foreground">Plan {superAdmin ? '' : '(sadece program admini)'}</label>
   357→                      <Select
   358→                        value={form.plan}
   359→                        onValueChange={(v) => setForm({ ...form, plan: v })}
   360→                        disabled={!superAdmin}
   361→                      >
   362→                        <SelectTrigger className="h-9"><SelectValue /></SelectTrigger>
   363→                        <SelectContent>
   364→                          <SelectItem value="free">free</SelectItem>
   365→                          <SelectItem value="business">business</SelectItem>
   366→                          <SelectItem value="enterprise">enterprise</SelectItem>
   367→                        </SelectContent>
   368→                      </Select>
   369→                    </div>
   370→                    <div className="space-y-1">
   371→                      <label className="text-xs font-medium text-muted-foreground">Para Birimi</label>
   372→                      <Select
   373→                        value={form.defaultCurrency}
   374→                        onValueChange={(v) => setForm({ ...form, defaultCurrency: v })}
   375→                      >
   376→                        <SelectTrigger className="h-9"><SelectValue /></SelectTrigger>
   377→                        <SelectContent>
   378→                          <SelectItem value="TRY">TRY ₺</SelectItem>
   379→                          <SelectItem value="USD">USD $</SelectItem>
   380→                          <SelectItem value="EUR">EUR €</SelectItem>
   381→                        </SelectContent>
   382→                      </Select>
   383→                    </div>
   384→                    <div className="space-y-1">
   385→                      <label className="text-xs font-medium text-muted-foreground">Ülke</label>
   386→                      <Input
   387→                        value={form.country}
   388→                        maxLength={2}
   389→                        onChange={(e) => setForm({ ...form, country: e.target.value.toUpperCase() })}
   390→                        className="h-9"
   391→                      />
   392→                    </div>
   393→                  </div>
   394→                  <div className="mt-3 flex justify-end">
   395→                    <Button
   396→                      size="sm"
   397→                      onClick={() => saveTenant.mutate({
   398→                        name: form.name,
   399→                        plan: form.plan,
   400→                        defaultCurrency: form.defaultCurrency,
   401→                        country: form.country,
   402→                      })}
   403→                      disabled={saveTenant.isPending}
   404→                    >
   405→                      {saveTenant.isPending
   406→                        ? <Loader2 className="h-4 w-4 mr-1.5 animate-spin" />
   407→                        : <Save className="h-4 w-4 mr-1.5" />}
   408→                      Kaydet
   409→                    </Button>
   410→                  </div>
   411→                </CardContent>
   412→              </Card>
   413→
   414→              {/* Modül sayaçları */}
   415→              <div className="grid grid-cols-2 sm:grid-cols-4 lg:grid-cols-7 gap-2">
   416→                <StatMini icon={Users} label="Kullanıcı" value={detail.counts.users} color="bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300" />
   417→                <StatMini icon={Store} label="Randevu İşletmesi" value={detail.counts.serviceProviders ?? detail.providers.length} color="bg-pink-50 text-pink-700 dark:bg-pink-950/40 dark:text-pink-300" />
   418→                <StatMini icon={CalendarClock} label="Randevu" value={detail.providers.reduce((s, p) => s + p._count.appointments, 0)} color="bg-amber-50 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300" />
   419→                <StatMini icon={Users} label="Müşteri" value={detail.counts.customers} color="bg-sky-50 text-sky-700 dark:bg-sky-950/40 dark:text-sky-300" />
   420→                <StatMini icon={ShoppingCart} label="Sipariş" value={detail.counts.orders} color="bg-violet-50 text-violet-700 dark:bg-violet-950/40 dark:text-violet-300" />
   421→                <StatMini icon={Receipt} label="Fatura" value={detail.counts.invoices} color="bg-teal-50 text-teal-700 dark:bg-teal-950/40 dark:text-teal-300" />
   422→                <StatMini icon={ListTree} label="Görev" value={detail.counts.tasks} color="bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300" />
   423→              </div>
   424→
   425→              {/* Kullanıcılar & Roller */}
   426→              <Card>
   427→                <CardHeader className="pb-2">
   428→                  <CardTitle className="text-sm flex items-center gap-1.5">
   429→                    <ShieldCheck className="h-4 w-4 text-emerald-600" />
   430→                    Kullanıcılar &amp; Roller
   431→                    <div className="ml-auto flex flex-wrap gap-1">
   432→                      {Object.entries(detail.roleGroups).map(([role, n]) => (
   433→                        <Badge key={role} variant="secondary" className="text-[10px]">
   434→                          {ROLE_LABELS[role as Role] ?? role}: {n}
   435→                        </Badge>
   436→                      ))}
   437→                    </div>
   438→                  </CardTitle>
   439→                  <CardDescription className="text-xs">
   440→                    Rolü değiştirmek için satırdaki rol menüsünü kullanın (kendi rolünüzü değiştiremezsiniz)
   441→                  </CardDescription>
   442→                </CardHeader>
   443→                <CardContent className="pt-0">
   444→                  <div className="rounded-lg border overflow-hidden">
   445→                    <div className="max-h-72 overflow-y-auto custom-scroll">
   446→                      <Table>
   447→                        <TableHeader>
   448→                          <TableRow className="bg-muted/50 hover:bg-muted/50">
   449→                            <TableHead className="pl-3 text-xs">Kullanıcı</TableHead>
   450→                            <TableHead className="text-xs">E-posta</TableHead>
   451→                            <TableHead className="text-xs">Rol</TableHead>
   452→                            <TableHead className="hidden md:table-cell text-xs">Durum</TableHead>
   453→                            <TableHead className="hidden lg:table-cell text-xs">Müşteri/Görev</TableHead>
   454→                          </TableRow>
   455→                        </TableHeader>
   456→                        <TableBody>
   457→                          {detail.users.map((u) => (
   458→                            <TableRow key={u.id}>
   459→                              <TableCell className="pl-3">
   460→                                <div className="text-sm font-medium">{u.name}</div>
   461→                                {u.title && <div className="text-[11px] text-muted-foreground">{u.title}</div>}
   462→                              </TableCell>
   463→                              <TableCell className="text-xs text-muted-foreground">{u.email}</TableCell>
   464→                              <TableCell>
   465→                                <Select
