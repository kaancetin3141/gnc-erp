   350→// Müşteri görünümü — public booking widget
   351→function PublicBooking({ provider, onBooked }: { provider: Provider; onBooked: () => void }) {
   352→  const [step, setStep] = useState(1)
   353→  const [selectedService, setSelectedService] = useState<Service | null>(null)
   354→  const [selectedStaff, setSelectedStaff] = useState<string>('')
   355→  const [selectedDate, setSelectedDate] = useState(new Date().toISOString().slice(0, 10))
   356→  const [selectedTime, setSelectedTime] = useState('')
   357→  const [customer, setCustomer] = useState({ name: '', phone: '', email: '', note: '' })
   358→  const [booking, setBooking] = useState(false)
   359→
   360→  // Basit slot üretimi (09:00-18:00, 30dk aralık)
   361→  const slots = useMemo(() => {
   362→    const result: string[] = []
   363→    for (let h = 9; h < 18; h++) {
   364→      result.push(`${String(h).padStart(2, '0')}:00`)
   365→      result.push(`${String(h).padStart(2, '0')}:30`)
   366→    }
   367→    return result
   368→  }, [])
   369→
   370→  const handleBook = async () => {
   371→    setBooking(true)
   372→    try {
   373→      const dateTime = new Date(`${selectedDate}T${selectedTime}:00`)
   374→      await apiPost(`/api/appointments/providers/${provider.id}/appointments`, {
   375→        serviceId: selectedService?.id,
   376→        staffId: selectedStaff || undefined,
   377→        customerName: customer.name,
   378→        customerPhone: customer.phone,
   379→        customerEmail: customer.email || undefined,
   380→        customerNote: customer.note || undefined,
   381→        date: dateTime.toISOString(),
   382→        source: 'web',
   383→      })
   384→      toast.success('Randevu oluşturuldu!', { description: `${formatDate(selectedDate)} ${selectedTime}` })
   385→      setStep(5) // success
   386→      onBooked()
   387→    } catch (e) {
   388→      toast.error('Randevu oluşturulamadı', { description: e instanceof Error ? e.message : '' })
   389→    } finally {
   390→      setBooking(false)
   391→    }
   392→  }
   393→
   394→  return (
   395→    <Card className="shadow-soft max-w-2xl mx-auto">
   396→      <CardContent className="p-6">
   397→        {/* Progress */}
   398→        <div className="flex items-center justify-between mb-6">
   399→          {['Hizmet', 'Personel', 'Tarih', 'Bilgiler', 'Tamam'].map((label, i) => (
   400→            <div key={i} className="flex items-center flex-1">
   401→              <div className={cn('w-7 h-7 rounded-full flex items-center justify-center text-xs font-semibold shrink-0',
