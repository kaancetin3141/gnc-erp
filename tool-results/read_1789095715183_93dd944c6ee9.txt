   123→   123→export function CafeOrderScreen({ cafeId }: { cafeId: string }) {
   124→   124→  const qc = useQueryClient()
   125→   125→  const [tab, setTab] = useState<'new' | 'active'>('new')
   126→   126→  const [selectedTableId, setSelectedTableId] = useState<string | null>(null)
   127→   127→  const [cart, setCart] = useState<CartItem[]>([])
   128→   128→  const [orderType, setOrderType] = useState<'dine_in' | 'takeaway' | 'delivery'>('dine_in')
   129→   129→  const [customerName, setCustomerName] = useState('')
   130→   130→  const [orderNotes, setOrderNotes] = useState('')
   131→   131→  const [sending, setSending] = useState(false)
   132→   132→  const [activeOrder, setActiveOrder] = useState<CafeOrder | null>(null)
   133→   133→  const [activeOrderDialog, setActiveOrderDialog] = useState(false)
   134→   134→  const [cancelTarget, setCancelTarget] = useState<CafeOrder | null>(null)
   135→   135→
   136→   136→  const { data: tablesData, isLoading: tablesLoading } = useQuery({
   137→   137→    queryKey: ['cafe-tables', cafeId],
   138→   138→    queryFn: () => apiGet<{ items: CafeTable[] }>(`/api/cafe/${cafeId}/tables`),
   139→   139→  })
   140→   140→  const { data: menuData, isLoading: menuLoading } = useQuery({
   141→   141→    queryKey: ['cafe-menu', cafeId],
   142→   142→    queryFn: () => apiGet<{ items: MenuCategory[] }>(`/api/cafe/${cafeId}/menu`),
   143→   143→  })
   144→   144→  const { data: ordersData, isLoading: ordersLoading } = useQuery({
   145→   145→    queryKey: ['cafe-orders', cafeId, 'active'],
   146→   146→    queryFn: () => apiGet<{ items: CafeOrder[] }>(`/api/cafe/${cafeId}/orders?status=acik&status=hazirlaniyor&status=hazir`),
   147→   147→    refetchInterval: 10_000,
   148→   148→  })
   149→   149→
   150→   150→  const tables = tablesData?.items ?? []
   151→   151→  const categories = menuData?.items ?? []
   152→   152→  const orders = ordersData?.items ?? []
   153→   153→
   154→   154→  // Flat item list with category info
   155→   155→  const allItems = useMemo(() => {
   156→   156→    const list: (MenuItem & { categoryName: string })[] = []
   157→   157→    for (const cat of categories) {
   158→   158→      for (const it of cat.items) {
   159→   159→        list.push({ ...it, categoryName: cat.name })
   160→   160→      }
   161→   161→    }
   162→   162→    return list.filter((i) => i.isAvailable)
   163→   163→  }, [categories])
   164→   164→
   165→   165→  const cartTotal = useMemo(() => cart.reduce((s, c) => s + c.menuItem.price * c.qty, 0), [cart])
   166→   166→
   167→   167→  function addToCart(item: MenuItem) {
   168→   168→    setCart((prev) => {
   169→   169→      const ex = prev.find((c) => c.menuItem.id === item.id && c.notes === '')
   170→   170→      if (ex) {
   171→   171→        return prev.map((c) =>
   172→   172→          c.menuItem.id === item.id && c.notes === '' ? { ...c, qty: c.qty + 1 } : c,
   173→   173→        )
   174→   174→      }
   175→   175→      return [...prev, { menuItem: item, qty: 1, notes: '' }]
   176→   176→    })
   177→   177→  }
   178→   178→
   179→   179→  function updateQty(idx: number, delta: number) {
   180→   180→    setCart((prev) => {
   181→   181→      const next = [...prev]
   182→   182→      const newQty = next[idx].qty + delta
   183→   183→      if (newQty <= 0) {
   184→   184→        next.splice(idx, 1)
   185→   185→      } else {
   186→   186→        next[idx] = { ...next[idx], qty: newQty }
   187→   187→      }
   188→   188→      return next
   189→   189→    })
   190→   190→  }
   191→   191→
   192→   192→  function updateNotes(idx: number, notes: string) {
   193→   193→    setCart((prev) => {
   194→   194→      const next = [...prev]
   195→   195→      next[idx] = { ...next[idx], notes }
   196→   196→      return next
   197→   197→    })
   198→   198→  }
   199→   199→
   200→   200→  function removeFromCart(idx: number) {
   201→   201→    setCart((prev) => {
   202→   202→      const next = [...prev]
   203→   203→      next.splice(idx, 1)
   204→   204→      return next
   205→   205→    })
   206→   206→  }
   207→   207→
   208→   208→  function clearCart() {
   209→   209→    setCart([])
   210→   210→    setOrderNotes('')
   211→   211→    setCustomerName('')
   212→   212→  }
   213→   213→
   214→   214→  async function handleSendOrder() {
   215→   215→    if (cart.length === 0) {
   216→   216→      toast.error('Sepet boş')
   217→   217→      return
   218→   218→    }
   219→   219→    if (orderType === 'dine_in' && !selectedTableId) {
   220→   220→      toast.error('Masa seçimi gerekli')
   221→   221→      return
   222→   222→    }
   223→   223→    setSending(true)
   224→   224→    try {
   225→   225→      await apiPost(`/api/cafe/${cafeId}/orders`, {
   226→   226→        tableId: orderType === 'dine_in' ? selectedTableId : null,
   227→   227→        type: orderType,
   228→   228→        customerName: customerName || undefined,
   229→   229→        notes: orderNotes || undefined,
   230→   230→        items: cart.map((c) => ({
   231→   231→          menuItemId: c.menuItem.id,
   232→   232→          qty: c.qty,
   233→   233→          notes: c.notes || undefined,
   234→   234→        })),
   235→   235→      })
   236→   236→      qc.invalidateQueries({ queryKey: ['cafe-orders', cafeId] })
   237→   237→      qc.invalidateQueries({ queryKey: ['cafe-tables', cafeId] })
   238→   238→      toast.success('Sipariş gönderildi')
   239→   239→      clearCart()
   240→   240→      setSelectedTableId(null)
   241→   241→      setTab('active')
   242→   242→    } catch (e) {
   243→   243→      toast.error(e instanceof Error ? e.message : 'Sipariş gönderilemedi')
   244→   244→    } finally {
   245→   245→      setSending(false)
   246→   246→    }
   247→   247→  }
   248→   248→
   249→   249→  async function openOrderDetail(o: CafeOrder) {
   250→   250→    try {
   251→   251→      const detail = await apiGet<CafeOrder>(`/api/cafe/${cafeId}/orders/${o.id}`)
   252→   252→      setActiveOrder(detail)
   253→   253→      setActiveOrderDialog(true)
   254→   254→    } catch (e) {
   255→   255→      toast.error(e instanceof Error ? e.message : 'Sipariş yüklenemedi')
   256→   256→    }
   257→   257→  }
   258→   258→
   259→   259→  async function handleCancelOrder() {
   260→   260→    if (!cancelTarget) return
   261→   261→    try {
   262→   262→      await apiDelete(`/api/cafe/${cafeId}/orders/${cancelTarget.id}`)
   263→   263→      qc.invalidateQueries({ queryKey: ['cafe-orders', cafeId] })
   264→   264→      qc.invalidateQueries({ queryKey: ['cafe-tables', cafeId] })
   265→   265→      toast.success('Sipariş iptal edildi')
   266→   266→      setCancelTarget(null)
   267→   267→      if (activeOrder?.id === cancelTarget.id) setActiveOrderDialog(false)
   268→   268→    } catch (e) {
   269→   269→      toast.error(e instanceof Error ? e.message : 'İptal başarısız')
   270→   270→    }
   271→   271→  }
   272→   272→
   273→   273→  return (
   274→   274→    <div className="space-y-4">
   275→   275→      <Tabs value={tab} onValueChange={(v) => setTab(v as 'new' | 'active')}>
   276→   276→        <TabsList>
   277→   277→          <TabsTrigger value="new" className="text-xs sm:text-sm">
   278→   278→            <Plus className="w-3.5 h-3.5 mr-1" />
   279→   279→            Yeni Sipariş
   280→   280→          </TabsTrigger>
   281→   281→          <TabsTrigger value="active" className="text-xs sm:text-sm">
   282→   282→            <Receipt className="w-3.5 h-3.5 mr-1" />
   283→   283→            Aktif Siparişler
   284→   284→            {orders.length > 0 && (
   285→   285→              <span className="ml-1 px-1.5 py-0.5 rounded-full bg-emerald-100 dark:bg-emerald-950/40 text-emerald-700 dark:text-emerald-300 text-[10px] font-bold">
   286→   286→                {orders.length}
   287→   287→              </span>
   288→   288→            )}
   289→   289→          </TabsTrigger>
   290→   290→        </TabsList>
   291→   291→      </Tabs>
   292→   292→
   293→   293→      {/* New order tab */}
   294→   294→      {tab === 'new' && (
   295→   295→        <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   296→   296→          {/* Left: menu */}
   297→   297→          <div className="lg:col-span-2 space-y-3">
   298→   298→            {/* Order type selector */}
   299→   299→            <Card>

   302→   302→                  <div className="grid grid-cols-3 gap-2">

   306→   306→                      { v: 'delivery', label: 'Gel-Al', icon: User },
   307→   307→                    ].map((t) => (
   308→   308→                      <button
   309→   309→                        key={t.v}

   311→   311→                        className={cn(
   312→   312→                          'flex flex-col items-center gap-1 p-2.5 rounded-lg border-2 transition-all',
   313→   313→                          orderType === t.v
   314→   314→                            ? 'border-emerald-500 bg-emerald-50 dark:bg-emerald-950/30 text-emerald-700 dark:text-emerald-300'

   316→   316→                        )}
   317→   317→                      >
   318→   318→                        <t.icon className="w-4 h-4" />

   332→   332→                        <Select value={selectedTableId ?? ''} onValueChange={setSelectedTableId}>
   333→   333→                          <SelectTrigger className="mt-1">

   336→   336→                          <SelectContent>

   338→   338→                              .filter((t) => t.status !== 'siparis' || t.id === selectedTableId)
   339→   339→                              .map((t) => (
   340→   340→                                <SelectItem key={t.id} value={t.id} disabled={t.status === 'siparis' && t.id !== selectedTableId}>
   341→   341→                                  Masa {t.number} {t.status === 'siparis' ? '(dolu)' : `· ${t.status}`}

   344→   344→                          </SelectContent>
   345→   345→                        </Select>
   346→   346→                      )}
   347→   347→                    </div>
   348→   348→                  )}
   349→   349→
   350→   350→                  {(orderType === 'takeaway' || orderType === 'delivery') && (
   351→   351→                    <div>
   352→   352→                      <Label className="text-xs text-muted-foreground">Müşteri Adı</Label>

   357→   357→                        className="mt-1 h-9"

   369→   369→                  Menü

   374→   374→                  </div>
   375→   375→                ) : allItems.length === 0 ? (
   376→   376→                  <div className="text-center py-8 text-sm text-muted-foreground">
   377→   377→                    <Coffee className="w-8 h-8 mx-auto mb-2 opacity-50" />
   378→   378→                    Menüde ürün yok
   379→   379→                  </div>
   380→   380→                ) : (

   382→   382→                    {categories.map((cat) => {

   385→   385→                      return (

   387→   387→                          <div className="flex items-center gap-1.5 text-xs font-semibold text-muted-foreground mb-2 px-1">

   397→   397→                                  key={item.id}
   398→   398→                                  onClick={() => addToCart(item)}

   400→   400→                                >
   401→   401→                                  <div className="flex items-start gap-2">
   402→   402→                                    <div className="w-10 h-10 rounded-md bg-muted flex items-center justify-center shrink-0 overflow-hidden">

   404→   404→                                        // eslint-disable-next-line @next/next/no-img-element
   405→   405→                                        <img src={item.photo} alt={item.name} className="w-full h-full object-cover" />
   406→   406→                                      ) : (
   407→   407→                                        <Icon className="w-4 h-4 text-muted-foreground" />
   408→   408→                                      )}
   409→   409→                                    </div>
   410→   410→                                    <div className="flex-1 min-w-0">
   411→   411→                                      <div className="text-xs font-medium leading-tight truncate">{item.name}</div>
   412→   412→                                      <div className="text-xs text-emerald-600 dark:text-emerald-400 font-semibold mt-0.5">
   413→   413→                                        {formatCurrency(item.price, item.currency)}
   414→   414→                                      </div>
   415→   415→                                    </div>
   416→   416→                                  </div>
   417→   417→                                </button>
   418→   418→                              )
   419→   419→                            })}
   420→   420→                          </div>
   421→   421→                        </div>
   422→   422→                      )

   429→   429→
   430→   430→          {/* Right: cart */}
   431→   431→          <Card className="lg:sticky lg:top-4 h-fit">
