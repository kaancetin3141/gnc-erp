    26→// ============================================================
    27→interface KpiData {
    28→  label: string
    29→  value: string | number
    30→  numericValue: number
    31→  sub: string
    32→  icon: typeof Users
    33→  color: string
    34→  onClick?: () => void
    35→}
    36→
    37→function KpiCard({ kpi }: { kpi: KpiData }) {
    38→  const animated = useCountUp(kpi.numericValue, 900)
    39→  const displayValue = typeof kpi.value === 'string' && kpi.value.includes('₺')
    40→    ? formatCurrency(animated, 'TRY')
    41→    : typeof kpi.value === 'number'
    42→      ? animated
    43→      : kpi.value
    44→  return (
    45→    <Card
    46→      className={cn(
    47→        'relative overflow-hidden transition-all duration-300 group hover:-translate-y-0.5',
    48→        kpi.onClick && 'cursor-pointer hover:shadow-md',
    49→      )}
    50→      onClick={kpi.onClick}
    51→    >
    52→      <CardContent className="p-5">
    53→        <div className="flex items-start justify-between mb-3">
    54→          <div className={cn('w-10 h-10 rounded-lg bg-gradient-to-br flex items-center justify-center shadow-sm', kpi.color)}>
    55→            <kpi.icon className="w-5 h-5 text-white" />
    56→          </div>
    57→        </div>
    58→        <div className="text-2xl font-bold tracking-tight">{displayValue}</div>
    59→        <div className="text-sm text-muted-foreground mt-0.5">{kpi.label}</div>
    60→        <div className="text-xs text-muted-foreground/80 mt-1.5">{kpi.sub}</div>
    61→      </CardContent>
    62→    </Card>
    63→  )
    64→}
    65→
    66→function EmptyState({ message }: { message: string }) {
    67→  return (
    68→    <div className="h-48 flex items-center justify-center text-sm text-muted-foreground">
    69→      {message}
    70→    </div>
    71→  )
    72→}
