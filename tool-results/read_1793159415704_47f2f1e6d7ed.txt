    70→export function AppointmentsView() {
    71→  const qc = useQueryClient()

    73→  const [tab, setTab] = useState('appointments')
    74→  const [bookOpen, setBookOpen] = useState(false)
    75→  const [selectedDate, setSelectedDate] = useState(new Date().toISOString().slice(0, 10))
    76→  const [statusFilter, setStatusFilter] = useState('all')
    77→  const [rejectTarget, setRejectTarget] = useState<Appointment | null>(null)
    78→  const [historyTarget, setHistoryTarget] = useState<{ name: string; phone: string } | null>(null)
    79→  const [customerDetailId, setCustomerDetailId] = useState<string | null>(null)
    80→  const [bookInitial, setBookInitial] = useState<{ name?: string; phone?: string; serviceId?: string; staffId?: string } | undefined>(undefined)
    81→
    82→  // Provider list
    83→  const { data: providersData } = useQuery({
    84→    queryKey: ['providers'],
    85→    queryFn: () => apiGet<{ items: Provider[] }>('/api/appointments/providers'),
    86→  })
    87→  const providers = providersData?.items ?? []
    88→  const currentProvider = providers[0] // first provider for now
    89→  const providerId = selectedProviderId || currentProvider?.id
    90→
    91→  // Provider detail (staff + services)
    92→  const { data: provider } = useQuery({
    93→    queryKey: ['provider', providerId],
    94→    queryFn: () => apiGet<Provider>(`/api/appointments/providers/${providerId}`),
    95→    enabled: !!providerId,
    96→  })
    97→
    98→  // Appointments for selected date
    99→  const { data: appointments = [], isLoading } = useQuery({
   100→    queryKey: ['appointments', providerId, selectedDate],
   101→    queryFn: () => apiGet<Appointment[]>(`/api/appointments/providers/${providerId}/appointments?date=${selectedDate}`),
   102→    enabled: !!providerId,
   103→  })
   104→
   105→  // Gün özeti + durum filtresi (program geliştirmesi)
   106→  const dayStats = useMemo(() => {
   107→    const pending = appointments.filter((a) => a.status === 'beklemede')
   108→    const approved = appointments.filter((a) => a.status === 'onaylandi')
   109→    const done = appointments.filter((a) => a.status === 'tamamlandi')
   110→    const cancelled = appointments.filter((a) => a.status === 'iptal' || a.status === 'gelmedi' || a.status === 'reddedildi')
   111→    const expectedRevenue = [...pending, ...approved].reduce((sum, a) => sum + (a.price || 0), 0)
   112→    return {
   113→      total: appointments.length,
   114→      pendingCount: pending.length,
   115→      approvedCount: approved.length,
   116→      doneCount: done.length,
   117→      cancelledCount: cancelled.length,
   118→      expectedRevenue,
   119→    }
   120→  }, [appointments])
   121→
   122→  const filteredAppointments = useMemo(() => {
   123→    if (statusFilter === 'all') return appointments
   124→    if (statusFilter === 'iptal') return appointments.filter((a) => a.status === 'iptal' || a.status === 'gelmedi' || a.status === 'reddedildi')
   125→    return appointments.filter((a) => a.status === statusFilter)
   126→  }, [appointments, statusFilter])
   127→
   128→  const approveAllPending = async () => {
   129→    const pending = appointments.filter((a) => a.status === 'beklemede')
   130→    if (pending.length === 0 || !providerId) return
   131→    try {
   132→      await Promise.all(pending.map((a) =>
   133→        apiPatch(`/api/appointments/providers/${providerId}/appointments/${a.id}`, { status: 'onaylandi' }),
   134→      ))
   135→      qc.invalidateQueries({ queryKey: ['appointments'] })
   136→      toast.success(`${pending.length} randevu onaylandı`, { description: 'Müşterilere WhatsApp ile bilgi gönderebilirsiniz' })
   137→    } catch (e) {
   138→      toast.error('Toplu onay başarısız', { description: e instanceof Error ? e.message : '' })
   139→    }
   140→  }
   141→
   142→  // Müşteri adına tıklanınca: kayıt defterinde kayıtlıysa tam kart, değilse telefonla geçmiş
   143→  const openCustomerInfo = (apt: Appointment) => {
   144→    if (apt.customerId) setCustomerDetailId(apt.customerId)
   145→    else setHistoryTarget({ name: apt.customerName, phone: apt.customerPhone })
   146→  }
   147→
   148→  // Tekrar planla — tamamlanan randevuyu aynı müşteri/hizmetle yeniden oluştur

   150→    const next = new Date()
   151→    next.setDate(next.getDate() + 7)
   152→    setBookInitial({
   153→      name: apt.customerName,
   154→      phone: apt.customerPhone,
   155→      serviceId: apt.service?.id,
   156→      staffId: apt.staff?.id,
   157→    })
   158→    setSelectedDate(next.toISOString().slice(0, 10))
   159→    setBookOpen(true)
   160→  }
   161→
   162→  // Otomatik onay — başlıkta hızlı anahtar (ayrıca İşletme Ayarları'nda da var)
   163→  const autoApprove = provider?.autoApprove ?? true
   164→  const [autoApproveSaving, setAutoApproveSaving] = useState(false)
   165→  const toggleAutoApprove = async (checked: boolean) => {
   166→    if (!providerId) return
   167→    setAutoApproveSaving(true)
   168→    try {
   169→      await apiPatch(`/api/appointments/providers/${providerId}`, { autoApprove: checked })
   170→      qc.invalidateQueries({ queryKey: ['provider', providerId] })
   171→      qc.invalidateQueries({ queryKey: ['appointment-provider', providerId] })
   172→      qc.invalidateQueries({ queryKey: ['providers'] })
   173→      qc.invalidateQueries({ queryKey: ['appointment-providers'] })
   174→      toast.success(
   175→        checked
   176→          ? 'Otomatik onay açıldı — yeni randevular onaysız kabul edilir'
   177→          : 'Otomatik onay kapatıldı — yeni randevular onay bekler',
   178→        { icon: checked ? '⚡' : '⏳' },
   179→      )
   180→    } catch (e) {
   181→      toast.error('Ayar değiştirilemedi', { description: e instanceof Error ? e.message : '' })
   182→    } finally {
   183→      setAutoApproveSaving(false)
   184→    }
   185→  }
   186→
   187→  if (!providersData) {
   188→    return <div className="p-8"><Skeleton className="h-32" /></div>
   189→  }
   190→
   191→  if (providers.length === 0) {
   192→    return (
   193→      <div className="space-y-5 animate-fade-in">
   194→        <h2 className="text-2xl font-bold">Randevu Sistemi</h2>
   195→        <Card><CardContent className="p-8 text-center">
   196→          <Store className="w-12 h-12 mx-auto mb-3 text-muted-foreground/30" />
   197→          <p className="text-sm text-muted-foreground mb-4">Henüz işletme kaydı yok. İlk işletmenizi oluşturun.</p>
   198→          <ProviderCreateDialog tenantId="" onCreated={() => qc.invalidateQueries({ queryKey: ['providers'] })} />
   199→        </CardContent></Card>
   200→      </div>
   201→    )
   202→  }
   203→
   204→  return (
   205→    <div className="space-y-5 animate-fade-in">
   206→      {/* Header */}
   207→      <div className="flex items-center justify-between flex-wrap gap-3">
   208→        <div>
   209→          <h2 className="text-2xl font-bold tracking-tight">Randevu Sistemi</h2>
   210→          <p className="text-sm text-muted-foreground mt-0.5">
   211→            {provider?.name} · {PROVIDER_TYPES.find((t) => t.value === provider?.type)?.label}
   212→          </p>
   213→        </div>
   214→        <div className="flex gap-2 items-center flex-wrap">
   215→          {/* Otomatik onay anahtarı — Randevu Ekle'nin yanı sıra */}
   216→          <div
   217→            className={cn(
   218→              'flex items-center gap-2 h-9 px-3 rounded-lg border transition-colors',
   219→              autoApprove
   220→                ? 'bg-emerald-50 border-emerald-200 dark:bg-emerald-950/30 dark:border-emerald-900/60'
   221→                : 'bg-amber-50 border-amber-200 dark:bg-amber-950/30 dark:border-amber-900/60',
   222→            )}
   223→          >
   224→            {autoApprove
   225→              ? <Zap className="w-4 h-4 text-emerald-600 shrink-0" />
   226→              : <ShieldCheck className="w-4 h-4 text-amber-600 shrink-0" />}
   227→            <span className={cn('text-xs font-medium whitespace-nowrap', autoApprove ? 'text-emerald-700 dark:text-emerald-400' : 'text-amber-700 dark:text-amber-400')}>
   228→              {autoApprove ? 'Otomatik onay' : 'Onay gerekli'}
   229→            </span>
   230→            <Switch
   231→              checked={autoApprove}
   232→              onCheckedChange={toggleAutoApprove}
   233→              disabled={autoApproveSaving}
   234→              aria-label="Randevuları otomatik onayla"
   235→            />
   236→          </div>
   237→          <Input type="date" value={selectedDate} onChange={(e) => setSelectedDate(e.target.value)} className="w-40" />
   238→          <Button size="sm" className="bg-emerald-600 hover:bg-emerald-700" onClick={() => setBookOpen(true)}>
   239→            <Plus className="w-4 h-4 mr-1.5" /> Randevu Ekle
   240→          </Button>
   241→        </div>
   242→      </div>
   243→
   244→      <Tabs value={tab} onValueChange={(v) => {
   245→        setTab(v)
   246→        // Personel/hizmet değişiklikleri listeye yansısın
   247→        if (v === 'appointments') qc.invalidateQueries({ queryKey: ['provider', providerId] })
   248→      }}>
   249→        <TabsList>
   250→          <TabsTrigger value="appointments" className="text-xs"><Calendar className="w-3.5 h-3.5 mr-1" /> Randevular</TabsTrigger>
   251→          <TabsTrigger value="calendar" className="text-xs"><CalendarDays className="w-3.5 h-3.5 mr-1" /> Takvim</TabsTrigger>
   252→          <TabsTrigger value="customers" className="text-xs"><Users className="w-3.5 h-3.5 mr-1" /> Müşteriler</TabsTrigger>
   253→          <TabsTrigger value="staff" className="text-xs"><Users className="w-3.5 h-3.5 mr-1" /> Personel</TabsTrigger>
   254→          <TabsTrigger value="services" className="text-xs"><Scissors className="w-3.5 h-3.5 mr-1" /> Hizmetler</TabsTrigger>
   255→          <TabsTrigger value="membership" className="text-xs"><Ticket className="w-3.5 h-3.5 mr-1" /> Üyelikler</TabsTrigger>
   256→          <TabsTrigger value="settings" className="text-xs"><Settings2 className="w-3.5 h-3.5 mr-1" /> İşletme Ayarları</TabsTrigger>
   257→        </TabsList>
   258→
   259→        {/* Randevular tab */}
   260→        <TabsContent value="appointments" className="mt-4 space-y-3">
   261→          {/* Gün özeti kartları */}
   262→          {!isLoading && appointments.length > 0 && (
   263→            <div className="grid grid-cols-2 md:grid-cols-4 gap-2">
   264→              <Card className="shadow-soft"><CardContent className="p-3">
   265→                <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground"><Calendar className="w-3.5 h-3.5" /> Bugün</div>
   266→                <div className="text-xl font-bold mt-0.5 tabular-nums">{dayStats.total}</div>
   267→                <div className="text-[10px] text-muted-foreground">randevu</div>
   268→              </CardContent></Card>
   269→              <Card className="shadow-soft"><CardContent className="p-3">
   270→                <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground"><Clock className="w-3.5 h-3.5 text-amber-500" /> Bekleyen</div>
   271→                <div className="text-xl font-bold mt-0.5 tabular-nums text-amber-600">{dayStats.pendingCount}</div>
   272→                <div className="text-[10px] text-muted-foreground">onay bekliyor</div>
   273→              </CardContent></Card>
   274→              <Card className="shadow-soft"><CardContent className="p-3">
   275→                <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground"><CheckCircle2 className="w-3.5 h-3.5 text-emerald-500" /> Onaylı</div>
   276→                <div className="text-xl font-bold mt-0.5 tabular-nums text-emerald-600">{dayStats.approvedCount}</div>
   277→                <div className="text-[10px] text-muted-foreground">{dayStats.doneCount} tamamlandı</div>
   278→              </CardContent></Card>
   279→              <Card className="shadow-soft"><CardContent className="p-3">
   280→                <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground"><Wallet className="w-3.5 h-3.5 text-teal-500" /> Beklenen</div>
   281→                <div className="text-xl font-bold mt-0.5 tabular-nums text-teal-600">{formatCurrency(dayStats.expectedRevenue)}</div>
   282→                <div className="text-[10px] text-muted-foreground">ciro potansiyeli</div>
   283→              </CardContent></Card>
   284→            </div>
   285→          )}
   286→
   287→          {/* Onay bekleyen uyarı bandı — otomatik onay kapalıyken */}
   288→          {!isLoading && !autoApprove && dayStats.pendingCount > 0 && (
   289→            <div className="flex items-center justify-between gap-3 flex-wrap rounded-lg border border-amber-200 bg-amber-50 dark:bg-amber-950/30 dark:border-amber-900/60 px-4 py-3">
   290→              <div className="flex items-center gap-2">
   291→                <AlertCircle className="w-4 h-4 text-amber-600 shrink-0" />
   292→                <span className="text-sm text-amber-800 dark:text-amber-300">
   293→                  <strong>{dayStats.pendingCount} randevu</strong> onayınızı bekliyor — otomatik onay kapalı
   294→                </span>
   295→              </div>
   296→              <Button size="sm" className="h-8 bg-emerald-600 hover:bg-emerald-700 text-white" onClick={approveAllPending}>
   297→                <CheckCheck className="w-3.5 h-3.5 mr-1.5" /> Tümünü Onayla
   298→              </Button>
   299→            </div>
   300→          )}
   301→
   302→          {/* Durum filtre çipleri */}
   303→          {!isLoading && appointments.length > 0 && (
   304→            <div className="flex items-center gap-1.5 flex-wrap">
   305→              {([
   306→                { key: 'all', label: 'Tümü', count: dayStats.total, cls: 'bg-primary text-primary-foreground border-transparent' },
   307→                { key: 'beklemede', label: 'Beklemede', count: dayStats.pendingCount, cls: 'bg-amber-100 text-amber-800 border-amber-300 dark:bg-amber-950/40 dark:text-amber-300 dark:border-amber-900' },
   308→                { key: 'onaylandi', label: 'Onaylı', count: dayStats.approvedCount, cls: 'bg-emerald-100 text-emerald-800 border-emerald-300 dark:bg-emerald-950/40 dark:text-emerald-300 dark:border-emerald-900' },
   309→                { key: 'tamamlandi', label: 'Tamamlanan', count: dayStats.doneCount, cls: 'bg-teal-100 text-teal-800 border-teal-300 dark:bg-teal-950/40 dark:text-teal-300 dark:border-teal-900' },
   310→                { key: 'iptal', label: 'İptal/Gelmedi', count: dayStats.cancelledCount, cls: 'bg-red-100 text-red-800 border-red-300 dark:bg-red-950/40 dark:text-red-300 dark:border-red-900' },
   311→              ] as const).map((c) => (
   312→                <button
   313→                  key={c.key}
   314→                  onClick={() => setStatusFilter(c.key)}
   315→                  className={cn(
   316→                    'h-7 px-2.5 rounded-full border text-[11px] font-medium transition-colors',
   317→                    statusFilter === c.key ? c.cls : 'bg-background text-muted-foreground border-border hover:bg-accent',
   318→                  )}
   319→                >
   320→                  {c.label} · {c.count}
   321→                </button>
   322→              ))}
   323→            </div>
   324→          )}
   325→
   326→          {isLoading ? (
   327→            <div className="space-y-2">{[...Array(5)].map((_, i) => <Skeleton key={i} className="h-16" />)}</div>
   328→          ) : filteredAppointments.length === 0 ? (
   329→            <Card><CardContent className="py-12 text-center">
   330→              <Calendar className="w-10 h-10 mx-auto mb-2 text-muted-foreground/30" />
   331→              <p className="text-sm text-muted-foreground">
   332→                {appointments.length === 0
   333→                  ? `${formatDate(selectedDate)} tarihinde randevu yok`
   334→                  : 'Bu filtreye uyan randevu yok'}
   335→              </p>
   336→            </CardContent></Card>
   337→          ) : (
   338→            <div className="space-y-2">
   339→              {filteredAppointments.map((apt) => {
   340→                const st = APPT_STATUS.find((s) => s.value === apt.status)
   341→                return (
   342→                  <Card key={apt.id} className="shadow-soft">
   343→                    <CardContent className="p-3 flex items-center gap-3">
   344→                      <div className="text-center shrink-0 w-16">
   345→                        <div className="text-lg font-bold tabular-nums">{formatTime(apt.date)}</div>
   346→                        <div className="text-[10px] text-muted-foreground">{apt.endTime ? formatTime(apt.endTime) : ''}</div>
   347→                      </div>
   348→                      <div className="flex-1 min-w-0">
   349→                        <div className="flex items-center gap-2">
   350→                          <button
   351→                            onClick={() => openCustomerInfo(apt)}
   352→                            className="font-medium text-sm hover:text-emerald-700 dark:hover:text-emerald-400 hover:underline text-left"
   353→                            title={apt.customerId ? 'Müşteri kartını aç (kayıt defteri)' : 'Müşteri geçmişini gör'}
   354→                          >
   355→                            {apt.customerName}
   356→                          </button>
   357→                          {apt.customerId && (
   358→                            <span
   359→                              className="inline-flex items-center gap-0.5 text-[9px] font-medium text-emerald-700 bg-emerald-50 border border-emerald-200 dark:bg-emerald-950/40 dark:text-emerald-300 dark:border-emerald-900 rounded px-1 py-px shrink-0"
   360→                              title="Kayıt defterinde kayıtlı müşteri"
   361→                            >
   362→                              <UserCheck className="w-2.5 h-2.5" /> kayıtlı
   363→                            </span>
   364→                          )}
   365→                          <Badge variant="outline" className={cn('text-[10px]', st?.color ?? '')}>{st?.label}</Badge>
   366→                        </div>
   367→                        <div className="text-xs text-muted-foreground mt-0.5">
   368→                          {apt.service?.name ?? '—'} · {apt.staff?.name ?? 'Herhangi'} · {formatCurrency(apt.price)}
   369→                        </div>
   370→                      </div>
   371→                      <div className="flex items-center gap-1 shrink-0 flex-wrap justify-end">
   372→                        {/* Beklemede → hızlı onay */}

   535→// Manual booking dialog (admin) — kayıt defteri otomatik tamamlamalı
   536→function ManualBookingDialog({ provider, open, onOpenChange, onSuccess, initial }: {
   537→  provider: Provider; open: boolean; onOpenChange: (v: boolean) => void; onSuccess: () => void
   538→  initial?: { name?: string; phone?: string; serviceId?: string; staffId?: string }
   539→}) {
   540→  const [form, setForm] = useState({
   541→    serviceId: initial?.serviceId ?? '', staffId: initial?.staffId ?? '',
   542→    date: new Date().toISOString().slice(0, 10),
   543→    time: '10:00', name: initial?.name ?? '', phone: initial?.phone ?? '', email: '', note: '',
   544→  })
   545→  const [saving, setSaving] = useState(false)
   546→
   547→  const handleSave = async () => {
   548→    setSaving(true)
   549→    try {
   550→      const dateTime = new Date(`${form.date}T${form.time}:00`)
   551→      await apiPost(`/api/appointments/providers/${provider.id}/appointments`, {
   552→        serviceId: form.serviceId || undefined,
   553→        staffId: form.staffId || undefined,
   554→        customerName: form.name,
   555→        customerPhone: form.phone,
   556→        customerEmail: form.email || undefined,
   557→        customerNote: form.note || undefined,
   558→        date: dateTime.toISOString(),
   559→        source: 'phone',
   560→      })
   561→      toast.success('Randevu oluşturuldu')
   562→      onSuccess()
   563→      onOpenChange(false)
   564→    } catch (e) {
   565→      toast.error('Randevu oluşturulamadı', { description: e instanceof Error ? e.message : '' })
   566→    } finally {
   567→      setSaving(false)
   568→    }
   569→  }
   570→
   571→  return (
   572→    <Dialog open={open} onOpenChange={onOpenChange}>
   573→      <DialogContent className="sm:max-w-md">
   574→        <DialogHeader><DialogTitle>Manuel Randevu</DialogTitle></DialogHeader>
   575→        {provider.autoApprove === false && (
   576→          <div className="flex items-start gap-2 rounded-lg border border-amber-200 bg-amber-50 dark:bg-amber-950/30 dark:border-amber-900/60 px-3 py-2 -mt-1">
   577→            <ShieldCheck className="w-4 h-4 text-amber-600 shrink-0 mt-0.5" />
   578→            <p className="text-xs text-amber-800 dark:text-amber-300">
   579→              Otomatik onay <strong>kapalı</strong> — randevu <strong>&quot;Beklemede&quot;</strong> oluşur, onayladıktan sonra kesinleşir.
   580→            </p>
   581→          </div>
   582→        )}
   583→        <div className="space-y-3">
   584→          <div><Label className="text-xs">Hizmet</Label>
   585→            <Select value={form.serviceId || '__none__'} onValueChange={(v) => setForm({ ...form, serviceId: v === '__none__' ? '' : v })}>
   586→              <SelectTrigger className="h-9"><SelectValue placeholder="Hizmet seç" /></SelectTrigger>
   587→              <SelectContent>
   588→                <SelectItem value="__none__">Hizmet seçilmedi</SelectItem>
   589→                {provider.services.map((s) => <SelectItem key={s.id} value={s.id}>{s.name} ({s.duration}dk)</SelectItem>)}
   590→              </SelectContent>
   591→            </Select>
   592→          </div>
   593→          <div><Label className="text-xs">Personel</Label>
   594→            <Select value={form.staffId || '__none__'} onValueChange={(v) => setForm({ ...form, staffId: v === '__none__' ? '' : v })}>
   595→              <SelectTrigger className="h-9"><SelectValue placeholder="Herhangi" /></SelectTrigger>
   596→              <SelectContent>
   597→                <SelectItem value="__none__">Herhangi biri</SelectItem>
   598→                {provider.staff.map((s) => <SelectItem key={s.id} value={s.id}>{s.name}</SelectItem>)}
   599→              </SelectContent>
   600→            </Select>
   601→          </div>
   602→          <div className="grid grid-cols-2 gap-2">
   603→            <div><Label className="text-xs">Tarih</Label><Input type="date" value={form.date} onChange={(e) => setForm({ ...form, date: e.target.value })} /></div>
   604→            <div><Label className="text-xs">Saat</Label><Input type="time" value={form.time} onChange={(e) => setForm({ ...form, time: e.target.value })} /></div>
   605→          </div>
   606→          <CustomerAutocomplete
   607→            providerId={provider.id}
   608→            name={form.name}
   609→            phone={form.phone}
   610→            onNameChange={(v) => setForm({ ...form, name: v })}
   611→            onPhoneChange={(v) => setForm({ ...form, phone: v })}
   612→          />
   613→          <div><Label className="text-xs">Not</Label><Textarea value={form.note} onChange={(e) => setForm({ ...form, note: e.target.value })} className="h-16" /></div>
   614→        </div>
   615→        <DialogFooter>
   616→          <Button variant="outline" onClick={() => onOpenChange(false)}>İptal</Button>
   617→          <Button className="bg-emerald-600 hover:bg-emerald-700" onClick={handleSave} disabled={saving || !form.name || !form.phone}>{saving ? 'Kaydediliyor...' : 'Randevu Oluştur'}</Button>
   618→        </DialogFooter>
   619→      </DialogContent>
   620→    </Dialog>
   621→  )
   622→}
