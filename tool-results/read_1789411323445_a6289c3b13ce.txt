   101→function activityDot(lastActivityAt: string | null) {
   102→  const d = daysSince(lastActivityAt)
   103→  if (d === null) return { color: 'bg-gray-400', label: 'İletişim yok' }
   104→  if (d <= 7) return { color: 'bg-emerald-500', label: `${d} gün önce` }
   105→  if (d <= 30) return { color: 'bg-amber-500', label: `${d} gün önce` }
   106→  return { color: 'bg-red-500', label: `${d} gün önce` }

   465→
   466→  // Filtre state

   472→  const [segment, setSegment] = useState('')
   473→  const [status, setStatus] = useState('')
   474→  const [ownerId, setOwnerId] = useState('')
   475→  const [customerType, setCustomerType] = useState<CustomerTypeKey | ''>('')
   476→  const [staleOnly, setStaleOnly] = useState(() => {
   477→    if (typeof window === 'undefined') return false
   478→    return new URL(window.location.href).searchParams.get('stale') === 'true'
   479→  })
   480→  const [filtersOpen, setFiltersOpen] = useState(false)
   481→  const [addOpen, setAddOpen] = useState(false)
   482→
   483→  // Query parametreleri
   484→  const params = useMemo(() => {
   485→    const p: Record<string, string> = {}
   486→    if (search) p.search = search
   487→    if (sector) p.sector = sector
   488→    if (city) p.city = city
   489→    if (district) p.district = district
   490→    if (country) p.country = country
   491→    if (segment) p.segment = segment
   492→    if (status) p.status = status
   493→    if (ownerId) p.ownerId = ownerId
   494→    if (customerType) p.customerType = customerType
   495→    if (staleOnly) p.stale = 'true'
   496→    p.limit = '100'
   497→    return p
   498→  }, [search, sector, city, district, country, segment, status, ownerId, customerType, staleOnly])
   499→
   500→  // Müşteri listesi
   501→  const { data, isLoading, isFetching, refetch } = useQuery({
   502→    queryKey: qk.customers(params),
   503→    queryFn: () => {
   504→      const qs = new URLSearchParams(params).toString()
   505→      return apiGet<CustomerListResponse>(`/api/customers?${qs}`)
   506→    },
   507→  })
   508→
   509→  // Kullanıcı listesi (sorumlu select'i için)
   510→  const { data: usersData } = useQuery({
   511→    queryKey: qk.users,
   512→    queryFn: () => apiGet<UsersResponse>('/api/users'),
   513→  })
   514→
   515→  const customers = data?.items ?? []
   516→  const total = data?.total ?? 0
   517→
   518→  // İstatistikler (mevcut listeye göre değil de global; yine de liste üzerinden hesaplayalım)
   519→  const stats = useMemo(() => {
   520→    const active = customers.filter((c) => c.status === 'aktif').length
   521→    const stale = customers.filter((c) => {
   522→      const d = daysSince(c.lastActivityAt)
   523→      return d === null || d > 30
   524→    }).length
   525→    const vip = customers.filter((c) => c.segment === 'vip').length
   526→    return { total, active, stale, vip }
   527→  }, [customers, total])
   528→
   529→  const canEdit = hasPermission(user as SessionUser | null, 'customers.edit')
   530→  const canExport = hasPermission(user as SessionUser | null, 'export.data')
   531→
   532→  // CSV dışa aktarma
   533→  const handleExport = () => {
   534→    if (!customers.length) {
   535→      toast.error('Dışa aktarılacak müşteri yok')
   536→      return
   537→    }
   538→    const rows = customers.map((c) => ({
   539→      Ad: c.name,
   540→      Sektör: c.sector,
   541→      Şehir: c.city || '',
   542→      Segment: getLabel(SEGMENTS, c.segment),
   543→      Telefon: formatPhone(c.phone),
   544→      'E-posta': c.email || '',
   545→      Durum: getLabel(CUSTOMER_STATUS, c.status),
   546→      Sorumlu: c.owner?.name || '',
   547→      'Son İletişim': c.lastActivityAt ? formatRelative(c.lastActivityAt) : '',
   548→    }))
   549→    const csv = toCSV(rows)
   550→    downloadFile(csv, `musteriler-${new Date().toISOString().slice(0, 10)}.csv`)
   551→    toast.success(`${customers.length} müşteri dışa aktarıldı`)
   552→  }
   553→
   554→  const handleClearFilters = () => {
   555→    setSearch(''); setSector(''); setCity(''); setDistrict(''); setCountry(''); setSegment(''); setStatus(''); setOwnerId(''); setCustomerType(''); setStaleOnly(false)
