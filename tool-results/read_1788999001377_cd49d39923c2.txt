    92→// ---- Aktivite renk noktası ----
    93→function activityDot(lastActivityAt: string | null) {
    94→  const d = daysSince(lastActivityAt)
    95→  if (d === null) return { color: 'bg-gray-400', label: 'İletişim yok' }
    96→  if (d <= 7) return { color: 'bg-emerald-500', label: `${d} gün önce` }
    97→  if (d <= 30) return { color: 'bg-amber-500', label: `${d} gün önce` }
    98→  return { color: 'bg-red-500', label: `${d} gün önce` }
    99→}

   411→export function CustomerList() {
   412→  const { user, openCustomer } = useAppStore()
   413→  const qc = useQueryClient()
   414→
   415→  // Filtre state
   416→  const [search, setSearch] = useState('')
   417→  const [sector, setSector] = useState('')
   418→  const [city, setCity] = useState('')
   419→  const [segment, setSegment] = useState('')
   420→  const [status, setStatus] = useState('')
   421→  const [ownerId, setOwnerId] = useState('')
   422→  const [staleOnly, setStaleOnly] = useState(() => {
   423→    if (typeof window === 'undefined') return false
   424→    return new URL(window.location.href).searchParams.get('stale') === 'true'
   425→  })
   426→  const [filtersOpen, setFiltersOpen] = useState(false)
   427→  const [addOpen, setAddOpen] = useState(false)
   428→
   429→  // Query parametreleri
   430→  const params = useMemo(() => {
   431→    const p: Record<string, string> = {}
   432→    if (search) p.search = search
   433→    if (sector) p.sector = sector
   434→    if (city) p.city = city
   435→    if (segment) p.segment = segment
   436→    if (status) p.status = status
   437→    if (ownerId) p.ownerId = ownerId
   438→    if (staleOnly) p.stale = 'true'
   439→    p.limit = '100'
   440→    return p
   441→  }, [search, sector, city, segment, status, ownerId, staleOnly])
   442→
   443→  // Müşteri listesi
   444→  const { data, isLoading, isFetching, refetch } = useQuery({
   445→    queryKey: qk.customers(params),
   446→    queryFn: () => {
   447→      const qs = new URLSearchParams(params).toString()
   448→      return apiGet<CustomerListResponse>(`/api/customers?${qs}`)
   449→    },
   450→  })
   451→
   452→  // Kullanıcı listesi (sorumlu select'i için)
   453→  const { data: usersData } = useQuery({
   454→    queryKey: qk.users,
   455→    queryFn: () => apiGet<UsersResponse>('/api/users'),
   456→  })
   457→
   458→  const customers = data?.items ?? []
   459→  const total = data?.total ?? 0
   460→
   461→  // İstatistikler (mevcut listeye göre değil de global; yine de liste üzerinden hesaplayalım)
   462→  const stats = useMemo(() => {
   463→    const active = customers.filter((c) => c.status === 'aktif').length
   464→    const stale = customers.filter((c) => {
   465→      const d = daysSince(c.lastActivityAt)
   466→      return d === null || d > 30
   467→    }).length
   468→    const vip = customers.filter((c) => c.segment === 'vip').length
   469→    return { total, active, stale, vip }
   470→  }, [customers, total])

   478→      toast.error('Dışa aktarılacak müşteri yok')
   479→      return
   480→    }
   481→    const rows = customers.map((c) => ({
   482→      Ad: c.name,
   483→      Sektör: c.sector,
   484→      Şehir: c.city || '',

   490→      'Son İletişim': c.lastActivityAt ? formatRelative(c.lastActivityAt) : '',
   491→    }))
   492→    const csv = toCSV(rows)
   493→    downloadFile(csv, `musteriler-${new Date().toISOString().slice(0, 10)}.csv`)
   494→    toast.success(`${customers.length} müşteri dışa aktarıldı`)
   495→  }
   496→
   497→  const handleClearFilters = () => {
   498→    setSearch(''); setSector(''); setCity(''); setSegment(''); setStatus(''); setOwnerId(''); setStaleOnly(false)
   499→  }
   500→
   501→  const activeFilterCount = [search, sector, city, segment, status, ownerId].filter(Boolean).length + (staleOnly ? 1 : 0)
   502→
   503→  return (
   504→    <div className="space-y-5 animate-fade-in">
   505→      {/* Header */}
   506→      <div className="flex items-start justify-between flex-wrap gap-3">

   509→            <Users className="w-6 h-6 text-emerald-600" />
   510→            Müşteri Portföyü
   511→          </h2>
   512→          <p className="text-sm text-muted-foreground mt-0.5">
   513→            Toplam <span className="font-semibold text-foreground">{total}</span> müşteri
   514→            {staleOnly && (
   515→              <span className="ml-2 text-amber-600 font-medium">· İletişimsiz filtresi aktif</span>
   516→            )}
   517→          </p>
   518→        </div>
   519→        <div className="flex gap-2 flex-wrap">
   520→          <Button
   521→            variant={staleOnly ? 'default' : 'outline'}
   522→            size="sm"
   523→            onClick={() => setStaleOnly((v) => !v)}
   524→            className={staleOnly ? 'bg-amber-600 hover:bg-amber-700' : ''}
   525→          >
   526→            <AlertTriangle className="w-4 h-4 mr-1.5" />
   527→            İletişimsizler
   528→            {stats.stale > 0 && (
   529→              <Badge variant="secondary" className="ml-1.5 h-5 px-1.5 text-[10px]">
   530→                {stats.stale}
   531→              </Badge>
   532→            )}
   533→          </Button>
   534→          {canExport && (
   535→            <Button variant="outline" size="sm" onClick={handleExport}>
   536→              <Download className="w-4 h-4 mr-1.5" />
   537→              Dışa Aktar
   538→            </Button>
   539→          )}
   540→          {canEdit && (
   541→            <Button size="sm" onClick={() => setAddOpen(true)}>
   542→              <Plus className="w-4 h-4 mr-1.5" />
   543→              Müşteri Ekle
   544→            </Button>
   545→          )}
   546→        </div>
   547→      </div>
   548→
   549→      {/* Stats row */}
   550→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   551→        <StatCard
   552→          label="Toplam Müşteri"
   553→          value={stats.total}
