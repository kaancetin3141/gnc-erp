    26→function rand<T>(arr: readonly T[]): T {
    27→  return arr[Math.floor(Math.random() * arr.length)]
    28→}

    30→function randInt(min: number, max: number): number {
    31→  return Math.floor(Math.random() * (max - min + 1)) + min
    32→}

    46→function randomEmail(name: string): string {
    47→  const slug = name.toLowerCase()
    48→    .replace('ı', 'i').replace('ş', 's').replace('ç', 'c')
    49→    .replace('ğ', 'g').replace('ü', 'u').replace('ö', 'o')
    50→    .replace(/[^a-z0-9]/g, '.')
    51→  return `${slug}@${rand(['gmail.com', 'outlook.com', 'hotmail.com', 'yandex.com'])}`
    52→}

    54→function daysAgo(days: number): Date {
    55→  const d = new Date()
    56→  d.setDate(d.getDate() - days)
    57→  d.setHours(randInt(8, 18), randInt(0, 59))
    58→  return d
    59→}

    61→function daysFromNow(days: number): Date {
    62→  const d = new Date()
    63→  d.setDate(d.getDate() + days)
    64→  return d
    65→}
