    24→    24→export async function getCafeReportsData(

    27→    27→  rangeEnd: Date,
    28→    28→): Promise<{
    29→    29→  sector: 'cafe'
    30→    30→  cafe: {
    31→    31→    cafeNames: string[]
    32→    32→    totalCafes: number
    33→    33→    totalRevenue: number
    34→    34→    totalOrders: number

    37→    37→    totalTables: number
    38→    38→    dailyRevenue: { date: string; revenue: number; orders: number }[]
    39→    39→    hourlyBreakdown: { hour: number; revenue: number; orders: number }[]
    40→    40→    orderTypeMix: { type: string; count: number; revenue: number }[]
    41→    41→    topItems: { name: string; qty: number; revenue: number }[]
    42→    42→    perCafe: { id: string; name: string; revenue: number; orders: number; avgTicket: number }[]
    43→    43→  }
    44→    44→}> {
    45→    45→  const cafes = await db.cafe.findMany({
    46→    46→    where: { tenantId },
    47→    47→    select: {
    48→    48→      id: true, name: true, tableCount: true,
    49→    49→      tables: { select: { id: true, status: true } },
    50→    50→    },
    51→    51→  })
    52→    52→
    53→    53→  const cafeIds = cafes.map((c) => c.id)
    54→    54→
    55→    55→  if (cafeIds.length === 0) {
    56→    56→    return {
    57→    57→      sector: 'cafe',
    58→    58→      cafe: {
    59→    59→        cafeNames: [],
    60→    60→        totalCafes: 0,
    61→    61→        totalRevenue: 0,
    62→    62→        totalOrders: 0,
    63→    63→        avgTicket: 0,
    64→    64→        activeTables: 0,
    65→    65→        totalTables: 0,
    66→    66→        dailyRevenue: [],
    67→    67→        hourlyBreakdown: [],
    68→    68→        orderTypeMix: [],
    69→    69→        topItems: [],
    70→    70→        perCafe: [],
    71→    71→      },
    72→    72→    }

    74→    74→
    75→    75→  const [orders, items, activeTables] = await Promise.all([
    76→    76→    db.cafeOrder.findMany({
    77→    77→      where: {
    78→    78→        cafeId: { in: cafeIds },

    80→    80→        status: 'odendi',

    86→    86→    }),

    88→    88→      where: {
    89→    89→        order: {
    90→    90→          cafeId: { in: cafeIds },
    91→    91→          createdAt: { gte: rangeStart, lt: rangeEnd },
    92→    92→          status: 'odendi',
    93→    93→        },
    94→    94→      },
    95→    95→      select: { name: true, qty: true, unitPrice: true },
    96→    96→    }),
    97→    97→    db.cafeTable.count({
    98→    98→      where: { cafeId: { in: cafeIds }, status: { in: ['dolu', 'siparis', 'rezerve'] } },
    99→    99→    }),
   100→   100→  ])
   101→   101→
   102→   102→  const totalRevenue = orders.reduce((s, o) => s + o.total, 0)
   103→   103→  const totalOrders = orders.length
   104→   104→  const avgTicket = totalOrders > 0 ? totalRevenue / totalOrders : 0
   105→   105→  const totalTables = cafes.reduce((s, c) => s + c.tableCount, 0)
   106→   106→
   107→   107→  // Daily revenue (last 30 days or full range — whichever is shorter)
   108→   108→  const dailyMap = new Map<string, { revenue: number; orders: number }>()
   109→   109→  for (let i = 29; i >= 0; i--) {
   110→   110→    const d = new Date(rangeEnd.getTime() - i * 24 * 60 * 60 * 1000)
   111→   111→    dailyMap.set(d.toISOString().slice(0, 10), { revenue: 0, orders: 0 })
   112→   112→  }
   113→   113→  for (const o of orders) {
   114→   114→    const day = new Date(o.createdAt).toISOString().slice(0, 10)
   115→   115→    const cur = dailyMap.get(day)
   116→   116→    if (cur) {
   117→   117→      cur.revenue += o.total
   118→   118→      cur.orders += 1
   119→   119→    }
   120→   120→  }
   121→   121→  const dailyRevenue = Array.from(dailyMap.entries()).map(([date, v]) => ({
   122→   122→    date, revenue: v.revenue, orders: v.orders,
   123→   123→  }))
   124→   124→
   125→   125→  // Hourly breakdown (08-23)
   126→   126→  const hourlyMap = new Map<number, { revenue: number; orders: number }>()
   127→   127→  for (let h = 8; h <= 23; h++) {
   128→   128→    hourlyMap.set(h, { revenue: 0, orders: 0 })
   129→   129→  }
   130→   130→  for (const o of orders) {
   131→   131→    const h = new Date(o.createdAt).getHours()
   132→   132→    const cur = hourlyMap.get(h)
   133→   133→    if (cur) {
   134→   134→      cur.revenue += o.total
   135→   135→      cur.orders += 1
   136→   136→    }
   137→   137→  }
   138→   138→  const hourlyBreakdown = Array.from(hourlyMap.entries()).map(([hour, v]) => ({
   139→   139→    hour, revenue: v.revenue, orders: v.orders,
   140→   140→  }))
   141→   141→
   142→   142→  // Order type mix
   143→   143→  const typeMap = new Map<string, { count: number; revenue: number }>()
   144→   144→  for (const o of orders) {
   145→   145→    const cur = typeMap.get(o.type) ?? { count: 0, revenue: 0 }
   146→   146→    cur.count += 1
   147→   147→    cur.revenue += o.total
   148→   148→    typeMap.set(o.type, cur)
   149→   149→  }
   150→   150→  const orderTypeMix = Array.from(typeMap.entries()).map(([type, v]) => ({
   151→   151→    type, count: v.count, revenue: v.revenue,
   152→   152→  }))
   153→   153→
   154→   154→  // Top items (by qty, then revenue)
   155→   155→  const itemMap = new Map<string, { name: string; qty: number; revenue: number }>()
   156→   156→  for (const it of items) {
   157→   157→    const cur = itemMap.get(it.name) ?? { name: it.name, qty: 0, revenue: 0 }
   158→   158→    cur.qty += it.qty
   159→   159→    cur.revenue += it.qty * it.unitPrice
   160→   160→    itemMap.set(it.name, cur)
   161→   161→  }
   162→   162→  const topItems = Array.from(itemMap.values())
   163→   163→    .sort((a, b) => b.revenue - a.revenue)
   164→   164→    .slice(0, 10)
   165→   165→
   166→   166→  // Per-cafe breakdown
   167→   167→  const cafeAggMap = new Map<string, { id: string; name: string; revenue: number; orders: number }>()
   168→   168→  for (const c of cafes) {
   169→   169→    cafeAggMap.set(c.id, { id: c.id, name: c.name, revenue: 0, orders: 0 })
   170→   170→  }
   171→   171→  for (const o of orders) {
   172→   172→    const cur = cafeAggMap.get(o.cafeId)
   173→   173→    if (cur) {
   174→   174→      cur.revenue += o.total
   175→   175→      cur.orders += 1
   176→   176→    }
   177→   177→  }
   178→   178→  const perCafe = Array.from(cafeAggMap.values()).map((c) => ({
   179→   179→    ...c,
   180→   180→    avgTicket: c.orders > 0 ? c.revenue / c.orders : 0,
   181→   181→  })).sort((a, b) => b.revenue - a.revenue)
   182→   182→
   183→   183→  return {
   184→   184→    sector: 'cafe',
   185→   185→    cafe: {
   186→   186→      cafeNames: cafes.map((c) => c.name),
   187→   187→      totalCafes: cafes.length,
   188→   188→      totalRevenue,
   189→   189→      totalOrders,
   190→   190→      avgTicket,
   191→   191→      activeTables,
   192→   192→      totalTables,
   193→   193→      dailyRevenue,
   194→   194→      hourlyBreakdown,
   195→   195→      orderTypeMix,
   196→   196→      topItems,
   197→   197→      perCafe,
   198→   198→    },
   199→   199→  }
   200→   200→}

   205→   205→// ============================================================
   206→   206→export async function getMarketReportsData(
   207→   207→  tenantId: string,
   208→   208→  rangeStart: Date,
   209→   209→  rangeEnd: Date,
   210→   210→): Promise<{
   211→   211→  sector: 'market'
   212→   212→  market: {
   213→   213→    marketNames: string[]
   214→   214→    totalMarkets: number
   215→   215→    totalRevenue: number
   216→   216→    totalTx: number
   217→   217→    avgBasket: number
   218→   218→    lowStockCount: number
   219→   219→    outOfStockCount: number
   220→   220→    dailyRevenue: { date: string; revenue: number; tx: number }[]
   221→   221→    paymentMethodMix: { method: string; count: number; revenue: number }[]
   222→   222→    topProducts: { name: string; qty: number; revenue: number; currentStock: number }[]
   223→   223→    stockTurnover: { name: string; soldQty: number; currentStock: number; daysOfCover: number | null }[]
   224→   224→    cashierPerformance: { userId: string; name: string; txCount: number; totalRevenue: number; avgBasket: number }[]
   225→   225→    lowStockProducts: { name: string; stock: number; minStock: number; category: string | null }[]
   226→   226→  }
   227→   227→}> {
   228→   228→  const markets = await db.market.findMany({
   229→   229→    where: { tenantId, isActive: true },
   230→   230→    select: { id: true, name: true },
   231→   231→  })
   232→   232→  const marketIds = markets.map((m) => m.id)
   233→   233→
   234→   234→  if (marketIds.length === 0) {
   235→   235→    return {
   236→   236→      sector: 'market',
   237→   237→      market: {
   238→   238→        marketNames: [],
   239→   239→        totalMarkets: 0,
   240→   240→        totalRevenue: 0,
   241→   241→        totalTx: 0,
   242→   242→        avgBasket: 0,
   243→   243→        lowStockCount: 0,
   244→   244→        outOfStockCount: 0,
   245→   245→        dailyRevenue: [],
   246→   246→        paymentMethodMix: [],
   247→   247→        topProducts: [],
   248→   248→        stockTurnover: [],
   249→   249→        cashierPerformance: [],
   250→   250→        lowStockProducts: [],
   251→   251→      },
   252→   252→    }
   253→   253→  }
   254→   254→
   255→   255→  const [sales, saleItems, products] = await Promise.all([
   256→   256→    db.marketSale.findMany({
   257→   257→      where: {
   258→   258→        marketId: { in: marketIds },
   259→   259→        createdAt: { gte: rangeStart, lt: rangeEnd },
   260→   260→        type: 'satis',
   261→   261→        status: 'tamamlandi',
   262→   262→      },
   263→   263→      select: {
   264→   264→        id: true, total: true, paymentMethod: true, userId: true, createdAt: true,
   265→   265→      },
   266→   266→    }),
   267→   267→    db.marketSaleItem.findMany({
   268→   268→      where: {
   269→   269→        sale: {
   270→   270→          marketId: { in: marketIds },
   271→   271→          createdAt: { gte: rangeStart, lt: rangeEnd },
   272→   272→          type: 'satis',
   273→   273→          status: 'tamamlandi',
   274→   274→        },
   275→   275→      },
   276→   276→      select: { name: true, qty: true, lineTotal: true },
   277→   277→    }),
   278→   278→    db.product.findMany({
   279→   279→      where: { tenantId },
   280→   280→      select: { id: true, name: true, stock: true, minStock: true, category: true, price: true },
   281→   281→    }),
   282→   282→  ])
   283→   283→
   284→   284→  // Cashier IDs — sales üzerinde grupla
   285→   285→  const cashierIds: string[] = Array.from(new Set(
   286→   286→    sales
   287→   287→      .map((s) => s.userId)
   288→   288→      .filter((u): u is string => !!u),
   289→   289→  ))
   290→   290→
   291→   291→  const totalRevenue = sales.reduce((s, x) => s + x.total, 0)
   292→   292→  const totalTx = sales.length
   293→   293→  const avgBasket = totalTx > 0 ? totalRevenue / totalTx : 0
   294→   294→
   295→   295→  // Low/out of stock
   296→   296→  const lowStockProducts = products
   297→   297→    .filter((p) => p.minStock > 0 && p.stock <= p.minStock)
   298→   298→    .map((p) => ({ name: p.name, stock: p.stock, minStock: p.minStock, category: p.category }))
   299→   299→  const outOfStockCount = products.filter((p) => p.stock === 0).length
   300→   300→  const lowStockCount = lowStockProducts.length
   301→   301→
   302→   302→  // Daily revenue (last 30 days)
   303→   303→  const dailyMap = new Map<string, { revenue: number; tx: number }>()
   304→   304→  for (let i = 29; i >= 0; i--) {
   305→   305→    const d = new Date(rangeEnd.getTime() - i * 24 * 60 * 60 * 1000)
   306→   306→    dailyMap.set(d.toISOString().slice(0, 10), { revenue: 0, tx: 0 })
   307→   307→  }
   308→   308→  for (const s of sales) {
   309→   309→    const day = new Date(s.createdAt).toISOString().slice(0, 10)
   310→   310→    const cur = dailyMap.get(day)
   311→   311→    if (cur) {
   312→   312→      cur.revenue += s.total
   313→   313→      cur.tx += 1
   314→   314→    }
   315→   315→  }
   316→   316→  const dailyRevenue = Array.from(dailyMap.entries()).map(([date, v]) => ({
   317→   317→    date, revenue: v.revenue, tx: v.tx,
   318→   318→  }))
   319→   319→
   320→   320→  // Payment method mix
   321→   321→  const methodMap = new Map<string, { count: number; revenue: number }>()
   322→   322→  for (const s of sales) {
   323→   323→    const cur = methodMap.get(s.paymentMethod) ?? { count: 0, revenue: 0 }
   324→   324→    cur.count += 1
   325→   325→    cur.revenue += s.total
   326→   326→    methodMap.set(s.paymentMethod, cur)
   327→   327→  }
   328→   328→  const paymentMethodMix = Array.from(methodMap.entries()).map(([method, v]) => ({
   329→   329→    method, count: v.count, revenue: v.revenue,
   330→   330→  }))
   331→   331→
   332→   332→  // Top products (by revenue) — matched with current stock
   333→   333→  const itemMap = new Map<string, { name: string; qty: number; revenue: number }>()
   334→   334→  for (const it of saleItems) {
   335→   335→    const cur = itemMap.get(it.name) ?? { name: it.name, qty: 0, revenue: 0 }
   336→   336→    cur.qty += it.qty
   337→   337→    cur.revenue += it.lineTotal
   338→   338→    itemMap.set(it.name, cur)
   339→   339→  }
   340→   340→  const topProductList = Array.from(itemMap.values())
   341→   341→    .sort((a, b) => b.revenue - a.revenue)
   342→   342→    .slice(0, 10)
   343→   343→  const topProducts = topProductList.map((p) => {
   344→   344→    const prodMatch = products.find((pr) => pr.name === p.name)
   345→   345→    return { ...p, currentStock: prodMatch?.stock ?? 0 }
   346→   346→  })
   347→   347→
   348→   348→  // Stock turnover (sold vs current stock) — days of cover
   349→   349→  const daysInRange = Math.max(
   350→   350→    1,
   351→   351→    Math.round((rangeEnd.getTime() - rangeStart.getTime()) / (1000 * 60 * 60 * 24)),
   352→   352→  )
   353→   353→  const stockTurnover = topProductList.map((p) => {
   354→   354→    const prodMatch = products.find((pr) => pr.name === p.name)
   355→   355→    const currentStock = prodMatch?.stock ?? 0
   356→   356→    const dailyRate = p.qty / daysInRange
   357→   357→    const daysOfCover = dailyRate > 0 ? Math.round(currentStock / dailyRate) : null
   358→   358→    return {
   359→   359→      name: p.name,
   360→   360→      soldQty: p.qty,
   361→   361→      currentStock,
   362→   362→      daysOfCover,
   363→   363→    }
   364→   364→  })
   365→   365→
   366→   366→  // Cashier performance
   367→   367→  const cashierUsers = cashierIds.length > 0
   368→   368→    ? await db.user.findMany({
   369→   369→      where: { id: { in: cashierIds } },
   370→   370→      select: { id: true, name: true },
   371→   371→    })
   372→   372→    : []
   373→   373→  const cashierMap = new Map<string, { userId: string; name: string; txCount: number; totalRevenue: number }>()
   374→   374→  for (const u of cashierUsers) {
   375→   375→    cashierMap.set(u.id, { userId: u.id, name: u.name, txCount: 0, totalRevenue: 0 })
   376→   376→  }
   377→   377→  for (const s of sales) {
   378→   378→    if (!s.userId) continue
   379→   379→    const cur = cashierMap.get(s.userId)
   380→   380→    if (cur) {
   381→   381→      cur.txCount += 1
   382→   382→      cur.totalRevenue += s.total
   383→   383→    }
   384→   384→  }
   385→   385→  const cashierPerformance = Array.from(cashierMap.values()).map((c) => ({
   386→   386→    ...c,
   387→   387→    avgBasket: c.txCount > 0 ? c.totalRevenue / c.txCount : 0,
   388→   388→  })).sort((a, b) => b.totalRevenue - a.totalRevenue)
   389→   389→
   390→   390→  return {
   391→   391→    sector: 'market',
   392→   392→    market: {
   393→   393→      marketNames: markets.map((m) => m.name),
   394→   394→      totalMarkets: markets.length,
   395→   395→      totalRevenue,
   396→   396→      totalTx,
   397→   397→      avgBasket,
   398→   398→      lowStockCount,
   399→   399→      outOfStockCount,
   400→   400→      dailyRevenue,
   401→   401→      paymentMethodMix,
   402→   402→      topProducts,
   403→   403→      stockTurnover,
   404→   404→      cashierPerformance,
   405→   405→      lowStockProducts,
   406→   406→    },
   407→   407→  }
   408→   408→}

   530→   530→        const cur = monthlyCollectionMap.get(key)

   565→   565→      if (hours >= 0) {
   566→   566→        cur.hoursSum += hours

   599→   599→      const cur = residentGrowthMap.get(key)
   600→   600→      if (cur !== undefined) residentGrowthMap.set(key, cur + 1)
   601→   601→    }
   602→   602→  }

   604→   604→    0,
   605→   605→    residents.filter((r) => new Date(r.createdAt) < new Date(rangeEnd.getFullYear(), rangeEnd.getMonth() - 11, 1)).length,
   606→   606→  )
   607→   607→  const residentGrowth = Array.from(residentGrowthMap.entries()).map(([month, newResidents]) => {
   608→   608→    runningTotal += newResidents
   609→   609→    return { month, newResidents, cumulative: runningTotal }
   610→   610→  })
   611→   611→
   612→   612→  // === Staff workload ===
   613→   613→  // Şemada Complaint.assigneeId olmadığı için, personel başına düşen
   614→   614→  // ortalama yükü site bazında eşit dağıtıyoruz (gerçek assigneeId yok).
   615→   615→  const staffWorkload = staff.map((s) => {

   617→   617→    const total = siteComplaints.length
   618→   618→    const resolved = siteComplaints.filter((c) => c.status === 'cozuldu').length
   619→   619→    const siteStaffCount = staff.filter((st) => st.siteId === s.siteId).length || 1

   627→   627→
   628→   628→  return {
   629→   629→    sector: 'site',
   630→   630→    site: {
   631→   631→      siteNames: sites.map((s) => s.name),
   632→   632→      totalSites: sites.length,
   633→   633→      residentsCount,
   634→   634→      apartmentsCount,
   635→   635→      staffCount,
   636→   636→      collectionRate,
   637→   637→      totalDuesExpected,
   638→   638→      totalDuesCollected,
   639→   639→      monthlyCollection,
   640→   640→      complaintStats: {
   641→   641→        total: totalComplaints,
   642→   642→        open: openComplaints,
   643→   643→        resolved: resolvedComplaints,
   644→   644→        avgResolutionHours,
   645→   645→        byCategory,
   646→   646→        byPriority,
   647→   647→      },
   648→   648→      residentGrowth,
   649→   649→      staffWorkload,

   801→   801→  return {

   822→   822→export function computeRange(range: string, now: Date = new Date()): { rangeStart: Date; rangeEnd: Date } {
   823→   823→  const rangeEnd = new Date(now)
   824→   824→  let rangeStart: Date
   825→   825→
   826→   826→  switch (range) {
   827→   827→    case '7d':
   828→   828→      rangeStart = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000)
   829→   829→      break
   830→   830→    case '30d':
   831→   831→      rangeStart = new Date(now.getTime() - 30 * 24 * 60 * 60 * 1000)
   832→   832→      break
   833→   833→    case '90d':
   834→   834→      rangeStart = new Date(now.getTime() - 90 * 24 * 60 * 60 * 1000)
   835→   835→      break
   836→   836→    case '6m':
   837→   837→      rangeStart = new Date(now.getFullYear(), now.getMonth() - 5, 1)
   838→   838→      break
   839→   839→    case '1y':
   840→   840→      rangeStart = new Date(now.getFullYear() - 1, now.getMonth(), now.getDate())
   841→   841→      break
   842→   842→    case 'all':
   843→   843→    default:
   844→   844→      // 10 yıl öncesinden başla — tüm veriyi kapsar
   845→   845→      rangeStart = new Date(now.getFullYear() - 10, 0, 1)
   846→   846→      break
   847→   847→  }
   848→   848→
   849→   849→  return { rangeStart, rangeEnd }
   850→   850→}
