   208→function SectorWelcomeBanner({
   209→  sector, sectorMeta, data,
   210→}: {
   211→  sector: string
   212→  sectorMeta: { label: string; emoji: string; gradient: string; shortLabel: string }

   218→  let summary = ''
   219→  let quickActions: { label: string; view: string; icon: typeof Users }[] = []
   220→

   222→    summary = data.customers.staleCount > 0
   223→      ? `${data.customers.staleCount} müşteri iletişimsiz — ilgilenmen gerek.`

   245→    summary = data.appointments.pendingCount > 0

   253→    ? 'from-slate-900 via-amber-900 to-slate-900'
   254→    : sector === 'market'
   255→      ? 'from-slate-900 via-emerald-900 to-slate-900'
   256→      : sector === 'site'
   257→        ? 'from-slate-900 via-violet-900 to-slate-900'

   259→          ? 'from-slate-900 via-pink-900 to-slate-900'
   260→          : 'from-slate-900 via-slate-800 to-emerald-900'
   261→
   262→  return (
   263→    <div className={cn(
   264→      'relative overflow-hidden rounded-xl bg-gradient-to-br text-white p-6 lg:p-7',

   290→              key={q.label}
   291→              size="sm"
   292→              variant="outline"

   311→function QuickStat({ sector, data }: { sector: string; data: DashboardData }) {
   312→  let stats: { label: string; value: string | number; accent?: string }[] = []
   313→
   314→  if (sector === 'crm') {
   315→    stats = [
   316→      { label: 'Müşteri', value: data.customers.total },
   317→      { label: 'Açık Fırsat', value: data.deals.activeCount },
   318→      { label: 'Açık Görev', value: data.tasks.openCount },
   319→      { label: 'Bu Ay Ciro', value: formatCurrency(data.deals.revenueThisMonth), accent: 'text-emerald-300' },
   320→    ]
   321→  } else if (sector === 'cafe' && data.cafe) {
   322→    stats = [
   323→      { label: 'Aktif Masa', value: data.cafe.activeTables },
   324→      { label: 'Bugünkü Sipariş', value: data.cafe.todayOrderCount },
   325→      { label: 'Toplam Masa', value: data.cafe.totalTables },
   326→      { label: 'Bugün Ciro', value: formatCurrency(data.cafe.todayRevenue), accent: 'text-emerald-300' },
   327→    ]
   328→  } else if (sector === 'market' && data.market) {
   329→    stats = [
   330→      { label: 'Bugünkü Fiş', value: data.market.todayTxCount },
   331→      { label: 'Aktif Vardiya', value: data.market.activeShifts },
   332→      { label: 'Stok Uyarısı', value: data.market.lowStockProducts, accent: data.market.lowStockProducts > 0 ? 'text-red-300' : 'text-emerald-300' },
   333→      { label: 'Bugün Satış', value: formatCurrency(data.market.todayRevenue), accent: 'text-emerald-300' },
   334→    ]
   335→  } else if (sector === 'site' && data.site) {
   336→    stats = [
   337→      { label: 'Toplam Sakin', value: data.site.residentsCount },
   338→      { label: 'Açık Şikayet', value: data.site.openComplaints, accent: data.site.openComplaints > 0 ? 'text-red-300' : 'text-emerald-300' },
   339→      { label: 'Tahsilat Oranı', value: `%${data.site.collectionRate.toFixed(0)}`, accent: 'text-emerald-300' },
   340→      { label: 'Bu Ay Tahsilat', value: formatCurrency(data.site.collectedThisMonth), accent: 'text-emerald-300' },
   341→    ]
   342→  } else if (sector === 'appointments' && data.appointments) {
   343→    stats = [
   344→      { label: 'Bugünkü Randevu', value: data.appointments.todayCount },
   345→      { label: 'Bu Hafta', value: data.appointments.weekCount },
   346→      { label: 'Bekleyen', value: data.appointments.pendingCount, accent: data.appointments.pendingCount > 0 ? 'text-amber-300' : 'text-emerald-300' },
   347→      { label: 'Bugün Ciro', value: formatCurrency(data.appointments.todayRevenue), accent: 'text-emerald-300' },
   348→    ]
   349→  }
   350→
   351→  return (
   352→    <>
   353→      {stats.map((s, i) => (
   354→        <div key={i}>
   355→          <div className="text-xs text-slate-400">{s.label}</div>
   356→          <div className={cn('text-lg font-bold tabular-nums', s.accent ?? '')}>{s.value}</div>
   357→        </div>
   358→      ))}
   359→    </>
   360→  )
   361→}

   406→      numericValue: data.deals.revenueThisMonth,
   407→      sub: `${data.deals.wonThisMonth} kazanıldı`,
   408→      icon: DollarSign,
   409→      color: 'from-emerald-500 to-teal-600',
   410→      onClick: () => setView('pipeline'),

   412→    {
   413→      label: 'Açık Görevler',
   414→      value: data.tasks.openCount,
   415→      numericValue: data.tasks.openCount,
   416→      sub: data.tasks.overdueCount > 0 ? `${data.tasks.overdueCount} gecikmiş!` : 'Zamanında',
   417→      icon: CheckSquare,
   418→      color: data.tasks.overdueCount > 0 ? 'from-red-500 to-rose-600' : 'from-amber-500 to-orange-600',
   419→      onClick: () => setView('tasks'),
   420→    },
   421→  ]

   423→  const chartData = DEAL_STAGES
   424→    .filter((s) => s.value !== 'kazanıldı' && s.value !== 'kaybedildi')
   425→    .map((s) => {
   426→      const found = data.pipelineByStage.find((p) => p.stage === s.value)
   427→      return { stage: s.label, count: found?.count ?? 0, value: found?.totalValue ?? 0 }
   428→    })

   430→  return (
   431→    <>
   432→      {/* KPI kartları */}
   433→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">

   435→          <CrmKpiCard key={kpi.label} kpi={kpi} />
   436→        ))}
   437→      </div>
   438→
   439→      {/* Bekleyen sevkiyatlar — sevk edilmemiş siparişler (ERP akışı) */}
   440→      {data.shipments && data.shipments.orders.length > 0 && (

   729→type TrendPeriod = (typeof TREND_PERIODS)[number]
   730→
   731→// Bağımsız trend ucu yanıtı — şirket filtresi için müşteri listesi de döner
   732→interface TrendFilterData {
   733→  trend: NonNullable<DashboardData['shipmentsTrend']>
   734→  customers: { id: string; name: string }[]
   735→  canSeeAmounts: boolean
   736→  restricted?: boolean
   737→}
   738→
   739→function ShipmentsTrendWidget({
   740→  trend,
   741→  showRevenue,
   742→}: {
   743→  trend: NonNullable<DashboardData['shipmentsTrend']>
   744→  showRevenue: boolean
   745→}) {
   746→  const { setView } = useAppStore()
   747→  const [period, setPeriod] = useState<TrendPeriod>(6)
   748→  // Şirket filtresi — bağımsız hafif uçtan (/api/dashboard/trend) çekilir

   750→  const { data: trendData, isFetching } = useQuery({
   751→    queryKey: ['shipments-trend', customerId],
   752→    queryFn: () => apiGet<TrendFilterData>(
   753→      `/api/dashboard/trend${customerId !== '__all__' ? `?customerId=${encodeURIComponent(customerId)}` : ''}`,
   754→    ),
   755→    placeholderData: keepPreviousData,
   756→    staleTime: 60_000,
   757→  })

   760→    ? (trendData?.trend ?? [])
   761→    : (trendData?.trend?.length ? trendData.trend : trend)

   763→  const selectedCustomer = customers.find((c) => c.id === customerId)
   764→  // API 12 ay döner — görünür dilim son N ay
   765→  const visible = activeTrend.slice(-period)

   768→  const totalRevenue = visible.reduce((s, t) => s + (t.revenue ?? 0), 0)

   770→  const hasAnyData = totalOrders > 0 || totalShipped > 0 || totalRevenue > 0

   782→              <p className="text-xs text-muted-foreground mt-0.5 flex items-center gap-1 min-w-0">
   783→                <span className="truncate max-w-[160px]">
   784→                  {customerId !== '__all__' && selectedCustomer ? selectedCustomer.name : 'Tüm müşteriler'}

   786→                <span className="text-muted-foreground/60">·</span>
   787→                <span className="shrink-0">Son {period} ay</span>

   986→  const [irsaliyePreview, setIrsaliyePreview] = useState<{ id: string; number: string } | null>(null)

   994→    try {
   995→      const res = await apiPost<{

   999→

  1003→        })
  1004→      }
  1005→      setIrsaliyePreview({ id: res.document.id, number: res.document.number })
  1006→      qc.invalidateQueries({ queryKey: ['documents-orders'] })
  1007→      qc.invalidateQueries({ queryKey: ['irsaliye'] })

  1009→      toast.error(e instanceof Error ? e.message : 'İrsaliye oluşturulamadı')

  1027→              </p>

  1041→              order.expectedDelivery != null && new Date(order.expectedDelivery) < today

  1117→      {irsaliyePreview && (

  1119→          irsaliyeId={irsaliyePreview.id}
  1120→          irsaliyeNumber={irsaliyePreview.number}
