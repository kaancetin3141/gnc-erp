   703→// IRSALİYE (DISPATCH NOTE) — Sevk irsaliyesi, ağırlık takibi, palet
   704→// ============================================================
   705→// Numara formatı: IRS-2026-001, IRS-2026-002 (yıl + tenant bazlı sıra)
   706→// Durum: taslak | hazir | sevk_edildi | teslim_edildi | iptal
   707→// Ağırlık hesabı:
   708→//   totalNetWeight       = sum(line.weightPerUnit * line.qty)
   709→//   totalPackagingWeight = sum(product.packagingWeight * line.qty) (opsiyonel)

   711→//   totalGrossWeight     = net + packaging + pallet

   718→  customerId           String
   719→  customer             Customer  @relation(fields: [customerId], references: [id])
   720→  number               String // IRS-2026-001
   721→  date                 DateTime  @default(now())
   722→  status               String    @default("taslak") // taslak | hazir | sevk_edildi | teslim_edildi | iptal
   723→  // Ağırlık takibi
   724→  totalNetWeight       Float? // sum of line.weightPerUnit * line.qty
   725→  totalPackagingWeight Float? // sum of product.packagingWeight * line.qty
   726→  palletWeight         Float? // user-entered pallet weight (default 20 kg)
   727→  totalGrossWeight     Float? // net + packaging + pallet
   728→  palletCount          Int?
   729→  palletType           String? // tahta | plastik | tek_kullanilik | eur_pallet | other
   730→  // Sevkiyat bilgileri
   731→  shippingAddress      String?
   732→  carrier              String? // yurtici | aras | mng | ptt | ups | fedex | dhl | other
   733→  trackingNo           String?
   734→  // Notlar
   735→  notes                String?
   736→  createdById          String?
   737→  createdAt            DateTime  @default(now())
   738→  updatedAt            DateTime  @updatedAt
   739→  deliveredAt          DateTime?
   740→
   741→  lines IrsaliyeLine[]
   742→
   743→  @@unique([tenantId, number])
