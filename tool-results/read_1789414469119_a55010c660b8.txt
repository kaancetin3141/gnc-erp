   102→function activityDot(lastActivityAt: string | null) {
   103→  const d = daysSince(lastActivityAt)
   104→  if (d === null) return { color: 'bg-gray-400', label: 'İletişim yok' }
   105→  if (d <= 7) return { color: 'bg-emerald-500', label: `${d} gün önce` }
   106→  if (d <= 30) return { color: 'bg-amber-500', label: `${d} gün önce` }
   107→  return { color: 'bg-red-500', label: `${d} gün önce` }
   108→}

   463→export function CustomerList() {
   464→  const { user, openCustomer } = useAppStore()
   465→  const qc = useQueryClient()
   466→
   467→  // Filtre state

   470→  const [city, setCity] = useState('')
   471→  const [district, setDistrict] = useState('')
   472→  const [country, setCountry] = useState('')
   473→  const [segment, setSegment] = useState('')
   474→  const [status, setStatus] = useState('')
   475→  const [ownerId, setOwnerId] = useState('')
   476→  const [customerType, setCustomerType] = useState<CustomerTypeKey | ''>('')
   477→  const [staleOnly, setStaleOnly] = useState(() => {
   478→    if (typeof window === 'undefined') return false
   479→    return new URL(window.location.href).searchParams.get('stale') === 'true'
   480→  })
   481→  const [filtersOpen, setFiltersOpen] = useState(false)
   482→  const [addOpen, setAddOpen] = useState(false)
   483→  const [importOpen, setImportOpen] = useState(false)
   484→
   485→  // Query parametreleri
   486→  const params = useMemo(() => {
   487→    const p: Record<string, string> = {}
   488→    if (search) p.search = search
   489→    if (sector) p.sector = sector
   490→    if (city) p.city = city
   491→    if (district) p.district = district
   492→    if (country) p.country = country
   493→    if (segment) p.segment = segment
   494→    if (status) p.status = status
   495→    if (ownerId) p.ownerId = ownerId
   496→    if (customerType) p.customerType = customerType
   497→    if (staleOnly) p.stale = 'true'
   498→    p.limit = '100'
   499→    return p
   500→  }, [search, sector, city, district, country, segment, status, ownerId, customerType, staleOnly])
   501→
   502→  // Müşteri listesi
   503→  const { data, isLoading, isFetching, refetch } = useQuery({
   504→    queryKey: qk.customers(params),
   505→    queryFn: () => {
   506→      const qs = new URLSearchParams(params).toString()
   507→      return apiGet<CustomerListResponse>(`/api/customers?${qs}`)
   508→    },
   509→  })
   510→
   511→  // Kullanıcı listesi (sorumlu select'i için)
   512→  const { data: usersData } = useQuery({
   513→    queryKey: qk.users,
   514→    queryFn: () => apiGet<UsersResponse>('/api/users'),
   515→  })
   516→
   517→  const customers = data?.items ?? []
   518→  const total = data?.total ?? 0
   519→
   520→  // İstatistikler (mevcut listeye göre değil de global; yine de liste üzerinden hesaplayalım)
   521→  const stats = useMemo(() => {
   522→    const active = customers.filter((c) => c.status === 'aktif').length
   523→    const stale = customers.filter((c) => {
   524→      const d = daysSince(c.lastActivityAt)
   525→      return d === null || d > 30
   526→    }).length
   527→    const vip = customers.filter((c) => c.segment === 'vip').length
   528→    return { total, active, stale, vip }
   529→  }, [customers, total])
   530→
   531→  const canEdit = hasPermission(user as SessionUser | null, 'customers.edit')
   532→  const canExport = hasPermission(user as SessionUser | null, 'export.data')
   533→
   534→  // CSV dışa aktarma
   535→  const handleExport = () => {
   536→    if (!customers.length) {
   537→      toast.error('Dışa aktarılacak müşteri yok')
   538→      return
   539→    }
   540→    const rows = customers.map((c) => ({
   541→      Ad: c.name,
   542→      Sektör: c.sector,
   543→      Şehir: c.city || '',
   544→      Segment: getLabel(SEGMENTS, c.segment),
   545→      Telefon: formatPhone(c.phone),
   546→      'E-posta': c.email || '',
   547→      Durum: getLabel(CUSTOMER_STATUS, c.status),
   548→      Sorumlu: c.owner?.name || '',
   549→      'Son İletişim': c.lastActivityAt ? formatRelative(c.lastActivityAt) : '',
   550→    }))
   551→    const csv = toCSV(rows)
   552→    downloadFile(csv, `musteriler-${new Date().toISOString().slice(0, 10)}.csv`)
   553→    toast.success(`${customers.length} müşteri dışa aktarıldı`)
   554→  }
