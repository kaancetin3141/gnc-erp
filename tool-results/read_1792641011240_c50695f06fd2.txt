   433→   433→function pageShell(o: { title: string; desc: string; body: string; scripts?: string[]; extraCss?: string[]; gwPort?: string | null }): string {
   434→   434→  // Sürüm parametresi: CSS/JS içeriğinin djb2 özeti — içerik değişince URL değişir,
   435→   435→  // tarayıcı/heuristik önbellek eski stili servis edemez (cache-busting).
   436→   436→  let _vh = 5381;
   437→   437→  const _verSrc = STYLE_CSS + '\u0000' + HOME_JS + '\u0000' + DETAIL_JS;
   438→   438→  for (let i = 0; i < _verSrc.length; i++) _vh = ((_vh << 5) + _vh + _verSrc.charCodeAt(i)) >>> 0;
   439→   439→  const VER = _vh.toString(36);
   440→   440→  // Gateway (XTransformPort) ile açıldıysa göreli varlık URL'lerine param ekle —
   441→   441→  // aksi hâlde /assets/* istekleri gateway'den CRM'e düşer ve 404 olur.
   442→   442→  const gw = (p: string) => {
   443→   443→    const withVer = p + (p.includes('?') ? '&' : '?') + 'v=' + VER;
   444→   444→    return o.gwPort ? withVer + '&XTransformPort=' + encodeURIComponent(o.gwPort) : withVer;
   445→   445→  };
   446→   446→  const scripts = (o.scripts || []).map((s) => `<script src="${gw(s)}" defer></script>`).join('\n  ');
   447→   447→  const extraCss = (o.extraCss || []).map((s) => `<link rel="stylesheet" href="${gw(s)}">`).join('\n  ');
   448→   448→  return `<!DOCTYPE html>
   449→   449→<html lang="tr">
   450→   450→<head>
   451→   451→<meta charset="UTF-8">
   452→   452→<meta name="viewport" content="width=device-width, initial-scale=1">
   453→   453→<title>${esc(o.title)}</title>
   454→   454→<meta name="description" content="${esc(o.desc)}">
   455→   455→<link rel="icon" href="${gw(FAVICON)}">
   456→   456→<link rel="stylesheet" href="${gw('/assets/style.css')}">
   457→   457→  ${extraCss}
   458→   458→</head>
   459→   459→<body>
   460→   460→<div class="page">
   461→   461→${o.body}
   462→   462→</div>
   463→   463→<div id="toast" role="status" aria-live="polite"></div>
   464→   464→  ${scripts}
   465→   465→</body>
   466→   466→</html>`;
   467→   467→}
