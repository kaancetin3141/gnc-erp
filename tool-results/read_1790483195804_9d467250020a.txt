    58→  { value: 'beklemede', label: 'Beklemede', color: 'text-amber-600 bg-amber-50 border-amber-200' },
    59→  { value: 'onaylandi', label: 'Onaylandı', color: 'text-emerald-600 bg-emerald-50 border-emerald-200' },

    66→export function AppointmentsView() {
    67→  const qc = useQueryClient()
    68→  const [selectedProviderId, setSelectedProviderId] = useState<string | null>(null)
    69→  const [tab, setTab] = useState('appointments')
    70→  const [bookOpen, setBookOpen] = useState(false)
    71→  const [selectedDate, setSelectedDate] = useState(new Date().toISOString().slice(0, 10))
    72→  const [statusFilter, setStatusFilter] = useState('all')
    73→  const [rejectTarget, setRejectTarget] = useState<Appointment | null>(null)
    74→  const [historyTarget, setHistoryTarget] = useState<{ name: string; phone: string } | null>(null)
    75→
    76→  // Provider list
    77→  const { data: providersData } = useQuery({
    78→    queryKey: ['providers'],
    79→    queryFn: () => apiGet<{ items: Provider[] }>('/api/appointments/providers'),
    80→  })
    81→  const providers = providersData?.items ?? []
    82→  const currentProvider = providers[0] // first provider for now
    83→  const providerId = selectedProviderId || currentProvider?.id
    84→
    85→  // Provider detail (staff + services)
    86→  const { data: provider } = useQuery({
    87→    queryKey: ['provider', providerId],
    88→    queryFn: () => apiGet<Provider>(`/api/appointments/providers/${providerId}`),
    89→    enabled: !!providerId,
    90→  })
    91→
    92→  // Appointments for selected date
    93→  const { data: appointments = [], isLoading } = useQuery({
    94→    queryKey: ['appointments', providerId, selectedDate],
    95→    queryFn: () => apiGet<Appointment[]>(`/api/appointments/providers/${providerId}/appointments?date=${selectedDate}`),
    96→    enabled: !!providerId,
    97→  })
    98→
    99→  // Gün özeti + durum filtresi (program geliştirmesi)
   100→  const dayStats = useMemo(() => {
   101→    const pending = appointments.filter((a) => a.status === 'beklemede')
   102→    const approved = appointments.filter((a) => a.status === 'onaylandi')
   103→    const done = appointments.filter((a) => a.status === 'tamamlandi')
   104→    const cancelled = appointments.filter((a) => a.status === 'iptal' || a.status === 'gelmedi' || a.status === 'reddedildi')
   105→    const expectedRevenue = [...pending, ...approved].reduce((sum, a) => sum + (a.price || 0), 0)
   106→    return {
   107→      total: appointments.length,
   108→      pendingCount: pending.length,
   109→      approvedCount: approved.length,
   110→      doneCount: done.length,
   111→      cancelledCount: cancelled.length,
   112→      expectedRevenue,
   113→    }
   114→  }, [appointments])
   115→
   116→  const filteredAppointments = useMemo(() => {
   117→    if (statusFilter === 'all') return appointments
   118→    if (statusFilter === 'iptal') return appointments.filter((a) => a.status === 'iptal' || a.status === 'gelmedi' || a.status === 'reddedildi')
   119→    return appointments.filter((a) => a.status === statusFilter)
   120→  }, [appointments, statusFilter])
   121→
   122→  const approveAllPending = async () => {
   123→    const pending = appointments.filter((a) => a.status === 'beklemede')
   124→    if (pending.length === 0 || !providerId) return
   125→    try {
   126→      await Promise.all(pending.map((a) =>
   127→        apiPatch(`/api/appointments/providers/${providerId}/appointments/${a.id}`, { status: 'onaylandi' }),
   128→      ))
   129→      qc.invalidateQueries({ queryKey: ['appointments'] })
   130→      toast.success(`${pending.length} randevu onaylandı`, { description: 'Müşterilere WhatsApp ile bilgi gönderebilirsiniz' })
   131→    } catch (e) {
   132→      toast.error('Toplu onay başarısız', { description: e instanceof Error ? e.message : '' })
   133→    }
   134→  }
   135→
   136→  // Otomatik onay — başlıkta hızlı anahtar (ayrıca İşletme Ayarları'nda da var)
   137→  const autoApprove = provider?.autoApprove ?? true
   138→  const [autoApproveSaving, setAutoApproveSaving] = useState(false)
   139→  const toggleAutoApprove = async (checked: boolean) => {
   140→    if (!providerId) return
   141→    setAutoApproveSaving(true)
   142→    try {
   143→      await apiPatch(`/api/appointments/providers/${providerId}`, { autoApprove: checked })
   144→      qc.invalidateQueries({ queryKey: ['provider', providerId] })
   145→      qc.invalidateQueries({ queryKey: ['appointment-provider', providerId] })
   146→      qc.invalidateQueries({ queryKey: ['providers'] })
   147→      qc.invalidateQueries({ queryKey: ['appointment-providers'] })
   148→      toast.success(
   149→        checked
   150→          ? 'Otomatik onay açıldı — yeni randevular onaysız kabul edilir'
   151→          : 'Otomatik onay kapatıldı — yeni randevular onay bekler',
   152→        { icon: checked ? '⚡' : '⏳' },
   153→      )
   154→    } catch (e) {
   155→      toast.error('Ayar değiştirilemedi', { description: e instanceof Error ? e.message : '' })
   156→    } finally {
   157→      setAutoApproveSaving(false)
   158→    }
   159→  }
   160→
   161→  if (!providersData) {
   162→    return <div className="p-8"><Skeleton className="h-32" /></div>
   163→  }
   164→
   165→  if (providers.length === 0) {
   166→    return (
   167→      <div className="space-y-5 animate-fade-in">
   168→        <h2 className="text-2xl font-bold">Randevu Sistemi</h2>
   169→        <Card><CardContent className="p-8 text-center">
   170→          <Store className="w-12 h-12 mx-auto mb-3 text-muted-foreground/30" />
   171→          <p className="text-sm text-muted-foreground mb-4">Henüz işletme kaydı yok. İlk işletmenizi oluşturun.</p>
   172→          <ProviderCreateDialog tenantId="" onCreated={() => qc.invalidateQueries({ queryKey: ['providers'] })} />
   173→        </CardContent></Card>
   174→      </div>
   175→    )
   176→  }
   177→
   178→  return (
   179→    <div className="space-y-5 animate-fade-in">
   180→      {/* Header */}
   181→      <div className="flex items-center justify-between flex-wrap gap-3">
   182→        <div>
   183→          <h2 className="text-2xl font-bold tracking-tight">Randevu Sistemi</h2>
   184→          <p className="text-sm text-muted-foreground mt-0.5">
   185→            {provider?.name} · {PROVIDER_TYPES.find((t) => t.value === provider?.type)?.label}
   186→          </p>
   187→        </div>
   188→        <div className="flex gap-2 items-center flex-wrap">
   189→          {/* Otomatik onay anahtarı — Randevu Ekle'nin yanı sıra */}
   190→          <div
   191→            className={cn(
   192→              'flex items-center gap-2 h-9 px-3 rounded-lg border transition-colors',
   193→              autoApprove
   194→                ? 'bg-emerald-50 border-emerald-200 dark:bg-emerald-950/30 dark:border-emerald-900/60'
   195→                : 'bg-amber-50 border-amber-200 dark:bg-amber-950/30 dark:border-amber-900/60',
   196→            )}
   197→          >
   198→            {autoApprove
   199→              ? <Zap className="w-4 h-4 text-emerald-600 shrink-0" />
   200→              : <ShieldCheck className="w-4 h-4 text-amber-600 shrink-0" />}
   201→            <span className={cn('text-xs font-medium whitespace-nowrap', autoApprove ? 'text-emerald-700 dark:text-emerald-400' : 'text-amber-700 dark:text-amber-400')}>
   202→              {autoApprove ? 'Otomatik onay' : 'Onay gerekli'}
   203→            </span>
   204→            <Switch
   205→              checked={autoApprove}
   206→              onCheckedChange={toggleAutoApprove}
   207→              disabled={autoApproveSaving}
   208→              aria-label="Randevuları otomatik onayla"
   209→            />
   210→          </div>
   211→          <Input type="date" value={selectedDate} onChange={(e) => setSelectedDate(e.target.value)} className="w-40" />
   212→          <Button size="sm" className="bg-emerald-600 hover:bg-emerald-700" onClick={() => setBookOpen(true)}>
   213→            <Plus className="w-4 h-4 mr-1.5" /> Randevu Ekle
   214→          </Button>
   215→        </div>
   216→      </div>
   217→
   218→      <Tabs value={tab} onValueChange={(v) => {
   219→        setTab(v)
   220→        // Personel/hizmet değişiklikleri Müşteri Görünümü'ne ve listeye yansısın
   221→        if (v === 'booking' || v === 'appointments') qc.invalidateQueries({ queryKey: ['provider', providerId] })
   222→      }}>
   223→        <TabsList>
   224→          <TabsTrigger value="appointments" className="text-xs"><Calendar className="w-3.5 h-3.5 mr-1" /> Randevular</TabsTrigger>
   225→          <TabsTrigger value="calendar" className="text-xs"><CalendarDays className="w-3.5 h-3.5 mr-1" /> Takvim</TabsTrigger>
   226→          <TabsTrigger value="staff" className="text-xs"><Users className="w-3.5 h-3.5 mr-1" /> Personel</TabsTrigger>
   227→          <TabsTrigger value="services" className="text-xs"><Scissors className="w-3.5 h-3.5 mr-1" /> Hizmetler</TabsTrigger>
   228→          <TabsTrigger value="booking" className="text-xs"><Store className="w-3.5 h-3.5 mr-1" /> Müşteri Görünümü</TabsTrigger>
   229→          <TabsTrigger value="settings" className="text-xs"><Settings2 className="w-3.5 h-3.5 mr-1" /> İşletme Ayarları</TabsTrigger>
   230→        </TabsList>
   231→
   232→        {/* Randevular tab */}
   233→        <TabsContent value="appointments" className="mt-4 space-y-3">
   234→          {/* Gün özeti kartları */}
   235→          {!isLoading && appointments.length > 0 && (
   236→            <div className="grid grid-cols-2 md:grid-cols-4 gap-2">
   237→              <Card className="shadow-soft"><CardContent className="p-3">
   238→                <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground"><Calendar className="w-3.5 h-3.5" /> Bugün</div>
   239→                <div className="text-xl font-bold mt-0.5 tabular-nums">{dayStats.total}</div>
   240→                <div className="text-[10px] text-muted-foreground">randevu</div>
   241→              </CardContent></Card>
   242→              <Card className="shadow-soft"><CardContent className="p-3">
   243→                <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground"><Clock className="w-3.5 h-3.5 text-amber-500" /> Bekleyen</div>
   244→                <div className="text-xl font-bold mt-0.5 tabular-nums text-amber-600">{dayStats.pendingCount}</div>
   245→                <div className="text-[10px] text-muted-foreground">onay bekliyor</div>
   246→              </CardContent></Card>
   247→              <Card className="shadow-soft"><CardContent className="p-3">
   248→                <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground"><CheckCircle2 className="w-3.5 h-3.5 text-emerald-500" /> Onaylı</div>
   249→                <div className="text-xl font-bold mt-0.5 tabular-nums text-emerald-600">{dayStats.approvedCount}</div>
   250→                <div className="text-[10px] text-muted-foreground">{dayStats.doneCount} tamamlandı</div>
   251→              </CardContent></Card>
   252→              <Card className="shadow-soft"><CardContent className="p-3">
   253→                <div className="flex items-center gap-1.5 text-[11px] text-muted-foreground"><Wallet className="w-3.5 h-3.5 text-teal-500" /> Beklenen</div>
   254→                <div className="text-xl font-bold mt-0.5 tabular-nums text-teal-600">{formatCurrency(dayStats.expectedRevenue)}</div>
   255→                <div className="text-[10px] text-muted-foreground">ciro potansiyeli</div>
   256→              </CardContent></Card>
   257→            </div>
   258→          )}
   259→
   260→          {/* Onay bekleyen uyarı bandı — otomatik onay kapalıyken */}
   261→          {!isLoading && !autoApprove && dayStats.pendingCount > 0 && (
   262→            <div className="flex items-center justify-between gap-3 flex-wrap rounded-lg border border-amber-200 bg-amber-50 dark:bg-amber-950/30 dark:border-amber-900/60 px-4 py-3">
   263→              <div className="flex items-center gap-2">
   264→                <AlertCircle className="w-4 h-4 text-amber-600 shrink-0" />
   265→                <span className="text-sm text-amber-800 dark:text-amber-300">
   266→                  <strong>{dayStats.pendingCount} randevu</strong> onayınızı bekliyor — otomatik onay kapalı
   267→                </span>
   268→              </div>
   269→              <Button size="sm" className="h-8 bg-emerald-600 hover:bg-emerald-700 text-white" onClick={approveAllPending}>
   270→                <CheckCheck className="w-3.5 h-3.5 mr-1.5" /> Tümünü Onayla
   271→              </Button>
   272→            </div>
   273→          )}
   274→
   275→          {/* Durum filtre çipleri */}
   276→          {!isLoading && appointments.length > 0 && (
   277→            <div className="flex items-center gap-1.5 flex-wrap">
   278→              {([
   279→                { key: 'all', label: 'Tümü', count: dayStats.total, cls: 'bg-primary text-primary-foreground border-transparent' },
   280→                { key: 'beklemede', label: 'Beklemede', count: dayStats.pendingCount, cls: 'bg-amber-100 text-amber-800 border-amber-300 dark:bg-amber-950/40 dark:text-amber-300 dark:border-amber-900' },
   281→                { key: 'onaylandi', label: 'Onaylı', count: dayStats.approvedCount, cls: 'bg-emerald-100 text-emerald-800 border-emerald-300 dark:bg-emerald-950/40 dark:text-emerald-300 dark:border-emerald-900' },
   282→                { key: 'tamamlandi', label: 'Tamamlanan', count: dayStats.doneCount, cls: 'bg-teal-100 text-teal-800 border-teal-300 dark:bg-teal-950/40 dark:text-teal-300 dark:border-teal-900' },
   283→                { key: 'iptal', label: 'İptal/Gelmedi', count: dayStats.cancelledCount, cls: 'bg-red-100 text-red-800 border-red-300 dark:bg-red-950/40 dark:text-red-300 dark:border-red-900' },
   284→              ] as const).map((c) => (
   285→                <button
   286→                  key={c.key}
   287→                  onClick={() => setStatusFilter(c.key)}
   288→                  className={cn(
   289→                    'h-7 px-2.5 rounded-full border text-[11px] font-medium transition-colors',
   290→                    statusFilter === c.key ? c.cls : 'bg-background text-muted-foreground border-border hover:bg-accent',
   291→                  )}
   292→                >
   293→                  {c.label} · {c.count}
   294→                </button>
   295→              ))}
   296→            </div>
   297→          )}
   298→
   299→          {isLoading ? (
   300→            <div className="space-y-2">{[...Array(5)].map((_, i) => <Skeleton key={i} className="h-16" />)}</div>
   301→          ) : filteredAppointments.length === 0 ? (
   302→            <Card><CardContent className="py-12 text-center">
   303→              <Calendar className="w-10 h-10 mx-auto mb-2 text-muted-foreground/30" />
   304→              <p className="text-sm text-muted-foreground">
   305→                {appointments.length === 0
   306→                  ? `${formatDate(selectedDate)} tarihinde randevu yok`
   307→                  : 'Bu filtreye uyan randevu yok'}
   308→              </p>
   309→            </CardContent></Card>
   310→          ) : (
   311→            <div className="space-y-2">
   312→              {filteredAppointments.map((apt) => {
   313→                const st = APPT_STATUS.find((s) => s.value === apt.status)
   314→                return (
   315→                  <Card key={apt.id} className="shadow-soft">
   316→                    <CardContent className="p-3 flex items-center gap-3">
   317→                      <div className="text-center shrink-0 w-16">
   318→                        <div className="text-lg font-bold tabular-nums">{formatTime(apt.date)}</div>
   319→                        <div className="text-[10px] text-muted-foreground">{apt.endTime ? formatTime(apt.endTime) : ''}</div>
   320→                      </div>
   321→                      <div className="flex-1 min-w-0">
   322→                        <div className="flex items-center gap-2">
   323→                          <button
   324→                            onClick={() => setHistoryTarget({ name: apt.customerName, phone: apt.customerPhone })}
   325→                            className="font-medium text-sm hover:text-emerald-700 dark:hover:text-emerald-400 hover:underline text-left"
   326→                            title="Müşteri geçmişini gör"
   327→                          >
   328→                            {apt.customerName}
   329→                          </button>
   330→                          <Badge variant="outline" className={cn('text-[10px]', st?.color ?? '')}>{st?.label}</Badge>
   331→                        </div>
   332→                        <div className="text-xs text-muted-foreground mt-0.5">
   333→                          {apt.service?.name ?? '—'} · {apt.staff?.name ?? 'Herhangi'} · {formatCurrency(apt.price)}
   334→                        </div>
   335→                      </div>
   336→                      <div className="flex items-center gap-1 shrink-0 flex-wrap justify-end">
   337→                        {/* Beklemede → hızlı onay */}
   338→                        {apt.status === 'beklemede' && (
   339→                          <Button size="sm" className="h-8 text-[11px] bg-emerald-600 hover:bg-emerald-700 text-white"
   340→                            onClick={async () => { await apiPatch(`/api/appointments/providers/${providerId}/appointments/${apt.id}`, { status: 'onaylandi' }); qc.invalidateQueries({ queryKey: ['appointments'] }); toast.success('Randevu onaylandı — WhatsApp ile bilgi gönderebilirsiniz') }}>
   341→                            <CheckCircle2 className="w-3.5 h-3.5 mr-1" /> Onayla
   342→                          </Button>
   343→                        )}
   344→                        {/* Beklemede → sebep ile reddet */}
   345→                        {apt.status === 'beklemede' && (
   346→                          <Button variant="ghost" size="icon" className="h-8 w-8 text-red-600" title="Reddet (sebep bildir)"
   347→                            onClick={() => setRejectTarget(apt)}>
   348→                            <Ban className="w-4 h-4" />
   349→                          </Button>
   350→                        )}
   351→                        {/* Onaylandı → WhatsApp ile bilgi gönder */}
   352→                        {apt.status === 'onaylandi' && (
   353→                          <Button
   354→                            size="sm"
   355→                            variant={apt.reminderSent ? 'outline' : 'default'}
   356→                            className={cn(
   357→                              'h-8 text-[11px]',
   358→                              !apt.reminderSent && 'bg-[#25D366] hover:bg-[#1fb857] text-white border-0',
   359→                              apt.reminderSent && 'text-[#25D366] border-[#25D366]/40',
   360→                            )}
   361→                            title="Müşteriye WhatsApp ile onay bilgisi gönder"
   362→                            onClick={async () => {
   363→                              const svc = apt.service?.name ? apt.service.name : 'randevu'
   364→                              const stf = apt.staff?.name ? ` (${apt.staff.name})` : ''
   365→                              const msg = `Merhaba ${apt.customerName}, 🎉\n\n${formatDateTime(apt.date)} tarihli ${svc}${stf} randevunuz ONAYLANDI. ✅\n\nGörüşmek üzere! 🙌`
   366→                              window.open(whatsappLink(apt.customerPhone, msg), '_blank', 'noopener,noreferrer')
   367→                              try {
   368→                                await apiPatch(`/api/appointments/providers/${providerId}/appointments/${apt.id}`, { reminderSent: true })
   369→                                qc.invalidateQueries({ queryKey: ['appointments'] })
