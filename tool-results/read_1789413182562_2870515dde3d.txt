   162→function TreePanel({
   163→  overview,
   164→  selection,
   165→  onSelect,
   166→}: {

   168→  selection: TreeSelection
   169→  onSelect: (s: TreeSelection) => void

   172→  const [usersOpen, setUsersOpen] = useState(true)
   173→
   174→  const totalCustomers = overview
   175→    ? Object.values(overview.customerByType).reduce((s, n) => s + n, 0)
   176→    : 0
   177→  const totalUsers = overview
   178→    ? Object.values(overview.userByRole).reduce((s, n) => s + n, 0)
   179→    : 0
   180→
   181→  return (
   182→    <div className="space-y-1">
   183→      {/* Tüm müşteriler */}

   185→        <div className="px-2 py-1.5 flex items-center gap-2 text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/80">
   186→          <Users className="w-3.5 h-3.5" />
   187→          <span className="flex-1">Müşteriler</span>
   188→          <span className="tabular-nums">{totalCustomers}</span>
   189→        </div>
   190→        <CollapsibleContent>
   191→          <div className="space-y-0.5 mb-2">
   192→            <TreeRow
   193→              active={selection.kind === 'all'}
   194→              onClick={() => onSelect({ kind: 'all' })}
   195→              icon={<BarChart3 className="w-4 h-4" />}
   196→              label="Tüm Müşteriler"
   197→              count={totalCustomers}
   198→              tone="text-slate-600 dark:text-slate-300"
   199→              dotClass="bg-slate-400"
   200→            />
   201→            {CUSTOMER_TYPES.map((t) => {
   202→              const Icon = t.icon
   203→              const count = overview?.customerByType[t.value] ?? 0
   204→              return (
   205→                <TreeRow
   206→                  key={t.value}
   207→                  active={selection.kind === 'customerType' && selection.value === t.value}
   208→                  onClick={() => onSelect({ kind: 'customerType', value: t.value })}
   209→                  icon={<Icon className="w-4 h-4" />}
   210→                  label={t.label}
   211→                  count={count}
   212→                  tone={t.color.split(' ')[0]}
   213→                  dotClass={t.dot}
   214→                />
   215→              )
   216→            })}
   217→          </div>
   218→        </CollapsibleContent>
   219→      </Collapsible>
   220→
   221→      <Separator className="my-2" />
   222→
   223→      {/* Kullanıcılar */}
   224→      <Collapsible open={usersOpen} onOpenChange={setUsersOpen}>
   225→        <CollapsibleTrigger asChild>
   226→          <button className="w-full px-2 py-1.5 flex items-center gap-2 text-[11px] font-semibold uppercase tracking-wider text-muted-foreground/80 hover:text-foreground transition-colors">
   227→            {usersOpen ? <ChevronDown className="w-3.5 h-3.5" /> : <ChevronRight className="w-3.5 h-3.5" />}
   228→            <CircleUser className="w-3.5 h-3.5" />

   230→            <span className="tabular-nums">{totalUsers}</span>
   231→          </button>
   232→        </CollapsibleTrigger>
   233→        <CollapsibleContent>
   234→          <div className="space-y-1.5">
   235→            {ROLE_TREE_GROUPS.map((group) => {
   236→              const GroupIcon = group.icon

   240→              )

   243→                  <div className={cn('px-2 py-1 flex items-center gap-1.5 text-[10px] font-medium uppercase tracking-wide', group.color)}>

   250→                      <TreeRow
   251→                        key={r.role}
   252→                        active={selection.kind === 'userRole' && selection.value === r.role}
   253→                        onClick={() => onSelect({ kind: 'userRole', value: r.role })}
   254→                        icon={<span className="text-sm leading-none">{r.emoji}</span>}
   255→                        label={ROLE_LABELS[r.role]}
   256→                        count={overview?.userByRole[r.role] ?? 0}
