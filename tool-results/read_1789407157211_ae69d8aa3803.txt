   177→function exportCSV(rows: Record<string, unknown>[], filename: string, sectionLabel: string) {
   178→  if (!rows.length) {
   179→    toast.error('Dışa aktarılacak veri yok', { description: `${sectionLabel} için kayıt bulunamadı.` })
   180→    return
   181→  }
   182→  const csv = toCSV(rows)
   183→  downloadFile(csv, filename, 'text/csv;charset=utf-8')
   184→  toast.success('CSV dışa aktarıldı', { description: `${filename} indirildi.` })
   185→}
   186→
   187→function exportXLSX(rows: Record<string, unknown>[], filename: string, sectionLabel: string) {
   188→  if (!rows.length) {
   189→    toast.error('Dışa aktarılacak veri yok', { description: `${sectionLabel} için kayıt bulunamadı.` })
   190→    return
   191→  }
   192→  // Excel'in açabileceği basit HTML tablosu
   193→  const headers = Object.keys(rows[0])
   194→  const escapeHtml = (v: unknown) => String(v ?? '').replace(/[&<>"']/g, (c) =>
   195→    ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' })[c] ?? c)
   196→  const html = `<!DOCTYPE html><html xmlns:o="urn:schemas-microsoft-com:office:office" xmlns:x="urn:schemas-microsoft-com:office:excel"><head><meta charset="UTF-8"></head><body><table border="1"><thead><tr>${headers.map((h) => `<th>${escapeHtml(h)}</th>`).join('')}</tr></thead><tbody>${rows.map((r) => `<tr>${headers.map((h) => `<td>${escapeHtml(r[h])}</td>`).join('')}</tr>`).join('')}</tbody></table></body></html>`
   197→  downloadFile('\uFEFF' + html, filename, 'application/vnd.ms-excel;charset=utf-8')
   198→  toast.success('XLSX dışa aktarıldı', { description: `${filename} indirildi.` })
   199→}

   201→// Ay kısaltması: "2024-08" → "Ağu 24"
   202→function formatMonthLabel(m: string): string {
   203→  const [y, mo] = m.split('-')
   204→  const monthIdx = parseInt(mo, 10) - 1
   205→  const names = ['Oca', 'Şub', 'Mar', 'Nis', 'May', 'Haz', 'Tem', 'Ağu', 'Eyl', 'Eki', 'Kas', 'Ara']
   206→  return `${names[monthIdx] ?? mo} ${y.slice(2)}`
   207→}
   208→
   209→// "2024-08-15" → "15.08"
   210→function formatDayLabel(d: string): string {
   211→  const [, m, day] = d.split('-')
   212→  return `${day}.${m}`
   213→}
