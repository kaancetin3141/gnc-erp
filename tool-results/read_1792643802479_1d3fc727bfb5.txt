   194→function exportCSV(rows: Record<string, unknown>[], filename: string, sectionLabel: string) {
   195→  if (!rows.length) {
   196→    toast.error('Dışa aktarılacak veri yok', { description: `${sectionLabel} için kayıt bulunamadı.` })
   197→    return
   198→  }
   199→  const csv = toCSV(rows)
   200→  downloadFile(csv, filename, 'text/csv;charset=utf-8')
   201→  toast.success('CSV dışa aktarıldı', { description: `${filename} indirildi.` })
   202→}
   203→
   204→function exportXLSX(rows: Record<string, unknown>[], filename: string, sectionLabel: string) {
   205→  if (!rows.length) {
   206→    toast.error('Dışa aktarılacak veri yok', { description: `${sectionLabel} için kayıt bulunamadı.` })
   207→    return
   208→  }
   209→  // Excel'in açabileceği basit HTML tablosu
   210→  const headers = Object.keys(rows[0])
   211→  const escapeHtml = (v: unknown) => String(v ?? '').replace(/[&<>"']/g, (c) =>
   212→    ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' })[c] ?? c)
   213→  const html = `<!DOCTYPE html><html xmlns:o="urn:schemas-microsoft-com:office:office" xmlns:x="urn:schemas-microsoft-com:office:excel"><head><meta charset="UTF-8"></head><body><table border="1"><thead><tr>${headers.map((h) => `<th>${escapeHtml(h)}</th>`).join('')}</tr></thead><tbody>${rows.map((r) => `<tr>${headers.map((h) => `<td>${escapeHtml(r[h])}</td>`).join('')}</tr>`).join('')}</tbody></table></body></html>`
   214→  downloadFile('\uFEFF' + html, filename, 'application/vnd.ms-excel;charset=utf-8')
   215→  toast.success('XLSX dışa aktarıldı', { description: `${filename} indirildi.` })
   216→}

   218→// Ay kısaltması: "2024-08" → "Ağu 24"
   219→function formatMonthLabel(m: string): string {
   220→  const [y, mo] = m.split('-')
   221→  const monthIdx = parseInt(mo, 10) - 1
   222→  const names = ['Oca', 'Şub', 'Mar', 'Nis', 'May', 'Haz', 'Tem', 'Ağu', 'Eyl', 'Eki', 'Kas', 'Ara']
   223→  return `${names[monthIdx] ?? mo} ${y.slice(2)}`
   224→}
   225→
   226→// "2024-08-15" → "15.08"
   227→function formatDayLabel(d: string): string {
   228→  const [, m, day] = d.split('-')
   229→  return `${day}.${m}`
   230→}
