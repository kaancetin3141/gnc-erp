   236→  function addToCart(product: Product, barcode: string | null) {
   237→    setCart((prev) => {
   238→      const existing = prev.find((c) => c.productId === product.id)
   239→      if (existing) {
   240→        if (existing.qty >= product.stock) {
   241→          toast.error(`Yetersiz stok: ${product.name} (stok: ${product.stock})`)
   242→          return prev
   243→        }
   244→        return prev.map((c) =>
   245→          c.productId === product.id ? { ...c, qty: c.qty + 1 } : c,
   246→        )
   247→      }
   248→      if (product.stock <= 0) {
   249→        toast.error(`Stokta yok: ${product.name}`)
   250→        return prev
   251→      }
   252→      return [
   253→        ...prev,
   254→        {
   255→          productId: product.id,
   256→          barcode,
   257→          name: product.name,
   258→          price: product.price,
   259→          qty: 1,
   260→          taxRate: product.taxRate,
   261→          stock: product.stock,
   262→        },
   263→      ]
   264→    })
   265→  }
   266→
   267→  function updateQty(productId: string, delta: number) {
   268→    setCart((prev) =>
   269→      prev
   270→        .map((c) => {
   271→          if (c.productId !== productId) return c
   272→          const newQty = c.qty + delta
   273→          if (newQty <= 0) return { ...c, qty: 0 }
   274→          if (newQty > c.stock) {
   275→            toast.error(`Stok aşıldı (max: ${c.stock})`)
   276→            return c
   277→          }
   278→          return { ...c, qty: newQty }
   279→        })
   280→        .filter((c) => c.qty > 0),
   281→    )
   282→  }
   283→
   284→  function removeFromCart(productId: string) {
   285→    setCart((prev) => prev.filter((c) => c.productId !== productId))
   286→  }

   288→  function clearCart() {

   299→  const taxableBase = Math.max(0, subtotal - discountAmount)
   300→  const taxTotal = cart.reduce((s, x) => {
   301→    const ratio = subtotal > 0 ? (x.price * x.qty) / subtotal : 0
   302→    const lineAfterDisc = taxableBase * ratio
   303→    return s + lineAfterDisc - lineAfterDisc / (1 + x.taxRate / 100)
   304→  }, 0)
   305→  const total = taxableBase
   306→  const willEarnPoints = Math.floor(total / 100)

   308→  const mixedShort = paymentMethod === 'mixed' ? Math.max(0, total - cashGiven - cardAmount) : 0

   313→  async function completeSale() {
   314→    if (cart.length === 0) {
   315→      toast.error('Sepet boş')
   316→      return
   317→    }
   318→    if (!openShift) {
   319→      toast.error('Önce vardiya açın')
   320→      setShiftDialog('open')
   321→      return
   322→    }
   323→    if (paymentMethod === 'cash' && cashGiven < total) {
   324→      toast.error('Nakit yetersiz')
   325→      return
   326→    }
   327→    if (paymentMethod === 'mixed' && cashGiven + cardAmount < total) {
   328→      toast.error('Ödeme tutarı yetersiz')
   329→      return
   330→    }
   331→    if (paymentMethod === 'veresiye' && !creditCustomerId) {
   332→      toast.error('Veresiye için müşteri seçin')
   333→      return
   334→    }
   335→
   336→    setProcessing(true)
   337→    try {
   338→      const sale = await apiPost<SaleResponse>(`/api/market/${marketId}/pos/sales`, {
   339→        items: cart.map((c) => ({
   340→          productId: c.productId,
   341→          barcode: c.barcode,
   342→          qty: c.qty,
   343→        })),
   344→        paymentMethod,
   345→        cashAmount: paymentMethod === 'cash' ? cashGiven : paymentMethod === 'mixed' ? cashGiven : 0,
   346→        cardAmount: paymentMethod === 'card' ? total : paymentMethod === 'mixed' ? cardAmount : 0,
   347→        discount: discountAmount,
   348→        customerName: customerName.trim() || undefined,
   349→        creditCustomerId: paymentMethod === 'veresiye' ? creditCustomerId : undefined,
   350→        creditDueDate: paymentMethod === 'veresiye' && creditDueDate ? new Date(creditDueDate + 'T23:59:59').toISOString() : undefined,
   351→      })
   352→
   353→      setLastSale(sale)
   354→      setReceiptOpen(true)
   355→      clearCart()
   356→      setCustomerName('')
   357→      setCreditCustomerId('')
   358→      setCreditDueDate('')
   359→      qc.invalidateQueries({ queryKey: ['market-shifts', marketId] })
   360→      qc.invalidateQueries({ queryKey: ['market-sales', marketId] })
   361→      qc.invalidateQueries({ queryKey: ['market-barcodes', marketId] })
   362→      qc.invalidateQueries({ queryKey: ['market-credit'] })
   363→      qc.invalidateQueries({ queryKey: ['market-credit-pos', marketId] })
   364→      qc.invalidateQueries({ queryKey: ['loyalty-lookup'] })
   365→      if (sale.earnedPoints > 0) {
   366→        toast.success(`Satış tamamlandı: ${sale.number} — ${sale.customerName} +${sale.earnedPoints} puan kazandı ⭐`)
   367→      } else {
   368→        toast.success(`Satış tamamlandı: ${sale.number}`)
   369→      }
   370→    } catch (e) {
   371→      toast.error(e instanceof Error ? e.message : 'Satış başarısız')
   372→    } finally {
   373→      setProcessing(false)
   374→    }
   375→  }
