   286→function NotificationsTab() {
   287→  const [settings, setSettings] = useState<Record<string, string>>({})
   288→  const [saving, setSaving] = useState(false)
   289→
   290→  const { data, isLoading } = useQuery({
   291→    queryKey: ['settings'],
   292→    queryFn: () => apiGet<Record<string, string>>(`/api/settings`),
   293→  })
   294→
   295→  useEffect(() => {
   296→    if (data) setSettings(data)
   297→  }, [data])
   298→
   299→  const updateSetting = async (key: string, value: string) => {
   300→    setSettings((s) => ({ ...s, [key]: value }))
   301→    setSaving(true)
   302→    try {
   303→      await apiPatch('/api/settings', { key, value })
   304→      toast.success('Ayar güncellendi')
   305→    } catch (e) {
   306→      toast.error('Ayar kaydedilemedi', { description: e instanceof Error ? e.message : '' })
   307→    } finally {
   308→      setSaving(false)
   309→    }
   310→  }
   311→
   312→  const emailNotif = settings['notification.email'] !== 'false'
   313→  const inAppNotif = settings['notification.inApp'] !== 'false'
   314→  const staleAlert = settings['notification.staleCustomerAlert'] !== 'false'
   315→  const staleDays = settings['automation.thresholdDays'] || '30'
   316→  const taskReminder = settings['notification.taskReminder'] !== 'false'
   317→
   318→  return (
   319→    <Card>
   320→      <CardHeader>
   321→        <CardTitle className="text-base flex items-center gap-2">
   322→          <Bell className="w-4 h-4 text-amber-600" />
   323→          Bildirim Tercihleri
   324→          {saving && <div className="w-3.5 h-3.5 border-2 border-amber-500 border-t-transparent rounded-full animate-spin" />}
   325→        </CardTitle>
   326→        <CardDescription>
   327→          E-posta ve uygulama içi bildirimlerinizi yönetin. Değişiklikler otomatik kaydedilir.
   328→        </CardDescription>
   329→      </CardHeader>
   330→      <CardContent className="space-y-1">
   331→        <NotificationRow
   332→          icon={<Bell className="w-4 h-4" />}
   333→          title="E-posta Bildirimleri"
   334→          desc="Önemli olaylar için e-posta alın (yeni müşteri, fırsat kazanıldı, görev hatırlatma)."
   335→          checked={emailNotif}
   336→          onChange={(v) => updateSetting('notification.email', String(v))}
   337→        />
   338→        <Separator />
   339→        <NotificationRow
   340→          icon={<Bell className="w-4 h-4" />}
   341→          title="Uygulama İçi Bildirimler"
   342→          desc="Tarayıcı içerinde anlık bildirimler göster."
   343→          checked={inAppNotif}
   344→          onChange={(v) => updateSetting('notification.inApp', String(v))}
   345→        />
   346→        <Separator />
   347→        <NotificationRow
   348→          icon={<AlertTriangle className="w-4 h-4" />}
   349→          title="İletişimsiz Müşteri Uyarısı"
   350→          desc="Belirli gün sayısı boyunca iletişim kurulmayan müşteriler için uyarı göster."
   351→          checked={staleAlert}
   352→          onChange={(v) => updateSetting('notification.staleCustomerAlert', String(v))}
   353→          extra={
   354→            <div className="flex items-center gap-2 mt-3">
   355→              <Label className="text-xs text-muted-foreground">Eşik (gün):</Label>
   356→              <Input
   357→                type="number"
   358→                value={staleDays}
   359→                onChange={(e) => updateSetting('automation.thresholdDays', e.target.value)}
   360→                className="w-20 h-8"
   361→                min={1}
   362→                max={365}
   363→                disabled={!staleAlert}
   364→              />
   365→            </div>
   366→          }
   367→        />
   368→        <Separator />
   369→        <NotificationRow
   370→          icon={<Clock className="w-4 h-4" />}
   371→          title="Görev Hatırlatma Süresi"
   372→          desc="Vade tarihinden belirli saat önce hatırlatma gönder."
   373→          checked={taskReminder}
   374→          onChange={(v) => updateSetting('notification.taskReminder', String(v))}
   375→          extra={
   376→            <div className="flex items-center gap-2 mt-3">
   377→              <Label className="text-xs text-muted-foreground">Hatırlatma (saat):</Label>
   378→              <Input
   379→                type="number"
   380→                value={settings['notification.taskReminderHours'] || '24'}
   381→                onChange={(e) => updateSetting('notification.taskReminderHours', e.target.value)}
   382→                className="w-20 h-8"
   383→                min={1}
   384→                max={168}
   385→                disabled={!taskReminder}
   386→              />
   387→            </div>
   388→          }
   389→        />
   390→        <Separator />
   391→        <div className="flex items-center justify-between pt-4 text-xs text-muted-foreground">
   392→          <span>{isLoading ? 'Ayarlar yükleniyor...' : 'Değişiklikler otomatik kaydedilir'}</span>
   393→        </div>
   394→      </CardContent>
   395→    </Card>
   396→  )
   397→}
