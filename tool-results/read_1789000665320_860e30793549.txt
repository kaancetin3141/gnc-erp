   139→   139→// Ürün stok durumu hesapla
   140→   140→function getStockStatus(p: { stock: number; minStock: number }) {
   141→   141→  if (p.stock === 0) {
   142→   142→    return {
   143→   143→      label: 'Tükendi',
   144→   144→      color: 'text-red-700 bg-red-50 border-red-200 dark:bg-red-950/30 dark:text-red-300',
   145→   145→      variant: 'out' as const,
   146→   146→    }
   147→   147→  }
   148→   148→  if (p.stock <= p.minStock) {
   149→   149→    return {
   150→   150→      label: 'Düşük Stok',
   151→   151→      color: 'text-amber-700 bg-amber-50 border-amber-200 dark:bg-amber-950/30 dark:text-amber-300',
   152→   152→      variant: 'low' as const,
   153→   153→    }
   154→   154→  }
   155→   155→  return {
   156→   156→    label: 'Stokta',
   157→   157→    color: 'text-emerald-700 bg-emerald-50 border-emerald-200 dark:bg-emerald-950/30 dark:text-emerald-300',
   158→   158→    variant: 'ok' as const,
   159→   159→  }
   160→   160→}

   227→   227→function ProductFormDialog({
   228→   228→  open, onOpenChange, editProduct,
   229→   229→}: {
   230→   230→  open: boolean
   231→   231→  onOpenChange: (v: boolean) => void
   232→   232→  editProduct?: Product | null
   233→   233→}) {
   234→   234→  const qc = useQueryClient()
   235→   235→  const [form, setForm] = useState<ProductForm>(EMPTY_FORM)
   236→   236→  const [submitting, setSubmitting] = useState(false)
   237→   237→
   238→   238→  useEffect(() => {
   239→   239→    if (open) {
   240→   240→      if (editProduct) {
   241→   241→        setForm({
   242→   242→          name: editProduct.name || '',
   243→   243→          sku: editProduct.sku || '',
   244→   244→          description: editProduct.description || '',
   245→   245→          price: String(editProduct.price ?? 0),
   246→   246→          currency: editProduct.currency || 'TRY',
   247→   247→          taxRate: String(editProduct.taxRate ?? 20),
   248→   248→          // Edit modunda stok doğrudan düzenlenmez — hareketler üzerinden yönetilir
   249→   249→          stock: String(editProduct.stock ?? 0),
   250→   250→          minStock: String(editProduct.minStock ?? 0),
   251→   251→          unit: editProduct.unit || 'adet',
   252→   252→          category: editProduct.category || '',
   253→   253→        })
   254→   254→      } else {
   255→   255→        setForm(EMPTY_FORM)
   256→   256→      }
   257→   257→    }
   258→   258→  }, [open, editProduct])
   259→   259→
   260→   260→  const handleSubmit = async () => {
   261→   261→    if (!form.name.trim()) {
   262→   262→      toast.error('Ürün adı gerekli')
   263→   263→      return
   264→   264→    }
   265→   265→    setSubmitting(true)
   266→   266→    try {
   267→   267→      if (editProduct) {
   268→   268→        // PATCH — stok dahil değil
   269→   269→        await apiPatch(`/api/products/${editProduct.id}`, {
   270→   270→          name: form.name.trim(),
   271→   271→          sku: form.sku.trim() || null,
   272→   272→          description: form.description.trim() || null,
   273→   273→          price: parseFloat(form.price) || 0,
   274→   274→          currency: form.currency,
   275→   275→          taxRate: parseFloat(form.taxRate) || 0,
   276→   276→          minStock: parseInt(form.minStock) || 0,
   277→   277→          unit: form.unit.trim() || 'adet',
   278→   278→          category: form.category.trim() || null,
   279→   279→        })
   280→   280→        toast.success('Ürün güncellendi')
   281→   281→        qc.invalidateQueries({ queryKey: ['product', editProduct.id] })
   282→   282→      } else {
   283→   283→        // POST — yeni ürün
   284→   284→        await apiPost('/api/products', {
   285→   285→          name: form.name.trim(),
   286→   286→          sku: form.sku.trim() || null,
   287→   287→          description: form.description.trim() || null,
   288→   288→          price: parseFloat(form.price) || 0,
   289→   289→          currency: form.currency,
   290→   290→          taxRate: parseFloat(form.taxRate) || 0,
   291→   291→          stock: parseInt(form.stock) || 0,
   292→   292→          minStock: parseInt(form.minStock) || 0,
   293→   293→          unit: form.unit.trim() || 'adet',
   294→   294→          category: form.category.trim() || null,
   295→   295→        })
   296→   296→        toast.success('Ürün eklendi')
   297→   297→      }
   298→   298→      qc.invalidateQueries({ queryKey: ['products'] })
   299→   299→      onOpenChange(false)
   300→   300→    } catch (e) {
   301→   301→      toast.error(e instanceof Error ? e.message : 'İşlem başarısız')
   302→   302→    } finally {
   303→   303→      setSubmitting(false)
   304→   304→    }
   305→   305→  }
   306→   306→
   307→   307→  return (
   308→   308→    <Dialog open={open} onOpenChange={onOpenChange}>
   309→   309→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto custom-scroll">
   310→   310→        <DialogHeader>
   311→   311→          <DialogTitle className="flex items-center gap-2">
   312→   312→            <Package className="w-5 h-5 text-emerald-600" />
   313→   313→            {editProduct ? 'Ürünü Düzenle' : 'Yeni Ürün'}
   314→   314→          </DialogTitle>
   315→   315→          <DialogDescription>
   316→   316→            {editProduct
   317→   317→              ? 'Ürün bilgilerini güncelleyin. Stok güncellemeleri için ürün detayındaki stok hareketi formunu kullanın.'
   318→   318→              : 'Yeni ürün ekleyin. Başlangıç stoğu için bir "Açılış stoğu" hareketi otomatik oluşturulur.'}
   319→   319→          </DialogDescription>
   320→   320→        </DialogHeader>
   321→   321→
   322→   322→        <div className="grid grid-cols-2 gap-4 py-2">
   323→   323→          {/* Ad */}
   324→   324→          <div className="space-y-1.5 col-span-2">
   325→   325→            <Label htmlFor="p-name" className="text-xs">Ürün Adı *</Label>
   326→   326→            <Input
   327→   327→              id="p-name"
   328→   328→              value={form.name}
   329→   329→              onChange={(e) => setForm({ ...form, name: e.target.value })}
   330→   330→              placeholder="Örn. CRM Pro Paket"
   331→   331→            />
   332→   332→          </div>
   333→   333→
   334→   334→          {/* SKU */}
   335→   335→          <div className="space-y-1.5">
   336→   336→            <Label htmlFor="p-sku" className="text-xs">SKU / Stok Kodu</Label>
   337→   337→            <Input
   338→   338→              id="p-sku"
   339→   339→              value={form.sku}
   340→   340→              onChange={(e) => setForm({ ...form, sku: e.target.value })}
   341→   341→              placeholder="CRM-PRO"
   342→   342→            />
   343→   343→          </div>
   344→   344→
   345→   345→          {/* Kategori */}
   346→   346→          <div className="space-y-1.5">
   347→   347→            <Label htmlFor="p-cat" className="text-xs">Kategori</Label>
   348→   348→            <Input
   349→   349→              id="p-cat"
   350→   350→              value={form.category}
   351→   351→              onChange={(e) => setForm({ ...form, category: e.target.value })}
   352→   352→              placeholder="Yazılım, Hizmet, Donanım..."
   353→   353→            />
   354→   354→          </div>
   355→   355→
   356→   356→          {/* Açıklama */}
   357→   357→          <div className="space-y-1.5 col-span-2">
   358→   358→            <Label htmlFor="p-desc" className="text-xs">Açıklama</Label>
   359→   359→            <Textarea
   360→   360→              id="p-desc"
   361→   361→              value={form.description}
   362→   362→              onChange={(e) => setForm({ ...form, description: e.target.value })}
   363→   363→              placeholder="Ürün açıklaması..."
   364→   364→              rows={2}
   365→   365→            />
   366→   366→          </div>
   367→   367→
   368→   368→          {/* Fiyat */}
   369→   369→          <div className="space-y-1.5">
   370→   370→            <Label htmlFor="p-price" className="text-xs">Fiyat</Label>
   371→   371→            <Input
   372→   372→              id="p-price"
   373→   373→              type="number"
   374→   374→              min="0"
   375→   375→              step="0.01"
   376→   376→              value={form.price}
   377→   377→              onChange={(e) => setForm({ ...form, price: e.target.value })}
   378→   378→            />
   379→   379→          </div>
   380→   380→
   381→   381→          {/* Para birimi */}
   382→   382→          <div className="space-y-1.5">
   383→   383→            <Label className="text-xs">Para Birimi</Label>
   384→   384→            <Select
   385→   385→              value={form.currency}
   386→   386→              onValueChange={(v) => setForm({ ...form, currency: v })}
   387→   387→            >
   388→   388→              <SelectTrigger className="w-full">
   389→   389→                <SelectValue />
   390→   390→              </SelectTrigger>
   391→   391→              <SelectContent>
   392→   392→                {CURRENCIES.map((c) => (
   393→   393→                  <SelectItem key={c.code} value={c.code}>
   394→   394→                    {c.symbol} {c.code} — {c.label}

   396→   396→                ))}
   397→   397→              </SelectContent>
   398→   398→            </Select>
   399→   399→          </div>
   400→   400→
   401→   401→          {/* KDV */}
   402→   402→          <div className="space-y-1.5">
   403→   403→            <Label htmlFor="p-tax" className="text-xs">KDV (%)</Label>
   404→   404→            <Input
   405→   405→              id="p-tax"
   406→   406→              type="number"
   407→   407→              min="0"
   408→   408→              max="100"
   409→   409→              step="1"
   410→   410→              value={form.taxRate}
   411→   411→              onChange={(e) => setForm({ ...form, taxRate: e.target.value })}
   412→   412→            />
   413→   413→          </div>
   414→   414→
   415→   415→          {/* Birim */}
   416→   416→          <div className="space-y-1.5">
   417→   417→            <Label htmlFor="p-unit" className="text-xs">Birim</Label>
   418→   418→            <Input
   419→   419→              id="p-unit"
   420→   420→              value={form.unit}
   421→   421→              onChange={(e) => setForm({ ...form, unit: e.target.value })}
   422→   422→              placeholder="adet, saat, kg, lt..."
   423→   423→            />
   424→   424→          </div>
   425→   425→
   426→   426→          {/* Min stok */}
   427→   427→          <div className="space-y-1.5">
   428→   428→            <Label htmlFor="p-min" className="text-xs">Min. Stok</Label>
   429→   429→            <Input
   430→   430→              id="p-min"
   431→   431→              type="number"
   432→   432→              min="0"
   433→   433→              step="1"
   434→   434→              value={form.minStock}
   435→   435→              onChange={(e) => setForm({ ...form, minStock: e.target.value })}
   436→   436→            />
   437→   437→          </div>
   438→   438→
   439→   439→          {/* Başlangıç stoğu (yalnızca yeni ürün) */}
   440→   440→          {!editProduct && (
   441→   441→            <div className="space-y-1.5">
   442→   442→              <Label htmlFor="p-stock" className="text-xs">Başlangıç Stoğu</Label>
   443→   443→              <Input
   444→   444→                id="p-stock"
   445→   445→                type="number"
   446→   446→                min="0"
   447→   447→                step="1"
   448→   448→                value={form.stock}
   449→   449→                onChange={(e) => setForm({ ...form, stock: e.target.value })}
   450→   450→              />
   451→   451→              <p className="text-[10px] text-muted-foreground">
   452→   452→                Bir &quot;Açılış stoğu&quot; hareketi oluşturulur.
   453→   453→              </p>
   454→   454→            </div>
   455→   455→          )}
   456→   456→          {editProduct && (
   457→   457→            <div className="space-y-1.5">
   458→   458→              <Label className="text-xs">Mevcut Stok</Label>
   459→   459→              <div className="h-9 flex items-center px-3 rounded-md border border-input bg-muted/40 text-sm font-medium">
   460→   460→                {editProduct.stock} {editProduct.unit}
   461→   461→                <span className="ml-auto text-[10px] text-muted-foreground">Stok hareketi ile güncellenir</span>
   462→   462→              </div>
   463→   463→            </div>
   464→   464→          )}
   465→   465→        </div>
   466→   466→
   467→   467→        <DialogFooter>
   468→   468→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={submitting}>
   469→   469→            İptal
   470→   470→          </Button>
   471→   471→          <Button onClick={handleSubmit} disabled={submitting || !form.name.trim()}>
   472→   472→            {submitting && <RefreshCw className="w-4 h-4 mr-1.5 animate-spin" />}
   473→   473→            {editProduct ? 'Güncelle' : 'Ürün Ekle'}
   474→   474→          </Button>
   475→   475→        </DialogFooter>
   476→   476→      </DialogContent>
   477→   477→    </Dialog>
   478→   478→  )
   479→   479→}

   485→   485→function StockMovementForm({

   490→   490→  currentStock: number
   491→   491→  onDone: () => void
   492→   492→}) {
   493→   493→  const qc = useQueryClient()
   494→   494→  const [type, setType] = useState<'giris' | 'cikis' | 'duzeltme'>('giris')
   495→   495→  const [quantity, setQuantity] = useState('')
   496→   496→  const [reason, setReason] = useState('')
   497→   497→  const [submitting, setSubmitting] = useState(false)
   498→   498→
   499→   499→  const handleSubmit = async () => {
   500→   500→    const q = parseInt(quantity)
   501→   501→    if (!Number.isFinite(q)) {
   502→   502→      toast.error('Miktar girin')
   503→   503→      return
   504→   504→    }
   505→   505→    if (type === 'giris' && q <= 0) {
   506→   506→      toast.error('Giriş miktarı pozitif olmalı')
   507→   507→      return
   508→   508→    }
   509→   509→    if (type === 'cikis' && q <= 0) {
   510→   510→      toast.error('Çıkış miktarı pozitif olmalı')
   511→   511→      return
   512→   512→    }

   514→   514→      toast.error(`Yetersiz stok (mevcut: ${currentStock} ${unit})`)
   515→   515→      return
   516→   516→    }
