   707→// IRSALİYE (DISPATCH NOTE) — Sevk irsaliyesi, ağırlık takibi, palet
   708→// ============================================================
   709→// Numara formatı: IRS-2026-001, IRS-2026-002 (yıl + tenant bazlı sıra)
   710→// Durum: taslak | hazir | sevk_edildi | teslim_edildi | iptal
   711→// Ağırlık hesabı:
   712→//   totalNetWeight       = sum(line.weightPerUnit * line.qty)
   713→//   totalPackagingWeight = sum(product.packagingWeight * line.qty) (opsiyonel)
   714→//   palletWeight         = kullanıcı girişi (varsayılan 20 kg)
   715→//   totalGrossWeight     = net + packaging + pallet
   716→model Irsaliye {
   717→  id                   String    @id @default(cuid())
   718→  tenantId             String
   719→  tenant               Tenant    @relation(fields: [tenantId], references: [id], onDelete: Cascade)
   720→  orderId              String?
   721→  order                Order?    @relation(fields: [orderId], references: [id], onDelete: SetNull)
   722→  customerId           String
   723→  customer             Customer  @relation(fields: [customerId], references: [id])
   724→  number               String // IRS-2026-001
   725→  date                 DateTime  @default(now())
   726→  status               String    @default("taslak") // taslak | hazir | sevk_edildi | teslim_edildi | iptal
   727→  // Ağırlık takibi
   728→  totalNetWeight       Float? // sum of line.weightPerUnit * line.qty
   729→  totalPackagingWeight Float? // sum of product.packagingWeight * line.qty
   730→  palletWeight         Float? // user-entered pallet weight (default 20 kg)
   731→  totalGrossWeight     Float? // net + packaging + pallet
   732→  palletCount          Int?
   733→  palletType           String? // tahta | plastik | tek_kullanilik | eur_pallet | other
   734→  // Sevkiyat bilgileri
   735→  shippingAddress      String?
   736→  carrier              String? // yurtici | aras | mng | ptt | ups | fedex | dhl | other
   737→  trackingNo           String?
   738→  // Notlar
   739→  notes                String?
   740→  createdById          String?
   741→  createdAt            DateTime  @default(now())
   742→  updatedAt            DateTime  @updatedAt
   743→  deliveredAt          DateTime?
   744→
   745→  lines IrsaliyeLine[]
   746→
   747→  @@unique([tenantId, number])
   748→  @@index([tenantId, status])
   749→}
