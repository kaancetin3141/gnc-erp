    33→} from '@/lib/appointment-utils'

    46→interface Staff {
    47→  id: string
    48→  name: string
    49→  title: string | null
    50→  photo: string | null
    51→}
    52→
    53→interface Service {
    54→  id: string
    55→  name: string
    56→  duration: number
    57→  price: number
    58→  currency: string
    59→}
    60→
    61→interface Appointment {
    62→  id: string
    63→  staffId: string | null
    64→  serviceId: string | null
    65→  customerName: string
    66→  customerPhone: string
    67→  customerEmail: string | null
    68→  customerNote: string | null
    69→  date: string
    70→  endTime: string | null
    71→  status: string
    72→  price: number
    73→  notes: string | null
    74→  source: string
    75→  staff?: Staff | null
    76→  service?: Service | null
    77→}
    78→
    79→type ApptsResponse = Appointment[]
    80→interface StaffResponse { items: Staff[] }
    81→interface ServiceResponse { items: Service[] }

    84→  autoApprove?: boolean
    85→}

    88→// Calendar (Day/Week view)
    89→// ============================================================
    90→
    91→interface CalendarProps {
    92→  providerId: string
    93→}
    94→
    95→export function AppointmentCalendar({ providerId }: CalendarProps) {
    96→  const qc = useQueryClient()
    97→  const [view, setView] = useState<'day' | 'week'>('day')
    98→  const [currentDate, setCurrentDate] = useState(() => {
    99→    const d = new Date()
   100→    d.setHours(0, 0, 0, 0)
   101→    return d
   102→  })
   103→  const [staffFilter, setStaffFilter] = useState<string>('all')
   104→
   105→  // Detail dialog
   106→  const [detailAppt, setDetailAppt] = useState<Appointment | null>(null)
   107→
   108→  // Provider (çalışma saatleri)
   109→  const { data: providerData } = useQuery({
   110→    queryKey: ['appointment-provider', providerId],
   111→    queryFn: () => apiGet<ProviderResponse>(`/api/appointments/providers/${providerId}`),
   112→    enabled: !!providerId,
   113→  })
   114→
   115→  const workingHours: WorkingHours = useMemo(() => {
   116→    if (!providerData?.workingHours) return {}
   117→    try {
   118→      return JSON.parse(providerData.workingHours) as WorkingHours
   119→    } catch {
   120→      return {}
   121→    }
   122→  }, [providerData])
   123→
   124→  const { data: staffData } = useQuery({
   125→    queryKey: ['appointment-staff', providerId],
   126→    queryFn: () => apiGet<StaffResponse>(`/api/appointments/providers/${providerId}/staff`),
   127→    enabled: !!providerId,
   128→  })
   129→  const staffList = staffData?.items ?? []
   130→
   131→  const { data: serviceData } = useQuery({
   132→    queryKey: ['appointment-services', providerId],
   133→    queryFn: () => apiGet<ServiceResponse>(`/api/appointments/providers/${providerId}/services`),
   134→    enabled: !!providerId,
   135→  })
   136→  const services = serviceData?.items ?? []
   137→
   138→  const dateRange = useMemo(() => {
   139→    const start = new Date(currentDate)
   140→    start.setHours(0, 0, 0, 0)
   141→    if (view === 'week') {
   142→      const day = (start.getDay() + 6) % 7
   143→      start.setDate(start.getDate() - day)
   144→    }
   145→    const days: Date[] = []
   146→    const numDays = view === 'week' ? 7 : 1
   147→    for (let i = 0; i < numDays; i++) {
   148→      const d = new Date(start)
   149→      d.setDate(start.getDate() + i)
   150→      days.push(d)
   151→    }
   152→    return days
   153→  }, [currentDate, view])
   154→
   155→  const rangeStart = dateRange[0]
   156→  const rangeEnd = new Date(dateRange[dateRange.length - 1].getTime() + 24 * 60 * 60_000)
   157→
   158→  const { data: apptData, isLoading } = useQuery({
   159→    queryKey: ['appointment-appointments', providerId, rangeStart.toISOString(), rangeEnd.toISOString(), staffFilter],
   160→    queryFn: () => {
   161→      const params = new URLSearchParams({
   162→        startDate: rangeStart.toISOString(),
   163→        endDate: rangeEnd.toISOString(),
   164→      })
   165→      if (staffFilter !== 'all') params.set('staffId', staffFilter)
   166→      return apiGet<ApptsResponse>(
   167→        `/api/appointments/providers/${providerId}/appointments?${params.toString()}`,
   168→      )
   169→    },
   170→    enabled: !!providerId,
   171→  })
   172→  const appointments = useMemo(() => (Array.isArray(apptData) ? apptData : (apptData as unknown as { items?: Appointment[] })?.items ?? []), [apptData])
   173→
   174→  const { visStart, visEnd } = useMemo(() => {
   175→    let minMin = 24 * 60
   176→    let maxMin = 0
   177→    for (const day of dateRange) {
   178→      const dk = dayKeyFromDate(day)
   179→      const sched: DaySchedule | undefined = workingHours[dk]
   180→      if (!sched || sched.closed || !sched.start || !sched.end) continue
   181→      minMin = Math.min(minMin, timeToMinutes(sched.start))
   182→      maxMin = Math.max(maxMin, timeToMinutes(sched.end))
   183→    }
   184→    if (minMin === 24 * 60) minMin = 9 * 60
   185→    if (maxMin === 0) maxMin = 19 * 60
   186→    return { visStart: minMin, visEnd: maxMin }
   187→  }, [dateRange, workingHours])
   188→
   189→  const hourSlots = useMemo(() => {
   190→    const slots: { min: number; label: string }[] = []
   191→    for (let m = visStart; m < visEnd; m += 30) {
   192→      slots.push({ min: m, label: minutesToTime(m) })
   193→    }
   194→    return slots
   195→  }, [visStart, visEnd])
   196→
   197→  const apptsByDay = useMemo(() => {
   198→    const map = new Map<string, Appointment[]>()
   199→    for (const appt of appointments) {
   200→      const d = new Date(appt.date)
   201→      const key = d.toDateString()
   202→      if (!map.has(key)) map.set(key, [])
   203→      map.get(key)!.push(appt)
   204→    }
   205→    return map
   206→  }, [appointments])
   207→
   208→  const [dialogOpen, setDialogOpen] = useState(false)
   209→  const [editAppt, setEditAppt] = useState<Appointment | null>(null)
   210→  const [form, setForm] = useState({
   211→    staffId: 'any',
   212→    serviceId: '',
   213→    customerName: '',
   214→    customerPhone: '',
   215→    customerEmail: '',
   216→    customerNote: '',
   217→    date: '',
   218→    time: '',
   219→    status: 'beklemede' as AppointmentStatus,
   220→  })
   221→  const [saving, setSaving] = useState(false)
   222→  const [deleteTarget, setDeleteTarget] = useState<Appointment | null>(null)
   223→
   224→  function prevPeriod() {
   225→    const d = new Date(currentDate)
   226→    d.setDate(d.getDate() - (view === 'week' ? 7 : 1))
   227→    setCurrentDate(d)
   228→  }
   229→  function nextPeriod() {
   230→    const d = new Date(currentDate)
   231→    d.setDate(d.getDate() + (view === 'week' ? 7 : 1))
   232→    setCurrentDate(d)
   233→  }
   234→  function goToday() {
   235→    const d = new Date()
   236→    d.setHours(0, 0, 0, 0)
   237→    setCurrentDate(d)
   238→  }
   239→
   240→  function openCreate(date?: Date, hour?: number, minute?: number) {
   241→    setEditAppt(null)
   242→    const targetDate = date ?? new Date()
   243→    targetDate.setHours(hour ?? 10, minute ?? 0, 0, 0)
   244→    setForm({
   245→      staffId: staffFilter !== 'all' ? staffFilter : 'any',
   246→      serviceId: services[0]?.id ?? '',
   247→      customerName: '',
   248→      customerPhone: '',
   249→      customerEmail: '',
   250→      customerNote: '',
   251→      date: targetDate.toISOString().slice(0, 10),
   252→      time: `${String(targetDate.getHours()).padStart(2, '0')}:${String(targetDate.getMinutes()).padStart(2, '0')}`,
   253→      status: 'beklemede',
   254→    })
   255→    setDialogOpen(true)
   256→  }
   257→
   258→  function openEdit(appt: Appointment) {
   259→    setDetailAppt(null)
   260→    setEditAppt(appt)
   261→    const d = new Date(appt.date)
   262→    setForm({
   263→      staffId: appt.staffId ?? 'any',
   264→      serviceId: appt.serviceId ?? '',
   265→      customerName: appt.customerName,
   266→      customerPhone: appt.customerPhone,
   267→      customerEmail: appt.customerEmail ?? '',
   268→      customerNote: appt.customerNote ?? '',
   269→      date: d.toISOString().slice(0, 10),
   270→      time: `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`,
   271→      status: appt.status as AppointmentStatus,
   272→    })
   273→    setDialogOpen(true)
   274→  }
   275→
   276→  async function handleSave() {
   277→    if (!form.customerName.trim()) {
   278→      toast.error('Müşteri adı gerekli')
   279→      return
   280→    }
   281→    if (!form.customerPhone.trim()) {
   282→      toast.error('Müşteri telefonu gerekli')
   283→      return
   284→    }
   285→    if (!form.serviceId) {
   286→      toast.error('Hizmet seçimi gerekli')
   287→      return
   288→    }
   289→    if (!form.date || !form.time) {
   290→      toast.error('Tarih ve saat gerekli')
   291→      return
   292→    }
   293→    setSaving(true)
   294→    try {
   295→      const [y, m, d] = form.date.split('-').map(Number)
   296→      const [hh, mm] = form.time.split(':').map(Number)
   297→      const start = new Date(y, m - 1, d, hh, mm, 0, 0)
   298→      const payload = {
   299→        staffId: form.staffId === 'any' ? undefined : form.staffId,
   300→        serviceId: form.serviceId,
   301→        customerName: form.customerName,
   302→        customerPhone: form.customerPhone,
   303→        customerEmail: form.customerEmail || undefined,
   304→        customerNote: form.customerNote || undefined,
   305→        date: start.toISOString(),
   306→        status: form.status,
   307→      }
   308→      if (editAppt) {
   309→        await apiPatch(
   310→          `/api/appointments/providers/${providerId}/appointments/${editAppt.id}`,
   311→          { ...payload, staffId: form.staffId },
   312→        )
   313→        toast.success('Randevu güncellendi')
   314→      } else {
   315→        await apiPost(
   316→          `/api/appointments/providers/${providerId}/appointments`,
   317→          payload,
   318→        )
   319→        toast.success('Randevu oluşturuldu')
   320→      }
   321→      qc.invalidateQueries({ queryKey: ['appointment-appointments', providerId] })
   322→      setDialogOpen(false)
   323→    } catch (e) {
   324→      toast.error(e instanceof Error ? e.message : 'İşlem başarısız')
   325→    } finally {
   326→      setSaving(false)
   327→    }
   328→  }
   329→
   330→  async function handleStatusChange(appt: Appointment, status: AppointmentStatus) {
   331→    try {
   332→      await apiPatch(
   333→        `/api/appointments/providers/${providerId}/appointments/${appt.id}`,
   334→        { status },
   335→      )
   336→      qc.invalidateQueries({ queryKey: ['appointment-appointments', providerId] })
   337→      setDetailAppt(null)
   338→      toast.success(`Durum: ${getStatusMeta(status).label}`)
   339→    } catch (e) {
   340→      toast.error(e instanceof Error ? e.message : 'Güncellenemedi')
   341→    }
   342→  }
   343→
   344→  async function handleDelete() {
   345→    if (!deleteTarget) return
   346→    try {
   347→      await apiDelete(
   348→        `/api/appointments/providers/${providerId}/appointments/${deleteTarget.id}`,
   349→      )
   350→      qc.invalidateQueries({ queryKey: ['appointment-appointments', providerId] })
   351→      toast.success('Randevu iptal edildi')
   352→      setDeleteTarget(null)
   353→    } catch (e) {
   354→      toast.error(e instanceof Error ? e.message : 'İptal edilemedi')
   355→    }
   356→  }
   357→
   358→  if (isLoading) {
   359→    return (
   360→      <Card>
   361→        <Skeleton className="h-96 w-full" />
   362→      </Card>
   363→    )
   364→  }
   365→
   366→  return (
   367→    <div className="space-y-3">
   368→      {/* Toolbar */}
   369→      <Card>
   370→        <CardContent className="p-3">
   371→          <div className="flex flex-col sm:flex-row sm:items-center gap-2 sm:gap-3">
   372→            <div className="flex items-center gap-1.5 shrink-0">
   373→              <Button size="icon" variant="outline" className="h-9 w-9" onClick={prevPeriod}>
   374→                <ChevronLeft className="w-4 h-4" />
   375→              </Button>
   376→              <Button size="sm" variant="outline" onClick={goToday}>
   377→                Bugün
   378→              </Button>
   379→              <Button size="icon" variant="outline" className="h-9 w-9" onClick={nextPeriod}>
   380→                <ChevronRight className="w-4 h-4" />
   381→              </Button>
   382→            </div>
   383→            <div className="flex items-center gap-2 flex-1 min-w-0">
   384→              <CalendarIcon className="w-4 h-4 text-emerald-600 shrink-0" />
   385→              <span className="font-semibold text-sm truncate">
   386→                {view === 'day'
   387→                  ? currentDate.toLocaleDateString('tr-TR', { day: 'numeric', month: 'long', year: 'numeric', weekday: 'long' })
   388→                  : `${rangeStart.toLocaleDateString('tr-TR', { day: 'numeric', month: 'short' })} - ${new Date(rangeEnd.getTime() - 86400000).toLocaleDateString('tr-TR', { day: 'numeric', month: 'short', year: 'numeric' })}`}
   389→              </span>
   390→            </div>
   391→            <div className="flex items-center gap-2 shrink-0">
   392→              <Select value={staffFilter} onValueChange={setStaffFilter}>
   393→                <SelectTrigger className="h-9 w-40 text-xs">
   394→                  <SelectValue placeholder="Tüm personel" />
   395→                </SelectTrigger>
   396→                <SelectContent>
   397→                  <SelectItem value="all">Tüm Personel</SelectItem>
   398→                  {staffList.map((s) => (
   399→                    <SelectItem key={s.id} value={s.id}>{s.name}</SelectItem>
   400→                  ))}
   401→                </SelectContent>
   402→              </Select>
   403→              <div className="flex border rounded-md overflow-hidden">
   404→                <Button
   405→                  size="sm"
   406→                  variant={view === 'day' ? 'default' : 'ghost'}
   407→                  className={cn('rounded-none h-9', view === 'day' && 'bg-emerald-600 hover:bg-emerald-700')}
   408→                  onClick={() => setView('day')}
   409→                >
   410→                  Gün
   411→                </Button>
   412→                <Button
   413→                  size="sm"
   414→                  variant={view === 'week' ? 'default' : 'ghost'}
   415→                  className={cn('rounded-none h-9', view === 'week' && 'bg-emerald-600 hover:bg-emerald-700')}
   416→                  onClick={() => setView('week')}
   417→                >
   418→                  Hafta
   419→                </Button>
   420→              </div>
   421→              <Button size="sm" onClick={() => openCreate()} className="bg-emerald-600 hover:bg-emerald-700">
   422→                <Plus className="w-4 h-4 mr-1" />
   423→                <span className="hidden sm:inline">Randevu</span>
   424→              </Button>
   425→            </div>
   426→          </div>
   427→        </CardContent>
   428→      </Card>
   429→
   430→      {/* Calendar Grid */}
   431→      <Card>
   432→        <CardContent className="p-0 overflow-x-auto">
   433→          <div className={cn('grid min-w-[640px]', view === 'week' ? 'grid-cols-[60px_repeat(7,1fr)]' : 'grid-cols-[60px_1fr]')}>
   434→            <div className="border-b border-r bg-muted/30 p-2 text-[10px] text-muted-foreground text-right">
   435→              S
   436→            </div>
   437→            {dateRange.map((d) => {
   438→              const isToday = d.toDateString() === new Date().toDateString()
   439→              const dk = dayKeyFromDate(d)
   440→              const sched = workingHours[dk]
   441→              const isClosed = !sched || sched.closed
   442→              return (
   443→                <div
   444→                  key={d.toISOString()}
   445→                  className={cn(
   446→                    'border-b border-r p-2 text-center',
   447→                    isToday && 'bg-emerald-50 dark:bg-emerald-950/20',
   448→                    isClosed && 'bg-muted/20',
   449→                  )}
   450→                >
   451→                  <div className="text-[10px] uppercase tracking-wide text-muted-foreground">
   452→                    {d.toLocaleDateString('tr-TR', { weekday: 'short' })}
   453→                  </div>
   454→                  <div className={cn('text-sm font-semibold', isToday && 'text-emerald-600')}>
   455→                    {d.getDate()}
   456→                  </div>
   457→                  {isClosed && (
   458→                    <div className="text-[9px] text-muted-foreground">kapalı</div>
   459→                  )}
   460→                </div>
   461→              )
   462→            })}
   463→
   464→            {hourSlots.map((slot) => (
   465→              <SlotRow
   466→                key={slot.min}
   467→                slot={slot}
   468→                dateRange={dateRange}
   469→                apptsByDay={apptsByDay}
   470→                workingHours={workingHours}
   471→                onSlotClick={(date) => openCreate(date, slot.min / 60, slot.min % 60)}
   472→                onApptClick={(appt) => setDetailAppt(appt)}
   473→              />
   474→            ))}
   475→          </div>
   476→        </CardContent>
   477→      </Card>
   478→
   479→      {/* Legend */}
   480→      <div className="flex items-center gap-3 flex-wrap text-[10px] text-muted-foreground">
   481→        <span>Durum:</span>
   482→        {APPOINTMENT_STATUSES.slice(0, 5).map((s) => (
   483→          <span key={s.value} className="flex items-center gap-1">
   484→            <span className={cn('w-2.5 h-2.5 rounded-sm', s.bg)} />
   485→            {s.label}
   486→          </span>
   487→        ))}
   488→      </div>
   489→
   490→      {/* Detail Dialog */}
   491→      <Dialog open={!!detailAppt} onOpenChange={(o) => !o && setDetailAppt(null)}>
   492→        <DialogContent className="sm:max-w-md">
   493→          {detailAppt && (() => {
   494→            const meta = getStatusMeta(detailAppt.status)
   495→            const d = new Date(detailAppt.date)
   496→            const end = detailAppt.endTime ? new Date(detailAppt.endTime) : null
   497→            const waLink = whatsappLink(
   498→              detailAppt.customerPhone,
   499→              `Merhaba ${detailAppt.customerName}, ${d.toLocaleDateString('tr-TR')} ${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')} tarihli randevunuz hakkında hatırlatma.`,
   500→            )
   501→            return (
   502→              <>
   503→                <DialogHeader>
   504→                  <DialogTitle className="flex items-center gap-2">
   505→                    <CalendarDays className="w-4 h-4 text-emerald-600" />
   506→                    Randevu Detayı
   507→                  </DialogTitle>
   508→                  <DialogDescription>
   509→                    {d.toLocaleDateString('tr-TR', { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' })}
   510→                    {' · '}
   511→                    {String(d.getHours()).padStart(2, '0')}:{String(d.getMinutes()).padStart(2, '0')}
   512→                    {end && ` - ${String(end.getHours()).padStart(2, '0')}:${String(end.getMinutes()).padStart(2, '0')}`}
   513→                  </DialogDescription>
   514→                </DialogHeader>
   515→                <div className="space-y-3">
   516→                  <div className={cn('rounded-lg p-3 border', meta.bg, meta.border)}>
   517→                    <div className="flex items-center justify-between">
   518→                      <div className="font-semibold text-base">{detailAppt.customerName}</div>
   519→                      <Badge className={cn(meta.bg, meta.color, 'border-0')}>{meta.label}</Badge>
   520→                    </div>
   521→                  </div>
   522→                  <div className="grid grid-cols-2 gap-3 text-sm">
   523→                    <div>
   524→                      <div className="text-xs text-muted-foreground">Hizmet</div>
   525→                      <div className="font-medium">{detailAppt.service?.name ?? '—'}</div>
   526→                      {detailAppt.service && (
   527→                        <div className="text-xs text-muted-foreground">
   528→                          {detailAppt.service.duration} dk · {formatCurrency(detailAppt.service.price, detailAppt.service.currency)}
   529→                        </div>
   530→                      )}
   531→                    </div>
   532→                    <div>
   533→                      <div className="text-xs text-muted-foreground">Personel</div>
   534→                      <div className="font-medium">{detailAppt.staff?.name ?? 'Herhangi biri'}</div>
   535→                      {detailAppt.staff?.title && (
   536→                        <div className="text-xs text-muted-foreground">{detailAppt.staff.title}</div>

   795→// Slot row — bir zaman dilimi için tüm günleri render eder

   797→
   798→function SlotRow({
   799→  slot,
   800→  dateRange,
   801→  apptsByDay,
   802→  workingHours,
   803→  onSlotClick,
   804→  onApptClick,
   805→}: {
   806→  slot: { min: number; label: string }
   807→  dateRange: Date[]
   808→  apptsByDay: Map<string, Appointment[]>
   809→  workingHours: WorkingHours
   810→  onSlotClick: (date: Date) => void
   811→  onApptClick: (appt: Appointment) => void
   812→}) {
   813→  const isHourLine = slot.min % 60 === 0
   814→  return (
   815→    <>
   816→      <div
   817→        className={cn(
   818→          'border-r border-b bg-muted/20 p-1 text-[10px] text-muted-foreground text-right align-top',
   819→          isHourLine ? 'border-b-border' : 'border-b-muted/40',
   820→        )}
   821→      >
   822→        {isHourLine ? slot.label : ''}
   823→      </div>
   824→      {dateRange.map((d) => {
   825→        const dk = dayKeyFromDate(d)
   826→        const sched = workingHours[dk]
   827→        const isClosed = !sched || sched.closed || !sched.start || !sched.end
   828→        const inHours = !isClosed &&
   829→          slot.min >= timeToMinutes(sched!.start!) &&
   830→          slot.min < timeToMinutes(sched!.end!)
   831→
   832→        const dayAppts = apptsByDay.get(d.toDateString()) ?? []
   833→        const slotAppts = dayAppts.filter((a) => {
   834→          const ad = new Date(a.date)
   835→          const aMin = ad.getHours() * 60 + ad.getMinutes()
   836→          return aMin >= slot.min && aMin < slot.min + 30
   837→        })
   838→
   839→        return (
   840→          <div
   841→            key={`${d.toISOString()}-${slot.min}`}
   842→            className={cn(
   843→              'border-r border-b relative h-7 group',
   844→              isHourLine ? 'border-b-border' : 'border-b-muted/40',
   845→              inHours ? 'bg-card hover:bg-emerald-50/30 dark:hover:bg-emerald-950/10 cursor-pointer' : 'bg-muted/10',
   846→            )}
   847→            onClick={() => inHours && onSlotClick(new Date(d.getFullYear(), d.getMonth(), d.getDate(), Math.floor(slot.min / 60), slot.min % 60))}
   848→          >
   849→            {slotAppts.map((appt) => {
   850→              const meta = getStatusMeta(appt.status)
   851→              const isCancelled = appt.status === 'iptal' || appt.status === 'reddedildi'
   852→              return (
   853→                <button
   854→                  key={appt.id}
   855→                  type="button"
   856→                  onClick={(e) => { e.stopPropagation(); onApptClick(appt) }}
   857→                  className={cn(
   858→                    'absolute inset-x-0.5 top-0.5 px-1.5 py-0.5 rounded text-[10px] text-left truncate transition-shadow hover:shadow-md z-10',
   859→                    meta.bg, meta.color,
   860→                    isCancelled && 'opacity-50 line-through',
   861→                  )}
   862→                  title={`${appt.customerName} · ${appt.service?.name ?? ''} · ${appt.staff?.name ?? 'Herhangi biri'}`}
   863→                >
   864→                  <span className="font-semibold">{appt.customerName}</span>
   865→                  {appt.service && <span className="opacity-80"> · {appt.service.name}</span>}
   866→                </button>
   867→              )
   868→            })}
   869→          </div>
   870→        )
   871→      })}
   872→    </>
   873→  )
   874→}
