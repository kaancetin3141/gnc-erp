    92→// ---- Aktivite renk noktası ----
    93→function activityDot(lastActivityAt: string | null) {
    94→  const d = daysSince(lastActivityAt)
    95→  if (d === null) return { color: 'bg-gray-400', label: 'İletişim yok' }
    96→  if (d <= 7) return { color: 'bg-emerald-500', label: `${d} gün önce` }
    97→  if (d <= 30) return { color: 'bg-amber-500', label: `${d} gün önce` }
    98→  return { color: 'bg-red-500', label: `${d} gün önce` }
    99→}

   411→export function CustomerList() {
   412→  const { user, openCustomer } = useAppStore()
   413→  const qc = useQueryClient()
   414→
   415→  // Filtre state
   416→  const [search, setSearch] = useState('')
   417→  const [sector, setSector] = useState('')
   418→  const [city, setCity] = useState('')
   419→  const [district, setDistrict] = useState('')
   420→  const [country, setCountry] = useState('')
   421→  const [segment, setSegment] = useState('')
   422→  const [status, setStatus] = useState('')
   423→  const [ownerId, setOwnerId] = useState('')
   424→  const [staleOnly, setStaleOnly] = useState(() => {
   425→    if (typeof window === 'undefined') return false
   426→    return new URL(window.location.href).searchParams.get('stale') === 'true'
   427→  })
   428→  const [filtersOpen, setFiltersOpen] = useState(false)
   429→  const [addOpen, setAddOpen] = useState(false)
   430→
   431→  // Query parametreleri
   432→  const params = useMemo(() => {
   433→    const p: Record<string, string> = {}
   434→    if (search) p.search = search
   435→    if (sector) p.sector = sector
   436→    if (city) p.city = city
   437→    if (district) p.district = district
   438→    if (country) p.country = country
   439→    if (segment) p.segment = segment
   440→    if (status) p.status = status
   441→    if (ownerId) p.ownerId = ownerId
   442→    if (staleOnly) p.stale = 'true'
   443→    p.limit = '100'
   444→    return p
   445→  }, [search, sector, city, district, country, segment, status, ownerId, staleOnly])
   446→
   447→  // Müşteri listesi
   448→  const { data, isLoading, isFetching, refetch } = useQuery({
   449→    queryKey: qk.customers(params),
   450→    queryFn: () => {
   451→      const qs = new URLSearchParams(params).toString()
   452→      return apiGet<CustomerListResponse>(`/api/customers?${qs}`)
   453→    },
   454→  })
   455→
   456→  // Kullanıcı listesi (sorumlu select'i için)
   457→  const { data: usersData } = useQuery({
   458→    queryKey: qk.users,
   459→    queryFn: () => apiGet<UsersResponse>('/api/users'),
   460→  })
   461→
   462→  const customers = data?.items ?? []
   463→  const total = data?.total ?? 0
   464→
   465→  // İstatistikler (mevcut listeye göre değil de global; yine de liste üzerinden hesaplayalım)
   466→  const stats = useMemo(() => {
   467→    const active = customers.filter((c) => c.status === 'aktif').length
   468→    const stale = customers.filter((c) => {
   469→      const d = daysSince(c.lastActivityAt)
   470→      return d === null || d > 30
   471→    }).length
   472→    const vip = customers.filter((c) => c.segment === 'vip').length
   473→    return { total, active, stale, vip }
   474→  }, [customers, total])
   475→
   476→  const canEdit = hasPermission(user as SessionUser | null, 'customers.edit')
   477→  const canExport = hasPermission(user as SessionUser | null, 'export.data')
   478→
   479→  // CSV dışa aktarma
   480→  const handleExport = () => {
   481→    if (!customers.length) {
   482→      toast.error('Dışa aktarılacak müşteri yok')
   483→      return
   484→    }
   485→    const rows = customers.map((c) => ({
   486→      Ad: c.name,
   487→      Sektör: c.sector,
   488→      Şehir: c.city || '',
   489→      Segment: getLabel(SEGMENTS, c.segment),
   490→      Telefon: formatPhone(c.phone),
   491→      'E-posta': c.email || '',
   492→      Durum: getLabel(CUSTOMER_STATUS, c.status),
   493→      Sorumlu: c.owner?.name || '',
   494→      'Son İletişim': c.lastActivityAt ? formatRelative(c.lastActivityAt) : '',
   495→    }))
   496→    const csv = toCSV(rows)
   497→    downloadFile(csv, `musteriler-${new Date().toISOString().slice(0, 10)}.csv`)
   498→    toast.success(`${customers.length} müşteri dışa aktarıldı`)
   499→  }
   500→
   501→  const handleClearFilters = () => {
   502→    setSearch(''); setSector(''); setCity(''); setDistrict(''); setCountry(''); setSegment(''); setStatus(''); setOwnerId(''); setStaleOnly(false)
   503→  }
   504→
   505→  const activeFilterCount = [search, sector, city, district, country, segment, status, ownerId].filter(Boolean).length + (staleOnly ? 1 : 0)
   506→
   507→  return (
   508→    <div className="space-y-5 animate-fade-in">
   509→      {/* Header */}
   510→      <div className="flex items-start justify-between flex-wrap gap-3">
   511→        <div>
   512→          <h2 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   513→            <Users className="w-6 h-6 text-emerald-600" />
   514→            Müşteri Portföyü
   515→          </h2>
   516→          <p className="text-sm text-muted-foreground mt-0.5">
   517→            Toplam <span className="font-semibold text-foreground">{total}</span> müşteri
   518→            {staleOnly && (
   519→              <span className="ml-2 text-amber-600 font-medium">· İletişimsiz filtresi aktif</span>
   520→            )}
   521→          </p>
   522→        </div>
   523→        <div className="flex gap-2 flex-wrap">
   524→          <Button
   525→            variant={staleOnly ? 'default' : 'outline'}
   526→            size="sm"
   527→            onClick={() => setStaleOnly((v) => !v)}
   528→            className={staleOnly ? 'bg-amber-600 hover:bg-amber-700' : ''}
   529→          >
   530→            <AlertTriangle className="w-4 h-4 mr-1.5" />
   531→            İletişimsizler
   532→            {stats.stale > 0 && (
   533→              <Badge variant="secondary" className="ml-1.5 h-5 px-1.5 text-[10px]">
   534→                {stats.stale}
   535→              </Badge>
   536→            )}
   537→          </Button>
   538→          {canExport && (
   539→            <Button variant="outline" size="sm" onClick={handleExport}>
   540→              <Download className="w-4 h-4 mr-1.5" />
   541→              Dışa Aktar
   542→            </Button>
   543→          )}
   544→          {canEdit && (
   545→            <Button size="sm" onClick={() => setAddOpen(true)}>
   546→              <Plus className="w-4 h-4 mr-1.5" />
   547→              Müşteri Ekle
   548→            </Button>
   549→          )}
   550→        </div>
   551→      </div>
