   590→// IRSALİYE (DISPATCH NOTE) — Sevk irsaliyesi, ağırlık takibi, palet
   591→// ============================================================
   592→// Numara formatı: IRS-2026-001, IRS-2026-002 (yıl + tenant bazlı sıra)
   593→// Durum: taslak | hazir | sevk_edildi | teslim_edildi | iptal
   594→// Ağırlık hesabı:
   595→//   totalNetWeight       = sum(line.weightPerUnit * line.qty)
   596→//   totalPackagingWeight = sum(product.packagingWeight * line.qty) (opsiyonel)
   597→//   palletWeight         = kullanıcı girişi (varsayılan 20 kg)
   598→//   totalGrossWeight     = net + packaging + pallet
   599→model Irsaliye {
   600→  id                 String   @id @default(cuid())
   601→  tenantId           String
   602→  tenant             Tenant   @relation(fields: [tenantId], references: [id], onDelete: Cascade)
   603→  orderId            String?
   604→  order              Order?   @relation(fields: [orderId], references: [id], onDelete: SetNull)
   605→  customerId         String
   606→  customer           Customer @relation(fields: [customerId], references: [id])
   607→  number             String   // IRS-2026-001
   608→  date               DateTime @default(now())
   609→  status             String   @default("taslak") // taslak | hazir | sevk_edildi | teslim_edildi | iptal
   610→  // Ağırlık takibi
   611→  totalNetWeight         Float?   // sum of line.weightPerUnit * line.qty
   612→  totalPackagingWeight    Float?   // sum of product.packagingWeight * line.qty
   613→  palletWeight            Float?   // user-entered pallet weight (default 20 kg)
   614→  totalGrossWeight        Float?   // net + packaging + pallet
   615→  palletCount             Int?
   616→  palletType              String?  // tahta | plastik | tek_kullanilik | eur_pallet | other
   617→  // Sevkiyat bilgileri
   618→  shippingAddress         String?
   619→  carrier                 String?  // yurtici | aras | mng | ptt | ups | fedex | dhl | other
   620→  trackingNo              String?
   621→  // Notlar
   622→  notes                   String?
   623→  createdById             String?
   624→  createdAt               DateTime @default(now())
   625→  updatedAt               DateTime @updatedAt
   626→  deliveredAt             DateTime?
   627→
   628→  lines IrsaliyeLine[]
   629→
   630→  @@unique([tenantId, number])
   631→  @@index([tenantId, status])
   632→}
