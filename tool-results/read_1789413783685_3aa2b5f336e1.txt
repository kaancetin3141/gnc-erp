   646→function CreateSiteDialog({ onCreated }: { onCreated: () => void }) {
   647→  const [open, setOpen] = useState(false)
   648→  const [form, setForm] = useState({ name: '', address: '', city: '', phone: '', managerName: '', dueDay: '5', defaultDueAmount: '500' })
   649→  const [saving, setSaving] = useState(false)
   650→
   651→  const handleSave = async () => {
   652→    setSaving(true)
   653→    try {
   654→      await apiPost('/api/site', form)
   655→      toast.success('Site oluşturuldu')
   656→      onCreated()
   657→      setOpen(false)
   658→    } catch (e) { toast.error(e instanceof Error ? e.message : 'Oluşturulamadı') }
   659→    finally { setSaving(false) }
   660→  }
   661→
   662→  return (
   663→    <>
   664→      <Button size="sm" className="bg-emerald-600 hover:bg-emerald-700" onClick={() => setOpen(true)}>
   665→        <Plus className="w-4 h-4 mr-1.5" /> Site Ekle
   666→      </Button>
   667→      <Dialog open={open} onOpenChange={setOpen}>
   668→        <DialogContent className="max-w-md">
   669→          <DialogHeader><DialogTitle>Yeni Site</DialogTitle></DialogHeader>
   670→          <div className="space-y-3">
   671→            <div><Label className="text-xs">Site Adı *</Label><Input value={form.name} onChange={e => setForm({ ...form, name: e.target.value })} /></div>
   672→            <div><Label className="text-xs">Adres</Label><Input value={form.address} onChange={e => setForm({ ...form, address: e.target.value })} /></div>
   673→            <div className="grid grid-cols-2 gap-2">

   687→        </DialogContent>
   688→      </Dialog>
   689→    </>
