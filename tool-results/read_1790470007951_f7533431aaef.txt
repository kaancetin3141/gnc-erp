   109→// ----------------------------------------------------------------------------
   110→// Map bounds + normalization + clustering
   111→// ----------------------------------------------------------------------------
   112→
   113→interface Bounds {
   114→  minLat: number
   115→  maxLat: number
   116→  minLng: number
   117→  maxLng: number
   118→}
   119→
   120→function computeBounds(results: MapsResult[]): Bounds | null {
   121→  if (results.length === 0) return null
   122→  let minLat = Infinity, maxLat = -Infinity, minLng = Infinity, maxLng = -Infinity
   123→  for (const r of results) {
   124→    minLat = Math.min(minLat, r.lat)
   125→    maxLat = Math.max(maxLat, r.lat)
   126→    minLng = Math.min(minLng, r.lng)
   127→    maxLng = Math.max(maxLng, r.lng)
   128→  }
   129→  const latRange = maxLat - minLat || 0.01
   130→  const lngRange = maxLng - minLng || 0.01
   131→  return {
   132→    minLat: minLat - latRange * 0.18,
   133→    maxLat: maxLat + latRange * 0.18,
   134→    minLng: minLng - lngRange * 0.18,
   135→    maxLng: maxLng + lngRange * 0.18,
   136→  }
   137→}
   138→
   139→function project(lat: number, lng: number, b: Bounds, zoom: number) {
   140→  const xRange = b.maxLng - b.minLng || 1
   141→  const yRange = b.maxLat - b.minLat || 1
   142→  let x = ((lng - b.minLng) / xRange) * 100
   143→  let y = ((b.maxLat - lat) / yRange) * 100 // invert Y
   144→  // zoom around center (50,50)
   145→  x = 50 + (x - 50) * zoom
   146→  y = 50 + (y - 50) * zoom
   147→  // clamp to viewport with padding
   148→  x = Math.max(4, Math.min(96, x))
   149→  y = Math.max(6, Math.min(94, y))
   150→  return { x, y }
   151→}

   153→interface MarkerItem {
   154→  type: 'cluster' | 'single'
   155→  x: number
   156→  y: number
   157→  count: number
   158→  results: MapsResult[]
   159→}
   160→
   161→function clusterMarkers(
   162→  positions: { x: number; y: number; result: MapsResult }[],
   163→  threshold: number,
   164→): MarkerItem[] {
   165→  const items: MarkerItem[] = []
   166→  const assigned = new Set<number>()
   167→  for (let i = 0; i < positions.length; i++) {
   168→    if (assigned.has(i)) continue
   169→    const group = [positions[i]]
   170→    assigned.add(i)
   171→    for (let j = i + 1; j < positions.length; j++) {
   172→      if (assigned.has(j)) continue
   173→      const dx = positions[i].x - positions[j].x
   174→      const dy = positions[i].y - positions[j].y
   175→      if (Math.sqrt(dx * dx + dy * dy) < threshold) {
   176→        group.push(positions[j])
   177→        assigned.add(j)
   178→      }
   179→    }
   180→    const cx = group.reduce((s, g) => s + g.x, 0) / group.length
   181→    const cy = group.reduce((s, g) => s + g.y, 0) / group.length
   182→    items.push({
   183→      type: group.length > 1 ? 'cluster' : 'single',
   184→      x: cx,
   185→      y: cy,
   186→      count: group.length,
   187→      results: group.map((g) => g.result),
   188→    })
   189→  }
   190→  return items
   191→}
