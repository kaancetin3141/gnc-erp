   193→function exportCSV(rows: Record<string, unknown>[], filename: string, sectionLabel: string) {
   194→  if (!rows.length) {
   195→    toast.error('Dışa aktarılacak veri yok', { description: `${sectionLabel} için kayıt bulunamadı.` })
   196→    return
   197→  }
   198→  const csv = toCSV(rows)
   199→  downloadFile(csv, filename, 'text/csv;charset=utf-8')
   200→  toast.success('CSV dışa aktarıldı', { description: `${filename} indirildi.` })
   201→}
   202→
   203→function exportXLSX(rows: Record<string, unknown>[], filename: string, sectionLabel: string) {
   204→  if (!rows.length) {
   205→    toast.error('Dışa aktarılacak veri yok', { description: `${sectionLabel} için kayıt bulunamadı.` })
   206→    return
   207→  }
   208→  // Excel'in açabileceği basit HTML tablosu
   209→  const headers = Object.keys(rows[0])
   210→  const escapeHtml = (v: unknown) => String(v ?? '').replace(/[&<>"']/g, (c) =>
   211→    ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' })[c] ?? c)
   212→  const html = `<!DOCTYPE html><html xmlns:o="urn:schemas-microsoft-com:office:office" xmlns:x="urn:schemas-microsoft-com:office:excel"><head><meta charset="UTF-8"></head><body><table border="1"><thead><tr>${headers.map((h) => `<th>${escapeHtml(h)}</th>`).join('')}</tr></thead><tbody>${rows.map((r) => `<tr>${headers.map((h) => `<td>${escapeHtml(r[h])}</td>`).join('')}</tr>`).join('')}</tbody></table></body></html>`
   213→  downloadFile('\uFEFF' + html, filename, 'application/vnd.ms-excel;charset=utf-8')
   214→  toast.success('XLSX dışa aktarıldı', { description: `${filename} indirildi.` })
   215→}

   217→// Ay kısaltması: "2024-08" → "Ağu 24"
   218→function formatMonthLabel(m: string): string {
   219→  const [y, mo] = m.split('-')
   220→  const monthIdx = parseInt(mo, 10) - 1
   221→  const names = ['Oca', 'Şub', 'Mar', 'Nis', 'May', 'Haz', 'Tem', 'Ağu', 'Eyl', 'Eki', 'Kas', 'Ara']
   222→  return `${names[monthIdx] ?? mo} ${y.slice(2)}`
   223→}
   224→
   225→// "2024-08-15" → "15.08"
   226→function formatDayLabel(d: string): string {
   227→  const [, m, day] = d.split('-')
   228→  return `${day}.${m}`
   229→}
