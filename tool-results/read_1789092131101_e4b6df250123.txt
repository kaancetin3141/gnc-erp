  1102→  1102→function formatFileSize(bytes: number): string {
  1103→  1103→  if (bytes === 0) return '0 B'
  1104→  1104→  const k = 1024
  1105→  1105→  const sizes = ['B', 'KB', 'MB', 'GB']
  1106→  1106→  const i = Math.floor(Math.log(bytes) / Math.log(k))
  1107→  1107→  return `${Math.round((bytes / Math.pow(k, i)) * 10) / 10} ${sizes[i]}`
  1108→  1108→}
