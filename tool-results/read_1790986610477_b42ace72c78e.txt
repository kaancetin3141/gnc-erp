   691→// IRSALİYE (DISPATCH NOTE) — Sevk irsaliyesi, ağırlık takibi, palet
   692→// ============================================================
   693→// Numara formatı: IRS-2026-001, IRS-2026-002 (yıl + tenant bazlı sıra)
   694→// Durum: taslak | hazir | sevk_edildi | teslim_edildi | iptal
   695→// Ağırlık hesabı:
   696→//   totalNetWeight       = sum(line.weightPerUnit * line.qty)
   697→//   totalPackagingWeight = sum(product.packagingWeight * line.qty) (opsiyonel)
   698→//   palletWeight         = kullanıcı girişi (varsayılan 20 kg)
   699→//   totalGrossWeight     = net + packaging + pallet
   700→model Irsaliye {
   701→  id                 String   @id @default(cuid())

   703→  tenant             Tenant   @relation(fields: [tenantId], references: [id], onDelete: Cascade)
   704→  orderId            String?

   706→  customerId         String
   707→  customer           Customer @relation(fields: [customerId], references: [id])
   708→  number             String   // IRS-2026-001
   709→  date               DateTime @default(now())
   710→  status             String   @default("taslak") // taslak | hazir | sevk_edildi | teslim_edildi | iptal
   711→  // Ağırlık takibi
   712→  totalNetWeight         Float?   // sum of line.weightPerUnit * line.qty
   713→  totalPackagingWeight    Float?   // sum of product.packagingWeight * line.qty
   714→  palletWeight            Float?   // user-entered pallet weight (default 20 kg)
   715→  totalGrossWeight        Float?   // net + packaging + pallet
   716→  palletCount             Int?
   717→  palletType              String?  // tahta | plastik | tek_kullanilik | eur_pallet | other
   718→  // Sevkiyat bilgileri
   719→  shippingAddress         String?
   720→  carrier                 String?  // yurtici | aras | mng | ptt | ups | fedex | dhl | other
   721→  trackingNo              String?
   722→  // Notlar
   723→  notes                   String?
   724→  createdById             String?
   725→  createdAt               DateTime @default(now())
   726→  updatedAt               DateTime @updatedAt
   727→  deliveredAt             DateTime?
   728→
   729→  lines IrsaliyeLine[]
   730→
   731→  @@unique([tenantId, number])
   732→  @@index([tenantId, status])
